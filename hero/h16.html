חוזרים לזוז בלי כאב, ובלי לחכות חודשיים לתור

טיפול אישי של 45 דקות ותוכנית תרגילים לבית. משאירים שם וטלפון, ואנחנו מתקשרים לקבוע.

מחזירים טלפון באותו יוםבלי התחייבות

תור פנוי השבועשלישי, 08:30

שורת הטופס היא ההבדל בין הירו שמבקש ללחוץ להירו שכבר מקבל את הפנייה: שני שדות בלבד, כי כל שדה נוסף מוריד השארת פרטים, ואת השאר שואלים בטלפון. היא רחבה מהעמודה שלה ב-112 פיקסלים, חוצה את הרווח ונכנסת לתמונה, וזה מה שמחבר את שני החצאים לאובייקט אחד; משטח וצל, בלי מסגרת. לכל שדה תווית גלויה בתוכו (לא placeholder במקום תווית), והטלפון הוא type=tel עם autocomplete, כך שהמכשיר ממלא אותו בנגיעה, והוא נכתב משמאל לימין. הבדיקה רצה רק בשליחה ולא בזמן ההקלדה: השגיאה נכתבת במילים ב-role=alert, השדה מסומן ב-aria-invalid והפוקוס חוזר אליו. אחרי שליחה השורה מתחלפת בתודה עם השם והפוקוס עובר אליה; בפרויקט התודה מוצגת רק אחרי שה-CRM ענה. הצ׳יפ על התמונה הוא עובדה (התור הפנוי הבא), לא קישוט. בטלפון התמונה הופכת לפס נמוך למעלה, והשדות יושבים זה לצד זה מעל כפתור ברוחב מלא. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
