H33 · הירו צילום עם נקודות מוצר

הירו CSS · JS הירו ופתיחהתמונות וגלריותhover ומיקרו ממתין

שתי עמודות: בצד אחד הכותרת, המשפט והכפתורים, ומתחתם רשימה קצרה של מה שרואים בצילום עם מחירים. בצד השני צילום גבוה של החלל עם נקודה על כל פריט. הרשימה והנקודות מחוברות: מעבר על שורה מדגיש את הנקודה שלה, ולחיצה על שורה או על נקודה פותחת כרטיס ליד הפריט עצמו.

מתי משתמשים: עסק שהעבודה שלו היא חלל שלם ובתוכו פריטים שאפשר להזמין: מטבחים, נגרות, עיצוב פנים, ריהוט, תאורה, אופנה בצילום אווירה, חנות כלי בית. הצילום מוכר את התמונה הגדולה, והרשימה עונה על "מה זה וכמה זה" גם למי שלא לוחץ על נקודות. לא ליותר מחמישה פריטים, ולא לצילום שבו הפריטים קטנים מכדי לזהות.

מטבח שנבנה סביב הדרך שבה אתם מבשלים

תכנון, ייצור בנגרייה שלנו והתקנה, עם החומרים והפרזול שבחרתם. כל מה שרואים בצילום אפשר להזמין.

צילום אווירה שהוא גם קטלוג, בשתי עמודות שמיושרות זו לזו: העמודה של הטקסט גבוהה בדיוק כמו הצילום, ההצעה צמודה לקצה העליון שלו והרשימה "בצילום הזה" לקצה התחתון. הרשימה והנקודות נכתבות מאותו מערך נתונים ומחוברות: מעבר עכבר או פוקוס מקלדת על שורה מגדיל את הנקודה שלה עם טבעת ומחליש את האחרות, וההפך; לחיצה על שורה או על נקודה פותחת את אותו כרטיס ליד הפריט, עם שורת פרטים, מחיר וקישור. כך מי שלא לוחץ על נקודות עדיין רואה מה יש ובכמה, וקורא מסך מקבל רשימה רגילה של כפתורים עם aria-expanded. למסגרת יש צורה קבועה (6/5), ולכן המיקום של כל נקודה הוא אחוזים מהצילום המלא (--x, --y), שעוברים דרך החיתוך עם --o ו---f, במיקום פיזי (left) שלא מתהפך עם השפה; הרוחב שלה תלוי גם בגובה המסך, כך שההירו כולו נכנס במסך הראשון. הכרטיס לא יוצא מקצה הצילום (clamp). כרטיס אחד פתוח בכל פעם, Escape או לחיצה בחוץ סוגרים, ו-Escape מחזיר את הפוקוס למה שפתח אותו. שתי טבעות רכות אחרי הכניסה אומרות שאפשר ללחוץ, ונעצרות לפני חמש שניות. בטלפון: ההצעה, צילום שנחתך סביב האי (כל הנקודות זזות לפי אותו חיתוך), והרשימה בשתי עמודות; הכרטיס הופך לפס ברוחב הצילום, למטה או למעלה לפי מקום הנקודה. בפרויקט מודדים את המיקומים על הצילום הסופי, ובחנות הכרטיס מקשר לעמוד המוצר. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
(function(){
  var root=document.getElementById("h33"), fr=root.querySelector(".h33-fr"), open=-1, opener=null, hov=-1;
  enter(root);
  function all(k){return root.querySelectorAll('[data-k="'+k+'"]');}
  // the item in focus is the hovered one, or else the open one; it lights up in the list and on the photo together
  function paint(){
    var k=hov>=0?hov:open;
    root.querySelectorAll(".hot").forEach(function(e){e.classList.remove("hot");});
    if(k>=0)all(k).forEach(function(e){e.classList.add("hot");});
    fr.classList.toggle("has-hot",k>=0);
  }
  function close(back){
    if(open<0)return;
    document.getElementById("h33-c"+open).hidden=true;
    all(open).forEach(function(e){if(e.hasAttribute("aria-expanded"))e.setAttribute("aria-expanded","false");});
    if(back&&opener)opener.focus();
    open=-1; opener=null; paint();
  }
  // a row and its point open the same card, by its item; one card at a time
  root.querySelectorAll(".h33-it,.h33-hs").forEach(function(b){
    var k=+b.dataset.k;
    b.addEventListener("click",function(){
      var same=open===k; close(false); if(same)return;
      document.getElementById("h33-c"+k).hidden=false;
      all(k).forEach(function(e){if(e.hasAttribute("aria-expanded"))e.setAttribute("aria-expanded","true");});
      open=k; opener=b; paint();
    });
    b.addEventListener("pointerenter",function(e){if(e.pointerType==="mouse"){hov=k;paint();}});
    b.addEventListener("pointerleave",function(){hov=-1;paint();});
    b.addEventListener("focus",function(){if(b.matches(":focus-visible")){hov=k;paint();}});
    b.addEventListener("blur",function(){hov=-1;paint();});
  });
  // Escape or a press anywhere else closes it, and Escape returns the focus to whatever opened it
  document.addEventListener("keydown",function(e){if(e.key==="Escape")close(true);});
  document.addEventListener("click",function(e){if(open>=0&&!e.target.closest(".h33-hs,.h33-it,.h33-card"))close(false);});
})();
