H9 · הירו כותרת עם תמונות בתוך השורה

כותרת גדולה וממורכזת, ובין המילים שלה שלוש תמונות קטנות בצורת גלולה, בגובה האותיות. המילים עולות בזו אחר זו, וכל תמונה נפתחת מהמרכז לצדדים מיד אחרי המילה שלפניה.

מתי משתמשים: עסק שיש לו צילומים קטנים וטובים של העבודה עצמה: מאפייה, בית קפה, סטודיו, קליניקה, חנות מוצר. עובד כשכל תמונה מראה את המילה שלידה. לא כשיש רק צילום אחד, ולא עם צילומי סטוק כלליים.

לתפריט הבוקראיך מגיעים
שלוש תמונות ולא יותר: אחת לכל חלק במשפט, וכל אחת מראה את המילה שלפניה (ידיים בבצק אחרי "לחם"). הגלולה נמדדת ב-em ולא בפיקסלים, ולכן היא תמיד בגובה האותיות, גם כשהכותרת מתכווצת בטלפון, ו-vertical-align של -.05em מושיב אותה על קו הבסיס של העברית. היא נפתחת ב-clip-path מהמרכז החוצה ולא ב-width, כדי שהשורה לא תקפוץ בזמן שהתמונה נפתחת. התמונות דקורטיביות (alt ריק ו-aria-hidden), כי הכותרת שלמה גם בלעדיהן וקורא מסך לא צריך לשמוע "ידיים לשות בצק" באמצע משפט. המילים עולות בהפרש של 60 מילישניות, והפתיחה כולה נגמרת ב-1.2 שניות. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

JS

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
