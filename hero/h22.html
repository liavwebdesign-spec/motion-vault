כותרת וכפתורים במרכז, ומתחתם חמישה כרטיסים שנפתחים כמו מניפה מתוך ערימה אחת ועולים מתחתית ההירו. בכל כרטיס צילום ושם קצר, והאמצעי הוא כרטיס טקסט בצבע המותג. ריחוף מרים כרטיס.

מתי משתמשים: עסק שיש לו כמה מוצרים או סוגים שכדאי לראות יחד: גלידרייה, קונדיטוריה, חנות קטנה, סדנאות, תפריט עונתי, מותג קוסמטיקה. לא כשיש רק מוצר אחד או פחות מארבעה צילומים; אז h3.

גלידה שמכינים כל בוקר, בטעמים שמתחלפים עם העונה

חלב מקיבוץ בגליל, פירות מהשוק, ושישה עשר טעמים בוויטרינה. לטעום לפני שבוחרים זה חלק מהעניין.

  • תות ושמנת
  • הטעם של השבועפיסטוק ומלח ים

כל הכרטיסים יושבים באותה נקודה ותלויים מציר אחד שנמצא הרחק מתחתם (transform-origin של 230%), כך שזווית שונה לכל אחד (--r) פותחת אותם למניפה, בלי לחשב מיקומים. הכניסה מתחילה בערימה סגורה שעולה מלמטה ונפתחת מהאמצע החוצה. הזווית וההזזה הן תכונות נפרדות (rotate ו-translate), ולכן הריחוף מרים כרטיס בלי לשבור את הזווית שלו ובלי להתנגש בכניסה. הריחוף לא נשען על :hover: הכרטיסים חופפים, וכרטיס שהורם ועלה לראש הערימה היה מכסה את השכן שלו וחוסם אותו. לכן הכרטיס נבחר לפי הזווית מהציר אל המצביע (הפרוסה שלו במניפה), ולא לפי מה שנמצא למעלה, והבחירה לפי הזווית הבסיסית (--r) יציבה ולא מהבהבת. השכנים נוטים 5 מעלות הצידה (--s) ומפנים לו מקום. אחרי הכניסה (המחלקה settled) מושהיות הכניסה יורדות, אחרת כרטיס שעוזבים נשאר למעלה עד שמגיע תורו, ו-z-index חוזר רק אחרי 0.2 שניות, כדי שכרטיס שיורד לא ייעלם מאחורי השכנים באמצע התנועה. הכרטיס האמצעי הוא טקסט בצבע המותג (הטעם של השבוע), ולכן המניפה אומרת משהו ולא רק מראה תמונות; התמונות דקורטיביות, והשם הקצר על כל כרטיס הוא הטקסט שלו, והוא יושב בקצה שהכרטיס הסמוך לא מכסה. הכרטיסים נחתכים בקצה התחתון של ההירו בכוונה, כמו קלפים ביד. בטלפון נשארים שלושה. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
(function(){
  var root=document.getElementById("h22"); enter(root);
  var fan=root.querySelector(".h22-fan"), cards=[].slice.call(fan.children), act=null;
  function settle(){root.classList.add("settled");}
  fan.addEventListener("transitionend",function(e){if(e.propertyName==="rotate")settle();});
  setTimeout(settle,3200);
  if(!matchMedia("(hover:hover) and (pointer:fine)").matches)return;
  function deg(c){return parseFloat(c.style.getPropertyValue("--r"));}
  function lift(c){
    if(c===act)return; act=c;
    // the lifted card gets room: the cards on each side lean five degrees away from it
    cards.forEach(function(o){
      o.classList.toggle("up",o===c);
      o.style.setProperty("--s",!c||o===c?"0deg":(deg(o)>deg(c)?"5deg":"-5deg"));
    });
  }
  // all cards hang from one pivot far below the fan: the angle from it to the pointer says which slice, and so which card, it is in
  fan.addEventListener("pointermove",function(e){
    if(!e.target.closest(".h22-card")){lift(null);return;}
    var b=fan.getBoundingClientRect(), cw=cards[0].offsetWidth||cards[1].offsetWidth, k=parseFloat(getComputedStyle(fan).getPropertyValue("--k"))||1;
    var px=b.left+b.width/2, py=b.bottom+cw*.3+cw*4/3*1.3;
    var a=Math.atan2(e.clientX-px,py-e.clientY)*180/Math.PI, best=null, gap=1e9;
    cards.forEach(function(c){if(!c.offsetWidth)return; var g=Math.abs(a-deg(c)*k); if(g<gap){gap=g;best=c;}});
    lift(best);
  });
  fan.addEventListener("pointerleave",function(){lift(null);});
})();
