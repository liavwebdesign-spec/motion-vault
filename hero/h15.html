H15 · הירו מצגת מסך מלא עם פסי התקדמות

הירו CSS · JS הירו ופתיחהתמונות וגלריותניווט ומבנה ממתין

שלוש תמונות במסך מלא שמתחלפות לבד, כל אחת עם שם קצר, ופס התקדמות לכל אחת כמו בסטורי. הכותרת והכפתור נשארים קבועים, נגיעה בשם עוברת לתמונה שלו, ויש כפתור עצירה.

מתי משתמשים: עסק שצריך להראות כמה מקומות או חלקים כבר במסך הראשון: מלון (חוף, חדרים, בר), נדל״ן (בניין, דירה, נוף), אולם, מסעדה עם כמה חללים, אתר תיירות. לא כשיש רק צילום אחד טוב; אז h3.

עשרים ושניים חדרים, כולם עם מרפסת לים, וארוחת בוקר עד הצהריים.

לבדיקת זמינות
הכותרת והכפתור לא מתחלפים עם התמונות. רק השם הקטן והתמונה מתחלפים, ולכן המסר נקרא פעם אחת ולא בורח באמצע קריאה. הפס של השקופית הנוכחית הוא השעון: אנימציית CSS של 6 שניות, ו-animationend שלה מביא את הבאה, כך שעצירה (animation-play-state) עוצרת גם את הזמן וגם את הזום האיטי בבת אחת, בלי טיימר מקביל שיברח. נגיעה בשם עוברת אליו. בהפחתת תנועה אין החלפה אוטומטית ואין זום: השמות עדיין מחליפים תמונה, וכפתור העצירה יורד כי אין מה לעצור. שכבת הקריאות אלכסונית מהפינה של הטקסט (התורה), וההיפוך שלה ל-LTR כתוב. רק התמונה הראשונה נשלפת בעדיפות; השתיים שמחכות נטענות רגיל ולא ב-lazy, כי הן כבר בתוך המסך ו-lazy רק מסכן מעבר למסגרת ריקה. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
(function(){
  var root=document.getElementById("h15"), pp=root.querySelector(".hpp"); enter(root);
  var sl=[].slice.call(root.querySelectorAll(".h15-sl")), tabs=[].slice.call(root.querySelectorAll(".h15-tab")), cur=0;
  // reduced motion: no autoplay, the names still switch; the button has nothing to stop, so it leaves
  var auto=matchMedia("(prefers-reduced-motion:no-preference)").matches;
  if(!auto){root.classList.add("still"); pp.hidden=true;}
  function go(n){
    cur=(n+sl.length)%sl.length;
    sl.forEach(function(s,k){s.classList.toggle("on",k===cur); s.setAttribute("aria-hidden",String(k!==cur));});
    tabs.forEach(function(t,k){t.setAttribute("aria-current",String(k===cur)); t.classList.toggle("done",k<cur);
      // restart the bar: drop the animation for one frame
      var i=t.querySelector("i"); i.style.animation="none"; void i.offsetWidth; i.style.animation="";});
  }
  // the bar of the current slide is the clock: when it fills, the next slide comes
  tabs.forEach(function(t,k){
    t.addEventListener("click",function(){go(k);});
    t.querySelector("i").addEventListener("animationend",function(){if(auto&&k===cur)go(cur+1);});
  });
  pp.addEventListener("click",function(){var on=!root.classList.contains("paused"); root.classList.toggle("paused",on);
    pp.setAttribute("aria-pressed",String(on)); pp.setAttribute("aria-label",on?"הפעלת המצגת":"עצירת המצגת");});
})();
