H19 · הירו לפני ואחרי עם ידית גרירה

הירו CSS · JS הירו ופתיחהתמונות וגלריותhover ומיקרו ממתין

טקסט בצד אחד, ובצד השני אותו חדר פעמיים, לפני ואחרי, עם ידית שגוררים לרוחב. אחרי הכניסה הידית זזה פעם אחת לבד כדי להראות שאפשר, ומשם היא של המבקר: בעכבר, במגע ובמקלדת.

מתי משתמשים: עסק שהתוצאה שלו נראית: תאורה, שיפוצים, ניקיון, צביעה, גינון, אסתטיקה, עריכת תמונות. עובד רק עם זוג צילומים מאותה זווית בדיוק. לא כשאין צילום "לפני" אמיתי; אז h11.

אותו חדר. רק עם אור שתוכנן נכון

תאורה בשכבות, עמעום לפי שעות היום, וגופים שנבחרו לחדר, במקום עוד ספוט בתקרה.

לתיאום פגישת תכנוןפרויקטים
אחרילפני
זה ההירו שמראה במקום לספר: אותו חדר בשני מצבים, והמבקר עושה את ההשוואה בעצמו. הידית היא input מסוג range אמיתי ושקוף שמכסה את כל התמונה, ולכן גרירה בעכבר, נגיעה בטלפון וחיצים במקלדת עובדים בלי קוד גרירה, והפוקוס שלו מצויר על העיגול. ההשוואה תמיד משמאל לימין, עם "לפני" בצד ימין, הצד שקורא עברית פוגש ראשון, והשכבה העליונה נחתכת ב-clip-path לפי --p, בלי לשנות רוחב. אחרי הכניסה הידית זזה פעם אחת לבד, ימינה וחזרה, וזה במקום הוראה כתובה; היא לא זזה אם המבקר כבר נגע בה, ולא בהפחתת תנועה. הקו של הידית הוא חלק מהפקד, לא קישוט. בדמו ה"לפני" הוא אותו צילום, מוחשך ודהוי ב-filter; בפרויקט שמים שני צילומים אמיתיים מאותה זווית, אחרת ההשוואה נראית מזויפת. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
(function(){
  var root=document.getElementById("h19"); enter(root);
  var cmp=root.querySelector(".h19-cmp"), rng=root.querySelector(".h19-rng"), touched=false;
  function set(v){cmp.style.setProperty("--p",v+"%");}
  function say(){rng.setAttribute("aria-valuetext",rng.value+" אחוז מהתמונה אחרי");}
  rng.addEventListener("input",function(){touched=true; set(rng.value); say();});
  say();
  // one hint, once: after the entrance the handle swings right and back, only when motion is allowed and nobody touched it
  if(!matchMedia("(prefers-reduced-motion:no-preference)").matches)return;

      if(k<1)requestAnimationFrame(f);
    });
  },1000);});
})();
