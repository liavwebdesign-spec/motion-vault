H10 · הירו טקסט במרכז ותמונות בכנפיים

הירו CSS · JS הירו ופתיחהתמונות וגלריות ממתין

טור טקסט צר במרכז, וארבע תמונות בגדלים ובגבהים שונים בשני הצדדים, חלקן נחתכות בקצה המסך. התמונות נכנסות מלמעלה ומלמטה, ובגלילה זזות כל אחת במהירות אחרת.

מתי משתמשים: עסק שמוכר אווירה ויש לו כמה צילומים באותה תאורה: בר, מסעדת ערב, מלון בוטיק, אולם אירועים, מותג אופנה. לא כשהטקסט ארוך משלוש שורות, כי הטור צר בכוונה.

להזמנת מקוםלתפריט
הטור במרכז צר בכוונה (עד 48% מהרוחב), כי הכנפיים צריכות אוויר: כותרת של שלוש שורות ומשפט אחד, לא יותר. הכנפיים יושבות על inset לוגי ולכן מתחלפות צד לבד ב-RTL וב-LTR, ושתיים מהן נחתכות בקצה המסגרת, וזה מה שהופך אותן ממסגרת סימטרית לקומפוזיציה. הן נכנסות מלמעלה ומלמטה לסירוגין (--fy) ולא מהצדדים, כי כניסה מהצד תלויה בכיוון הקריאה. הסחיפה בגלילה יושבת על התמונה שבפנים (עם scale של 1.18 כדי שלא ייפתח חור), כי הכנף עצמה מחזיקה את טרנספורם הכניסה, ונעצרת בהפחתת תנועה. בטלפון נשארות שתי כנפיים קטנות שמציצות מהפינות התחתונות. פונט התצוגה (Bellefair) הוא דוגמה: בפרויקט נכנס פונט התצוגה של העור. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
(function(){
  var root=document.getElementById("h10"); enter(root);
  // scroll drift: each wing at its own speed, on the image inside (the wing itself owns the entrance transform)
  if(!matchMedia("(prefers-reduced-motion:no-preference)").matches)return;
  var imgs=[].slice.call(root.querySelectorAll(".h10-wing img")), rates=[-.08,.05,-.05,.1], raf=0;

    imgs.forEach(function(im,k){im.style.setProperty("--py",(-r.top*rates[k]).toFixed(1)+"px");});}
  addEventListener("scroll",function(){if(!raf)raf=requestAnimationFrame(upd);},{passive:true});
})();
