טקסט בצד אחד, תמונה גבוהה בצד השני, ושני כרטיסי נתון שיושבים על קצוות התמונה ונכנסים אחרונים. בלי קישוט: התמונה והנתון עושים את העבודה.

מתי משתמשים: עסק שיש לו נתון אמיתי להראות (שנים, לקוחות, זמן תגובה): מערכת, שירות מקצועי, קליניקה, חברת השמה. לא כשהמספרים מומצאים, שם הם מורידים אמון.

בדיקה ראשונה של 40 דקות, צילום, ותוכנית טיפול מודפסת עם מחירים. בלי הפתעות בכיסא.

גרסה ראשונה ישבה על כתם צבע אורגני (blob) מאחורי התמונה, וליאב פסל אותו: נישתי ולא מתחבר (23.9.2026). עכשיו אין קישוט בכלל. התמונה גבוהה (4:4.4) כדי לאזן עמודת טקסט של חמש שורות, והכרטיסים יוצאים מקצה התמונה ב-32 וב-24 פיקסלים, מספיק כדי להיראות מונחים עליה ולא כדי לצוף באוויר. הם יושבים על inset לוגי ולכן מתהפכים לבד ב-RTL וב-LTR, נעלמים במובייל, ונכנסים אחרונים (--i 4 ו-5, הכרטיס השני נגמר ב-1.15 שניות) כי הם הפרט, לא המסר. אם אין נתון אמיתי, בוחרים הירו אחר. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

JS

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
