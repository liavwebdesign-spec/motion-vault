ספירה לאחור שמוכרת כרטיסים ולא מלחיצה: ימים, שעות ודקות, בלי שניות, כך ששום דבר לא מהבהב כל שנייה ואין צורך בכפתור עצירה; הספרה שהשתנתה מחליקה פנימה בחצי שנייה, ובהפחתת תנועה פשוט מתחלפת. הספרות טבלאיות (tabular-nums) כדי שמספר שמשתנה לא יזיז את השכנים, ובלי אפס מוביל. היעד הוא שעה בשעון ישראל (UTC), כך שמבקר מחו"ל רואה את אותה ספירה. בלי JS הספירה מוסתרת ונשאר התאריך הכתוב ב-time עם datetime, כך שאין מצב של אפסים או מספר שגוי. role=timer בלי הכרזות חיות, כדי שקורא מסך לא יקריא כל דקה. כשהאירוע התחיל, בפרויקט (data-roll="0") הספירה מתחלפת בשורה שאומרת שהוא כבר התחיל; הדמו מגלגל שנה קדימה. הצילום מתהפך בעברית כדי שהזמרת תעמוד בצד הרחוק מהטקסט, ונשאר כפי שצולם באנגלית; לא הופכים צילום עם כיתוב או לוגו. שכבת הקריאות מהפינה התחתונה בצד של הטקסט. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
