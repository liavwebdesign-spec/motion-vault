H24 · הירו דירוג למעלה ורצועת ביקורות

הירו CSS · JS הירו ופתיחהטעינה ופידבקרקע ואווירה ממתין

כותרת ממורכזת שמעליה שורת דירוג: עיגולי ראשי תיבות, כוכבים ומספר הביקורות. מתחת לכפתורים רצועה של כרטיסי ביקורת שגולשת לאט לרוחב, עם קצוות נמוגים, עצירה בריחוף וכפתור עצירה גלוי.

מתי משתמשים: עסק שהלקוח בוחר בו לפי ביקורות ויש לו הרבה כאלה: הובלות, שיפוצים, ניקיון, מרפאה, מוסך, בית ספר לנהיגה, קייטרינג. הביקורות הן הטיעון, ולכן הן במסך הראשון ולא בתחתית העמוד. לא לעסק עם פחות מעשרים ביקורות אמיתיות, ולעולם לא עם ביקורות ממוצאות.

הדירוג עולה לפני הכותרת, כי מי שבוחר מוביל, שיפוצניק או מרפאה בודק קודם כמה כוכבים יש לו, ורק אחר כך קורא מה הוא מבטיח. עיגולי ראשי התיבות הם רמז לאנשים אמיתיים בלי צילומי פנים מבנק תמונות, והכוכבים בצבע הטקסט ולא בצהוב, כדי שיחזיקו בכל עור. רצועת הביקורות כתובה פעמיים והאנימציה מזיזה אותה בחצי ועוד חצי רווח, כך שהלולאה בלי קפיצה; הכיוון מתהפך באנגלית. זו תנועה של יותר מחמש שניות, ולכן יש כפתור עצירה גלוי, ריחוף עוצר אותה, ובהפחתת תנועה היא לא זזה בכלל: נשאר עותק אחד והרצועה נגללת ביד, והיא יכולה לקבל פוקוס כדי שגם המקלדת תגלול. העותק השני מוסתר מקורא מסך, כדי שכל ביקורת תיקרא פעם אחת. בפרויקט הביקורות מגיעות מגוגל או מהפאנל, עם שם פרטי ועיר, ולעולם לא נכתבות בשביל הדמו. פתיחת העמוד (motion.md 2): ההירו מוסתר רק מאחורי html.open-anim, שהשורה הראשונה במארקאפ קובעת רק כשתנועה מותרת (בפרויקט היא עוברת ל-head). בלי JS הכל גלוי, ואם הסקריפט לא רץ ה-CSS מציג הכל אחרי 2.5 שניות. בעמוד עם הדר קובעים --t0 של 0.2 שניות כדי שההדר ייכנס ראשון, ומקצרים כך שהפתיחה כולה לא עוברת 1.2 שניות. המסך הראשון בפרויקט: min-height:100svh, תוכן ממורכז לגובה וריפוד עליון בגובה ההדר; hero-fit ו-hero-vcenter מכריעים, לא הגובה של הדמו.

// in view = fire. Measured directly on scroll and resize, not through IntersectionObserver:
// an observer does not update while the tab is throttled (and on a real site the hero is usually
// on screen at load anyway), and an entrance that never fires leaves the hero invisible.
function inView(el,f){
  var ro;
  function chk(){
    var r=el.getBoundingClientRect();
    if(r.top<innerHeight*.9&&r.bottom>0){off();f();}
  }
  function off(){removeEventListener("scroll",chk);removeEventListener("resize",chk);if(ro)ro.disconnect();}
  addEventListener("scroll",chk,{passive:true}); addEventListener("resize",chk);
  // a layout change with no scroll (a bar above it closes, a font swaps in) can also bring it into view
  if(window.ResizeObserver){ro=new ResizeObserver(chk);ro.observe(document.body);}
  requestAnimationFrame(chk); setTimeout(chk,300);
}
function enter(root){
  var els=root.querySelectorAll(".rv");
  els.forEach(function(el,i){if(!el.style.getPropertyValue("--i"))el.style.setProperty("--i",i);});
  // a script that ran late, after the 2.5s CSS failsafe already showed the hero: keep it shown, no second entrance
  if(els.length&&getComputedStyle(els[0]).opacity==="1")root.classList.add("is-in");
  root.classList.add("rv-on");
  inView(root,function(){root.classList.add("is-in");});
}
