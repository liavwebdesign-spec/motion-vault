CSS34 · קצה שמתעמעם רק כשיש עוד

CSS טהור CSS · scroll-driven animation ניווט ומבנהטעינה ופידבק ממתין

קופסה שנגללת (שורת צ'יפים, טבלת השוואה רחבה, רשימה ארוכה) מעמעמת את הקצה שלה רק בכיוון שבו יש עוד תוכן: בתחילה ההתחלה חדה, ככל שגוללים ההתחלה דוהה בהדרגה, ובקצה האחרון הדהייה בסוף נעלמת. כשהכל נכנס ואין מה לגלול, אין דהייה בכלל.

מתי משתמשים: שורת קטגוריות או תגיות שגולשת בטלפון, טבלת מחירים או השוואה רחבה בתוך עטיפה אופקית, רשימה ארוכה בתוך פאנל בגובה קבוע. בכל מקום שבו חיתוך חד של תוכן נראה כמו באג ולא כמו הזמנה לגלול.

מקור: shadcn/ui, תוסף scroll-fade, https://github.com/shadcn-ui/ui (packages/shadcn/src/tailwind.css), רישיון MIT, והרעיון של Shu Ding, https://codepen.io/shuding/pen/WNLGQor (CodePen, MIT). הקוד נכתב מחדש בלי Tailwind: שני משתנים רשומים (@property, מסוג length) --sf-s ו---sf-e שמייצגים את גודל הדהייה בהתחלה ובסוף, שני keyframes שמגדלים את הראשון מ-0 לגודל המלא ומקטינים את השני לאפס, ושתי אנימציות שכל אחת מחוברת ל-scroll(self inline) (או block לאנכי) עם animation-range משלה: 0 עד 96px בהתחלה, ו-calc(100% - 96px) עד 100% בסוף. המסכה היא linear-gradient אחד שקורא את שני המשתנים. הערך 96px הוא כמה גוללים עד שהדהייה מלאה, וה---sf-size הוא רוחב הדהייה עצמה (48 עד 56 פיקסלים), והם בכוונה נפרדים. כשאין גלישה, ציר הגלילה לא פעיל והאנימציות לא רצות, ולכן אין דהייה בכלל, בלי שורה אחת של קוד לבדיקה. RTL: הכיוון נקבע ב-:dir(rtl) ו-:dir(ltr) (ברירת המחדל עברית: to left), וציר ה-inline של הגלילה מתחיל מהצד הימני, כך שההתחלה בעברית היא הימנית ונבדקה בכרום: בטעינה הצד הימני חד והשמאלי דוהה. מלכודות: animation בקיצור מאפס את animation-timeline ואת animation-range ולכן הם מוגדרים אחריו; המסכה נחתכת לפי תיבת הגבול ולכן אין צל חיצוני על האלמנט המוסתר (הרקע נשאר, הצל לא); scroll-padding בצ'יפים מונע ממקלדת להביא צ'יפ אל תוך הדהייה. מסלול נפילה: ב-@supports not (animation-timeline: scroll()) נשארת דהייה סטטית קטנה בסוף ו-JS קטן מחשב את אותם שני משתנים בגלילה, כך שגם דפדפן בלי התמיכה מקבל את אותה התנהגות, והתוכן אף פעם לא מוסתר. תנועה מופחתת: זה מחוון מצב ולא קישוט, ולכן לא מכבים אותו. בלי JS: הדהייה עצמה בלי JS בכלל בדפדפן תומך; רק בחירת הצ'יפ דורשת JS. שלוש התשובות: (1) הגרסה הבנאלית היא דהייה קבועה בשני הצדדים, או צל פנימי שנשאר גם כשהגעתם לסוף ומסתיר את הפריט האחרון, או חיתוך חד בלי שום רמז; כאן הדהייה מדווחת על המצב האמיתי של הקופסה בכל רגע. (2) המיקרו-אינטראקציה שנושאת מצב: הדהייה עצמה, שגדלה ונעלמת עם הגלילה, ובחירת צ'יפ שמרכזת אותו ומעדכנת את שני הקצוות. (3) הפעולה שזה מקל: הבנה שיש עוד בלי לנחש, ותנועה לצד הנכון בלי חצים.

(function(){
  const reduce=()=>matchMedia("(prefers-reduced-motion: reduce)").matches;
  // צ'יפ שנבחר מתמרכז: שני הקצוות של השורה מתעדכנים לבד, כי הם נקראים מהגלילה
  document.querySelectorAll(".sf-chips").forEach(row=>{
    const chips=[...row.querySelectorAll(".sf-chip")];
    chips.forEach(c=>c.addEventListener("click",()=>{
      chips.forEach(x=>x.setAttribute("aria-pressed",String(x===c)));
      c.scrollIntoView({inline:"center",block:"nearest",behavior:reduce()?"auto":"smooth"});
    }));
  });
  // דפדפן בלי scroll-driven animation: אותם שני משתנים, מחושבים בגלילה. בדפדפן תומך אין שורה אחת שרצה.
  if(window.CSS&&CSS.supports&&CSS.supports("animation-timeline: scroll()"))return;
  const clamp=v=>Math.max(0,Math.min(1,v));
  document.querySelectorAll(".sf").forEach(el=>{
    const x=el.classList.contains("sf-x"),size=parseFloat(getComputedStyle(el).getPropertyValue("--sf-size"))||48,reveal=96;
    function sync(){
      const pos=x?Math.abs(el.scrollLeft):el.scrollTop,max=x?el.scrollWidth-el.clientWidth:el.scrollHeight-el.clientHeight;
      if(max<=1){el.style.setProperty("--sf-s","0px");el.style.setProperty("--sf-e","0px");return;}
      el.style.setProperty("--sf-s",size*clamp(pos/reveal)+"px");
