CSS48 · שני פסי מילים שמצטלבים באיקס

CSS טהור CSS keyframes · שכפול מדוד · IntersectionObserver טקסט וכותרותרקע ואווירה ממתין

שני פסים רחבים ומוטים שחוצים את הסקשן באיקס, אחד בצבע המותג ואחד כהה. בכל פס מילים גדולות שזורמות בלולאה, והפסים זזים בכיוונים מנוגדים. הפס הכהה עובר מעל הצבעוני באמצע. ריחוף עוצר, ויש כפתור עצירה.

מתי משתמשים: מעבר בין סקשנים באתר עם אנרגיה: בית ספר למוזיקה, סטודיו לריקוד, חדר כושר, פסטיבל, מותג לצעירים. מילים קצרות שאומרות מה יש אצלכם, לא משפטים. פעם אחת בעמוד. לא באתר שקט ורציני (משפט, רפואה, פיננסים), ולא מתחת להירו שכבר זז; שם b01 ישר ועדין.

שחזור. השראה: animtd.store (26.10.2026), סרטון של שני פסי טקסט שמצטלבים וזזים בכיוונים מנוגדים. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: b01 הוא פס אחד, ישר ושקט, לנקודות אמון. כאן שני פסים מוטים בזוויות מנוגדות (מינוס 6 ופלוס 5 מעלות) שחוצים זה את זה, בצבע מלא, עם מילים גדולות, והם הסקשן עצמו ולא שוליים שלו. הלולאה: אותו מנגנון כמו ב-b01. ה-JS משכפל את קבוצת המילים עד שהיא מכסה את רוחב הפס ועוד אחת, ומזיז ברוחב קבוצה אחת בפיקסלים, כך שהתפר לא נראה. המהירות נקבעת בפיקסלים לשנייה, ולכן שני הפסים זזים באותו קצב גם כשאחד ארוך יותר. הפס השני רץ ב-animation-direction:reverse. כיוון: ב-RTL העותקים נערמים שמאלה וההזזה ימינה; ב-LTR הסימן מתהפך (--wm-sign). הפסים רחבים ב-30% מהסקשן כדי שהקצוות המוטים לא ייראו, והסקשן חותך אותם רק לרוחב (overflow-x:clip עם overflow-y:visible): חיתוך בשני הצירים היה קוטע את הצל של הפסים בקצה העליון והתחתון כשהסקשן נמוך. נגישות: הפסים aria-hidden (עותקים חוזרים), ולצדם רשימה מוסתרת של כל המילים פעם אחת. עוצר בריחוף, בכפתור, ומחוץ למסך. הפחתת תנועה: הפסים עומדים והכפתור לא מופיע. בטלפון אותו דבר, באותיות קטנות יותר.

JS

(function(){
  const root=document.querySelector("[data-wm]");
  if(!root)return;
  const bands=[...root.querySelectorAll(".wm-band")];
  // כמה עותקים צריך: אחרי הזזה של קבוצה אחת, מה שנשאר חייב עדיין לכסות את כל הפס
  const build=b=>{
    const track=b.querySelector(".wm-track"),proto=track.firstElementChild;
    [...track.children].slice(1).forEach(c=>c.remove());
    const w=proto.offsetWidth;if(!w)return;
    const need=Math.ceil(b.offsetWidth/w)+1;
    for(let i=1;i<need;i++)track.appendChild(proto.cloneNode(true));
    track.style.setProperty("--wm-shift",w+"px");
    track.style.setProperty("--wm-dur",(w/70).toFixed(2)+"s");   // אותו קצב בפיקסלים בשני הפסים
  };
  const all=()=>bands.forEach(build);
  all();
  // מדידה לפני שהפונט נטען יוצאת קצרה, והלולאה קופצת בכל סיבוב
  if(document.fonts)document.fonts.ready.then(all);
  let t;addEventListener("resize",()=>{clearTimeout(t);t=setTimeout(all,200);});
  const pp=root.querySelector("button[data-pp]");
  if(!matchMedia("(prefers-reduced-motion: reduce)").matches)pp.hidden=false;
  pp.addEventListener("click",()=>{
    const on=!root.classList.contains("is-paused");
    root.classList.toggle("is-paused",on);
    pp.querySelector("span").textContent=on?"המשך התנועה":"עצירת התנועה";
  });
  // מחוץ למסך התנועה עומדת
  if("IntersectionObserver" in window)new IntersectionObserver(es=>es.forEach(e=>root.classList.toggle("is-off",!e.isIntersecting))).observe(root);
})();
