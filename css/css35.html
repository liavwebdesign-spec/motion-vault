CSS35 · תמונה שגולשת בתוך המסגרת בהחלקה

CSS טהור CSS scroll-snap · scroll() timeline תמונות וגלריותכרטיסים ממתין

קרוסלת תמונות נטיבית עם scroll-snap, שבה בכל החלקה התמונה גולשת בתוך המסגרת שלה לאט יותר מהמסגרת, כמו הצצה דרך חלון. התנועה קשורה למיקום הגלילה, בלי JS. יש חצים, גרירה בעכבר, מקלדת, מונה ופס התקדמות.

מתי משתמשים: גלריית עבודות, מוצרים, צוות או מקרי בוחן בתמונות גדולות, כשהתמונה היא הסיפור והגולש מחליק ביד אחת. לא לצילומי מסמכים או תמונות שחשוב לראות כולן בלי חיתוך, ולא יותר מקרוסלה אחת בעמוד.

מקור: Preethi Sam, https://codepen.io/rpsthecoder/pen/JoXbNbg (CodePen, MIT). בקוד המקורי הרקע של כל שקופית מונפש ב-background-position-x תחת animation-timeline: view(inline). כאן נכתב מחדש עם תמונה אמיתית (img) שרחבה 130 אחוז מהמסגרת, ממורכזת, ו-translate שרץ מצד לצד לפי ה-view timeline, מה שזול הרבה יותר מ-object-position כי translate רץ על הקומפוזיטור ולא מצייר מחדש. הכיוון: המסגרת נעה בכיוון ההחלקה, והתמונה זזה בכיוון ההפוך ביחס אליה, ולכן במסך היא נעה לאט יותר מהמסגרת. חשוב: בכוונה לא השתמשתי ב-view-timeline של הפרויקט המקורי. בדקתי בכרום 155 ב-RTL, והפאזה של view(inline) שם לא נכונה: ההתקדמות של כל שקופית מחושבת לפי מיקום השקופית ההפוכה בשורה, כך שהראשונה והאחרונה עומדות במקום בזמן שהן על המסך והאמצעיות זזות. ב-LTR זה תקין. לכן כל תמונה רצה על scroll-timeline של המסלול (scroll-timeline: --slgt inline), עם animation-range משלה: מרחק שקופית i משפת ההתחלה הוא i * (רוחב השקופית + הרווח), והטווח הוא מהרגע שהמסגרת נכנסת מהצד הרחוק (מרחק מינוס רוחב המסלול) עד שהיא יוצאת מהצד הקרוב (מרחק ועוד רוחב השקופית). זו אותה נוסחה בעברית ובאנגלית, כי היא מודדת מרחק משפת ההתחלה ולא כיוון פיזי, והיא נבדקה בכרום בשני הכיוונים: תמונה במרכז המסלול ללא הזזה, ובצדדים בדיוק פלוס ומינוס 11.5 אחוז. רוחב השקופית מוגדר פעם אחת ב---slg-fw (min(340px,78cqw)) והמסלול הוא container-type: inline-size, ולכן ה-cqw בטווח מתייחס לרוחב המסלול ועובד גם כשהרוחב משתנה. לכל שקופית יש --i באינליין (0 עד 4). הסימן ב---slg-s הוא 1 בעברית ו--1 ב-:dir(ltr), כי כיוון התנועה על המסך מתהפך. האנימציה כולה בתוך @supports (animation-timeline: scroll()) ו-@media (prefers-reduced-motion: no-preference); בלעדיהם התמונה פשוט עומדת במרכז המסגרת ולא נחתכת. הקרוסלה שלמה: scroll-snap mandatory, המסלול הוא region עם tabindex ו-aria-label וחצי מקלדת, חצי הקודם והבא מגללים שקופית אחת עם scrollBy לפי מרחק בין שפות ההתחלה (כך הסימן נכון גם ב-RTL, שבו scrollLeft שלילי), הם מקבלים aria-disabled בקצוות כדי שהפוקוס לא ייעלם, ויש גרירה בעכבר בלבד (במגע הדפדפן עושה את זה לבד). פס ההתקדמות הוא גם scroll timeline (scroll-timeline על המסלול ו-timeline-scope על העטיפה) ומוצג רק בדפדפן שתומך. בלי JS: ההחלקה, ה-snap והתנועה של התמונה עובדים, הכפתורים מוסתרים. תנועה מופחתת: אין גלישה של התמונה, וההחלקה עצמה נשארת. מגע: כפתורים של 44 פיקסלים, בלי hover. הפרש מ-b31: b31 הוא סליידר המלצות (כרטיסי טקסט עם נקודות); כאן מוסיפים את התנועה הפנימית של התמונה שמתאימה לקרוסלות תמונה, ואת מונה המיקום, ותנועת התמונה היא CSS בלבד. שלוש התשובות: (1) הגרסה הבנאלית היא תמונה קבועה בתוך כרטיס שמחליק, שנראית כמו מדבקה; כאן התמונה חיה בתוך המסגרת בכל פיקסל של החלקה, ולכן ההחלקה מרגישה כמו עומק ולא כמו החלפת כרטיסים, בלי ספרייה ובלי סקריפט. (2) המשוב: מונה וקו התקדמות שזזים עם האצבע, וחצים שנכבים בקצה. (3) הפעולה שזה מקל: לעבור בין עבודות עם החלקה אחת, ולהרגיש כמה נשאר בלי לספור.

document.querySelectorAll(".slg").forEach(root=>{
  const track=root.querySelector(".slg-track"),slides=[...track.children];
  const prev=root.querySelector(".slg-btn.prev"),next=root.querySelector(".slg-btn.next"),num=root.querySelector(".slg-count b");
  const rtl=getComputedStyle(track).direction==="rtl";
  const reduce=()=>matchMedia("(prefers-reduced-motion: reduce)").matches;
  // שפת ההתחלה: ימין בעברית, שמאל באנגלית. כל החישובים לפי מרחק בין שפות, ולכן הסימן נכון גם כש-scrollLeft שלילי
  const edge=el=>{const r=el.getBoundingClientRect();return rtl?r.right:r.left;};
  const pos=()=>Math.abs(track.scrollLeft),max=()=>track.scrollWidth-track.clientWidth;
  const atStart=()=>pos()<4,atEnd=()=>max()-pos()<4;
  function current(){
    const s=edge(track);let best=0,d=1e9;
    slides.forEach((c,i)=>{const k=Math.abs(edge(c)-s);if(k<d){d=k;best=i;}});
    return best;
  }
  function go(i){
    i=Math.max(0,Math.min(slides.length-1,i));
    track.scrollBy({left:edge(slides[i])-edge(track),behavior:reduce()?"auto":"smooth"});
  }
  // המונה מראה עד איזו שקופית כבר ראיתם במלואה: בדסקטופ נראות כמה יחד, ובטלפון אחת, ובשני המקרים בסוף המסלול זה המספר האחרון
  function seen(){
    const t=track.getBoundingClientRect();let n=1;
    slides.forEach((c,i)=>{const r=c.getBoundingClientRect();if(r.left>=t.left-2&&r.right<=t.right+2)n=i+1;});
    return atEnd()?slides.length:n;
  }
  function sync(){
    num.textContent=seen();
    prev.setAttribute("aria-disabled",String(atStart()));
    next.setAttribute("aria-disabled",String(atEnd()));
  }
  // aria-disabled ולא disabled: כפתור שנכבה תוך כדי שהוא בפוקוס לא משאיר את המקלדת באוויר
  prev.addEventListener("click",()=>{if(!atStart())go(current()-1);});
  next.addEventListener("click",()=>{if(!atEnd())go(current()+1);});
  track.addEventListener("scroll",()=>requestAnimationFrame(sync),{passive:true});
  track.addEventListener("keydown",e=>{
    if(e.key!=="ArrowLeft"&&e.key!=="ArrowRight")return;
    e.preventDefault();
    const forward=(e.key==="ArrowLeft")===rtl;   // בעברית השקופית הבאה היא משמאל
    go(current()+(forward?1:-1));
  });
  // גרירה בעכבר בלבד. במגע הדפדפן כבר גולל לבד.
  let down=false,sx=0,sl=0;
  track.addEventListener("pointerdown",e=>{if(e.pointerType!=="mouse"||e.button!==0)return;down=true;sx=e.clientX;sl=track.scrollLeft;track.classList.add("drag");track.setPointerCapture(e.pointerId);});
  track.addEventListener("pointermove",e=>{if(down)track.scrollLeft=sl-(e.clientX-sx);});
  const up=()=>{if(!down)return;down=false;track.classList.remove("drag");go(current());};
  ["pointerup","pointercancel"].forEach(ev=>track.addEventListener(ev,up));
  root.classList.add("slg-js");   // רק עם JS מציגים חצים ומונה: בלעדיו ההחלקה והתנועה של התמונה עובדות כמו שהן
  sync();
});
