const btn=document.querySelector(".rbtn"),wrap=document.querySelector(".rframe");
btn.addEventListener("click",()=>{
  if(wrap.classList.contains("on")){wrap.classList.remove("on");wrap.innerHTML="";btn.textContent="טען דמו חי כאן";return;}
  wrap.innerHTML='<iframe loading="lazy" src="https://reactbits.dev/text-animations/split-text"></iframe>';
  wrap.classList.add("on");btn.textContent="סגור דמו";
});
