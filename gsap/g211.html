G211 · שורות שנוספות אחת אחרי השנייה והישנות מתעמעמות

GSAP GSAP · ScrollTrigger scrub · sticky טקסט וכותרותתהליך ושלבים ממתין

משפט פתיחה אחד נעוץ במסך, ובכל גלילה נוספת שורה חדשה מתחתיו. השורה שנכנסה זה עתה היא היחידה בצבע הטקסט המלא, והקודמת מתעמעמת לאפור. כך הרשימה נבנית כמו משפט מצטבר, והעין תמיד יודעת איפה הקורא נמצא.

מתי משתמשים: שירותים, שלבי עבודה או התחייבויות של עסק, בארבע עד שש שורות קצרות שכל אחת משלימה את הקודמת. מתאים לעמוד שירות, לאודות ולתהליך עבודה. פעם אחת בעמוד, בסקשן הנעוץ היחיד. לא לשורות ארוכות ולא כשהסדר ביניהן לא חשוב.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שני סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם שורות נוספות אחת אחת והישנות הופכות אפורות. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: כל השורות מוצגות, הישנות באפור (--muted) והאחרונה בצבע המלא. בלי GSAP ובהפחתת תנועה רואים את זה, בלי גובה נוסף. כשהמהלך רץ .cs מקבל is-live וגובה של מסך ועוד 56% מסך לכל שורה (--n נקבע מכמות השורות), והבמה נעוצה (sticky). אותו קוד עובד גם בטלפון, כי הגובה וההצמדה לא תלויים ברוחב. הצבעים נקראים מהעור: האפור משורה ראשונה והצבע המלא מהאחרונה (getComputedStyle), כך ש-GSAP מעביר בין שני ערכי rgb ולא מנסה לפתור var(). כל צעד: השורה החדשה עולה בשקיפות ובתנועה של 28 פיקסלים, והקודמת מתעמעמת באותו רגע. אין תנועה אופקית, ולכן אין מה להפוך באתר משמאל לימין. נגישות: רק opacity ולא visibility, כך שקורא מסך שומע את כל השורות גם לפני שהן מופיעות. מספר השורות: ארבע עד שש. יותר מזה הופך את הגלילה לארוכה.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP כל השורות מוצגות, הישנות באפור והאחרונה במלואה
  const root=document.querySelector(".cs"),ls=[...root.querySelectorAll(".cs-l")];
  // בהפחתת תנועה הפונקציה לא רצה, והרשימה נשארת במצב הסופי
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const muted=getComputedStyle(ls[0]).color,ink=getComputedStyle(ls[ls.length-1]).color;
    root.classList.add("is-live");root.style.setProperty("--n",ls.length);
    gsap.set(ls,{color:ink});gsap.set(ls.slice(1),{opacity:0,y:28});
    const tl=gsap.timeline({defaults:{ease:"power2.out"},
      scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.5,invalidateOnRefresh:true}});
    // בכל צעד: השורה החדשה עולה והקודמת מתעמעמת באותו רגע
    ls.slice(1).forEach((l,i)=>{
      tl.to(ls[i],{color:muted,duration:.5,ease:"none"},i+.2).to(l,{opacity:1,y:0,duration:.6},i+.2);
    });
    tl.to({},{duration:.5});      // עצירה בסוף, כדי שהשורה האחרונה תישאר במלואה רגע לפני שהעמוד ממשיך
    return()=>{if(tl.scrollTrigger)tl.scrollTrigger.kill();tl.kill();root.classList.remove("is-live");root.style.removeProperty("--n");gsap.set(ls,{clearProps:"color,opacity,transform"});};
  });
})();
