G271 · דמות שמציצה מקצה המסך ומחליפה צד בכל סקשן

GSAP GSAP · ScrollTrigger · sticky overlay · SVG תמונות וגלריותתהליך ושלבים ממתין

דמות איורית ידידותית מציצה מתחתית המסך ואוחזת בקצה. כשסקשן חדש מגיע היא צוללת, עוברת לצד הנגדי לטקסט ועולה שוב, מביטה אל התוכן ומצמצת. כשהסקשן מגיע לאמצע המסך נפתחת מעליה בועת דיבור קצרה בגוף ראשון שמתאימה לו, והדמות מזיזה את הפה בזמן שהמילים נכנסות. לפני הסקשן הראשון ואחרי האחרון היא מסתתרת. השכבה לא חוסמת לחיצות והטקסט נשאר תוכן רגיל.

מתי משתמשים: עמוד תהליך, אודות או שירות עם שלושה עד חמישה סקשנים, כל אחד בגובה של כשבעים אחוז מהמסך עם טקסט בצד אחד ותמונה בצד השני, ודמות אחת שמייצגת את העסק: קמע של המותג או איור של בעל העסק. הדמות מלווה את הקורא לאורך העמוד ומדברת בגוף ראשון. לא בעמוד קצר, לא עם אנשים שאינם חלק מהמותג, ולא כשהטקסט בסקשן מגיע עד הקצה התחתון שלו.

מסירה שאפשר לגור בה מיד

ביום המסירה עוברים יחד על כל חדר, רושמים הערות ומסיימים אותן לפני שאתם נכנסים.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון בגלריה של prompt-motion.com (21.10.2026) שבו דמות מציצה מקצה המסך בכל סקשן ומתחלפת בין הצדדים, לפעמים עם בועת דיבור. נבנה מאפס מהצפייה. מה שונה בגרסה הזו (25.10): הגרסה הראשונה הייתה תמונה אמיתית חתוכה במסכה עם קו מתאר עבה, והיא נראתה כמו גזירה ולא כמו דמות, וגם לא התאימה לכל עסק. עכשיו הדמות איור SVG שנבנה מאפס: ראש עגול, ידיים שאוחזות בקצה המסך וקוקו קטן שמתנדנד, בצבעי העור (--accent לגוף, --ink לעיניים ולפה, --bg לקו המדבקה סביבה). יש לה חיים משלה: היא מביטה אל הצד של התוכן, מצמצת מדי כמה שניות ומזיזה את הפה בזמן שהמילים נכנסות לבועה. היא קטנה יותר, ולכן לא מכסה חצי תמונה. התזמון: הדמות מחליפה צד כשהסקשן הבא עובר את קו 75% מהמסך, אבל הבועה נפתחת רק כשהסקשן מגיע לאמצע המסך (שני ScrollTrigger לכל סקשן). כך הבועה לא מדברת על סקשן שעדיין בקושי נראה: קודם הדמות מציצה, ורק אחר כך היא מדברת. המארקאפ הוא המצב הסופי: סקשנים של כותרת ופסקה עם תמונה אמיתית מהמאגר בצד הנגדי לטקסט (.pk-ph, גובה קבוע וחיתוך object-fit), והשכבה של הדמות מוסתרת (display:none). בלי GSAP ובהפחתת תנועה זה מה שרואים, וכל המידע נשאר בטקסט. כשהמהלך רץ הסקריפט מוסיף is-live על העוטף והשכבה נעשית position:sticky בגובה המסך, עם margin-bottom שלילי באותו גובה כדי שלא תוסיף גובה לעמוד (העוטף חותך עם overflow:clip, כדי שה-sticky ימשיך לעבוד). המעבר: פונקציה אחת, show(i), הורגת את המעבר הקודם ובונה חדש: הדמות יורדת (yPercent), הצד והמבט מוחלפים בזמן שהיא מוסתרת, והיא עולה מעט מעבר לנקודת המנוחה ונוחה בה, כך שהידיים נשארות על קצה המסך. בטלפון הקו הוא 50% לשני הטריגרים, ולכן תמיד סקשן אחד בדיוק פעיל. על הדמות יש תנועת ציפה איטית (sine.inOut) שנעצרת כשהיא מוסתרת, וצמצום עיניים ב-delayedCall שנעצר יחד איתה. הצד נקבע לפי הכיווניות: הסקריפט קורא את direction של העוטף ובוחר צד פיזי (data-pos), והטקסט בסקשנים מתחלף לפי justify-self לוגי, כך שהדמות תמיד בצד שמנגד לטקסט. באתר LTR אין מה להחליף, רק לוודא שהסקשן הראשון משאיר את הצד הנגדי פנוי. להחליף בדמות של העסק: מחליפים את ה-SVG בתוך .pk-fig בתמונה חתוכה על רקע שקוף (PNG או WebP) של בעל העסק או הקמע. קו מתאר אפשר לקבל עם כמה drop-shadow על .pk-fig, ואם התמונה אינה חתוכה, במסכה. אז מוחקים את ההתייחסויות לעיניים ולפה בסקריפט. בועה בצבע --ink: כך היא נראית גם על סקשן בגוון --card. הטקסט שלה נכתב במילים של העסק בגוף ראשון ובלשון שמתאימה לדמות. התמונה והדמות: הדמות יושבת בצד התמונה ומציצה מעליה, ולכן הפינה החיצונית התחתונה של התמונה מוסתרת חלקית; לא שמים שם פרט חשוב. מובייל: העמודות הופכות לעמודה אחת עם התמונה מתחת לטקסט, הדמות קטנה יותר והבועה נשארת בתחתית המסך, מעל התמונה ורחוק מהטקסט הפעיל. נגישות: הדמות והבועה aria-hidden, והטקסט של הבועה נשמר בסקשן כ-data-say ולא מוסיף מידע שאין בגוף העמוד.

      gsap.set(pupils,{x:0,y:2});
    }
    // הבועה נפתחת רק כשהדמות למעלה והסקשן שלה הגיע לאמצע המסך
    function talk(){
      if(!ready||cur<0||said!==cur||open===cur)return;
      open=cur;
      const txt=secs[cur].getAttribute("data-say")||"";
      say.textContent="";
      const ws=txt.split(" ").filter(Boolean).map((w,k,a)=>{const s=document.createElement("span");s.className="pk-wd";s.textContent=w;say.appendChild(s);if(k<a.length-1)say.appendChild(document.createTextNode(" "));return s;});
      if(bt)bt.kill();
      bt=gsap.timeline();
      bt.fromTo(bub,{autoAlpha:0,scale:.8},{autoAlpha:1,scale:1,duration:.35,ease:"power2.out"})
        .fromTo(ws,{autoAlpha:0,y:6},{autoAlpha:1,y:0,duration:.25,ease:"power2.out",stagger:.06},"-=.15")
        .fromTo(mouth,{scaleY:.55},{scaleY:1,duration:.08,ease:"sine.inOut",yoyo:true,repeat:ws.length*2-1},"<");
    }
    function hush(){
      open=-1;
      if(bt)bt.kill();
      bt=gsap.to(bub,{autoAlpha:0,scale:.85,duration:.18,ease:"power1.in"});
      gsap.to(mouth,{scaleY:.55,duration:.12,overwrite:"auto"});
    }
    // פונקציה אחת לכל החילופים: הורגת את המעבר הקודם, יורדת אם צריך, מחליפה צד ומבט, עולה ואז מדברת
    function show(i){
      if(i===cur)return;cur=i;ready=false;
      if(tl)tl.kill();
      if(blinkCall)blinkCall.kill();
      if(open>=0)hush();
      tl=gsap.timeline();
      const y=gsap.getProperty(peek,"yPercent");
      if(y<HIDE-1)tl.to(peek,{yPercent:HIDE,duration:.2+.3*(HIDE-y)/(HIDE-REST),ease:"power2.in"},0);
      if(i<0){tl.add(()=>{bob.pause();sway.pause();});return;}
      tl.add(()=>{place(i);bob.play();sway.play();})
        .fromTo(peek,{yPercent:HIDE},{yPercent:REST-5,duration:.6,ease:"power3.out"})
        .to(peek,{yPercent:REST,duration:.28,ease:"power2.inOut"})
        .to(pupils,{x:look*7,y:0,duration:.45,ease:"power2.inOut"},"-=.5")
        .add(()=>{ready=true;blinkCall=gsap.delayedCall(1.2,blink);talk();});
    }
    function hear(i){said=i;if(open>=0&&open!==i)hush();talk();}
    secs.forEach((s,i)=>{
      ScrollTrigger.create({trigger:s,start:()=>mob()?"top 50%":"top 75%",end:()=>mob()?"bottom 50%":"bottom 75%",invalidateOnRefresh:true,
        onEnter:()=>show(i),onEnterBack:()=>show(i),
        onLeave:()=>{if(i===secs.length-1)show(-1);},
        onLeaveBack:()=>{if(i===0)show(-1);}});
      ScrollTrigger.create({trigger:s,start:"top 50%",end:"bottom 50%",
        onEnter:()=>hear(i),onEnterBack:()=>hear(i),
        onLeave:()=>{if(i===secs.length-1)hear(-1);},
        onLeaveBack:()=>{if(i===0)hear(-1);}});
    });
    return()=>{
      if(tl)tl.kill();if(bt)bt.kill();if(blinkCall)blinkCall.kill();
      root.classList.remove("is-live");layer.removeAttribute("data-pos");
      gsap.set([peek,fig,bub,mouth,tuft].concat(eyes,pupils),{clearProps:"all"});
      say.textContent=first;
    };
  });
})();
