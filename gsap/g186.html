עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני התקדמות בגלריה של prompt-motion.com (21.10.2026), שבהם איור איזומטרי מתמלא בפריטים בזמן שמונה ושלבים עולים. נבנה מאפס מהצפייה. האיור הוא SVG אחד שחושב כקוביות איזומטריות (שלוש פאות, 30 מעלות) ומשתמש בצבעי ערכת הנושא בלבד, ולכן הוא מתלבש על כל עור. כל שלב הוא קבוצת g משלו (il-l1 עד il-l5), ואחריהן קבוצת הניצוצות. המארקאפ הוא מצב הסיום: איור שלם, כל העיגולים סגורים, 100%. בלי GSAP ובהפחתת תנועה רואים אותו, וה-ol של השלבים קריא לקורא מסך. כשהמהלך רץ הסצנה נעוצה ב-position:sticky בתוך אזור גלילה של 380vh (רק עם is-live, כדי שבלעדיו לא יהיה רווח ריק). ציר זמן אחד מוצמד לגלילה עם scrub של חצי שנייה: כל שכבה נכנסת בנקודה שלה, שתיים ראשונות בהקטנה קלה ושלוש אחרונות בנפילה של 18 יחידות. פונקציה אחת paint(t) קובעת מזמן הציר את מצב העיגולים (todo, cur, done), את שם השלב בכרטיס (דרך data-cur על השורש ו-CSS), את הפס ואת האחוזים, ולכן היא נכונה גם בגלילה למעלה. להתאים: מספר השלבים הוא מספר השכבות, וזמני ההתחלה הם המערך starts. כל איור אחר מתאים כל עוד הוא מחולק לקבוצות לפי שלבים. RTL: הפס מתמלא מימין, ב-LTR משנים את transform-origin של .il-fill. העיגולים מסתדרים לפי כיוון העמוד והקו ביניהם משתמש ב-inset-inline-start, כך שאין מה להפוך. בנייד הטקסט, האיור והכרטיס נערמים במסך אחד.

    steps=[...root.querySelectorAll(".il-step")],fill=root.querySelector(".il-fill"),pct=root.querySelector(".il-pct"),sn=root.querySelector(".il-sn");
  const n=steps.length,D=9,dur=.9,starts=[.3,1.95,3.6,5.25,6.9],span=starts[n-1]+dur-starts[0];   // זמני ההתחלה של השכבות בציר
  // כל מה שבכרטיס ובעיגולים נגזר מזמן הציר בלבד, ולכן נכון בגלילה קדימה ואחורה
  const paint=t=>{
    let cur=0,done=0;
    steps.forEach((s,i)=>{const v=t>=starts[i]+dur?"done":t>=starts[i]?"cur":"todo";if(v==="done")done++;if(t>=starts[i])cur=i+1;if(s.dataset.s!==v)s.dataset.s=v;});
    const p=Math.min(1,Math.max(0,(t-starts[0])/span));
    root.dataset.cur=done===n?n+1:Math.max(1,cur);
    sn.textContent=Math.min(n,Math.max(1,cur));
    pct.textContent=Math.round(p*100)+"%";
    gsap.set(fill,{scaleX:p});
  };
  // בהפחתת תנועה לא רצים, והאיור נשאר במצב הסיום
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");

    layers.forEach((g,i)=>tl.fromTo(g,{autoAlpha:0,y:i<2?0:-18,scale:i<2?.94:1},{autoAlpha:1,y:0,scale:1,duration:dur},starts[i]));
    sparks.forEach((p,i)=>tl.fromTo(p,{autoAlpha:0,scale:.3},{autoAlpha:1,scale:1,duration:.5,ease:"power2.out"},starts[n-1]+dur+.1+i*.08));
    tl.set({},{},D);                          // מותח את הציר ל-D כדי להשאיר רגע מנוחה בסוף
    tl.eventCallback("onUpdate",()=>paint(tl.time()));
    paint(0);
    return()=>{tl.scrollTrigger.kill();tl.kill();root.classList.remove("is-live");gsap.set([...layers,...sparks,fill],{clearProps:"all"});
      steps.forEach(s=>{s.dataset.s="done";});root.dataset.cur=n+1;sn.textContent=n;pct.textContent="100%";};

})();
