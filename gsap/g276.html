G276 · טבעות מקטעים שמסתובבות בקצבים שונים

GSAP GSAP · ScrollTrigger · SVG dasharray תהליך ושלביםרקע ואווירה ממתין

חמש טבעות מקטעים בגדלים שונים מסתובבות סביב עיגול במרכז, כל אחת בקצב ובכיוון משלה, ובכניסה פורצים מהמרכז קווים דקים החוצה. גלילה מהירה מאיצה את כולן לרגע. לצד הגרפיקה רשימה שמסבירה מה כל טבעת מייצגת.

מתי משתמשים: מחזורים שחוזרים בקצבים שונים: תחזוקה, גבייה, ביקורות, טיפולים, עדכוני אתר. חמישה פריטים לכל היותר, כל אחד עם שם וקצב. מתאים לסקשן תהליך או שירותים בעסק שמנהל דברים לאורך זמן. לא מתאים כשצריך להשוות כמויות, כי הטבעות מראות קצב ולא גודל.

כל מערכת בבניין נבדקת במחזור משלה

אנחנו מנהלים את כל המחזורים בלוח אחד ושולחים לוועד תזכורת לפני כל ביקור, כך שאף בדיקה לא נשכחת.

  • ניקיון חדרי מדרגות ותאורהכל שבוע
  • גינון והשקיהכל חודש
  • בדיקת מעליות ומערכות כיבוי אשכל רבעון
  • ניקוי מרזבים ובדיקת הגגפעמיים בשנה
  • בדיקת תקינות החשמל והמיםפעם בשנה
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם טבעות שבורות למקטעים מסתובבות סביב עיגול מרכזי בכיוונים מנוגדים, עם קווים דקים שפורצים החוצה בהתחלה ועם חוגה של שנתות. נבנה מאפס מהצפייה. כל טבעת היא circle אחד: pathLength=360 לכל העיגולים, כך שדפוס המקטעים (stroke-dasharray) נכתב במעלות, וסכום כל דפוס הוא בדיוק 360 כדי שהמקטעים ייסגרו בלי פער חסר. הטבעת החיצונית היא חוגת שנתות (72 שנתות, דפוס 1 4). הסיבוב: כל טבעת נעה בטווין אינסופי משלה סביב מרכז ה-SVG (svgOrigin:"300 300" שנקבע פעם אחת לכל הטבעות לפני כל טווין: אם הוא מופיע רק בחלק מהטוויינים, GSAP מתחיל מפינת התיבה ומשאיר הסטה קבועה במרכז), משך ממוצע של 90 שניות לטבעת החיצונית עד 12 לפנימית, בכיוונים מתחלפים, כך שהחיצונית כמעט עומדת והפנימית ברורה. הטווינים עוצרים כשהגרפיקה מחוץ למסך. תאוצת הגלילה: מהירות הגלילה (getVelocity()) מוחלקת בטיקר ומכפילה את timeScale של כל הסיבובים עד פי שש, ודועכת בחזרה. פריצת הקווים: 36 קווים דקים נמשכים מהמרכז החוצה ואז נמחקים החוצה (strokeDashoffset), פעם אחת בכניסה, והטבעות נכנסות בקנה מידה בהדרגה. הערך הסופי: בלי GSAP ובהפחתת תנועה רואים טבעות סטטיות וקווים נסתרים. הצבעים הם accent ו-ink בשקיפויות שונות מהטוקנים, והעיגול המרכזי הוא div ב-accent עם --accent-ink. נגישות: הגרפיקה aria-hidden, והמשמעות כולה ברשימה שליד. בחרו עד חמישה פריטים, ולכל טבעת פריט.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הטבעות סטטיות והקווים נסתרים (במארקאפ)
  const root=document.querySelector(".rn"),fig=root.querySelector(".rn-fig"),rings=gsap.utils.toArray(".rn-ring",root),lines=gsap.utils.toArray(".rn-burst line",root);
  const SPIN=[[90,1],[36,-1],[28,1],[22,-1],[17,1],[12,-1]];     // [שניות לסיבוב, כיוון] לכל טבעת, מהחיצונית לפנימית
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    // מרכז הסיבוב נקבע פעם אחת לכל הטבעות, לפני כל טווין. אם svgOrigin מופיע רק בחלק מהטוויינים,
    // GSAP מתחיל מהפינה של התיבה ומשאיר הסטה קבועה במרכז
    gsap.set(rings,{svgOrigin:"300 300"});
    const spins=rings.map((el,i)=>gsap.to(el,{rotation:SPIN[i][1]*360,duration:SPIN[i][0],ease:"none",repeat:-1,paused:true}));
    // כניסה חד פעמית: הטבעות גדלות אחת אחת, והקווים נמשכים מהמרכז החוצה ואז נמחקים החוצה
    gsap.timeline({scrollTrigger:{trigger:fig,start:"top 78%",once:true}})
      .fromTo(rings,{opacity:0,scale:.82},{opacity:1,scale:1,duration:.9,ease:"power3.out",stagger:.09},0)
      .set(lines,{opacity:.7,strokeDashoffset:190},.15)
      .to(lines,{strokeDashoffset:0,duration:.7,ease:"power2.out",stagger:{each:.012,from:"random"}},.15)
      .to(lines,{strokeDashoffset:-190,duration:.7,ease:"power2.in",stagger:{each:.012,from:"random"}},.75)
      .set(lines,{opacity:0});
    // הסיבובים רצים רק כשהגרפיקה על המסך
    ScrollTrigger.create({trigger:fig,start:"top bottom",end:"bottom top",onToggle:s=>spins.forEach(t=>s.isActive?t.play():t.pause())});
    // גלילה מהירה מאיצה את כל הטבעות לרגע, ודועכת בחזרה לקצב הרגיל
    let boost=1,target=1;
    ScrollTrigger.create({trigger:root,start:"top bottom",end:"bottom top",onUpdate:s=>{target=Math.max(target,1+Math.min(5,Math.abs(s.getVelocity())/450));}});
    const tick=()=>{boost+=(target-boost)*.12;target=1+(target-1)*.92;spins.forEach(t=>t.timeScale(boost));};
    gsap.ticker.add(tick);
    return()=>{gsap.ticker.remove(tick);spins.forEach(t=>t.kill());gsap.set(rings,{clearProps:"all"});gsap.set(lines,{clearProps:"opacity,strokeDashoffset"});};
  });
})();
