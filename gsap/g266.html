שחזור. השראה: שני סרטונים בגלריה (21.10.2026), בכל אחד אריחים קטנים נופלים אחד אחרי השני לתאים של רשת, בלוח שנה ובמשחק מילים. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: לוח עם תשעת האריחים בתאים והמספר 9, וליד הלוח רשימה (ol) של אותם ימים. בלי GSAP או בהפחתת תנועה זה מה שרואים. כשהמהלך רץ כל אריח מוסט למעלה אל מחוץ לגריד (overflow:clip על הגריד, כך שהאריח נכנס מהקצה העליון), ו-ScrollTrigger עם once מפעיל timeline אחד כשהלוח מגיע ל-70% מהמסך. הנפילה: power2.in עד הנחיתה, מעיכה קצרה (scaleY כלפי מטה מהבסיס), ויישור ב-power2.out. בלי back ו-bounce, שנחסמים, והמעיכה היא שתי תנועות רגילות. משך הנפילה גדל עם השורה, כדי שהמהירות תהיה דומה. הסדר הוא סדר התאריכים, כמה שרוצים: אפשר גם לפי השורה. המונה עולה בכל נחיתה ולא ברצף נפרד, ולכן תמיד מסכים עם מה שנוחת. הלוח נבנה מ-35 תאים (נובמבר 2026 מתחיל ביום ראשון, ולכן אין תאים ריקים בהתחלה), והוא מוסתר מקורא מסך כי הרשימה מכילה את אותו מידע. מסומן data-order כי הסדר הוא מידע. כיוון: יום ראשון בצד ימין, כמו לוח עברי. באתר LTR הלוח כבר מתהפך מעצמו. בטלפון התאים קטנים (כ-40px), לכן מספר היום נעלם מהאריחים והרשימה מתחת מוסרת את המידע. בפרויקט: התאריכים והצילומים מהיומן האמיתי של הלקוח, ואפשר להחליף את הרשימה בקישורים.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP כל האריחים כבר בתאים והמונה על 9
  const root=document.querySelector(".fc"),grid=root.querySelector(".fc-grid"),tiles=[...grid.querySelectorAll(".fc-tile")];
  const num=root.querySelector(".fc-count b"),N=tiles.length;
  // בהפחתת תנועה הפונקציה לא רצה, והאריחים נשארים בתאים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    num.textContent="0";
    const tl=gsap.timeline({paused:true});
    tiles.forEach((t,i)=>{
      const cell=t.parentElement,at=i*.36;
      // הנפילה מתחילה מעל הקצה העליון של הגריד (הגריד חותך), והמשך שלה גדל עם השורה
      const row=()=>Math.round(cell.offsetTop/(cell.offsetHeight||1)),fall=.34+.06*(Math.round(cell.offsetTop/(cell.offsetHeight||1)));
      tl.fromTo(t,{y:()=>-(cell.offsetTop+cell.offsetHeight+8)},{y:0,duration:fall,ease:"power2.in",immediateRender:true},at)
        .to(t,{scaleY:.88,scaleX:1.05,transformOrigin:"50% 100%",duration:.09,ease:"power1.out"},at+fall)
        .to(t,{scaleY:1,scaleX:1,duration:.24,ease:"power2.out"},at+fall+.09)
        .call(()=>{num.textContent=String(i+1);},null,at+fall);
    });
    ScrollTrigger.create({trigger:grid,start:"top 70%",once:true,onEnter:()=>tl.play()});
    return()=>{gsap.set(tiles,{clearProps:"all"});num.textContent=String(N);};
  });
})();
