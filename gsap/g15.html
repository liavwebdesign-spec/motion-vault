מתי משתמשים: אתרי פרימיום כבדי מדיה בלבד. פרילודר על אתר מהיר הוא עיכוב מיותר.

לא באתר מהיר, בדף נחיתה או בחנות; פרילודר על אתר קל רק מעכב גולשים.

הדמו כאן עיצובי-ניטרלי בכוונה. כשהמהלך נכנס לפרויקט, מיובאת רק ההתנהגות: הצבעים, הרדיוסים, הפונטים והצללים יורשים את העיצוב של אותו פרויקט.

הדמו

כאן יושב הלוגו או לוטי של המותג
HOLD בקוד הוא השהיה של הדמו בלבד, כדי שתספיק לראות. באתר אמיתי HOLD=0, והמסך עולה ברגע ה-load. בהפחתת תנועה המסך יורד בלי החלקה. אם GSAP לא נטען הוא יורד מיד, ואם הסקריפט לא רץ בכלל, ה-CSS מוריד אותו אחרי 2.5 שניות.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

(function(){
  const p=document.getElementById("pre");
  if(typeof gsap==="undefined"){p.style.display="none";return;}   // CDN חסום: המסך יורד מיד ולא מכסה את האתר
  p.style.animation="none";                  // מכאן הסקריפט אחראי, כולל הגיבוי של 3 שניות למטה
  const HOLD=.9;                             // דמו בלבד, כדי שתספיק לראות. באתר אמיתי 0
  const rm=matchMedia("(prefers-reduced-motion: reduce)");
  let run;
  function runPre(){
    if(run)run.kill();
    p.style.display="flex";gsap.set(p,{yPercent:0});
    // הפחתת תנועה: בלי החלקה, המסך פשוט יורד
    if(rm.matches){run=gsap.delayedCall(HOLD,()=>p.style.display="none");return;}
    run=gsap.to(p,{yPercent:-100,duration:.6,ease:"power3.inOut",delay:HOLD,
      onComplete:()=>p.style.display="none"});
  }
  let done=false;
  function hide(){
    if(done)return;done=true;
    if(rm.matches){p.style.display="none";return;}
    runPre();
  }
  if(document.readyState==="complete")hide();
  else{addEventListener("load",hide);setTimeout(hide,3000);}
  document.querySelector(".replay").addEventListener("click",runPre);
})();
