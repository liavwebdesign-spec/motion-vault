שתי תמונות זו לצד זו שעוברות אחת על פני השנייה ומחליפות צדדים בגלילה: אחת נסוגה וקטנה, והשנייה גדלה ובסוף עולה קדימה.

מתי משתמשים: השוואה בין שתי אפשרויות, לפני ואחרי, שני קהלים, שני שירותים. מספר ויזואלית שיש כאן שתי אפשרויות שמתחלפות ביניהן.

גלול. השתיים מחליפות צדדים, וזו שגדלה עולה קדימה.

הפרט שקובע הוא רגע החלפת ה-zIndex: הוא קורה בסוף המהלך, כשהתמונות כבר כמעט לא חופפות, ולכן העין לא תופסת קפיצה של שכבה. באמצע, כשהן חופפות לגמרי, זו שנסוגה עדיין מלפנים. ההחלפה נעשית ב-set על הטיימליין ולא ב-onUpdate, כי בטיימליין מסונכרן ב-scrub זה עובד גם בגלילה אחורה בלי קוד נוסף. לכל טווין משך מפורש של 1, כך שהמיקומים בציר הם אחוזי הגלילה ואפשר לקרוא אותם. הגודל משתנה יחד עם המעבר: זו שנסוגה קטנה וזו שמתקדמת גדלה, כך שכשההחלפה מגיעה בסוף, העין כבר מצפה שהגדולה תהיה מלפנים. הכיתובים נעלמים לרגע סביב נקודת החפיפה, אחרת "אפשרות א" ו"אפשרות ב" נכתבים זה על זה.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;
  const a=document.querySelector(".sx-a"),b=document.querySelector(".sx-b");
  const tags=gsap.utils.toArray(".sx-tag");
  // בהפחתת תנועה שתי התמונות עומדות זו לצד זו
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const tl=gsap.timeline({scrollTrigger:{trigger:".sx",start:"top top",end:"bottom bottom",scrub:.6}});
    // התנועה באחוזים של האלמנט עצמו, ולכן היא מדויקת בכל רוחב מסך. משך 1, כך שהמיקומים בציר הם אחוזי הגלילה
    tl.fromTo(a,{xPercent:0,scale:1,zIndex:2},{xPercent:-92,scale:.86,ease:"power1.inOut",duration:1},0)
      .fromTo(b,{xPercent:0,scale:.86,zIndex:1},{xPercent:92,scale:1,ease:"power1.inOut",duration:1},0)
      // הכיתובים נעלמים לרגע סביב נקודת החפיפה, אחרת "אפשרות א" ו"אפשרות ב" נכתבים זה על זה
      .to(tags,{opacity:0,duration:.08,ease:"none"},.34)
      .to(tags,{opacity:1,duration:.08,ease:"none"},.58)
      // החלפת השכבות בסוף המהלך, כשהתמונות כבר כמעט לא חופפות, ולכן העין לא תופסת קפיצה של שכבה
      .set(a,{zIndex:1},1)
      .set(b,{zIndex:2},1);
  });
})();
