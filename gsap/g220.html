G220 · סרט פסים עבה שנמתח על המסך וחושף את הסקשן הבא

GSAP GSAP · ScrollTrigger scrub · SVG stroke + mask תהליך ושלביםתמונות וגלריות ממתין

סרט עבה בכמה פסי צבע, עם מסגרת כהה וראש מעוגל, נכנס מימין המסך ושוחה במסלול מתעקל עד הקצה השמאלי. מאחורי הראש נחשף הסקשן הבא, והזנב ממשיך לחלוף עד שהסרט יוצא מהמסך. המעבר מרגיש כמו משיכת קו אחת שמעבירה אתכם הלאה.

מתי משתמשים: מעבר בין שני סקשנים חזקים באתר שיש לו אופי גרפי ושובב: סטודיו, מותג לילדים, אירוע, מרפאה ידידותית. פעם אחת בעמוד, כשיש סקשן אחד שנגמר ואחד שמתחיל, ושניהם עם תוכן. באתר רציני וקפדני עדיף לוח שקט.

מתחילים בשיחה קצרה

אתם מספרים מה מטריד, אנחנו מסבירים מה אפשר לעשות. בלי בדיקות ובלי למהר.

ואז מגיעה הבדיקה

הבדיקה נעשית בחדר שקט, אצל הרופא שהכרתם בשיחה. הוא מסביר כל שלב לפני שהוא מתחיל.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטוני קידום ב-prompt-motion.com (21.10.2026) שבהם צינור עבה בפסי צבע עם מסגרת כהה נמתח על פני הפריים בקשת. נבנה מאפס מהצפייה. ארבעה נתיבים באותו מסלול: מסגרת כהה, פס צבע, פס בהיר וקו דק במרכז, כל אחד בעובי קטן יותר, עם stroke-linecap:round. הנתיבים נבנים בפיקסלים של הבמה: הסקריפט מחשב את המסלול מהרוחב והגובה (נבנה מחדש ב-refresh), ולכן הסרט עובר מקצה לקצה בכל מסך, ובטלפון הוא לא נחתך. כל הנתיבים עם pathLength=1, כך שההתקדמות היא מספר בין 0 ל-1 ולא אורך בפיקסלים. נחש ולא קו: הקטע הנראה הוא stroke-dasharray באורך קבוע שזז לאורך המסלול, עם ראש וזנב, ולכן הסרט גם נמתח וגם יוצא מהמסך. החשיפה: הסקשן השני מוסתר ב-mask:url(#rb-m) שמצויר מאותו מסלול בקו עבה מאוד (פי 1.9 מגובה הבמה, כדי לכסות את כל הגובה), ומתקדם מעט אחרי ראש הסרט (LAG). שים לב: mask שמפנה ל-SVG עובד ב-Chrome וב-Firefox, ב-Safari צריך לבדוק; חלופה בטוחה היא clip-path עם קצה משופע כמו ב-g219. בלי GSAP ובהפחתת תנועה שני הסקשנים זה מתחת לזה והסרט לא נוצר בכלל. כיוון: הסרט נכנס מימין ויוצא בשמאל (קריאה בעברית). באתר משמאל לימין הופכים את הסדר של נקודות המסלול. נגישות: ה-SVG קישוט (aria-hidden), והתוכן של שני הסקשנים הוא טקסט ותמונה רגילים.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP שני הסקשנים זה מתחת לזה
  const root=document.querySelector(".rb"),pin=root.querySelector(".rb-pin");
  // בהפחתת תנועה הפונקציה לא רצה, והסרט לא נוצר
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const NS="http://www.w3.org/2000/svg";
    const mk=(n,a,p)=>{const e=document.createElementNS(NS,n);for(const k in a)e.setAttribute(k,a[k]);if(p)p.appendChild(e);return e;};
    const svg=mk("svg",{"class":"rb-svg","aria-hidden":"true","focusable":"false"});
    const mask=mk("mask",{id:"rb-m",maskUnits:"userSpaceOnUse"},mk("defs",{},svg));
    const mp=mk("path",{fill:"none",stroke:"white",pathLength:1,"stroke-linecap":"butt"},mask);
    const L=["rb-l0","rb-l1","rb-l2","rb-l3"].map(c=>mk("path",{"class":"rb-l "+c,pathLength:1},svg));
    pin.appendChild(svg);root.classList.add("is-live");
    const o={h:0},SEG=.5,LAG=.09;let w=0,h=0;
    // המסלול נבנה מהמידות של הבמה: נכנס מימין, יורד, עולה ויוצא משמאל
    const layout=()=>{
      w=pin.clientWidth;h=pin.clientHeight;
      const P=[[w+90,h*.3],[w*.85,h*.3],[w*.8,h*.7],[w*.58,h*.66],[w*.4,h*.62],[w*.38,h*.26],[w*.18,h*.32],[w*.08,h*.35],[-w*.04,h*.6],[-90,h*.6]];      // המסלול נגמר במשיק אופקי מחוץ למסך, כדי שהמסכה תכסה את כל הפינה
      const d="M"+P[0]+"C"+P.slice(1).map(p=>p.join(" ")).join(" ");
      const t=Math.min(Math.max(w*.045,26),60);
      svg.setAttribute("viewBox","0 0 "+w+" "+h);
      [mp].concat(L).forEach(p=>p.setAttribute("d",d));
      [1,.8,.5,.16].forEach((f,i)=>L[i].setAttribute("stroke-width",(t*f).toFixed(1)));
      mp.setAttribute("stroke-width",(h*1.9).toFixed(0));
      ["x","y"].forEach(a=>mask.setAttribute(a,0));mask.setAttribute("width",w);mask.setAttribute("height",h);
    };
    const draw=()=>{
      const head=o.h,tail=Math.max(head-SEG,0),vis=Math.max(Math.min(head,1)-tail,0);
      L.forEach(p=>{p.setAttribute("stroke-dasharray",vis+" 3");p.setAttribute("stroke-dashoffset",-tail);p.style.opacity=vis>.002?1:0;});
      mp.setAttribute("stroke-dasharray",Math.min(Math.max(head-LAG,0),1)+" 3");
    };
    const size=()=>{layout();draw();};
    ScrollTrigger.addEventListener("refresh",size);size();
    gsap.timeline({scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}})
      .to({},{duration:.12})
      .to(o,{h:1+SEG,duration:.76,ease:"power1.inOut",onUpdate:draw})
      .to({},{duration:.12});
    return()=>{ScrollTrigger.removeEventListener("refresh",size);root.classList.remove("is-live");svg.remove();};
  });
})();
