G176 · איור שמצטייר, נצבע ומקבל הערות

GSAP GSAP · ScrollTrigger scrub · SVG תמונות וגלריותטקסט וכותרות ממתין

איור קווי של צמח עם שורשים באדמה, ענן וטיפות מצטייר קו אחרי קו כמו ביד, כשהגלילה מזיזה אותו. כשהקווים נגמרים הצבע נכנס לעלים, לעלי הכותרת ולאדמה, ואז צצות ארבע הערות קצרות ליד החלקים, וכל מילה בשורת הכיתוב למטה נדלקת בתורה עד שהמשפט מלא.

מתי משתמשים: סקשן הסבר שמלמד רעיון אחד דרך תמונה: איך תוכן מצטבר לפניות, איך לקוח קטן הופך לקבוע, איך תהליך מתחיל בבסיס ונגמר בתוצאה. מתאים כשיש משפט אחד שאפשר להמחיש בציור פשוט, וארבע הערות לכל היותר. איור שצריך שמונה הערות כבר מתאים יותר לדיאגרמה.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני explainer ב-prompt-motion.com (21.10.2026) שבהם איור קווי של צמח, שורשים וטיפות מצטייר פס אחרי פס, נצבע, ואחריו מופיעות הערות כתובות ליד חלקיו ושורת כיתוב שמילותיה נדלקות. נבנה מאפס מהצפייה. מבנה: ה-SVG נבנה בזמן הבנייה מרשימות נתיבים (בראש הקובץ, il*). כל צורה סגורה נכתבת פעמיים: נתיב מילוי (.il-f, מונפש ב-opacity) ונתיב קו (.il-s, pathLength=1 ו-stroke-dashoffset), כך שהצבע והקו נפרדים בלי להניע תכונות שאינן opacity. אחד ScrollTrigger עם scrub על שורש הסקשן (start:top 75%, end:bottom 45%), וציר זמן אחד: אדמה, שורשים, גזע, עלים, פרח, ענן וטיפות, ואז מילויים, קווי הנחיה, הערות, ובמקביל המילים נדלקות באופן אחיד לאורך כל הציור. ההערות הן ul של HTML ממוקם באחוזים מעל ה-SVG (--x,--y לפי ה-viewBox), בקו תחתון גלי כתחושת כתב יד, ולכן נשארות בקריאות ובתוכן הנגיש. אפשר להחליף את האיור בכל SVG אחר: כל קבוצה מקבלת data-g, וה-JS מסדר אותן לפי שם הקבוצה. המצב הסופי במארקאפ: איור צבוע, ארבע הערות והמשפט מלא. בלי GSAP או בהפחתת תנועה זה מה שרואים. כיוון: האיור עצמו אינו מתהפך. ב-LTR מחליפים את grid-template-areas כך שהטקסט משמאל, ובטקסטי ההערות את הכיוון. ביצועים: opacity, transform ו-stroke-dashoffset בלבד, וה-ScrollTrigger נהרג בניקוי. נגישות: ה-SVG aria-hidden, ההערות והכיתוב טקסט רגיל.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  const root=document.querySelector(".il");if(!root)return;
  if(typeof gsap==="undefined")return;       // בלי GSAP: האיור צבוע, ההערות והמשפט מלאים
  gsap.registerPlugin(ScrollTrigger);
  const q=s=>[...root.querySelectorAll(s)],mm=gsap.matchMedia();
  mm.add("(prefers-reduced-motion: no-preference)",()=>{
    const g=n=>q('.il-s[data-g="'+n+'"]'),f=n=>q('.il-f[data-g="'+n+'"]');
    const lines=q(".il-s"),fills=q(".il-f"),leads=q(".il-ld"),labs=q(".il-labs span"),words=q(".il-cap span");
    gsap.set([...lines,...leads],{strokeDashoffset:1});
    gsap.set(fills,{autoAlpha:0});
    gsap.set(words,{opacity:.22});
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top 75%",end:"bottom 45%",scrub:.6}});
    const draw=(els,at,each,gap)=>els.forEach((el,i)=>tl.to(el,{strokeDashoffset:0,duration:each},at+i*gap));
    const fade=(els,at,each,gap)=>els.forEach((el,i)=>tl.to(el,{autoAlpha:1,duration:each},at+i*gap));
    draw(g("soil"),0,1.2,0);
    draw(g("roots"),1,.9,.25);
    draw(g("stem"),3,1,0);
    draw(g("leaves"),3.8,.6,.25);
    draw([...g("petals"),...g("core")],5,.5,.15);
    draw(g("cloud"),5.6,.8,0);
    draw(g("rain"),6.2,.5,.2);
    fade(f("soil"),6.6,.8,0);
    fade(f("leaves"),6.9,.6,.2);
    fade(f("petals"),7.4,.5,.1);
    fade(f("core"),7.9,.4,0);
    fade(f("cloud"),7.8,.6,0);
    fade(f("rain"),7.9,.5,.15);
    draw(leads,8,.3,.4);
    labs.forEach((el,i)=>tl.fromTo(el,{autoAlpha:0,scale:.8,rotation:-9},{autoAlpha:1,scale:1,rotation:-3,duration:.4,ease:"power3.out"},8.3+i*.4));
    tl.to(words,{opacity:1,duration:.4,stagger:{each:9.4/(words.length-1)}},1);
    return()=>{if(tl.scrollTrigger)tl.scrollTrigger.kill();tl.kill();gsap.set([...lines,...leads,...fills,...labs,...words],{clearProps:"all"});};
  });
})();
