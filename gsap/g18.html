G18 · סצנת סיפור מוצמדת (Master Timeline)

GSAP GSAP · pin + timeline הירו ופתיחהתמונות וגלריות ממתין

הסקשן ננעל והופך לבמה: כותרת נכנסת ונמוגה, שנייה מחליפה אותה, והמדיה נפתחת מגלולה זעירה למסך כמעט מלא. גלול לאט, גם אחורה.

מתי משתמשים: סקשן שיא אחד באתר סטוריטלינג. אחד לאתר: זה מהלך חתימה, לא תבנית.

הווידאו שלך כאן
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
הטריק המבני: הטיימליין מתחיל לפני הנעיצה ומסתיים אחריה, שני ScrollTrigger נפרדים. המדיה בגודלה הסופי כל הזמן, והגלולה היא חיתוך clip-path שנשלט במשתנה --pill יחד עם scale, כך שה-scrub לא מנפיש גובה או רדיוס. אורך הנעילה יחסי לגובה המסך. בטלפון המדיה רחבה יותר (88vw) ונפתחת מגלולה עבה יותר. בהפחתת תנועה, וכש-GSAP לא נטען, אין נעילה: הכותרות והמדיה מוצגות זו מתחת לזו.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger, SplitText);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הסצנה סטטית: שתי הכותרות ואז המדיה
  const sc=document.querySelector(".scene");
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    sc.classList.add("gsap-live");
    const t1=new SplitText(".st1",{type:"words"}),t2=new SplitText(".st2",{type:"words"});
    // אורך יחסי לגובה המסך: נעילה של 2.2 מסכים, בתוך התקרה של 2.5 מסכים לסצנה
    const vh=k=>()=>"+="+Math.round(innerHeight*k);
    const tl=gsap.timeline({scrollTrigger:{trigger:sc,start:"top 90%",end:vh(2.6),scrub:1}});
    ScrollTrigger.create({trigger:sc,start:"top top",end:vh(2.2),pin:true,scrub:1,anticipatePin:1});
    tl.from(".bgA",{y:"-10vw",duration:1},0)
      .from(".bgB",{y:"10vw",duration:1},0)
      .from(t1.words,{y:50,opacity:0,stagger:.05,duration:.5},"-=0.5")
      .to(".st1",{opacity:0,duration:.5},"+=0.3")
      .from(t2.words,{y:50,opacity:0,stagger:.05,duration:.5})
      .to(".st2",{opacity:0,duration:.4},"+=0.3")
      .from(".smedia",{opacity:0,duration:.3})
      .fromTo(".smedia",{scale:.2,"--pill":1},{scale:1,"--pill":0,duration:.8});
    return()=>{sc.classList.remove("gsap-live");t1.revert();t2.revert();};
  });
})();
