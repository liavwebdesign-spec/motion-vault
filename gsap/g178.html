G178 · קשתות שמחברות מילים קשורות במשפט

GSAP GSAP · ScrollTrigger scrub · SVG טקסט וכותרותתמונות וגלריות ממתין

משפט אחד מוצג בכותרת גדולה, ומתחת לכל מילה פס קצר. תוך כדי גלילה קשתות דקות ואפורות נמתחות מעל המשפט בין זוגות מילים קשורות, ואז קשת אחת עבה וצבעונית נמתחת מהמילה אותה אל ההצעה שהיא מתייחסת אליה, והפס מתחת להצעה מתארך וצובע. בטלפון המשפט נערם אנכית והקשתות עוברות לפס צד.

מתי משתמשים: סקשן הסבר ליכולת שקוראת טקסט: תשובות אוטומטיות, סיכום שיחות, סינון פניות, תרגום, תיוג. מתאים כשיש משפט אמיתי אחד מתוך הלקוחות של העסק, בין שמונה לעשר מילים, ואפשר להראות באמצעותו מה המערכת הבינה. לא מתאים לפסקאות ארוכות.

המערכת מבינה למה מתייחסת כל מילה

בהודעה של לקוח, המילה אותה מחזירה אל ההצעה. כך התשובה האוטומטית עונה על ההצעה הנכונה ולא על משהו אחר.

ההצעה המעודכנת נשלחה אתמול, והלקוח כבר קרא אותה פעמיים

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני הסבר ב-prompt-motion.com (21.10.2026) שבהם משפט בכתב כבד עם פס קטן מתחת לכל מילה, קשתות דקות מעליו בין מילים קשורות, וקשת אחת שהופכת עבה וצבעונית כדי לחבר כינוי לשם העצם שלו. נבנה מאפס מהצפייה. המדידה: layout() מודד את מרכז כל מילה (getBoundingClientRect יחסית לקופסה), קובע viewBox לגודל הקופסה בפיקסלים, וכותב את d של כל קשת ומיקום כל נקודת עוגן. הוא רץ לפני בדיקת ה-GSAP, ולכן בלי GSAP הקשתות והפסים מוצגים, וגם אחרי שינוי גודל, טעינת פונט ושינוי מצב (ResizeObserver ו-document.fonts.ready). שני מצבים: בדסקטופ השורה אחת, והקשתות מעל המילים עם גובה לפי המרחק (עד 80). בטלפון (max-width:767px) המילים בעמודה, בפס צד מימין לכל מילה נקודת עוגן, והקשתות מתנפחות הצידה. כל הקשת מצוירת בפיקסלים ולא מתכווצת. הגלילה: ScrollTrigger אחד עם scrub על הקופסה, וציר זמן של 10 יחידות: המילים נדלקות, הקשתות הדקות נמתחות (dashoffset), הקשת העבה נמתחת, ואז הפס של שם העצם מתארך והפס של הכינוי מתארך חלקית (scaleX ו-opacity של שכבת צבע). לשינוי: מחליפים את AW_WORDS ואת AW_PAIRS/AW_ACT (אינדקסים של מילים) בראש הקובץ, ובמארקאפ של הקשתות data-a ו-data-b. המצב הסופי מוצג: כל הקשתות, הקשת העבה והפסים. כיוון: המילה הראשונה בצד ימין. לאתר LTR הקשתות בדסקטופ עובדות כמות שהן, כי המדידה לפי מיקום בפועל. בטלפון ה-padding-inline-start מתהפך לבד, ובתוך layout() מחליפים את w-40 ב-40 ואת A.x+hh*2 ב-A.x-hh*2, כדי שפס הצד יעבור לשמאל. נגישות: ה-SVG aria-hidden, והמשפט טקסט רגיל. ביצועים: dashoffset, opacity ו-transform בלבד, וניקוי מלא.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  const box=document.querySelector(".aw-box");if(!box)return;
  const svg=box.querySelector(".aw-svg"),words=[...box.querySelectorAll(".aw-w")],
        arcs=[...svg.querySelectorAll(".aw-t,.aw-h")],dots=[...svg.querySelectorAll(".aw-d")],
        mq=matchMedia("(max-width:767px)");
  const num=v=>Math.round(v*10)/10;
  // מודד את מרכז כל מילה וכותב את הקשתות. רץ גם בלי GSAP, ובכל שינוי גודל
  function layout(){
    const cb=box.getBoundingClientRect(),w=cb.width,h=cb.height,mob=mq.matches;
    svg.setAttribute("viewBox","0 0 "+num(w)+" "+num(h));
    const P=words.map(el=>{const r=el.querySelector("b").getBoundingClientRect();
      return mob?{x:w-40,y:r.top-cb.top+r.height/2}:{x:r.left-cb.left+r.width/2,y:r.top-cb.top-8};});
    dots.forEach(d=>{const p=P[+d.dataset.i];d.setAttribute("cx",num(p.x));d.setAttribute("cy",num(p.y));});
    arcs.forEach(a=>{const A=P[+a.dataset.a],B=P[+a.dataset.b];
      if(mob){const hh=Math.min(32,8+Math.abs(A.y-B.y)*.12);a.setAttribute("d","M"+num(A.x)+" "+num(A.y)+" Q"+num(A.x+hh*2)+" "+num((A.y+B.y)/2)+" "+num(B.x)+" "+num(B.y));}
      else{const hh=Math.min(80,18+Math.abs(A.x-B.x)*.28);a.setAttribute("d","M"+num(A.x)+" "+num(A.y)+" Q"+num((A.x+B.x)/2)+" "+num(A.y-hh*2)+" "+num(B.x)+" "+num(B.y));}});
  }
  layout();
  if(window.ResizeObserver)new ResizeObserver(layout).observe(box);else addEventListener("resize",layout);
  if(document.fonts&&document.fonts.ready)document.fonts.ready.then(layout);
  if(typeof gsap==="undefined")return;       // בלי GSAP: הקשתות והפסים מוצגים כבר במצב הסופי
  gsap.registerPlugin(ScrollTrigger);
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const wd=words.map(w=>w.querySelector("b")),bars=words.map(w=>w.querySelector(".aw-bar")),
          thin=arcs.filter(a=>a.classList.contains("aw-t")),hot=svg.querySelector(".aw-h"),
          soft=dots.filter(d=>!d.classList.contains("is-hot")),hotDots=dots.filter(d=>d.classList.contains("is-hot")),
          noun=box.querySelector(".is-n .aw-bar"),pron=box.querySelector(".is-p .aw-bar"),fills=[noun,pron].map(b=>b.querySelector("u"));
    gsap.set(wd,{opacity:.3});
    gsap.set([...thin,hot],{strokeDashoffset:1});
    gsap.set(dots,{opacity:0});
    gsap.set(bars,{scaleX:.22});
    gsap.set(fills,{opacity:0});
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:box,start:"top 75%",end:"bottom 35%",scrub:.6,invalidateOnRefresh:true}});
    tl.to(wd,{opacity:1,duration:.6,stagger:.15},0)
      .to(thin,{strokeDashoffset:0,duration:.6,stagger:.3},1.5)
      .to(soft,{opacity:1,duration:.3,stagger:.25},1.5)
      .to(hot,{strokeDashoffset:0,duration:2},4.5)
      .to(hotDots,{opacity:1,duration:.3,stagger:1.7},4.5)
      .to(noun,{scaleX:1,duration:2},6.5)
      .to(pron,{scaleX:.6,duration:1.5},6.5)
      .to(fills,{opacity:1,duration:1.5},6.5)
      .to({},{duration:1.5},8.5);
    return()=>{if(tl.scrollTrigger)tl.scrollTrigger.kill();tl.kill();gsap.set([...wd,...thin,hot,...dots,...bars,...fills],{clearProps:"all"});};
  });
})();
