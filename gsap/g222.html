G222 · שם שמתכווץ לקו דק ונפתח לסקשן הבא

GSAP GSAP · ScrollTrigger scrub · clip-path הירו ופתיחהתמונות וגלריות ממתין

שם הסטודיו בגדול על רקע כהה. בגלילה הוא נמעך לגובה אפס ונהפך לקו אור דק, כמו מסך ישן שנכבה. הקו מתארך לרוחב המסך, ואז נפתח מהמרכז למעלה ולמטה כמו עפעפיים, ובמרווח שנפתח מופיע הסקשן הבא עם התמונה והטקסט שלו.

מתי משתמשים: פתיחה או מעבר ראשון של אתר סטודיו, צילום, עיצוב או אדריכלות, שמתחיל בשם ואז נכנס לעבודה. מעבר דרמטי, לכן פעם אחת בעמוד ובין שני סקשנים שיש להם משקל.

סטודיו נור

צילום אדריכלות ופנים

מצלמים את הדירה כשהאור נכנס אליה

כל צילום נקבע לשעה שבה החדר נראה הכי טוב. בדרך כלל זה בוקר, ולפעמים הרבה לפני שהדיירים קמים.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שני סרטוני הצגה ב-prompt-motion.com (21.10.2026), באחד שם נמעך לקו אור והולך לשחור ובשני קו דק נפתח לפאנל בהיר. נבנה מאפס מהצפייה, כמהלך אחד שמחבר את שני החצאים. שלושה שלבים בטיימליין אחד: (1) השם נמעך ב-scaleY עד כמעט אפס, והשורה שמתחתיו דוהה. (2) קו אור (.so-line) לוקח את מקומו ברוחב השם ומתארך לרוחב המסך, עם box-shadow זוהר בצבע המותג. (3) הסקשן השני נחשף ב-clip-path:inset(50% 0) שנפתח ל-inset(0), והקו נמוג בפתיחה. הרוחב ההתחלתי של הקו נמדד מהשם עצמו (offsetWidth), כדי שהמעבר מהשם לקו לא יקפוץ. בלי GSAP ובהפחתת תנועה שני הסקשנים זה מתחת לזה והקו לא קיים. כשהמהלך רץ הבמה נעוצה ויש לה מסלול של כשלושה מסכים. כפרילודר: אותו קו, בלי גלילה: טיימליין של שנייה וחצי שמופעל בטעינה ומתנקה מיד (ראו g15b לעקרון של פרילודר שלא חוסם). כיוון: המהלך אנכי ולכן אין מה להפוך באתר משמאל לימין. נגישות: השם נשאר טקסט אמיתי (h2), והקו הוא קישוט (aria-hidden).

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP שני הסקשנים זה מתחת לזה
  const root=document.querySelector(".so"),pin=root.querySelector(".so-pin"),A=root.querySelector(".so-a"),B=root.querySelector(".so-b"),
    word=root.querySelector(".so-word"),tag=root.querySelector(".so-tag"),line=root.querySelector(".so-line"),img=B.querySelector("img");
  // בהפחתת תנועה הפונקציה לא רצה, והסקשנים נשארים זה מתחת לזה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    // הקו מתחיל ברוחב השם עצמו, כך שהמעבר מהשם לקו לא קופץ
    const ratio=()=>Math.min(word.offsetWidth/pin.clientWidth,1);
    gsap.set(line,{scaleX:ratio,opacity:0});
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    tl.to({},{duration:.12})
      .to(tag,{opacity:0,duration:.1})
      .to(word,{scaleY:.012,scaleX:1.04,duration:.26,ease:"power3.in"},"<")
      .to(word,{opacity:0,duration:.04},">-.04")
      .to(line,{opacity:1,duration:.04},"<")
      .to(line,{scaleX:1,duration:.22,ease:"power2.out"},">")
      .to({},{duration:.1})
      .fromTo(B,{clipPath:"inset(50% 0% 50% 0%)"},{clipPath:"inset(0% 0% 0% 0%)",duration:.5,ease:"power2.inOut"})
      .fromTo(img,{scale:1.18},{scale:1,duration:.5},"<")
      .to(line,{opacity:0,duration:.14},"<.04")
      .to({},{duration:.1});
    return()=>{root.classList.remove("is-live");gsap.set([word,tag,line,B,img],{clearProps:"all"});};
  });
})();
