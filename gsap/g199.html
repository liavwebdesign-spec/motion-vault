G199 · דמות חתוכה מול רקעים מתחלפים

GSAP GSAP · ScrollTrigger scrub · clip-path הירו ופתיחהתמונות וגלריות ממתין

דמות חתוכה עומדת במקום במרכז הבמה, והעולם שמאחוריה מתחלף בגלילה: שמיים עם שמש, פודיום עם קרני אור, נקודות של קומיקס, קווי מהירות ורדיד מבריק. כל חילוף הוא חיתוך נקי שנפתח מימין לשמאל, עם מילה גדולה חדשה שעולה מאחורי הדמות ושכבת עיטור משלה, והרשימה שלצד הבמה מדגישה את התחום הפעיל.

מתי משתמשים: סקשן שירותים או אודות שבו אדם אחד (בעל העסק, מרצה, יועץ) מייצג כמה כיוונים. הדמות היא הקבוע והרקע הוא הרעיון, וכך אפשר להראות חמישה תחומים בלי חמש תמונות. פעם אחת בעמוד, עם תמונה חתוכה באיכות טובה. לא כשהדמות אינה חלק מהמותג.

שחזור. השראה: סרטון בגלריה (21.10.2026) של דמות חתוכה מסרטון שמדבר למצלמה, כשסביבה הרקע, המילה הגדולה והעיטורים מתחלפים לפי הרעיון. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: הדמות על הרקע הראשון, ורשימת חמשת התחומים בצד (הראשון מודגש). בלי GSAP ובהפחתת תנועה זה מה שרואים. כשהמהלך רץ הסקריפט מוסיף is-live (במה נעוצה בגובה 380vh) ו-timeline אחד עם scrub. הרקעים מוערמים והדמות (z-index גבוה) לא זזה בכלל. כל חילוף הוא חיתוך נקי בצורת וייפ: clip-path:inset(0 0 0 100%) שנפתח ל-inset(0) ב-power3.inOut, והמילה הגדולה של הרקע החדש עולה מעט ונכנסת באותו רגע. כל רקע מוסיף שכבת עיטור משלו ב-CSS בלבד (שמש, קרני פודיום, נקודות קומיקס, קווי מהירות, רדיד), והכול מצבעי הערכה (--accent, --ink, --card), ולכן מתלבש על כל עור. הרשימה בצד: מחלקה is-on על הפריט הפעיל, לפי זמן ה-timeline. הדמות: תמונה אמיתית מהמאגר (dive/portrait.webp) שאין לה גזרה על רקע שקוף, ולכן היא נחתכת במסכה: mask-image של SVG שקו המתאר שלו נמדד על התמונה (קצה מטושטש מעט, ודהייה רכה בחלק העליון, שם התמונה עצמה חתוכה). קו המדבקה הוא ארבעה drop-shadow בצבע --card על המעטפת ולא על התמונה, כי סדר העיבוד של הדפדפן הוא filter, אחר כך clip, ואז mask, והצל על אלמנט עם מסכה היה נחתך. התמונה בנויה ביחס 3:2 בלי חיתוך, כדי שהמסכה תיפול בדיוק על האישה. עם תמונה חתוכה אמיתית על רקע שקוף מוחקים את המסכה ומשאירים את המעטפת והצל. מובייל: הבמה נעוצה בראש המסך והתחומים הופכים לשבבים מתחתיה. כיווניות: הוייפ נפתח מימין לשמאל. באתר LTR הופכים ל-inset(0 100% 0 0). נגישות: הרקעים, המילים והדמות הם איור (aria-hidden), והתחומים נשארים טקסט רגיל ברשימה.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשארים הרקע הראשון והרשימה
  const root=document.querySelector(".bqd"),bgs=[...root.querySelectorAll(".bqd-bg")],lis=[...root.querySelectorAll(".bqd-li")];
  // בהפחתת תנועה הפונקציה לא רצה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const HOLD=1.2,SWAP=.5,starts=[0];
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.5}});
    tl.to({},{duration:HOLD});
    for(let i=1;i<bgs.length;i++){
      starts.push(tl.duration());
      // וייפ מימין לשמאל: הקצה השמאלי של החיתוך זז מ-100% ל-0. immediateRender:false כי ה-CSS כבר מסתיר את הרקע
      tl.fromTo(bgs[i],{clipPath:"inset(0% 0% 0% 100%)"},{clipPath:"inset(0% 0% 0% 0%)",duration:SWAP,ease:"power3.inOut",immediateRender:false},">")
        .fromTo(bgs[i].querySelector(".bqd-word"),{yPercent:16,autoAlpha:0},{yPercent:0,autoAlpha:1,duration:SWAP,ease:"power3.out",immediateRender:false},"<")
