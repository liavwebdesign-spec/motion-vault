עמוד אתר שלם מונח לאחור בפרספקטיבה כמו רצפה, והמצלמה גולשת לאורכו: הקצה הקרוב מיטשטש, הקצה הרחוק נמוג, וארבעה סמלי אפליקציה צפים בפינות. בסוף העמוד כולו מתכווץ לפס צר שרואים בו את כל המבנה, ואז חוזר להיות עמוד בית ישר, בדיוק איפה שהתחיל.

מתי משתמשים: סקשן שמציג עמוד אחד שלם, כמו תיק עבודות, אתר שנבנה ללקוח או מוצר: עמוד עם כמה מקטעים ברורים, שמלמד לראות את כל המבנה לפני שנכנסים לפרטים. מתאים לסטודיו, סוכנות או מוצר שהעמוד שלו הוא ההוכחה. לא באתר שמרני, ולא כשהקהל צריך לקרוא מהר. פעם אחת בעמוד.

שחזור. השראה: סרטון מוצר ב-prompt-motion.com (21.10.2026), שבו עמוד נחיתה שלם נראה בפרספקטיבה והמצלמה גולשת עליו עד שהוא מתכווץ לפס. נבנה מאפס מהצפייה. המבנה: שכבה דביקה (sticky) בגובה המסך בתוך סקשן גבוה, ובתוכה .fp-cam עם perspective, מעליה .fp-tilt שמסתובב ב-rotationX מהקצה העליון, ובתוכו .fp-plane שהוא העמוד עצמו. ההזזה לאורך העמוד היא y של .fp-plane בלבד, כלומר בתוך המישור המוטה, ולכן התנועה מתקצרת בפרספקטיבה כמו על רצפה אמיתית. אם מזיזים את .fp-tilt במקום, זה פשוט גלילה. העמוד: נבנה ביחידות em שנגזרות מרוחבו (--pw), ולכן הוא מתכווץ ומתרחב כמו תמונה, והטקסטים בו הם צילום מסך ולא נועדו לקריאה. בטלפון הוא נבנה מחדש בעמודה אחת עם @container. כדי להציג עמוד אמיתי מחליפים את .fp-plane בתמונה גבוהה, או בקופסה עם iframe, והשאר נשאר. הזמנים: נטייה (22 אחוז), גלישה (48), התכווצות לפס (15) וחזרה (15). את גובה העמוד נמדדים מהאלמנט ב-refresh (invalidateOnRefresh), והפס הצר מחושב כך שכל העמוד נכנס ב-84 אחוז מגובה המסך. עומק שדה: שכבה בתחתית עם backdrop-filter ומסכה שמחלישה את הטשטוש כלפי מעלה. הטשטוש קבוע, ולכן אין עבודה בגלילה. הדהייה בקצה העליון נחשפת עם הנטייה. הסמלים: צפים ב-CSS בלבד (translate, cubic-bezier(.4,0,.2,1)), וגם נכנסים ויוצאים עם ה-scrub. הציפה כבויה בהפחתת תנועה. מצב סטטי: בלי GSAP ובהפחתת תנועה רואים את תחילת העמוד, ישר ובגובה מסך, עם דהייה בתחתית. כיוון: העמוד עצמו בעברית מימין לשמאל. הנטייה אנכית ולכן אין מה להפוך לאתר משמאל לימין חוץ מהעמוד שבתוכה.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשארת תחילת העמוד, ישרה ובגובה מסך
  const root=document.querySelector(".fp"),pin=root.querySelector(".fp-pin"),tilt=root.querySelector(".fp-tilt"),
    plane=root.querySelector(".fp-plane"),fade=root.querySelector(".fp-fade"),icos=[...root.querySelectorAll(".fp-ico")];
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const H=()=>pin.clientHeight,PH=()=>plane.offsetHeight;
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    const mob=()=>innerWidth<768;
    tl.to(tilt,{rotationX:()=>mob()?54:60,scale:()=>mob()?.64:.74,y:()=>H()*.1,duration:.22,ease:"power2.inOut"},0)   // העמוד נשכב
      .to(fade,{opacity:1,duration:.22},0)
      .fromTo(icos,{opacity:0,scale:.6},{opacity:1,scale:1,duration:.14,stagger:.03,ease:"power2.out"},.1)
      .to(plane,{y:()=>-(PH()-H()*.7),duration:.48,ease:"power1.inOut"},.22)                       // המצלמה גולשת לאורך העמוד
      .to(tilt,{rotationX:0,scale:()=>.84*H()/PH(),y:()=>.08*H(),duration:.15,ease:"power2.inOut"},.7) // כל העמוד כפס צר
      .to(plane,{y:0,duration:.15,ease:"power2.inOut"},.7)
      .to([fade,...icos],{opacity:0,duration:.12},.7)
      .to(tilt,{scale:1,y:0,duration:.15,ease:"power2.inOut"},.85);                                // וחוזר לעמוד הבית
    return()=>{root.classList.remove("is-live");gsap.set([tilt,plane,fade,...icos],{clearProps:"all"});};
  });
})();
