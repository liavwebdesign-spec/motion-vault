G251 · קשקוש על תמונה שמוחק את האובייקט

GSAP GSAP · stroke-dashoffset · ScrollTrigger once תמונות וגלריותטעינה ופידבק ממתין

תמונת מוצר עם שלושה גביעי גלידה בשורה. קו קשקוש נצבע בזיגזג על הגביע הימני, והגביע נמחק והרקע משלים את עצמו. אחר כך אותו דבר על השמאלי. בסוף קופצת תווית קטנה שמספרת מה הוסר.

מתי משתמשים: שירות של עריכת תמונה, ניקוי רקע, צילום מוצר או שיפור תמונות בבינה מלאכותית: ״מסמנים מה מפריע והוא נעלם״. מתאים לדמו בראש עמוד או לסקשן של ״לפני ואחרי״. לא לשירות שלא מנקה תמונה בפועל: המחיקה כאן היא הבטחה שמישהו יבדוק על תמונה שלו.

מסמנים מה מפריע, והוא נעלם

מקשקשים על מה שלא צריך להיות בתמונה, והמערכת מוחקת אותו ומשלימה את הרקע. התמונה נשארת נקייה, בלי עריכה ידנית.

בתמונה הוסרו שני גביעים, ונשאר גביע אחד באמצע.
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון בגלריה של prompt-motion.com (21.10.2026) שבו קשקוש ורוד נצבע על אובייקט בתוך תמונה, האובייקט נעלם והרקע משלים את עצמו, ותווית קטנה קופצת לידו. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: התמונה נקייה (נשאר גביע אחד) והתווית נראית. איך זה בנוי: תמונה אחת עם שלוש גביעים, ומעליה שתי שכבות תיקון (sc-pl, sc-pr), כל אחת תמונת WebP שקופה וקטנה (כ-2.7KB) שבה האזור של גביע אחד מולא מחדש בצבע הרקע ועם קצה רך. בהתחלה הסקריפט מוריד את שתי השכבות לשקיפות אפס, כך שרואים את התמונה המקורית. הקשקוש הוא מסלול SVG עם pathLength=1, והציור הוא stroke-dashoffset מ-1 ל-0 (בלי תוסף). אחרי כל קשקוש שכבת התיקון עולה והקשקוש דועך. שכבות התיקון הן דוגמה: הן חושבו מהצבע הרקע החלק של התמונה הזאת. בעבודה אמיתית שכבת התיקון היא התמונה המעובדת בפועל, עם אותו גודל ואותו יחס, ואז אין שום תלות בצבע הרקע. באתר LTR התווית עוברת לצד השני אוטומטית (inset-inline-start), והקשקושים בקואורדינטות פיזיות של התמונה. נגישות: לתמונה יש טקסט חלופי שמתאר את המצב הסופי, הקשקוש והתווית aria-hidden, ומשפט לקוראי מסך אומר שהוסרו שני גביעים. בהפחתת תנועה נשארת התמונה הנקייה והתווית. טלפון: התמונה ממלאת את הרוחב, והקו עבה מספיק (3.4 יחידות מתוך 140) כדי להיראות גם בקטן.

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP: התמונה הנקייה והתווית (במארקאפ)
  const root=document.querySelector(".sc"),fig=root.querySelector(".sc-img"),pl=root.querySelector(".sc-pl"),pr=root.querySelector(".sc-pr"),
    skR=root.querySelector(".sc-skr"),skL=root.querySelector(".sc-skl"),chip=root.querySelector(".sc-chip"),
    lines=[...root.querySelectorAll(".sc-sk path")];
  // בהפחתת תנועה הפונקציה לא רצה, והתמונה נשארת נקייה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    gsap.set([pl,pr],{opacity:0});gsap.set([skR,skL],{opacity:1});gsap.set(lines,{strokeDashoffset:1});
    gsap.set(chip,{autoAlpha:0,scale:.85,y:8});
    const tl=gsap.timeline({paused:true});
    // קשקוש נצבע, שכבת התיקון עולה והקשקוש דועך
    const erase=(sk,patch,at)=>{
      tl.to(sk.querySelectorAll("path"),{strokeDashoffset:0,duration:1.05,ease:"power1.inOut"},at);
      tl.to(patch,{opacity:1,duration:.6,ease:"power1.out"},at+1.1);
      tl.to(sk,{opacity:0,duration:.45,ease:"power1.in"},at+1.2);
    };
    erase(skR,pr,.4);
    erase(skL,pl,2.5);
    tl.to(chip,{autoAlpha:1,scale:1,y:0,duration:.5,ease:"power3.out"},">-=.1");
    const st=ScrollTrigger.create({trigger:fig,start:"top 65%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();gsap.set([pl,pr,skR,skL,chip,...lines],{clearProps:"all"});};
  });
})();
