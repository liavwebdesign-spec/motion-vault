שחזור. השראה: סרטון בגלריה של prompt-motion.com (21.10.2026) שבו סליל דו-גדילי עובר סריקה והגדילים נצבעים בצבע חם בעקבות קו אנכי. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: סליל סטטי שמחצית ממנו כבר דולקת וקו הסריקה במקומו, כולל כל הנתיבים, כך שבלי GSAP ובהפחתת תנועה רואים איור גמור. הגיאומטריה בפונקציה אחת, helix(W,H,ph,N), שמחזירה חמישה נתיבים: שני הגדילים, השלבים, והנקודות שבחזית ובעומק (נקודה היא קטע באורך אפס עם stroke-linecap:round). אותה פונקציה בונה את המארקאפ הסטטי בזמן הבנייה ורצה בסקריפט, וכך אין שתי גרסאות להחזיק. עומק בלי תלת מימד: הנקודה שבחזית גדולה ובצבע מלא, שבעומק קטנה ועמומה, ובכל שלב נבחר מי בחזית לפי קוסינוס הפאזה. שני עותקים של אותו איור: הקבוצה #dx-g מוגדרת פעם אחת ב-defs, ומוצגת פעמיים עם use: האפור כמו שהוא, והצבעוני מעליו, חתוך ב-clipPath שהמלבן שלו זז עם קו הסריקה. הצבע מגיע מה-use (stroke עובר בירושה לתוך העותק), ולכן אין שכפול של הנתיבים בצבעים. ריצה: ResizeObserver קובע את ה-viewBox לפי המידות האמיתיות (כך שעובי הקו הוא פיקסלים אמיתיים), מספר השלבים נגזר מהרוחב (W/26), ו-ticker יחיד מקדם את הפאזה ומצייר מחדש. הסריקה היא timeline עם tween של p מ-0 ל-1 ב-power1.inOut, השהיה, דעיכה ואיפוס. קל: כשהסקשן מחוץ למסך ה-ticker וה-timeline מושהים (IntersectionObserver), ומוזזים רק transform ו-d. כיווניות: הסריקה מתחילה מקצה ההתחלה, כלומר מימין ל-RTL ומשמאל ל-LTR, והגל נע לאותו כיוון. הסקריפט קורא את direction, ובאתר LTR הסליל הסטטי במארקאפ עדיין מציג מצב סביר. נגישות: האיור aria-hidden ובלי כיתוב, כי הוא קישוט לכותרת. מובייל: יחס 390:220 ושני סיבובים בלבד, כך שהסליל לא נראה שטוח.

  function helix(W,H,ph,N){
  const pad=W*.03,w=W-pad*2,A=H*.34,cy=H/2,turns=W<600?2.4:3.6,f=n=>Math.round(n*10)/10;
  const y=(x,o)=>cy+A*Math.sin((x-pad)/w*turns*6.2832+ph+o);
  const z=(x,o)=>Math.cos((x-pad)/w*turns*6.2832+ph+o);
  let a="",b="",r="",nf="",nb="";
  const S=Math.max(40,Math.round(W/8));
  for(let i=0;i<=S;i++){const x=pad+w*i/S;a+=(i?"L":"M")+f(x)+" "+f(y(x,0));b+=(i?"L":"M")+f(x)+" "+f(y(x,3.1416));}
  for(let i=0;i<=N;i++){
    const x=pad+w*i/N,ya=y(x,0),yb=y(x,3.1416),front=z(x,0)>=0?ya:yb,back=z(x,0)>=0?yb:ya;
    r+="M"+f(x)+" "+f(ya)+"L"+f(x)+" "+f(yb);
    nf+="M"+f(x)+" "+f(front)+"h0";nb+="M"+f(x)+" "+f(back)+"h0";
  }
  return{a:a,b:b,r:r,nf:nf,nb:nb};
}
