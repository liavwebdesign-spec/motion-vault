G233 · פרוסות של מילה שקופצות הצידה וחוזרות למקום

GSAP GSAP · ScrollTrigger once · clip-path · background-clip:text טקסט וכותרותתמונות וגלריותהירו ופתיחה ממתין

מילה גדולה שממולאת בתמונה נחתכת לשבע פרוסות אופקיות. כשהיא נכנסת למסך הפרוסות קופצות הצידה בצעדים חדים, בכיוונים מתחלפים, ופסים דקים בצבע המותג מחליקים עליהן. אחר כך כל הפרוסות נשמטות חזרה ליישור והמילה נשארת שלמה. אותו דבר קורה שוב בכל ריחוף בעכבר. בלי GSAP ובהפחתת תנועה המילה מוצגת שלמה.

מתי משתמשים: מילה אחת שמסמלת את הרגע שבו העמוד נפתח, באתר של עסק שרוצה חדות ותנועה: הפלגות, מוזיקה, אירועים, מותג לצעירים. התמונה שממלאת את המילה חייבת להיות צבעונית וברורה, והמילה עבה מספיק כדי שהתמונה תיראה בתוכה.

לבדיקת תאריכים פנויים
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם אות או מילה נחתכות לפרוסות שקופצות הצידה וחוזרות ליישור, באחד מהם כשהמילה ממולאת בתמונה. נבנה מאפס מהצפייה. המארקאפ: ה-h2 מחזיק sr-only עם המילה, ועטיפה aria-hidden (.gs-stack) שמכילה את המילה השלמה (.gs-w). מילוי בתמונה: background:url(...) 50% 60% / cover עם background-clip:text ו--webkit-text-fill-color:transparent. הפרוסות: כשהמהלך רץ ה-JS מוסיף שבעה עותקים של המילה (.gs-b), כל אחד position:absolute; inset:0 בדיוק על המילה, עם אותו רקע, ו-clip-path: inset(i/7 0 (6-i)/7 0). מכיוון שלכולם אותה תיבה, התמונה ממשיכה ברצף ביניהם. הרצף: המילה השלמה נעלמת, והפרוסות מופיעות עם xPercent שנקבע בשלוש קפיצות קשיחות (‏set בזמנים .0, ‏.1, ‏.2, בין 7 ל-26 אחוז מרוחב המילה, בסימנים מתחלפים בין פרוסה לפרוסה), ואז כל פרוסה חוזרת ל-0 ב-power4.out (‏.3 שניות, ‏.022 בין פרוסות). בסוף המילה השלמה חוזרת והפרוסות מוסתרות. החלפה בסוף: המעבר לעותק השלם חשוב, כי קצוות חיתוך על פיקסלים חצויים משאירים קווי שיער בין פרוסות. שלושה פסים דקים (.gs-bar, גובה 3px, --accent) נכנסים בהחלקה ודוהים, ואין הבהוב בשום שלב. הפעלה: פעם אחת בכניסה למסך ובכל pointerenter של עכבר (לא של מגע). עד הכניסה המילה מוסתרת, והפרוסות הן שבונות אותה. RTL: הכיוון של הקפיצות אקראי ואין מה להחליף. התמונה: assets/media/real/s08/hero.webp; לתמונה כהה מאוד, האותיות נבלעות בעור כהה.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP המילה מוצגת שלמה
  const root=document.querySelector(".gs"),stack=root.querySelector(".gs-stack"),base=stack.querySelector(".gs-w");
  // בהפחתת תנועה הפונקציה לא רצה, והמילה נשארת שלמה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const N=7,txt=base.textContent,bands=[],bars=[],R=gsap.utils.random;
    for(let i=0;i<N;i++){                                         // שבע פרוסות: אותו טקסט, אותה תיבה, חיתוך אחר
      const b=document.createElement("span");b.className="gs-b";b.setAttribute("aria-hidden","true");b.textContent=txt;
      b.style.clipPath="inset("+(i*100/N)+"% 0 "+((N-1-i)*100/N)+"% 0)";stack.appendChild(b);bands.push(b);
    }
    for(let i=0;i<3;i++){
      const r=document.createElement("span");r.className="gs-bar";r.setAttribute("aria-hidden","true");
      r.style.top=R(22,78,1)+"%";r.style.width=R(22,40,1)+"%";r.style.insetInlineStart=R(0,60,1)+"%";stack.appendChild(r);bars.push(r);
    }
    let tl=null;
    const run=()=>{
      if(tl&&tl.isActive())return;
      if(tl)tl.kill();
      tl=gsap.timeline({onComplete:()=>{gsap.set(base,{opacity:1});gsap.set(bands,{visibility:"hidden",xPercent:0});}});
      tl.set(base,{opacity:0},0).set(bands,{visibility:"visible"},0);
      bands.forEach((b,i)=>{
        const s=i%2?1:-1;
        for(let j=0;j<3;j++)tl.set(b,{xPercent:s*(j%2?-1:1)*R(7,26,1)},j*.1+i*.014);   // שלוש קפיצות קשיחות הצידה
        tl.to(b,{xPercent:0,duration:.3,ease:"power4.out"},.42+i*.022);                    // ושמיטה חזרה ליישור
      });
      bars.forEach((r,j)=>{
        tl.fromTo(r,{xPercent:R(-140,140,1),opacity:0},{xPercent:0,opacity:.9,duration:.16,ease:"power2.out"},.04+j*.1)
          .to(r,{opacity:0,duration:.3,ease:"none"},.55+j*.05);
      });
      tl.set(base,{opacity:1},.95).set(bands,{visibility:"hidden"},.95);                   // הפרוסות מוחלפות במילה השלמה, בלי קווי שיער
    };
    gsap.set(base,{opacity:0});                                    // עד הכניסה הפרוסות הן שבונות את המילה
    const st=ScrollTrigger.create({trigger:stack,start:"top 85%",once:true,onEnter:run});
    const over=e=>{if(e.pointerType==="mouse")run();};
    stack.addEventListener("pointerenter",over);
    return()=>{
      st.kill();if(tl)tl.kill();stack.removeEventListener("pointerenter",over);
      bands.concat(bars).forEach(n=>n.remove());gsap.set(base,{clearProps:"opacity"});
    };
  });
})();
