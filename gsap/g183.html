G183 · כרטיס מוצר שנבנה שלב אחר שלב

הודעה קטנה יורדת מלמעלה מטושטשת ומתחדדת, ומתחתיה נבנה כרטיס של דוח שבועי בסדר קבוע: מתג נדלק, מספר הפניות רץ עד 38, שבעה עמודים עולים בזה אחר זה והאחרון בצבע המותג, שורת אישור מציירת וי, והכפתור מתחלף מ״שליחה עכשיו״ ל״נשלח״. כל הדוח מצטייר בתוך שש שניות בערך.

מתי משתמשים: הצגת תוצאה אחת של מוצר או שירות: דוח, אישור, סיכום, התראה. סקשן הוכחה או ״מה מקבלים״ באתר של כלי, מערכת או שירות דיגיטלי. צריך תוצאה אחת ברורה ולא יותר מחמישה רכיבים בכרטיס. פעם אחת בעמוד.

הדוח מגיע לבד, בכל יום ראשון

בשבוע האחרון נכנסו 38 פניות חדשות. סיכום קצר נשלח לשלושה אנשים בצוות, וכל אחד רואה מיד איפה העסק עומד.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני מוצר בגלריה של prompt-motion.com (21.10.2026), שבהם התראה נופלת ומתחתיה כרטיס נבנה רכיב אחרי רכיב. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: ההודעה חדה, המתג דולק, המונה על 38, כל העמודים מלאים, שורת האישור מוצגת והכפתור במצב ״נשלח״. בלי GSAP ובהפחתת תנועה זה מה שמוצג. כשהמהלך רץ ה-JS מאפס הכל, ובכניסה לעמוד (top 65%, פעם אחת) ציר זמן אחד בונה את הכרטיס בסדר קבוע: ההודעה יורדת ומתחדדת (filter: blur, רק על רכיב קטן וקצר), הכרטיס עולה, המתג נדלק, המונה רץ, העמודים עולים ב-scaleY מהבסיס, הווי מצטייר ב-stroke-dashoffset והכפתור נלחץ. הכרטיס כולו aria-hidden, והמספר 38 כתוב גם בפסקה שלידו, כך שקורא המסך לא מפסיד כלום. להתאים: התראה, כותרת הכרטיס, המתג, המונה, העמודים ותווית הכפתור הם התוכן. העמוד האחרון הוא התוצאה, ולכן רק הוא בצבע המותג. RTL: הכרטיס בצד ימין של הטקסט, והעמודים מסודרים מימין לשמאל (הימים עוברים בכיוון הקריאה). ב-LTR מחליפים את סדר העמודות ב-.pc ואת --sx.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הכרטיס כבר שלם: מתג דולק, מספר סופי, כפתור במצב נשלח (במארקאפ)
  const root=document.querySelector(".pc"),note=root.querySelector(".pc-note"),card=root.querySelector(".pc-card"),
    sw=root.querySelector(".pc-sw"),num=root.querySelector(".pc-num"),bars=[...root.querySelectorAll(".pc-bars i")],
    done=root.querySelector(".pc-done"),tick=done.querySelector("path"),send=root.querySelector(".pc-send");
  const total=parseInt(num.textContent,10)||0;
  const finish=()=>{sw.classList.add("is-on");num.textContent=total;send.classList.add("is-sent");};
  // בהפחתת תנועה הפונקציה לא רצה, והכרטיס נשאר במצב הסיום
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    sw.classList.remove("is-on");send.classList.remove("is-sent");num.textContent="0";
    gsap.set(note,{autoAlpha:0,y:-28,filter:"blur(14px)"});
    gsap.set(card,{autoAlpha:0,y:24});
    gsap.set(bars,{scaleY:0});
    gsap.set(done,{autoAlpha:0,y:10});
    gsap.set(tick,{strokeDashoffset:1});
    const c={v:0},tl=gsap.timeline({paused:true});
    tl.to(note,{autoAlpha:1,y:0,filter:"blur(0px)",duration:.8,ease:"power3.out"})
      .to(card,{autoAlpha:1,y:0,duration:.6,ease:"power2.out"},"-=.35")
      .call(()=>sw.classList.add("is-on"),null,">+=.2")
      .to(c,{v:total,duration:1,ease:"power2.out",onUpdate:()=>{num.textContent=Math.round(c.v);}},">+=.1")
      .to(bars,{scaleY:1,duration:.5,stagger:.12,ease:"power3.out"},"<.15")
      .to(done,{autoAlpha:1,y:0,duration:.45,ease:"power2.out"},">+=.1")
      .to(tick,{strokeDashoffset:0,duration:.35,ease:"power1.inOut"},"<.15")
      .to(send,{scale:.96,duration:.1,ease:"power1.out"},">+=.35")
      .call(()=>send.classList.add("is-sent"),null,">")
      .to(send,{scale:1,duration:.2,ease:"power1.out"},">");
    const st=ScrollTrigger.create({trigger:root,start:"top 65%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();gsap.set([note,card,bars,done,tick,send],{clearProps:"all"});finish();};
  });
})();
