בתחילת הגלילה רואים טלפון אחד במרכז. כשגוללים, שני טלפונים נוספים מחליקים החוצה מאחוריו לשני הצדדים, גדלים מעט ונעמדים לידו, והכיתוב שמתחת לכל אחד מהם נכנס אחריו. כל טלפון מראה מסך אחר של אותו מוצר. הגולש מבין בגלילה אחת שזו מערכת שלמה ולא מסך בודד.

מתי משתמשים: מוצר או שירות שיש לו שלושה צדדים שכדאי להראות יחד: אפליקציה של עסק מקומי (הזמנה, מעקב, מועדון), פורטל לקוחות, מערכת תורים. מתאים לסקשן ״מה מקבלים״ בעמוד נחיתה או בעמוד מוצר, פעם אחת בעמוד. לא למוצר שיש בו מסך אחד בלבד, ולא כשאין מסכי דוגמה אמיתיים להציג.

שחזור. השראה: כמה סרטוני מוצר בגלריה של prompt-motion.com (21.10.2026) שבהם מכשירי דוגמה נכנסים אל המסגרת אחד אחרי השני. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: שלושה טלפונים זה לצד זה, והכיתובים מתחתם ברשימה אחת. בלי GSAP ובהפחתת תנועה זה מה שמוצג. כשהמהלך רץ הסקשן נעוץ (sticky בתוך אזור גלילה של 260vh), ו-p מ-0 עד 1 מזין פונקציה אחת paint(p): בין 12% ל-72% הטלפונים הצדדיים עוברים מהמרכז אל מקומם (x מ-dx ל-0), גדלים מ-84% ל-100% ומתבהרים, והטלפון המרכזי מתכווץ קצת מ-106% ל-100%. ההיסט dx נמדד ב-offsetLeft של כל מכשיר ביחס לשורה (לא מושפע מ-transform) ונמדד מחדש בכל refresh, ולכן אותו קוד נכון בכל רוחב ובכל כיוון. הכיתובים של הצדדיים מתבהרים אחרי שהטלפון כבר ממוקם, והאמצעי מוצג מההתחלה. RTL: הכל סימטרי סביב המרכז, אין מה להפוך ב-LTR. בנייד הטלפונים חופפים כמו מניפה והכיתובים הופכים לרשימה אחת. המסכים בנויים ב-container-type:inline-size וכל מידה בהם ב-cqw, לכן הם מתכווצים עם הטלפון בלי לשבור. כל השורה aria-hidden (מסכי דוגמה), והכיתובים ברשימה הם התוכן שקורא מסך שומע. תמונות: מהמאגר.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשארים שלושה טלפונים זה לצד זה עם הכיתובים (במארקאפ)
  const root=document.querySelector(".dvm"),run=root.querySelector(".dvm-run"),mid=root.querySelector(".dvm-mid"),
    sides=[...root.querySelectorAll(".dvm-side")],caps=[...root.querySelectorAll(".dvm-caps li")];
  const ease=gsap.parseEase("power3.inOut"),clamp=v=>Math.min(1,Math.max(0,v));
  // בהפחתת תנועה הפונקציה לא רצה, והשורה נשארת במצב הסיום
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    let dx=[],last=0;
    // כל מה שעל המסך נגזר מ-p בלבד, לכן אותו קוד נכון בגלילה קדימה ואחורה
    const paint=p=>{
      last=p;
      const a=clamp((p-.12)/.6),e=ease(a),c=clamp((a-.55)/.45);
      sides.forEach((s,i)=>gsap.set(s,{x:(dx[i]||0)*(1-e),scale:.84+.16*e,autoAlpha:Math.min(1,a*3)}));
      gsap.set(mid,{scale:1.06-.06*e});
      caps.forEach((el,i)=>{if(i!==1)gsap.set(el,{autoAlpha:c});});     // הכיתוב האמצעי מוצג מההתחלה
    };
    // המרחק מכל טלפון צדדי למרכז, נמדד ב-offsetLeft ולכן לא מושפע מה-transform
    const meas=()=>{const m=mid.offsetLeft+mid.offsetWidth/2;dx=sides.map(s=>m-(s.offsetLeft+s.offsetWidth/2));paint(last);};
    ScrollTrigger.addEventListener("refresh",meas);
    meas();
    const o={p:0};
    gsap.to(o,{p:1,ease:"none",onUpdate:()=>paint(o.p),
      scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.4,invalidateOnRefresh:true}});
    return()=>{
      ScrollTrigger.removeEventListener("refresh",meas);root.classList.remove("is-live");
      gsap.set([mid,...sides,...caps],{clearProps:"all"});
    };
  });
})();
