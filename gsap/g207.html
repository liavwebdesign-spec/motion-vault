G207 · אותיות וספרות שמתערבלות ואז ננעלות

GSAP GSAP · ScrollTrigger · פיצול ידני טקסט וכותרותמספרים ונתונים ממתין

כותרת ושלושה מספרי נתון נטענים כרצף של סימנים אקראיים בצבע המותג, ונועלים לערך האמיתי תו אחרי תו: האותיות בעברית מימין לשמאל, והספרות מהגדולה לקטנה. כל תו שומר על הרוחב הסופי שלו בזמן הערבול, אז השורה לא קופצת. בסוף נשאר טקסט רגיל.

מתי משתמשים: כותרת הירו של מוצר דיגיטלי, מערכת או דשבורד, ורצועת נתונים עם שלושה מספרים. עובד כשיש מספרים אמיתיים שמרגישים כמו מדידה. פעם אחת בעמוד, כי ערבול חוזר מעייף. לא לכותרת רגשית ולא לעסק שמבקש חמימות.

שחזור. השראה: כמה סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם כותרת או מספר מתפענחים מסימנים אקראיים. נבנה מאפס מהצפייה. המארקאפ: כל מטרה היא זוג: span.sr-only עם הטקסט הסופי לקורא מסך, וspan[data-sc][aria-hidden] עם אותו טקסט למי שרואה. בלי GSAP ובהפחתת תנועה מוצג הטקסט הסופי, ואין ערבול. פיצול ידני ובטוח: רק בעת ההפעלה נבנים תווים בתוך מילים (inline-block, בלי מילים שנשברות באמצע), וכל תו נועל את הרוחב שנמדד לו. כך אין קפיצת שורה כשגליף אקראי רחב מהאות האמיתית. בסיום הכול חוזר ל-textContent רגיל, בלי תווים נפרדים. סדר הנעילה: לפי הסדר הלוגי של הטקסט. בעברית התו הראשון הוא הימני ביותר, כלומר מצד הקריאה, ובמספר הספרה הראשונה היא הגדולה ביותר. מילה שמתחילה בספרה עטופה ב-direction:ltr; unicode-bidi:isolate, כדי שהספרות לא יתהפכו ב-RTL. קצב: גליף חדש כל 55 מילישניות בערך, בצבע --accent, וכל תו שננעל חוזר לצבע הטקסט. אין הבהוב בהיר-כהה בשדה גדול. הגליפים: אותיות עברית לתווים עבריים, ספרות לספרות, וסימנים (פסיק, נקודה, אחוז) לא מתערבלים. באתר משמאל לימין מחליפים את מאגר האותיות HEB במאגר של השפה, וסדר הנעילה נשאר: התו הראשון תמיד. נגישות: קורא מסך שומע רק את הטקסט הסופי.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הטקסט והמספרים כבר במצב הסופי
  const root=document.querySelector(".sc"),targets=[...root.querySelectorAll("[data-sc]")];
  const HEB="אבגדהוזחטיכלמנסעפצקרשת",FIN="ךםןףץ",DIG="0123456789";
  const pick=s=>s[Math.floor(Math.random()*s.length)];
  // בהפחתת תנועה הפונקציה לא רצה, והטקסט נשאר במצב הסופי
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const finals=new Map(targets.map(el=>[el,el.textContent])),sts=[],tws=[];
    gsap.set(targets,{opacity:0});      // לפני ההפעלה השורה מוסתרת, כך שהטקסט הסופי לא מבהיב לפני הערבול
    targets.forEach((el,k)=>{
      const final=finals.get(el);
      const run=()=>{
        // פיצול ידני לתווים בתוך מילים. כל תו נועל את הרוחב הסופי שלו, והמילה לא נשברת באמצע
        el.textContent="";
        const cs=[];
        final.split(" ").forEach((w,wi,arr)=>{
          const ws=document.createElement("span");ws.className="sc-w";
          if(DIG.indexOf(w[0])>=0)ws.classList.add("is-num");
          [...w].forEach(ch=>{
            const c=document.createElement("span");c.className="sc-c";c.textContent=ch;ws.appendChild(c);
            cs.push({el:c,ch:ch,done:false,pool:DIG.indexOf(ch)>=0?DIG:(HEB.indexOf(ch)>=0||FIN.indexOf(ch)>=0)?HEB:""});
          });
          el.appendChild(ws);
          if(wi<arr.length-1)el.appendChild(document.createTextNode(" "));
        });
        cs.forEach(c=>{c.w=c.el.getBoundingClientRect().width;});
        cs.forEach(c=>{c.el.style.width=c.w+"px";});
        const n=cs.length,o={p:0};let last=-1;
        const paint=()=>{
          const t=Math.floor(performance.now()/55),locked=Math.floor(o.p*(n+2));
          cs.forEach((c,i)=>{
            if(i<locked){if(!c.done){c.done=true;c.el.textContent=c.ch;c.el.classList.remove("is-s");}}
            else if(c.pool&&t!==last){c.el.textContent=pick(c.pool);c.el.classList.add("is-s");}
          });
          last=t;
        };
        paint();      // כבר בפריים הראשון הכול מעורבל, בלי הבהוב של הטקסט הסופי
        gsap.to(el,{opacity:1,duration:.25,ease:"power1.out"});
        tws.push(gsap.to(o,{p:1,duration:.9+n*.05,delay:k*.18,ease:"none",onUpdate:paint,onComplete:()=>{el.textContent=final;}}));
      };
      sts.push(ScrollTrigger.create({trigger:el,start:"top 88%",once:true,onEnter:run}));
    });
    return()=>{sts.forEach(s=>s.kill());tws.forEach(t=>t.kill());targets.forEach(el=>{gsap.killTweensOf(el);gsap.set(el,{clearProps:"opacity"});el.textContent=finals.get(el);});};
  });
})();
