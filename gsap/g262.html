שחזור. השראה: שני סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם נוף מאויר עובר בגלילה משקיעה ללילה, השמיים מתכהים והירח עולה. נבנה מאפס מהצפייה, עם תמונה אמיתית מהמאגר במקום איור. המארקאפ הוא הלילה: בלי GSAP ובהפחתת תנועה רואים את הנוף החשוך, הירח למעלה, הכוכבים והשעה 22:00, והשלבים כרשימה לידו. כשהמהלך רץ נוסף is-live: הבמה נעוצה בעמודה השנייה (בטלפון נעוצה למעלה בחצי המסך) והשלבים גוללים בעמודה הראשונה. ההתקדמות נמדדת ממרכז השלב הראשון ועד מרכז האחרון, כמו ב-g158. השכבות: התמונה (גולשת בין גודל 1.18 ל-1.06 סביב קו האופק, ולכן האופק לא זז), שכבת לילה שהיא גרדיאנט של --accent מעורבב בשחור במצב multiply (שחור ולא צבע הטקסט, כדי שהלילה יישאר כהה גם בעורות כהים), ירח, פס השתקפות ו-46 כוכבים. הירח הוא גרדיאנט רדיאלי (צד מואר למעלה מצד ההתחלה, שפה מעט כהה וכחלחלה) עם שני הילות, קרובה וצפופה ורחבה וקלושה, כדי שישב בתוך השמיים ולא יראה כמדבקה. הירח והפס יושבים בעטיפה ממוקמת באחוזים, ו-GSAP מזיז אותה במרחק שמחושב מגודל הבמה (פונקציה עם invalidateOnRefresh), כך שהמסלול נכון בכל גודל. הירח עולה מימין למטה לשמאל למעלה. צבע האור: --ls-light הוא --accent מעורבב בלבן, כדי שהירח והכוכבים יישארו בהירים בכל עור, כולל עור שהמבטא שלו כהה. השעון הוא תווית (aria-hidden) והשעות עצמן מופיעות בטקסט של השלבים. הוא מחושב בקטעים, 18:30 עד 20:00 בחצי הראשון של הגלילה ו-20:00 עד 22:00 בחצי השני, כדי שכל שלב יתפוס את השעה שלו. כוכב אחד מכל ארבעה מנצנץ ב-CSS על המאפיין scale, נפרד מהשקיפות ש-GSAP מזיז. הכוכבים מתחילים כבויים ב-gsap.set לפני הטיימליין, מאותה סיבה שמוסברת ב-g261 (בטווין עם stagger רק הראשון מקבל את מצב ההתחלה). מגבלה: זו תמונה אחת ולא שכבות איור, ולכן הקרוב והרחוק לא נעים במהירויות שונות. לעומק אמיתי מחליפים אותה בשלוש שכבות חתוכות.     const W=()=>scene.offsetWidth,H=()=>scene.offsetHeight,two=n=>String(n).padStart(2,"0");
    // השעה: 18:30 עד 20:00 בחצי הראשון של הגלילה, 20:00 עד 22:00 בחצי השני, בקפיצות של חמש דקות
    const prog={p:0},clock=()=>{const p=prog.p,m=Math.round((p<.5?1110+p*180:1200+(p-.5)*240)/5)*5;chip.textContent=two(Math.floor(m/60))+":"+two(m%60);};
    // ההתקדמות נמדדת ממרכז השלב הראשון ועד מרכז האחרון, על קו הקריאה: מרכז המסך, ובטלפון מרכז השטח שמתחת לבמה
    const list=root.querySelector(".ls-steps"),steps=list.querySelectorAll(".ls-step"),line=()=>matchMedia("(max-width:767px)").matches?"74%":"center";
    gsap.set(stars,{opacity:0});      // הכוכבים מתחילים כבויים (בטווין עם stagger רק הראשון מקבל את מצב ההתחלה מראש)
    const tl=gsap.timeline({scrollTrigger:{trigger:list,scrub:.6,invalidateOnRefresh:true,
      start:()=>"top+="+(steps[0].offsetTop-list.offsetTop+steps[0].offsetHeight/2)+" "+line(),
      end:()=>"bottom-="+steps[steps.length-1].offsetHeight/2+" "+line()}});
    tl.fromTo(img,{scale:1.18},{scale:1.06,ease:"none",duration:1},0)
      .fromTo(night,{opacity:0},{opacity:nightEnd,ease:"power1.inOut",duration:1},0)
      .fromTo(moonP,{x:()=>W()*.42,y:()=>H()*.5},{x:0,y:0,ease:"power1.inOut",duration:1},0)
      .fromTo(moon,{opacity:0},{opacity:1,ease:"power1.out",duration:.35},0)
      .fromTo(reflP,{x:()=>W()*.42},{x:0,ease:"power1.inOut",duration:1},0)
      .fromTo(refl,{opacity:0},{opacity:1,ease:"power1.in",duration:.7},.3)
      .to(stars,{opacity:i=>fin[i],ease:"none",duration:.4,stagger:{each:.6/stars.length,from:"random"}},.4)
      .fromTo(prog,{p:0},{p:1,ease:"none",duration:1,onUpdate:clock},0);
    clock();
    return()=>{root.classList.remove("is-live");chip.textContent=endText;};
  });
})();
