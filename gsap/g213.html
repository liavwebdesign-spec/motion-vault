G213 · פסים ושורות דירוג שמתמלאים לערכים

GSAP GSAP · ScrollTrigger once · scaleX מספרים ונתוניםכרטיסיםטעינה ופידבק ממתין

כרטיס עם חמש שורות דירוג, ובכל אחת שם, פס ואחוז. בכניסה למסך הפסים נמתחים מימין לאורך שלהם בזה אחר זה, והאחוזים סופרים מאפס באותו קצב. המקום הראשון בצבע המותג ושאר השורות באפור, ולכן רואים מיד מי מוביל ובכמה.

מתי משתמשים: תוצאות, השוואה או דירוג עם ארבע עד שבע שורות: מאיפה הגיעו הפניות, איזה שירות הכי מבוקש, חלוקת התקציב, תוצאות סקר. סקשן הוכחה או ״במספרים״. פעם אחת בעמוד, על הנתון שהכי חשוב להראות.

שחזור. השראה: כמה סרטונים בגלריה (21.10.2026) שבהם פסים אופקיים גדלים מאפס לערכים, המוביל בצבע המותג, והמספר סופר לידם. נבנה מאפס מהצפייה. מהווריאנטים (עמודות אנכיות, מד חום עם קווי סף, פסים עם טווח ביטחון) נבנה רק הפס האופקי, כי הוא הצורה שמתרגמת ישר לכרטיס באתר עסקי. כיוון (RTL): אין ציר זמן. הפס גדל מצד ההתחלה, שבעברית הוא ימין (transform-origin:100% 50%), והאחוז יושב בקצה השני. באתר משמאל לימין משנים ל-0% 50%. הערך הסופי במארקאפ: אורך הפס הוא --s על השורה (הערך חלקי הגבוה ביותר, כך שהמוביל מלא), ו-transform:scaleX(var(--s)) ב-CSS. בלי GSAP ובהפחתת תנועה רואים את הפסים והאחוזים הסופיים. כשהמהלך רץ הוא מאפס את שניהם ומעלה אותם יחד, תמיד במספר שלם, ו-clearProps בניקוי מחזיר את ה-CSS. נגישות: הפסים והמספר הרץ מוסתרים מקורא מסך, ולכל שורה יש טקסט מוסתר עם הערך הסופי (״34 אחוז, 141 פניות״), כך שקורא מסך לא שומע מספרי ביניים. הרשימה היא ol כי הסדר הוא מידע. ביצועים: scaleX בלבד וטריגר אחד על הכרטיס, פעם אחת (once). בטלפון השם עולה מעל הפס והאחוז נשאר בקצה.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הפסים והאחוזים כבר בערך הסופי (במארקאפ)
  const root=document.querySelector(".rk"),rows=[...root.querySelectorAll(".rk-row")];
  const bars=rows.map(r=>r.querySelector(".rk-b")),vals=rows.map(r=>r.querySelector(".rk-v"));
  const to=vals.map(v=>parseInt(v.textContent,10)),sc=rows.map(r=>parseFloat(r.style.getPropertyValue("--s"))||1);
  // בהפחתת תנועה הפונקציה לא רצה, והערך הסופי מוצג מיד
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const o=to.map(()=>({v:0})),paint=()=>vals.forEach((v,i)=>{v.textContent=Math.round(o[i].v)+"%";});
    gsap.set(bars,{scaleX:0});paint();
    const tl=gsap.timeline({paused:true,defaults:{ease:"power3.out",duration:1.1}});
    rows.forEach((r,i)=>{tl.to(bars[i],{scaleX:sc[i]},i*.09).to(o[i],{v:to[i],onUpdate:paint},i*.09);});
    const st=ScrollTrigger.create({trigger:root,start:"top 78%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();gsap.set(bars,{clearProps:"transform"});vals.forEach((v,i)=>{v.textContent=to[i]+"%";});};
  });
})();
