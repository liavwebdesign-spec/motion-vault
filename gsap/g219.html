G219 · לוח צבע אלכסוני שעובר ומחליף תמונה

GSAP GSAP · ScrollTrigger scrub · clip-path תמונות וגלריותהירו ופתיחה ממתין

שתי תמונות מלאות זו על זו, וכשגוללים לוח צבע אלכסוני רחב עובר על המסך מימין לשמאל, ומאחוריו נחשפת התמונה השנייה. מיד אחריו חולף לוח שני, דק ובהיר יותר. המעבר מכסה את הראשונה ומגלה את הבאה בתנועה אחת, בלי קפיצה.

מתי משתמשים: מעבר בין שני פרקים של אותו סיפור שלכל אחד תמונה גדולה משלו: בית קפה ואז המאפייה, לפני ואחרי, שני מוצרים, שני מקומות. פעם אחת בעמוד, ורק כשהתמונות חזקות מספיק כדי לשאת את המעבר.

הדמו

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני קידום ב-prompt-motion.com (21.10.2026) שבהם לוח צבע או רצועת אור אלכסונית חולפת על הפריים ומחליפה סצנה. נבנה מאפס מהצפייה. שלוש שכבות, כולן clip-path: התמונה השנייה נחשפת בפוליגון שקצהו המשופע נע עם הלוחות, והלוחות עצמם הם שני פוליגונים מקבילים לאותו קצה. משתנה אחד (o.p) מזיז את כולם, ולכן הלוח לא נפרד מקצה התמונה. הצד החיצוני של הפוליגון רחוק מהמסך תמיד: אחרת הצורה מתהפכת כשהקצה עובר את שולי המסך. המשפט של הקצה: ההטיה נקבעת מגובה הבמה (k = h * .2), לא מהרוחב, כך שבטלפון הזווית לא נהיית שטוחה. בלי GSAP ובהפחתת תנועה שתי הסצנות זו מתחת לזו והלוחות לא קיימים. כשהמהלך רץ הבמה נעוצה, לכן צריך מסלול של כשלושה מסכים בגובה: height:280svh על העוטף. כיוון: הלוח נכנס מהקצה הימני ויוצא בשמאלי, כך שהמחיקה קוראת בכיוון הקריאה בעברית. באתר משמאל לימין מחליפים בין הקצה הימני לשמאלי (מראת הציר: T0 מתחיל משמאל). גרסה רכה (רצועת אור): מחליפים את שני הלוחות בשכבה אחת עם filter:blur(24px), גרדיאנט פסטל ו-mix-blend-mode:screen, עם אותו פוליגון. נגישות: הלוחות הם קישוט (aria-hidden), והטקסט של כל סצנה נשאר בתוך התמונה שלה.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP שתי הסצנות זו מתחת לזו
  const root=document.querySelector(".ls"),pin=root.querySelector(".ls-pin"),A=root.querySelector(".ls-a"),B=root.querySelector(".ls-b"),
    s1=root.querySelector(".ls-s1"),s2=root.querySelector(".ls-s2"),ia=A.querySelector("img"),ib=B.querySelector("img");
  // בהפחתת תנועה הפונקציה לא רצה, והסצנות נשארות זו מתחת לזו
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const o={p:0};let w=0,h=0;
    const quad=(a,b,k)=>"polygon("+(a+k)+"px 0,"+(b+k)+"px 0,"+(b-k)+"px 100%,"+(a-k)+"px 100%)";
    const draw=()=>{
      const k=h*.2,w1=Math.max(w*.2,150),w2=Math.max(w*.035,26),gap=w2*.7;
      const t0=w+w1+w2+gap+k*2,T=t0+(-k-t0)*o.p;        // הקצה נע מימין למסך אל שמאל לו, ההטיה k נשארת קבועה
      // הצד הימני של הפוליגון תמיד מעבר לשולי המסך, אחרת הצורה מתהפכת כשהקצה עובר אותם
      B.style.clipPath="polygon("+(T-2+k)+"px 0,"+(t0+k)+"px 0,"+(t0+k)+"px 100%,"+(T-2-k)+"px 100%)";
      s2.style.clipPath=quad(T-w2,T,k);
      const b=T-w2-gap;
      s1.style.clipPath=quad(b-w1,b,k);
      gsap.set(ib,{scale:1.1-.1*o.p});
      gsap.set(ia,{scale:1.1,xPercent:-4*o.p});
    };
    const size=()=>{w=pin.clientWidth;h=pin.clientHeight;draw();};
    ScrollTrigger.addEventListener("refresh",size);size();
    gsap.timeline({scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}})
      .to({},{duration:.14})
      .to(o,{p:1,duration:.72,ease:"power2.inOut",onUpdate:draw})
      .to({},{duration:.14});
    return()=>{ScrollTrigger.removeEventListener("refresh",size);root.classList.remove("is-live");
      [B,s1,s2].forEach(e=>{e.style.clipPath="";});gsap.set([ia,ib],{clearProps:"transform"});};
  });
})();
