בהפחתת תנועה, וכש-GSAP לא נטען, אין נעילה: הרצועה נגללת לצד ביד, הקו מלא וכל התחנות דולקות. הקו והמסילה באורך הרצועה (inset-inline:0 על .hcont ברוחב max-content), ולכן מוסיפים או מורידים תחנה בלי לגעת ב-CSS.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הרצועה נגללת לצד וכל התחנות דולקות
  const wrap=document.querySelector(".hwrap"),hc=wrap.querySelector(".hcont");
  const steps=gsap.utils.toArray(".step",hc);
  const rails=[hc.querySelector(".tline-rail"),hc.querySelector(".tline")];
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    wrap.classList.add("gsap-live");
    const total=()=>hc.scrollWidth-wrap.clientWidth;
    gsap.set(steps.slice(1),{opacity:.22});
    gsap.set(".tline",{scaleX:0});
    // הכל יושב על טיימליין אחד שמחובר לפין. ScrollTrigger נפרד לכל תחנה,
    // שמכוון לאותו אלמנט מפונן, מחשב טווחים על מיקום שמשתנה מתחתיו והשלבים נשארים שקופים.
    const tl=gsap.timeline({scrollTrigger:{
      trigger:wrap,pin:true,scrub:.5,start:"top top",
      end:()=>"+="+total()*1.6,anticipatePin:1,invalidateOnRefresh:true}});
    tl.to([...steps,...rails],{x:()=>total(),ease:"none",duration:1},0)   // אתר עברי: x חיובי
      .to(".tline",{scaleX:1,ease:"none",duration:1},0);
    // הקו מגיע למרכז תחנה i ב-(i+0.5)/n מהמהלך, והתחנה מסיימת להידלק בדיוק שם ולא לפני
    steps.forEach((el,i)=>{
      if(!i)return;
      tl.to(el,{opacity:1,ease:"none",duration:.04},(i+.5)/steps.length-.04);
    });
    return()=>wrap.classList.remove("gsap-live");
  });
})();
