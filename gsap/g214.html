G214 · סימונים ותוויות שנוספים אחד אחרי השני

פס אחד מחולק לארבעה חלקים לפי המחיר של כל אחד. בגלילה מצטיירים סוגריים מסולסלים מתחת לכל חלק, ולידם מופיעה תווית עם השם והסכום, אחד אחרי השני. בסוף נפתח סוגר גדול אחד מעל כל הפס, והסכום הכולל סופר מאפס.

מתי משתמשים: הסבר שמורכב מכמה חלקים: מה כלול במחיר, איך מתחלקת תוכנית, שלבי פרויקט, מרכיבי שירות. סקשן מחירים, תהליך או הוכחה. שלושה עד חמישה חלקים, ולכל אחד שם קצר ומספר. הסיכום מופיע רק אחרי שהקורא ראה את החלקים.

שחזור. השראה: כמה סרטונים בגלריה (21.10.2026) שבהם סוגריים מסולסלים וקווי מידה נוספים בזה אחר זה על ציר מספרים, נוסחה או תרשים, ולידם תוויות. נבנה מאפס מהצפייה. מהווריאנטים (הערות בכתב יד, חצים מעוקלים ומילה מחוקה סביב צילום מסך) נבנו רק הסוגריים וקווי המידה, כי הם הצורה שמתרגמת למחיר או לשלבים באתר עסקי. כיוון (RTL): הפס מסודר מימין לשמאל, החלק הראשון בימין, והסימון נחשף מימין לשמאל. באתר משמאל לימין מעבירים ב-clip-path את ההיסט מצד שמאל (left) לצד ימין, והחלק הראשון יהיה משמאל. לא DrawSVG: הסוגר נמתח לרוחב החלק שלו (עד פי שישה בין הצר לרחב), ולכן הוא preserveAspectRatio="none" עם vector-effect:non-scaling-stroke כדי שהקו לא יתעבה, ושילוב כזה שובר את חישוב הקו המקווקו. במקומו הסימון נחשף במסכה (clip-path שמונפש דרך המשתנה --w), ונראה כמו ציור מימין לשמאל. כל סימון בתוך העמודה שלו: העמודות נקבעות ב-fr לפי הסכום, והפס, הסוגר והתווית יושבים באותה עמודה, ולכן הכול נשאר מיושר בכל רוחב, בלי שכבת SVG גלובלית. הערך הסופי במארקאפ: בלי GSAP ובהפחתת תנועה רואים הכול מסומן. כשהמהלך רץ הסצנה נעוצה (position:sticky, is-live) כדי שאפשר יהיה להגיע לסוף, כמו ב-g186, ובאותו כלל ה-CSS המשתנה --w מאופס ל-0 (לא ב-gsap.set, שלא כותב משתנה שעדיין לא הוגדר). בטלפון התוויות הופכות לרשימה מתחת לפס, ולכל שורה ריבוע בצבע החלק שלה, כי תווית לא נכנסת מתחת לחלק של 7%. נגישות: הסוגריים והפס מוסתרים מקורא מסך, והתוויות והסכום הכולל הם טקסט אמיתי, כך שקורא מסך שומע את כל הפירוט.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הכול מסומן ומתויג (במארקאפ)
  const root=document.querySelector(".an"),run=root.querySelector(".an-run");
  const blks=[...root.querySelectorAll(".an-blk")],brs=[...root.querySelectorAll(".an-br:not(.an-tb)")],tb=root.querySelector(".an-tb");
  const labs=[...root.querySelectorAll(".an-lab")],tot=root.querySelector(".an-tl"),tn=root.querySelector(".an-tn");
  const TOTAL=14850,fmt=new Intl.NumberFormat("he-IL"),wipes=[...brs,tb];
  // בהפחתת תנועה הפונקציה לא רצה, וכל הסימונים נשארים במקומם
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const o={v:0},paint=()=>{tn.textContent="₪"+fmt.format(Math.round(o.v));};
    gsap.set(blks,{opacity:.25});gsap.set([...labs,tot],{opacity:0,y:12});paint();   // opacity ולא visibility: התוויות נשארות בעץ הנגישות
    const tl=gsap.timeline({defaults:{ease:"power1.inOut"},scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.5,invalidateOnRefresh:true}});
    tl.to({},{duration:.4});
    brs.forEach((b,i)=>{
      const t=.4+i*1.1;
      tl.to(blks[i],{opacity:1,duration:.45},t).to(b,{"--w":1,duration:.6},t).to(labs[i],{opacity:1,y:0,duration:.45,ease:"power1.out"},t+.35);
    });
    const t=.4+brs.length*1.1;
    tl.to(tb,{"--w":1,duration:.8},t).to(tot,{opacity:1,y:0,duration:.5,ease:"power1.out"},t+.4)
      .to(o,{v:TOTAL,duration:.9,ease:"power1.out",onUpdate:paint},t+.4).to({},{duration:.7});
    return()=>{
      tl.scrollTrigger.kill();tl.kill();root.classList.remove("is-live");
      gsap.set([...blks,...labs,tot],{clearProps:"opacity,transform"});wipes.forEach(w=>w.style.removeProperty("--w"));
      tn.textContent="₪"+fmt.format(TOTAL);
    };
  });
})();
