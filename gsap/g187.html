G187 · מילה ענקית שנכנסת במריחה ומתחדדת

GSAP GSAP · ScrollTrigger scrub · SVG feGaussianBlur טקסט וכותרותהירו ופתיחה ממתין

מילה ענקית אחת נכנסת מהצד במריחת תנועה, מתחדדת בזמן שהיא מאטה ונעצרת, ובצבע המותג היא מסתיימת בנקודה. המשפט שלה עולה מתחתיה, ואז המילה נמרחת החוצה והבאה נכנסת. בכל רגע על המסך מילה אחת וברורה.

מתי משתמשים: תהליך עבודה, עקרונות או שלבי טיפול של ארבעה עד שישה צעדים, ולכל אחד מילה אחת חזקה ומשפט אחד. סקשן אחד בעמוד, באתר שמרשה לעצמו דרמה. פחות מתאים למילים ארוכות או כשיש הרבה טקסט בכל שלב.

כך נראה הטיפול אצלנו, משלב לשלב

  1. אבחון.

    פגישה ראשונה של שעה: בדיקה מלאה ושיחה על מה שמפריע לכם ביום יום.

  2. תוכנית.

    מקבלים תוכנית כתובה עם מטרות, מספר המפגשים והסבר על כל תרגיל.

  3. טיפול.

    מפגשים קבועים עם אותו מטפל, והתוכנית מתעדכנת לפי ההתקדמות שלכם.

  4. מעקב.

    אחרי הסיום בודקים איך אתם מרגישים, ואם צריך קובעים מפגש תחזוקה.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם מילה אחת גדולה נכנסת במריחה אופקית ומתחדדת כשהיא נעצרת, ונגמרת בנקודה צבעונית. נבנה מאפס מהצפייה. המריחה: פילטר SVG עם feGaussianBlur ו-stdDeviation="X 0", כלומר טשטוש לרוחב בלבד. המספר X רץ מ-90 ל-0 יחד עם התנועה (expo.out). הפילטרים נוצרים ב-JS ומוצמדים למילה רק כל עוד X גדול מ-0.4, ולכן המילה הסופית חדה לגמרי ובמנוחה אין שום פילטר פעיל. המארקאפ הוא רשימה מסודרת: כל שלב הוא li עם המילה והמשפט שלו, ובלי GSAP ובהפחתת תנועה זה מה שרואים. כשהמהלך רץ הסקריפט מוסיף is-live: הסקשן נעשה גבוה פי ארבעה, הבמה נדבקת למסך (position:sticky) וכל השלבים נערמים באותו מקום. המילה נכנסת מימין, המשפט עולה מתחתיה, והמילה יוצאת שמאלה בטשטוש עד שהבאה נכנסת. השלב האחרון נשאר. התחלה מוקדמת: הטריגר מתחיל ב-top 40%, כך שהמילה הראשונה כבר בדרך כשהסקשן עולה ולפני שהוא נדבק. באתר משמאל לימין מחליפים את הסימן של xPercent: מינוס 100 לכניסה ו-100 ליציאה. נגישות: הרשימה האמיתית נשארת בעץ הנגישות כל הזמן (שקיפות אינה הסתרה), והנקודה הצבעונית מוסתרת מקורא מסך. בטלפון אותה במה, והמילה היא 72px לפחות. מילים ארוכות מ-7 אותיות כדאי לקצר. פילטר SVG על אלמנט HTML חלש יותר ב-Safari, ושם המריחה עשויה להיראות כטשטוש עדין. התנועה עצמה נשארת.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP נשארת הרשימה המסודרת
  const root=document.querySelector(".mb"),lis=[...root.querySelectorAll(".mb-li")];
  if(!lis.length)return;
  // בהפחתת תנועה הפונקציה לא רצה, והרשימה נשארת
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const NS="http://www.w3.org/2000/svg",svg=document.createElementNS(NS,"svg");
    svg.setAttribute("width","0");svg.setAttribute("height","0");svg.setAttribute("aria-hidden","true");svg.style.position="absolute";
    const lines=lis.map(li=>li.querySelector(".mb-line")),subs=lis.map(li=>li.querySelector(".mb-s"));
    // פילטר טשטוש אופקי לכל מילה: stdDeviation "X 0" מותח את המילה לרוחב בלבד
    const B=lines.map((ln,i)=>{
      const f=document.createElementNS(NS,"filter"),g=document.createElementNS(NS,"feGaussianBlur");
      f.setAttribute("id","mb-f"+i);f.setAttribute("x","-70%");f.setAttribute("y","-10%");f.setAttribute("width","240%");f.setAttribute("height","120%");
      f.setAttribute("color-interpolation-filters","sRGB");g.setAttribute("stdDeviation","0 0");f.appendChild(g);svg.appendChild(f);
      return{v:0,g:g,ln:ln,id:"mb-f"+i};
    });
    const blur=o=>{o.g.setAttribute("stdDeviation",o.v.toFixed(1)+" 0");o.ln.style.filter=o.v>.4?"url(#"+o.id+")":"none";};
    root.appendChild(svg);root.classList.add("is-live");
    const tl=gsap.timeline({scrollTrigger:{trigger:root,start:"top 40%",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    let t=0;
    lis.forEach((li,i)=>{
      const last=i===lis.length-1;
      // כניסה: המילה מאטה (expo.out) והטשטוש יורד איתה, המשפט עולה באמצע התנועה
      tl.fromTo(lines[i],{xPercent:100},{xPercent:0,duration:.9,ease:"expo.out"},t)
        .fromTo(lines[i],{opacity:0},{opacity:1,duration:.3,ease:"power1.out"},t)
        .fromTo(B[i],{v:90},{v:0,duration:.9,ease:"expo.out",onUpdate:()=>blur(B[i])},t)
        .fromTo(subs[i],{opacity:0,y:16},{opacity:1,y:0,duration:.5,ease:"power2.out"},t+.55);
      t+=1.8;
      if(!last){
        // יציאה: נמרחת שמאלה בזמן שהבאה כבר נכנסת
        tl.to(subs[i],{opacity:0,duration:.3,ease:"power1.in"},t)
          .to(lines[i],{xPercent:-100,duration:.55,ease:"power2.in"},t)
          .to(lines[i],{opacity:0,duration:.3,ease:"power1.in"},t+.25)
          .to(B[i],{v:70,duration:.55,ease:"power2.in",onUpdate:()=>blur(B[i])},t);
        t+=.5;
      }
    });
    tl.to({},{duration:.7},t);
    // אחרי רענון (שינוי גודל חלון) הטיימליין חוזר למקומו בלי להפעיל onUpdate, אז מסנכרנים את הטשטוש מהערכים
    const sync=()=>B.forEach(blur);
    ScrollTrigger.addEventListener("refresh",sync);
    return()=>{ScrollTrigger.removeEventListener("refresh",sync);root.classList.remove("is-live");svg.remove();gsap.set([...lines,...subs],{clearProps:"all"});lines.forEach(l=>{l.style.filter="";});};
  });
})();
