G235 · כותרת עם אווטארים שמתחלפים ומילה שמתגלגלת

GSAP GSAP · IntersectionObserver · תמונות אמיתיות טקסט וכותרותהירו ופתיחהטעינה ופידבק ממתין

בתוך משפט הכותרת יושבת חבילה של ארבעה עיגולים חופפים עם פנים של אנשים, בין שתי מילים. כל כמה שניות הפנים מתחלפות בגלגול אנכי, עיגול אחרי עיגול, והמילה האחרונה במשפט מתגלגלת יחד איתן למילה אחרת. כשהכותרת נכנסת למסך החבילה צומחת עיגול אחרי עיגול. בלי GSAP ובהפחתת תנועה מוצגות ארבע פנים והמילה הראשונה.

מתי משתמשים: כותרת הירו של סוכנות, קליניקה או שירות שמאחוריו אנשים אמיתיים, כשיש ארבע עד שמונה תמונות פנים באיכות טובה. המילה המתגלגלת היא מי שהעסק משרת. כשאין תמונות פנים אמיתיות עדיף לוותר על המהלך.

לשיחת היכרות
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם משפט כותרת מכיל חבילה קטנה של אווטארים חופפים, והפנים מתחלפות בין המילים. נבנה מאפס מהצפייה. המארקאפ: ה-h2 מחזיק עותק sr-only של המשפט עם המילה הראשונה בלבד, והמשפט החזותי (.ga-vis) הוא aria-hidden. החבילה (.ga-av) היא inline-flex עם vertical-align:middle וגודלה .9em, ולכן היא גדלה עם הכותרת וחלק מהשורה. כל עיגול (.ga-s) חופף את קודמו ב-.3em, עם טבעת בצבע הרקע. המילה המתגלגלת (.ga-roll) היא inline-grid שבו כל המילים באותו תא ומסכה overflow:hidden בגובה השורה, ורוחבה הוא רוחב המילה הרחבה ביותר. חיתוך הפנים: התמונות רחבות, ולכן כל תמונה מקבלת שלושה משתנים (--x ו---y הם מיקום הפנים כשבר מהתמונה, ‏--z הגדלה) ו---ar יחס רוחב לגובה. ה-CSS מציב את התמונה כך שהפנים נופלות במרכז העיגול: גובה var(--s) * var(--z) ו-left של var(--s) * (.5 - x*z*ar). כשמחליפים תמונה מודדים מחדש את x ו-y. התנועה: מאגר של שמונה פנים. בכל סבב כל עיגול מקבל את הפנים ה'הבאות' במאגר (עיגול i מציג את (i+t) mod 8, ולכן ארבע פנים שונות תמיד), בגלגול אנכי של yPercent עם power3.inOut, ‏.7 שניות, ‏.12 בין עיגולים. המילה מתגלגלת באותו סבב. הסבב חוזר כל 2.8 שניות רק בזמן שהכותרת על המסך (IntersectionObserver, בלי ScrollTrigger, כי זה לא מהלך גלילה), ובכל כניסה למסך התמונות נטענות מראש. ההחלפה שומרת על שורה אחת שלמה: הפנים החדשות נכנסות מלמטה והישנות יוצאות למעלה. RTL: העיגול הראשון הוא הימני, והעיגולים הבאים נערמים שמאלה (margin-inline-start שלילי). באתר LTR ההערמה מתהפכת לבד. בטלפון הכותרת קטנה והחבילה והמילה שומרות על אותו יחס em.

(function(){
  if(typeof gsap==="undefined"||typeof IntersectionObserver==="undefined")return;      // בלי GSAP מוצגות ארבע פנים והמילה הראשונה
  const root=document.querySelector(".ga"),slots=[...root.querySelectorAll(".ga-s")],words=[...root.querySelectorAll(".ga-roll>span")];
  const IMG="https://liavwebdesign-spec.github.io/motion-vault/assets/media/";
  // [תמונה, רוחב, גובה, x, y, z]: x ו-y הם מיקום הפנים כשבר מהתמונה, z ההגדלה. ארבעת הראשונים הם מה שבמארקאפ
  const POOL=[["real/s02/hero-panel",2200,1468,.3,.45,3.2],["real/s09/step-2-welcome",1500,1200,.65,.27,2.8],["real/s12/hero-window",1100,1100,.48,.33,2.5],["real/s06/hug",1500,1100,.35,.5,2.8],
    ["real/s12/hero-window",1100,1100,.2,.37,2.5],["real/s09/hero-corridor",2000,1666,.26,.22,3],["real/s09/step-1-booking",1500,1200,.22,.28,2.8],["real/s12/hero-window",1100,1100,.86,.32,2.5]];
  // בהפחתת תנועה הפונקציה לא רצה, והחבילה והמילה נשארות במצב הראשון
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const init=slots.map(s=>s.innerHTML);
    let t=0,w=0,loop=null,warm=false,intro=false;
    const mk=p=>{
      const i=document.createElement("img");
      i.src=IMG+p[0]+".webp";i.alt="";i.width=p[1];i.height=p[2];i.decoding="async";
      i.style.cssText="--x:"+p[3]+";--y:"+p[4]+";--z:"+p[5]+";--ar:"+(p[1]/p[2]).toFixed(3);return i;
    };
    const step=()=>{
      t++;
      slots.forEach((s,i)=>{
        const old=s.querySelector("img:last-child"),nu=mk(POOL[(i+t)%POOL.length]),d=i*.12;
        s.appendChild(nu);
        gsap.fromTo(nu,{yPercent:100},{yPercent:0,duration:.7,ease:"power3.inOut",delay:d});
        gsap.to(old,{yPercent:-100,duration:.7,ease:"power3.inOut",delay:d,onComplete:()=>old.remove()});
      });
      const cur=words[w];w=(w+1)%words.length;
      gsap.fromTo(words[w],{yPercent:110},{yPercent:0,duration:.7,ease:"power3.inOut",delay:.1});
      gsap.to(cur,{yPercent:-110,duration:.7,ease:"power3.inOut",delay:.1});
    };
    const tick=()=>{step();loop=gsap.delayedCall(2.8,tick);};
    gsap.set(words.slice(1),{yPercent:110,opacity:1});
    gsap.set(slots,{scale:.5,opacity:0});
    // הסבב רץ רק כשהכותרת על המסך: IntersectionObserver במקום ScrollTrigger
    const io=new IntersectionObserver(es=>{es.forEach(e=>{
      if(loop){loop.kill();loop=null;}
      if(!e.isIntersecting)return;
      if(!warm){warm=true;POOL.forEach(p=>{new Image().src=IMG+p[0]+".webp";});}
      if(!intro){intro=true;gsap.to(slots,{scale:1,opacity:1,duration:.6,ease:"power3.out",stagger:.08});}
      loop=gsap.delayedCall(1.2,tick);
    });},{threshold:.2});
    io.observe(root);
    return()=>{
      io.disconnect();if(loop)loop.kill();
      gsap.killTweensOf(slots.concat(words));slots.forEach(s=>gsap.killTweensOf(s.querySelectorAll("img")));
      slots.forEach((s,i)=>{s.innerHTML=init[i];});gsap.set(slots.concat(words),{clearProps:"all"});
    };
  });
})();
