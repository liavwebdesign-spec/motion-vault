G182 · רשימת שלבים עם שבבי סטטוס שמתחלפים

GSAP GSAP · ScrollTrigger · sticky תהליך ושלביםטעינה ופידבק ממתין

כרטיס עם חמישה שלבים, ובכל שלב שבב סטטוס: ״ממתין״ באפור, ״בטיפול״ עם ספינר, ״הושלם״ בצבע המותג עם וי שמצטייר. הכרטיס נעוץ במסך והגלילה מקדמת אותו: שלב אחרי שלב הסטטוס מתחלף, פס ההתקדמות מתמלא והמונה למעלה עולה מאפס לחמש. זה תהליך שרואים מתקדם, לא רשימה שקוראים.

מתי משתמשים: תהליך אמיתי של שלושה עד שישה שלבים שיש להם סטטוס: קליטת לקוח, מעבר דירה או משרד, הקמת חשבון, הזמנה עם כמה שלבי טיפול. סקשן ״איך זה עובד״ או ״מה קורה אחרי שמצטרפים״. השלבים והסטטוסים צריכים להיות אמיתיים ובאים בסדר הזה. פעם אחת בעמוד.

שחזור. השראה: כמה סרטוני תהליך בגלריה של prompt-motion.com (21.10.2026), שבהם רשימת שלבים מחליפה סטטוסים בזה אחר זה ופס ההתקדמות עולה איתה. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: ol אמיתי, בכל שורה data-s="done", והמונה על 5. בלי GSAP ובהפחתת תנועה זו רשימה רגילה בזרימת העמוד. הקורא מסך מקבל את שם השלב ואת ״הושלם״, והתוויות האחרות של השבב מסומנות aria-hidden. כשהמהלך רץ הכרטיס נעוץ ב-position:sticky בתוך אזור גלילה גבוה (300vh, רק כשה-JS חי, כדי שבלעדיו לא יהיה רווח ריק). פונקציה אחת paint(p) מקבלת את ההתקדמות וקובעת לכל שורה את המצב (pending, busy, done) לפי הטווח שלה, ולכן אותו קוד עובד בגלילה למעלה ולמטה, ואין צורך בשום callback של ציר זמן. את המעבר בין שבבים עושה ה-CSS: שלוש תוויות חופפות שמתחלפות באטימות, ווי שמצטייר ב-stroke-dashoffset. להתאים: שלבים, שמות וסטטוסים הם התוכן, והטווח של כל שלב הוא T בקוד. RTL: הפס מתמלא מימין. ב-LTR משנים את transform-origin של .cl-fill ל-left center. בנייד הטקסט נמצא מעל הכרטיס, ושניהם נכנסים במסך אחד.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP זו רשימה רגילה, וכל השלבים במצב הושלם (במארקאפ)
  const root=document.querySelector(".cl"),run=root.querySelector(".cl-run"),rows=[...root.querySelectorAll(".cl-row")],
    fill=root.querySelector(".cl-fill"),cnt=root.querySelector(".cl-count b");
  const n=rows.length,T=.17,L=.06,fin=T*.65;     // כל שלב תופס T מההתקדמות, הראשון מתחיל ב-L, ו-fin הוא הנקודה שבה הוא מושלם
  const end=L+(n-1)*T+fin;
  // מצב כל שורה נגזר מההתקדמות בלבד, ולכן אותו קוד נכון בגלילה קדימה ואחורה
  const paint=p=>{
    let done=0;
    rows.forEach((r,i)=>{const a=L+i*T,s=p>=a+fin?"done":p>=a?"busy":"pending";if(s==="done")done++;if(r.dataset.s!==s)r.dataset.s=s;});
    cnt.textContent=done;
    gsap.to(fill,{scaleX:Math.min(1,Math.max(0,(p-L)/(end-L))),duration:.35,ease:"power2.out",overwrite:true});
  };
  // בהפחתת תנועה הפונקציה לא רצה, והרשימה נשארת במצב הסיום
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    gsap.set(fill,{scaleX:0});paint(0);
    const st=ScrollTrigger.create({trigger:run,start:"top top",end:"bottom bottom",invalidateOnRefresh:true,onUpdate:s=>paint(s.progress)});
    return()=>{st.kill();root.classList.remove("is-live");rows.forEach(r=>{r.dataset.s="done";});cnt.textContent=n;gsap.killTweensOf(fill);gsap.set(fill,{clearProps:"all"});};
  });
})();
