G231 · מילה גדולה עם הדים שנגררים אחריה

GSAP GSAP · ScrollTrigger velocity · pointer · text-stroke טקסט וכותרותהירו ופתיחה ממתין

מילה אחת גדולה מצוירת כמה פעמים מאחורי עצמה: חמישה עותקים בקו מתאר בצבע המותג, כל אחד מוסט מעט יותר מהקודם ובהיר פחות. כשהעמוד נכנס למסך העותקים נפתחים מהמילה אחד אחרי השני. כשגוללים מהר או מזיזים את העכבר המילה זזה, והעותקים נגררים אחריה בהשהיה, אחד אחרי השני, ואחרי רגע חוזרים ונשארים כערימה מוסטת מאחוריה. בלי GSAP ובהפחתת תנועה הערימה עומדת במקומה.

מתי משתמשים: כותרת הירו או פתיח של עמוד שכולו מילה אחת קצרה וחזקה, באתר עם אופי נועז: סטודיו, מותג צעיר, אירוע או קולקציה. מילה של עד שש אותיות, פעם אחת בעמוד. כשצריך לקרוא משפט שלם ההדים רק מפריעים.

לקביעת פגישת היכרות
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם מילה אחת ענקית מצוירת שוב ושוב מאחורי עצמה, בעותקים בקו מתאר ובהיסטים קטנים. נבנה מאפס מהצפייה. המארקאפ: ה-h2 מחזיק טקסט sr-only עם המילה ועטיפה aria-hidden עם שישה עותקים שלה: חמישה הדים (.ge-e, ולכל אחד --k מ-1 עד 5) ומילה מלאה אחת (.ge-m) שיושבת אחרונה ב-DOM, כדי שתצויר מעליהם. ההדים הם -webkit-text-stroke בלי מילוי. המצב הסופי הוא CSS בלבד: היסט של --k כפול --dx ו---dy (באורך em, ולכן גדל עם המילה; --dx מוגבל ל-1.15% מרוחב המסך כדי שההדים לא ייחתכו בשולי הטלפון) ושקיפות שיורדת עם --k. כשהמהלך רץ (.is-live) ה-CSS מוותר על ההיסט וה-JS כותב transform ישירות. חמישה עותקים בכל פריים לא צריכים את GSAP set. הלולאה היא gsap.ticker שפועל רק כשיש תנועה ונעצר כשהכול התייצב. שלושה כוחות מזיזים את המילה: (1) מצביע: המילה נמשכת עד כ-15% מגודל האות לכיוון העכבר, ביחס למרכז הסקשן ולא למרכז המילה, כדי שלא תיווצר לולאת משוב. (2) מהירות גלילה: ScrollTrigger.getVelocity() מתורגם להיסט אנכי שדועך ב-12% בכל פריים. (3) כניסה: כל הד נפתח מהמילה בעיכוב משלו (power3.out, ‏1.1 שניות). ההדים נגררים בשרשרת: הד 1 עוקב אחרי המילה, הד 2 אחרי הד 1 וכן הלאה, וכל חוליה עם מקדם השהיה קטן מהקודמת, ולכן ככל שההד רחוק יותר הוא מפגר יותר. RTL: ההדים נערמים ימינה ולמעלה, כלומר מאחורי כיוון הקריאה. לאתר LTR מחליפים את הסימן של --dx (כבר מוגדר ב-:dir(ltr), וה-JS קורא את direction בעצמו). לשימוש: מילה אחת של עד שש אותיות, צבע ההדים הוא --accent, ועובי הקו max(1.5px, .01em). בטלפון אין מצביע והגלילה היא שמניעה את ההדים.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשארת ערימת ההדים הסטטית
  const root=document.querySelector(".ge"),main=root.querySelector(".ge-m"),E=[...root.querySelectorAll(".ge-e")].reverse();   // E[0] הוא ההד הקרוב ביותר
  // בהפחתת תנועה הפונקציה לא רצה, וההדים נשארים במקומם
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const N=E.length,sgn=getComputedStyle(root).direction==="rtl"?1:-1;
    const rest=E.map((e,k)=>1-(k+1)*.17);                       // השקיפות במנוחה, כמו ב-CSS
    const sp=E.map(()=>({v:0}));                                 // כמה כל הד כבר נפתח מהמילה (0 עד 1)
    const m={x:0,y:0},p=E.map(()=>({x:0,y:0}));                  // מיקום המילה, ומיקום כל חוליה בשרשרת
    let fs=200,dxp=10,px=0,py=0,sy=0,on=false,entered=false;
    const measure=()=>{fs=parseFloat(getComputedStyle(main).fontSize)||200;dxp=Math.min(.05*fs,.0115*innerWidth);};   // כמו --dx ב-CSS: ההדים לא חורגים מהשוליים בטלפון
    const draw=()=>{
      main.style.transform="translate3d("+m.x+"px,"+m.y+"px,0)";
      for(let k=0;k<N;k++){
        const a=sp[k].v;
        E[k].style.transform="translate3d("+(p[k].x+sgn*(k+1)*dxp*a)+"px,"+(p[k].y-(k+1)*.045*fs*a)+"px,0)";
        E[k].style.opacity=rest[k]*a;
      }
    };
    const tick=()=>{
      sy*=.88;
      const tx=px,ty=py+sy;
      m.x+=(tx-m.x)*.16;m.y+=(ty-m.y)*.16;
      let prev=m,energy=Math.abs(tx-m.x)+Math.abs(ty-m.y);
      for(let k=0;k<N;k++){                                      // כל חוליה רודפת אחרי הקודמת, ופחות מהקודמת
        const f=.24-k*.035;
        p[k].x+=(prev.x-p[k].x)*f;p[k].y+=(prev.y-p[k].y)*f;
        energy+=Math.abs(prev.x-p[k].x)+Math.abs(prev.y-p[k].y);prev=p[k];
      }
      draw();
      if(energy<.05&&Math.abs(sy)<.05&&sp.every(o=>o.v===1))stop();
    };
    const go=()=>{if(!on){on=true;gsap.ticker.add(tick);}};
    const stop=()=>{if(on){on=false;gsap.ticker.remove(tick);}};
    const enter=()=>{
      if(entered)return;entered=true;go();
      sp.forEach((o,i)=>gsap.to(o,{v:1,duration:1.1,delay:.12+i*.09,ease:"power3.out"}));
    };
    const clamp=gsap.utils.clamp;
    const st=ScrollTrigger.create({trigger:root,start:"top 85%",end:"bottom top",onEnter:enter,onUpdate:s=>{   // מהירות הגלילה מושכת את המילה, וההדים נגררים אחריה
      sy=clamp(-fs*.3,fs*.3,s.getVelocity()*fs*.00006);go();
    }});
    const onMove=e=>{
      if(e.pointerType==="touch")return;
      const r=root.getBoundingClientRect();
      px=clamp(-1,1,(e.clientX-(r.left+r.width/2))/(innerWidth/2))*fs*.15;
      py=clamp(-1,1,(e.clientY-(r.top+r.height/2))/(innerHeight/2))*fs*.08;go();
    };
    const onLeave=()=>{px=py=0;go();};
    root.addEventListener("pointermove",onMove);root.addEventListener("pointerleave",onLeave);
    ScrollTrigger.addEventListener("refresh",measure);
    measure();draw();
    return()=>{
      stop();st.kill();sp.forEach(o=>gsap.killTweensOf(o));
      root.removeEventListener("pointermove",onMove);root.removeEventListener("pointerleave",onLeave);
      ScrollTrigger.removeEventListener("refresh",measure);
      root.classList.remove("is-live");main.style.transform="";E.forEach(e=>{e.style.transform="";e.style.opacity="";});
    };
  });
})();
