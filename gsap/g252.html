G252 · גרף עקומת האטה עם נקודה נעה

גרף גדול של עקומת קצב, ונקודה שרצה עליה בקצב הגלילה עם קווי עזר לשני הצירים. לידו שש שורות, ובכל אחת נקודה שעוברת אותו מרחק באותו זמן בקצב אחר: אחיד, מתחיל לאט, נגמר לאט, זינוק, קפיצות. אחרי כל עשירית מהזמן נשארת בשורה נקודה קטנה, והמרווחים ביניהן מראים איפה התנועה מהירה ואיפה איטית. לא קוראים את העקומה, מרגישים אותה.

מתי משתמשים: הסבר על קצב, בלי קשר לתחום: איך אנחנו עובדים (שלב איטי ויסודי בהתחלה, מהיר באמצע, גימור רגוע בסוף), סטודיו לאנימציה או עיצוב ממשקים שמסביר למה תנועה אחת מרגישה נכונה, קורס, או כל שירות שמתקדם בקצב לא אחיד. פעם אחת בעמוד, ורק כשהקצב הוא באמת חלק מהסיפור.

שחזור. השראה: כמה סרטוני תנועה ב-prompt-motion.com (21.10.2026), בכל אחד גרף של עקומת האטה לצד שורות של נקודות שמתחרות בקצבים שונים. נבנה מאפס מהצפייה. מקור אמת אחד: ששת הקצבים הם מערך E של פונקציות בקוד, והוא הדבר היחיד שמזיז הכול. הגרף, הנקודות והמספרים מתעדכנים מאותו ערך התקדמות (P, בין 0 ל-1), ולכן הם תמיד מסונכרנים. הקצב הרביעי הוא עקומת בזייה (0.76, 0, 0.24, 1), הקצב החתום של הספרייה, ונפתר מ-x ל-y בחיפוש בינארי. הערך הסופי בסימון: כל נקודה יושבת בקצה השורה דרך --p:1 ברירת מחדל, הנקודות הקטנות (.eg-gh) כולן מוצגות, והסמן בפינה העליונה של הגרף. רק כשהמהלך רץ הוא מאפס אותם ומקדם בגלילה. בלי מדידות ב-JS: מיקום הנקודה בשורה הוא translate ב-CSS מתוך --p וממידת השורה (cqw), ולכן שינוי רוחב המסך לא דורש חישוב מחדש. הגרף: SVG בלי טקסט (הכיתובים הם HTML), כדי שיהפוך בלי לקלקל אותיות. RTL: הזמן זורם מימין לשמאל, והגרף הוא SVG שמצויר משמאל לימין ומתהפך ב-:dir(rtl). באתר LTR מוחקים את שורת ההיפוך, והנקודות בשורות ממשיכות מעצמן כי הכיוון נקרא מ-:dir(). נגישות: הנקודות והעקומות מוסתרות מקורא מסך. מה שנקרא הוא טקסט מוסתר: עד כמה כל קצב התקדם בחצי הזמן, ומשפט שמתאר את העקומה (בנוי מאותה פונקציה, לא כתוב ביד). בטלפון הגרף מוגבל בגובה כדי ששתי הכרטיסיות נכנסות יחד במסך נעוץ, והשורות עוברות לשתי שורות: כיתוב ועקומה קטנה מעל, ומסלול ברוחב מלא מתחת.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP כל הנקודות בסוף והגרף שלם (בסימון)
  const root=document.querySelector(".eg"),run=root.querySelector(".eg-run");
  const mk=root.querySelector(".eg-mk"),halo=root.querySelector(".eg-halo"),gx=root.querySelector(".eg-gx"),gy=root.querySelector(".eg-gy");
  const rt=root.querySelector("[data-t]"),rd=root.querySelector("[data-d]");
  const rows=[...root.querySelectorAll(".eg-tr")].map(r=>({dot:r.querySelector(".eg-dot"),gh:[...r.querySelectorAll(".eg-gh")].map(g=>({el:g,t:Number(g.dataset.t)}))}));
  // ששת הקצבים: מקור האמת היחיד. הרביעי הוא העקומה שבגרף
  const E=[t => t,t => t * t * t,t => 1 - Math.pow(1 - t, 3),t => { let lo = 0, hi = 1, s = .5; for (let i = 0; i < 28; i++) { s = (lo + hi) / 2; const x = 3 * (1 - s) * (1 - s) * s * .76 + 3 * (1 - s) * s * s * .24 + s * s * s; if (x < t) lo = s; else hi = s; } return 3 * (1 - s) * s * s + s * s * s; },t => (1 - Math.pow(2, -10 * t)) / (1 - Math.pow(2, -10)),t => t >= 1 ? 1 : Math.floor(t * 6) / 6];
  const L=28,R=388,T=32,B=296;      // אזור הציור בתוך ה-viewBox של ה-SVG
  const draw=P=>{
    const d=E[3](P),x=L+(R-L)*P,y=B-(B-T)*d;
    mk.setAttribute("cx",x);mk.setAttribute("cy",y);halo.setAttribute("cx",x);halo.setAttribute("cy",y);
    gx.setAttribute("x1",x);gx.setAttribute("x2",x);gx.setAttribute("y1",y);
    gy.setAttribute("x2",x);gy.setAttribute("y1",y);gy.setAttribute("y2",y);
    rt.textContent=Math.round(P*100)+"%";rd.textContent=Math.round(d*100)+"%";
    rows.forEach((r,i)=>{r.dot.style.setProperty("--p",E[i](P).toFixed(4));r.gh.forEach(g=>g.el.classList.toggle("on",P>=g.t-.001));});
  };
  // בהפחתת תנועה הפונקציה לא רצה, ורואים את המצב הסופי מהסימון
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const st={p:0};
    draw(0);
    // עצירה קצרה בהתחלה ובסוף, כדי שמרגישים את נקודת היציאה ואת ההגעה
    const tl=gsap.timeline({scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    tl.to({},{duration:.1}).to(st,{p:1,duration:1,ease:"none",onUpdate:()=>draw(st.p)}).to({},{duration:.1});
    return()=>{
      root.classList.remove("is-live");
      mk.setAttribute("cx",R);mk.setAttribute("cy",T);halo.setAttribute("cx",R);halo.setAttribute("cy",T);
      gx.setAttribute("x1",R);gx.setAttribute("x2",R);gx.setAttribute("y1",T);
      gy.setAttribute("x2",R);gy.setAttribute("y1",T);gy.setAttribute("y2",T);
      rt.textContent="100%";rd.textContent="100%";
      rows.forEach(r=>{r.dot.style.removeProperty("--p");r.gh.forEach(g=>g.el.classList.remove("on"));});
    };
  });
})();
