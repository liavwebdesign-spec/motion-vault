G232 · אותיות שנוחתות במתיחה ובמעיכה

GSAP GSAP · ScrollTrigger once · פיצול לאותיות בלי SplitText טקסט וכותרותהירו ופתיחה ממתין

כותרת נבנית אות אחרי אות, לפי סדר הקריאה. כל אות יורדת מלמעלה ארוכה וצרה, עם צל בצבע המותג שנגרר מעליה, נמעכת לרגע כשהיא פוגעת בקו הבסיס, קופצת מעט למעלה ומתיישבת בגודל הרגיל. הקפיצה והמעיכה בנויות משלוש תנועות עוצמה רגילות בזו אחר זו, בלי עקומות קפיץ. בלי GSAP ובהפחתת תנועה הכותרת מוצגת סגורה.

מתי משתמשים: כותרת הירו קצרה, עד עשרים וארבעה תווים בשורה, בעמוד של מותג עם אופי שובב או חזק: מטבחים, קייטרינג, מותג לילדים, חדר כושר. פעם אחת בעמוד, בכותרת הראשית בלבד.

לתיאום פגישת תכנון
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם אותיות של כותרת נכנסות מעוותות, ארוכות או רחבות מדי, וחוזרות לפרופורציה שלהן. נבנה מאפס מהצפייה. הפיצול בלי SplitText: פונקציה קטנה (gr) מפרקת את הטקסט לאותיות וחוברת אליהן כל סימן ניקוד או טעם (תווים 1425 עד 1479), כך שאות עם ניקוד נשארת יחידה אחת. כל מילה עטופה ב-.gk-w עם nowrap, והאותיות הן .gk-c מסוג inline-block, כדי שהשורה תישבר רק בין מילים. אסור לפצל כך כתב שמתחבר (ערבית). הטקסט לקורא מסך: ה-h2 מחזיק עותק sr-only, והטקסט המפוצל יושב בעטיפה aria-hidden. הצל הוא ::after על כל אות עם content:attr(data-c) ושני משתנים (--gh שקיפות, --gy הזזה אנכית) ש-GSAP מנפיש. התנועה לכל אות, בסדר זה: נפילה (power2.in, ‏.28 שניות) שבה האות מתכווצת מרוחב 1.3 וגובה .6 אחרי שהייתה גבוהה וצרה; קפיצה (power2.out, ‏.2) לגובה 1.12; התיישבות (power2.inOut, ‏.26). מקור הטרנספורם הוא קו הבסיס (50% 100%) כדי שהמעיכה תהיה על הקו. ההשהיה בין אותיות היא .05 שניות. למה לא עקומת קפיץ: back, ‏elastic ו-bounce אסורים בתורה, ושלוש תנועות עוצמה רגילות נותנות את אותו קצב. ההפעלה: פעם אחת בכניסה למסך (once). עד אז האותיות מוסתרות, והכותרת מופיעה לקורא מסך מה-sr-only. RTL: האות הראשונה היא הימנית, כי הסדר הוא סדר ה-DOM. באתר LTR אין מה להחליף.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הכותרת מוצגת סגורה
  const root=document.querySelector(".gk"),vis=root.querySelector(".gk-vis");
  // ניקוד וטעמים (1425 עד 1479) נשארים צמודים לאות שלפניהם. עברית לא מתחברת, ולכן פיצול לאותיות בטוח
  const isMark=ch=>{const c=ch.charCodeAt(0);return c>=1425&&c<=1479;};
  const gr=s=>{const o=[];for(const ch of s){if(o.length&&isMark(ch))o[o.length-1]+=ch;else o.push(ch);}return o;};
  // בהפחתת תנועה הפונקציה לא רצה, והכותרת נשארת כמו שהיא
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const orig=vis.textContent;vis.textContent="";
    const L=[],words=orig.split(" ");
    words.forEach((w,wi)=>{
      const ws=document.createElement("span");ws.className="gk-w";
      gr(w).forEach(g=>{const c=document.createElement("span");c.className="gk-c";c.textContent=g;c.setAttribute("data-c",g);ws.appendChild(c);L.push(c);});
      vis.appendChild(ws);if(wi<words.length-1)vis.appendChild(document.createTextNode(" "));
    });
    gsap.set(L,{yPercent:-130,scaleY:1.9,scaleX:.6,opacity:0,"--gh":.55,"--gy":-.55});
    const tl=gsap.timeline({paused:true});
    L.forEach((c,i)=>{
      const t=i*.05;
      tl.to(c,{yPercent:0,scaleY:.6,scaleX:1.3,duration:.28,ease:"power2.in"},t)             // נפילה: נמתחת ואז נמעכת
        .to(c,{opacity:1,duration:.08,ease:"none"},t)
        .to(c,{"--gh":0,"--gy":0,duration:.28,ease:"power1.in"},t)                              // הצל נבלע באות
        .to(c,{yPercent:-9,scaleY:1.12,scaleX:.94,duration:.2,ease:"power2.out"},t+.28)         // קפיצה קטנה
        .to(c,{yPercent:0,scaleY:1,scaleX:1,duration:.26,ease:"power2.inOut"},t+.48);           // התיישבות
    });
    const st=ScrollTrigger.create({trigger:vis,start:"top 85%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();vis.textContent=orig;};
  });
})();
