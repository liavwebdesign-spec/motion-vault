G226 · תמונה מוטה שעפה אל הצופה עם טשטוש רדיאלי

GSAP GSAP · ScrollTrigger scrub · עותקי תמונה + קרניים תמונות וגלריותמספרים ונתוניםהירו ופתיחה ממתין

תמונה קטנה ומוטה במרכז המסך. בגלילה היא מסתובבת ישר ומתקרבת אל הצופה, ומסביבה נמתחים עותקים חיוורים וקרני אור שיוצאות מהמרכז, כמו צילום בזום מהיר. כשהיא נוחתת במקומה, הקרניים נעלמות והמספר שלצדה כבר קפץ לערך גדול.

מתי משתמשים: סקשן הוכחה או פתיחה של אתר מוצר, צילום או מותג עם תמונה אחת חזקה ונתון אחד גדול: כמה לקוחות, כמה מוצרים, כמה שנים. פעם אחת בעמוד. בטלפון זה רק התקרבות וסיבוב, בלי הטשטוש.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון מותג ב-prompt-motion.com (21.10.2026) שבו אריח מוטה מסתובב וזוכה לזום רדיאלי כשמספר קופץ לערך ענק. נבנה מאפס מהצפייה, בלי canvas ובלי WebGL: הטשטוש הרדיאלי הוא שישה עותקי תמונה חיוורים מאחורי התמונה, כל אחד מוגדל מעט יותר ושקוף יותר, ושכבת קרניים (repeating-conic-gradient עם מסכה רדיאלית) שסביבה. כל זה transform ו-opacity בלבד. העוצמה נגזרת מהקטע האמצעי של התנועה (sin(π·t)): אפס בהתחלה ובסוף, ולכן התמונה נוחתת חדה לגמרי. עדין בכוונה: העותק החיצוני מוגדל ב-21% לכל היותר, והשקיפות של כל עותק נמוכה. המספר עולה בשלב האמצעי ומוצג תמיד כמספר שלם. בלי GSAP ובהפחתת תנועה התמונה במנוחה והמספר הסופי, ואין עותקים וקרניים. בטלפון (עד 767 פיקסלים) אין עותקים וקרניים, כדי לחסוך ציור, והאריח רק מסתובב ומתקרב. נגישות: העותקים והקרניים קישוט (aria-hidden), והמספר הרץ מוסתר מקורא מסך ובמקומו המספר הסופי בטקסט מוסתר.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP התמונה במנוחה והמספר הסופי
  const root=document.querySelector(".zb"),tile=root.querySelector(".zb-tile"),img=tile.querySelector(".zb-img"),rays=root.querySelector(".zb-rays"),
    num=root.querySelector(".zb-count b"),to=parseInt(num.getAttribute("data-to"),10)||0;
  // בהפחתת תנועה הפונקציה לא רצה, והתמונה במנוחה
  gsap.matchMedia().add({desk:"(min-width:768px) and (prefers-reduced-motion: no-preference)",mob:"(max-width:767px) and (prefers-reduced-motion: no-preference)"},ctx=>{
    const mob=ctx.conditions.mob,N=mob?0:6,s0=mob?.6:.34,r0=mob?-8:-14;
    root.classList.add("is-live");
    const ghosts=[];
    for(let i=0;i<N;i++){const g=img.cloneNode();g.className="zb-ghost";g.alt="";g.setAttribute("aria-hidden","true");tile.insertBefore(g,img);ghosts.push(g);}
    const o={t:0,c:0};
    const draw=()=>{
      const t=o.t,s=Math.sin(Math.PI*t);      // העוצמה היא אפס בהתחלה ובסוף, התמונה נוחתת חדה
      gsap.set(tile,{scale:s0+(1-s0)*t,rotate:r0*(1-t)});
      ghosts.forEach((g,i)=>gsap.set(g,{scale:1+(i+1)*.035*s,opacity:s*(.2-i*.025)}));
      if(!mob)gsap.set(rays,{opacity:s*.6,scale:.8+.5*t,rotate:t*18});
      num.textContent=Math.round(o.c*to).toLocaleString("en-US");
    };
    draw();
    gsap.timeline({scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}})
      .to({},{duration:.1})
      .to(o,{t:1,duration:.7,ease:"power2.inOut",onUpdate:draw})
      .to(o,{c:1,duration:.5,ease:"power2.out",onUpdate:draw},"<.2")
      .to({},{duration:.2});
    return()=>{root.classList.remove("is-live");ghosts.forEach(g=>g.remove());gsap.set([tile,rays],{clearProps:"all"});num.textContent=to.toLocaleString("en-US");};
  });
})();
