G212 · רצועת שמות שנעה והשם במרכזה מואר

GSAP GSAP · ticker · ScrollTrigger טקסט וכותרותכרטיסים ממתין

רצועה של שמות לקוחות זורמת בלי הפסקה, כשכל השמות מתווה ריק חוץ מזה שנמצא במרכז: הוא נצבע במלואו בצבע המותג, ובדרך החוצה חוזר להיות מתווה. השם המואר מתחלף ברציפות, ומתחת לרצועה עומד נתון אחד גדול.

מתי משתמשים: רצועת לקוחות, שותפים או מותגים שאין להם לוגו מוכן, או שרוצים להאיר בהם אחד בכל פעם. מתאים לסקשן הוכחה, מתחת להירו או לפני הצעה. פעם אחת בעמוד. לא כשיש פחות משישה שמות, ולא כשהשמות הם הבטחה משפטית.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון בגלריה של prompt-motion.com (21.10.2026) עם רצועת שמות חברות שבה השם שבמרכז מואר ושאר השמות מתווה. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: רשימת שמות רגילה (ul) שעוטפת לכמה שורות במצב ניגודיות בינוני. כך זה נראה בלי GSAP ובהפחתת תנועה, ושם גם קורא מסך שומע אותה. כשהמהלך רץ הרשימה הופכת ל-sr-only, והסקריפט משכפל את הפריטים פעמיים לרצועה אחת (aria-hidden) שנעה ימינה: ההמשך של משפט בעברית נכנס מצד שמאל, כמו בכיוון הקריאה. מחזור אחד הוא רוחב סדרה אחת, ולכן אין קפיצה. ההארה: לכל פריט מחושבת מרחק מהמרכז ממידות שנשמרו בזמן המדידה (בלי getBoundingClientRect בכל פריים), ומוגדר --l בין 0 ל-1. ה-CSS מערבב מ---l את הצבע ואת קו המתאר ב-color-mix. ההדרגה ריבועית, כך שההארה חדה במרכז ונגמרת מהר. --l מתעדכן רק כשהערך זז, ולכן הדפדפן לא מחשב סגנון מחדש בכל פריים. ביצועים: עצירה כשהרצועה מחוץ למסך, והמדידה מחדש בשינוי גודל ובטעינת הפונטים. באתר משמאל לימין מחליפים את הסימן של התנועה (x מחושב כ--ph במקום ph-W) כך שהרצועה זזה שמאלה. מספר שמות: שבעה עד שנים עשר. פחות מזה והרצועה לא מכסה את הרוחב.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשארת רשימה רגילה של השמות
  const root=document.querySelector(".ml"),ul=root.querySelector(".ml-ul"),view=root.querySelector(".ml-vw"),track=root.querySelector(".ml-tr");
  // בהפחתת תנועה הפונקציה לא רצה, והשמות נשארים ברשימה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const items=[...ul.children];
    // שתי סדרות זהות ברצף, והמסלול נע רוחב סדרה אחת ומתחיל מחדש בלי קפיצה
    const sets=[0,1].map(()=>items.map(li=>{const c=li.cloneNode(true);track.appendChild(c);return c;}));
    root.classList.add("is-live");
    let W=1,half=0,ph=0,on=true,pos=[];
    const measure=()=>{
      half=view.clientWidth/2;W=track.scrollWidth/2||1;
      pos=[].concat(...sets).map(c=>({c:c,x:c.offsetLeft,w:c.offsetWidth,l:-1}));
    };
    const tick=(t,dt)=>{
      if(!on)return;
      ph=(ph+Math.min(dt,64)/1000*70)%W;      // ימינה: כיוון הקריאה בעברית נכנס משמאל
      const x=ph-W,R=Math.max(160,half*.34);
      track.style.transform="translate3d("+x.toFixed(1)+"px,0,0)";
      for(let i=0;i<pos.length;i++){
        const p=pos[i],d=Math.abs(p.x+x+p.w/2-half),l=Math.max(0,1-d/R),v=Math.round(l*l*50)/50;
        if(v!==p.l){p.l=v;p.c.style.setProperty("--l",v);}
      }
    };
    measure();tick(0,0);
    if(document.fonts&&document.fonts.ready)document.fonts.ready.then(measure);
    gsap.ticker.add(tick);
    window.addEventListener("resize",measure);ScrollTrigger.addEventListener("refresh",measure);
    const st=ScrollTrigger.create({trigger:view,start:"top bottom",end:"bottom top",onToggle:s=>{on=s.isActive;}});
    return()=>{gsap.ticker.remove(tick);st.kill();window.removeEventListener("resize",measure);ScrollTrigger.removeEventListener("refresh",measure);
      sets.forEach(s=>s.forEach(c=>c.remove()));root.classList.remove("is-live");track.style.transform="";};
  });
})();
