9 תורים השבוע, אף אחד לא כפול

שחזור. השראה: animtd.store (26.10.2026), סרטון של כרטיסים מפוזרים שמסתדרים לרשת בגלילה. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: ב-g266 אריחים נופלים מלמעלה אחד אחרי השני, פעם אחת (once). כאן כל הכרטיסים מתחילים יחד בערימה מבולגנת, והגלילה עצמה (scrub, עם הצמדה ב-sticky) מסדרת אותם; גוללים אחורה והבלגן חוזר. זה סיפור של לפני ואחרי, לא של מילוי. המארקאפ הוא היומן המסודר: גריד של 5 ימים על 4 שעות, וכל תור יושב בתא שלו (grid-column, grid-row). בלי GSAP, בהפחתת תנועה או במסך נמוך מ-600 זה מה שרואים. הערימה: לכל כרטיס נקודה בתוך היומן (שבר מהרוחב ומהגובה) וזווית. ההפרש בין הנקודה למקום שלו נמדד מ-offsetLeft ו-offsetTop, שלא מושפעים מה-transform, ולכן החישוב נכון גם כשהוא מתבצע מחדש אחרי שינוי גודל (invalidateOnRefresh). הגלילה: הסקשן גבוה פי 2.4 מהמסך והתוכן מוצמד ב-sticky; כל כרטיס נוסע למקום עם power2.inOut והשהיה קטנה בין כרטיס לכרטיס, ושורת הסיכום מופיעה בסוף. tween נפרד לכל כרטיס, לא stagger: ב-fromTo אחד עם stagger רק הכרטיס הראשון מצויר בערימה מראש, והשאר יושבים ביומן ונזרקים רק כשהתור שלהם מגיע, כך שהגלילה מפרקת את היומן במקום לסדר אותו (נתפס בצילומי הגלילה). נגישות: התורים הם רשימה, ולכל אחד טקסט מוסתר עם היום והשעה, כך שקורא מסך שומע "ראשון, 9:00: נועה כהן, טיפול ראשון". הימים, השעות והמשבצות aria-hidden. בלי קווי מסגרת: המשבצות הן משטח בהיר והכרטיסים כרטיסים. בטלפון אותו יומן בעמודות צרות, ובכרטיס נשאר רק השם.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP: כל התורים כבר ביומן (במארקאפ)
  const root=document.querySelector("[data-ms]");if(!root)return;
  gsap.registerPlugin(ScrollTrigger);
  const run=root.querySelector(".ms-run"),cal=root.querySelector(".ms-cal"),aps=[...root.querySelectorAll(".ms-ap")],sum=root.querySelector(".ms-sum");
  // לאן כל כרטיס נזרק בערימה: שבר מרוחב היומן ומגובהו, וזווית
  const PX=[.3,.62,.45,.72,.38,.55,.26,.68,.5],PY=[.35,.3,.55,.62,.72,.45,.58,.42,.25],PR=[-12,9,-6,14,7,-15,11,-8,4];
  // המרחק מהנקודה בערימה למקום ביומן, לפי הפריסה בלי transform, כך שגם חישוב מחדש אחרי שינוי גודל נכון
  const dx=i=>PX[i%PX.length]*cal.offsetWidth-(aps[i].offsetLeft+aps[i].offsetWidth/2);
  const dy=i=>PY[i%PY.length]*cal.offsetHeight-(aps[i].offsetTop+aps[i].offsetHeight/2);
  const mm=gsap.matchMedia();
  mm.add("(prefers-reduced-motion: no-preference) and (min-height: 600px)",()=>{
    root.classList.add("is-live");
    const tl=gsap.timeline({scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.5,invalidateOnRefresh:true}});
    // tween לכל כרטיס ולא stagger אחד: ב-fromTo עם stagger רק הכרטיס הראשון מצויר בערימה מראש, והשאר יושבים ביומן עד שהתור שלהם מגיע
    aps.forEach((a,i)=>tl.fromTo(a,{x:()=>dx(i),y:()=>dy(i),rotation:PR[i%PR.length],scale:1.04},
        {x:0,y:0,rotation:0,scale:1,ease:"power2.inOut",duration:.6},.05+i*.05));
    tl.fromTo(sum,{autoAlpha:0,y:12},{autoAlpha:1,y:0,ease:"power2.out",duration:.12});
    return()=>root.classList.remove("is-live");
  });
})();
