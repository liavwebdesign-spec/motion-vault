G208 · שורות טקסט שזורמות על מסלולים מעוקלים

GSAP GSAP · ticker · ScrollTrigger · SVG textPath טקסט וכותרותרקע ואווירה ממתין

שלוש שורות של שירותים מופרדים בנקודות מתעקלות על שלושה מסלולים בעקומות שונות, כל אחת במשקל ובצבע אחר: מלאה, מתווה וצבועה. השורות זורמות לכיוונים מנוגדים, ומהירות הזרימה גדלה כשגוללים. יחד הן נראות כסרט טיפוגרפי שחולף בעמוד.

מתי משתמשים: רצועת מעבר בין שני סקשנים, או רקע של הירו, כשיש שישה עד עשרה שירותים או תחומים קצרים להזכיר. מתאים לסטודיו, למשרד, לקבלן ולמרפאה עם כמה תחומי עיסוק. פעם אחת בעמוד. לא כשהמילים צריכות להיקרא לעומק, כי זה גם קישוט.

הדמו

גלול למטה, הדמו מגיע ↓

בנייה, שיפוצים, תכנון, עיצוב פנים, ליווי הנדסי ופיקוח

שיפוצים ובנייה מהתכנון ועד המפתח, עם אדם אחד שאחראי על כל הפרויקט.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים בגלריה של prompt-motion.com (21.10.2026) עם שורות מילים שמתעקלות על מסלולים ונוסעות. נבנה מאפס מהצפייה. המארקאפ: שלושה path בתוך defs, ועל כל אחד text עם textPath. הטקסט חוזר ארבע פעמים, כך שגם בלי JS המסלולים מלאים והשורות עומדות במקום. עברית על מסלול: ל-text יש direction:ltr, והמילים העבריות מסתדרות בעצמן מימין לשמאל לאורך המסלול (נבדק בדפדפן). עם direction:rtl הטקסט יוצא מהמסלול ונעלם. הזרימה: startOffset מוזז ב-ticker של GSAP. הסקריפט מודד את אורך יחידה אחת (data-u) ומשכפל אותה עד שהשורה מכסה את המסלול ועוד שתי יחידות, ואז המיקום חוזר ב-modulo לאורך יחידה אחת, ולכן אין קפיצה בסיבוב. שורה ראשונה ושלישית זורמות ימינה (ההמשך של משפט בעברית נכנס משמאל), והשנייה שמאלה. גלילה: ScrollTrigger.getVelocity() מוסיף מהירות שדועכת מעצמה. כשהרצועה מחוץ למסך ה-ticker לא עושה כלום. מובייל: ה-SVG רחב פי 2.3 מהמסך וממורכז, כך שרואים את אמצע העקומות באותיות קריאות. באתר משמאל לימין מוחקים direction:ltr מה-CSS והופכים את סימן dir של השורות (i%2?-1:1). נגישות: הסרט כולו aria-hidden, והשירותים בפסקה מוסתרת. בהפחתת תנועה הוא עומד.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הסרטים עומדים במקום, עם הטקסט במלואו
  const root=document.querySelector(".rb"),view=root.querySelector(".rb-v"),texts=[...root.querySelectorAll(".rb-svg text")];
  // בהפחתת תנועה הפונקציה לא רצה, והסרטים נשארים עומדים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const saved=texts.map(t=>t.querySelector("textPath").textContent);
    const rows=texts.map((t,i)=>({t:t,tp:t.querySelector("textPath"),unit:t.getAttribute("data-u"),
      len:root.querySelector(t.querySelector("textPath").getAttribute("href")).getTotalLength(),
      P:1,ph:0,dir:i%2?-1:1,sp:[64,42,88][i]||56}));
    // אורך יחידה אחת נמדד, והטקסט מוארך עד שהוא מכסה את המסלול ועוד שתי יחידות, כדי שהסיבוב לא יקפוץ
    const measure=()=>rows.forEach(r=>{
      r.tp.textContent=r.unit;r.P=r.tp.getComputedTextLength()||1;
      r.tp.textContent=r.unit.repeat(Math.ceil((r.len+2*r.P)/r.P)+1);
    });
    measure();
    if(document.fonts&&document.fonts.ready)document.fonts.ready.then(measure);
    let boost=0,on=true;
    const tick=(time,dt)=>{
      if(!on)return;
      const s=Math.min(dt,64)/1000;boost*=.92;
      rows.forEach(r=>{
        r.ph=((r.ph+r.dir*(r.sp+boost)*s)%r.P+r.P)%r.P;
        r.tp.setAttribute("startOffset",(r.ph-r.P).toFixed(1));
      });
    };
    gsap.ticker.add(tick);
    const st=ScrollTrigger.create({trigger:view,start:"top bottom",end:"bottom top",
      onToggle:s=>{on=s.isActive;},
      onUpdate:s=>{boost=Math.min(Math.abs(s.getVelocity())*.25,360);}});
    return()=>{gsap.ticker.remove(tick);st.kill();rows.forEach((r,i)=>{r.tp.textContent=saved[i];r.tp.setAttribute("startOffset","0");});};
  });
})();
