ארבעה שלבים סביב מעגל, עם חצים מעוגלים ביניהם. הדגשה עוברת משלב לשלב: השלב נדלק, פס ההתקדמות שבתוכו מתמלא, החץ אל השלב הבא מצטייר, והשלב הבא נדלק. אחרי הרביעי המחזור מתאפס וחוזר מההתחלה, והתנועה נעצרת כשהסקשן יוצא מהמסך. בטלפון המעגל הופך לרשימה אנכית עם חץ חזרה בצד.

מתי משתמשים: שירות שחוזר על עצמו: תחזוקה חודשית של אתר, מחזור קמפיין, בדיקה תקופתית, ריטיינר. מתאים כשיש ארבעה עד חמישה שלבים בסדר קבוע והמחזור חוזר מההתחלה, כי המעגל מראה את זה בלי להסביר. לא לתהליך שיש לו סוף, כמו פרויקט חד פעמי. פעם אחת בעמוד.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון אחד בגלריה של prompt-motion.com (21.10.2026) שבו צמתים יושבים על טבעת עם חצים ביניהם, הדגשה עוברת ביניהם ופסי התקדמות מתמלאים לכל צומת. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: כל ארבעת השלבים דולקים, כל הפסים מלאים וכל החצים בצבע. בלי GSAP ובהפחתת תנועה זה מה שמוצג, ואפשר לקרוא את הסדר מהרשימה. הגיאומטריה נחשבת בזמן בנייה: אליפסה 880 על 420, ארבעה כרטיסים בזוויות 45 מעלות, וכל קשת נחתכת בנקודה שבה היא יוצאת מהכרטיס שלה ונכנסת לכרטיס הבא (דגימת נקודות מול מלבן הכרטיס), עם ראש חץ בכיוון המשיק. הכרטיסים ממוקמים באחוזים ובמאפייני inset-inline-start. כשהמהלך רץ טיימליין אחד עם repeat:-1 מזיז משתני CSS: --act על הכרטיס (דולק או לא), --b על הפס (מילוי), ו-stroke-dashoffset על קשת ההדגשה (כל קשת עם pathLength=1 וראש החץ באותו נתיב, ולכן הוא מופיע אחרון). בסוף המחזור הכל מתאפס בחצי שנייה. הטיימליין נעצר כשהסקשן מחוץ למסך (ScrollTrigger עם onToggle). בטלפון (פחות מ-1000) אותה רשימה ol הופכת לעמודה, החצים בין הכרטיסים הם SVG קטן בתוך כל פריט, וחץ החזרה מהרביעי לראשון הוא סוגר בצד ההתחלה של העמודה (גובה כל כרטיס 96 וכל מרווח 36, ולכן ה-SVG בגובה קבוע). סדר השלבים הוא סדר ה-DOM, והחצים והפסים aria-hidden. ל-LTR: מיקום הכרטיסים לוגי (inset-inline-start) וה-SVG מתהפך ב-:dir(ltr), כך שהמעגל זורם נגד כיוון השעון מימין למעלה ב-RTL, ובכיוון השעון משמאל למעלה ב-LTR, תמיד לפי כיוון הקריאה. להחליף: מספר השלבים דורש גם זוויות חדשות במערך lpAng בקוד הבנייה, וקצב ההדגשה הוא SLOT.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP: כל השלבים דולקים, הפסים מלאים והחצים בצבע (במארקאפ)
  const root=document.querySelector(".lp");if(!root)return;
  gsap.registerPlugin(ScrollTrigger);
  const nodes=[...root.querySelectorAll(".lp-n")],bars=[...root.querySelectorAll(".lp-bar")];
  const SLOT=2.6,mm=gsap.matchMedia();
  // mob: רשימה אנכית. הקשת הרביעית (חזרה לראשון) היא הסוגר שבצד
  const build=mob=>{
    const arcs=mob?[...root.querySelectorAll(".lp-am .lp-h"),root.querySelector(".lp-br .lp-h")]:[...root.querySelectorAll(".lp-svg .lp-h")];
    gsap.set(nodes,{"--act":0});gsap.set(bars,{"--b":0});gsap.set(arcs,{strokeDashoffset:1});
    const tl=gsap.timeline({repeat:-1,paused:true}),end=nodes.length*SLOT;
    nodes.forEach((n,i)=>{
      const t=i*SLOT;
      tl.to(n,{"--act":1,duration:.35,ease:"power1.out"},t)
        .to(bars[i],{"--b":1,duration:SLOT-.6,ease:"none"},t+.15)
        .to(arcs[i],{strokeDashoffset:0,duration:.7,ease:"power1.inOut"},t+SLOT-.75);
      if(i<nodes.length-1)tl.to(n,{"--act":0,duration:.35,ease:"power1.in"},t+SLOT);
    });
    // סוף המחזור: הכל מתאפס, והשלב הראשון נדלק מחדש בהתחלה הבאה
    tl.to(nodes[nodes.length-1],{"--act":0,duration:.5},end+.2).to(bars,{"--b":0,duration:.5},end+.2).to(arcs,{strokeDashoffset:1,duration:.5},end+.2);
    // רץ רק כשהסקשן על המסך
    const st=ScrollTrigger.create({trigger:root,start:"top bottom",end:"bottom top",onToggle:s=>{if(s.isActive)tl.play();else tl.pause();}});
    if(st.isActive)tl.play();
    return()=>{
      st.kill();tl.kill();
      nodes.forEach(n=>n.style.removeProperty("--act"));bars.forEach(b=>b.style.removeProperty("--b"));
      arcs.forEach(a=>a.removeAttribute("style"));
    };
  };
  mm.add("(prefers-reduced-motion: no-preference) and (min-width: 1000px)",()=>build(false));
  mm.add("(prefers-reduced-motion: no-preference) and (max-width: 999px)",()=>build(true));
})();
