G169 · אותו איור עובר בין עונות עם חלקיקים

GSAP GSAP · ScrollTrigger scrub · שכבות SVG · canvas תמונות וגלריותרקע ואווירה ממתין

איור נוף אחד של בית עם חצר, בארבע שכבות מוערמות שמתחלפות בגלילה: אביב בפריחה, קיץ ירוק, סתיו כתום והעצים מתפשטים, חורף עם שדה לבן. מעל האיור חלקיקים שמתחלפים עם העונה, עלי כותרת, אבקה, עלים ושלג, והאור בחלון ואדי הארובה מתחזקים ככל שהיום קצר יותר.

מתי משתמשים: סקשן שמספר סיפור שנמשך לאורך השנה, כמו גינון ותחזוקה, אירוח, אירועים או כל שירות שהעבודה בו חוזרת בכל עונה. מתאים כשצריך להסביר ללקוח שהקשר לא נגמר אחרי הביקור הראשון, וכשיש איור או צילום אחד שאפשר לצבוע מחדש בשלושה או ארבעה מצבים. לא כשהסיפור לא משתנה בין המצבים, ולא באתר שמרני. פעם אחת בעמוד.

שחזור. השראה: איור שטוח אחד של כפר שעובר בין עונות, עם עלווה שמכתימה בכתום, עצים שמשילים עלים, שדות שמלבינים וחלקיקים שמתחלפים מעלי כותרת לעלים ולשלג (21.10.2026). נבנה מאפס מהצפייה. המבנה: .se-art מחזיק ארבע שכבות .se-l זו על זו, וכל אחת היא עותק מלא של אותו איור עם מחלקת עונה (sp, su, au, wi). הצבעים, השמש, הזוהר בחלון, האדים והשלג הם משתני CSS על מחלקת העונה, ולכן הוספת עונה היא עוד בלוק צבעים ולא עוד איור. ב-HTML נמצאת רק העונה הראשונה, ו-JS משכפל אותה בעת ההרצה. ההחלפה: שכבה חדשה מתגלה ב-opacity מעל הקודמת, בין s = k + .3 ל-s = k + .7, כך שהשכבות שמתחת נשארות מלאות ואין שום רגע של שקיפות כפולה. השמיים הם linear-gradient ב-HTML ולא מעבר צבע בתוך ה-SVG, כי מעבר צבע בתוך ה-SVG משתכפל עם אותו מזהה ויקבל את הצבעים של העותק הראשון. החלקיקים: canvas אחד מעל האיור עם ארבעה מאגרים (כ-34, 26, 32 ו-80 חלקיקים, ופחות במסך קטן). משקל כל מאגר נגזר מאותה אטימות של השכבות, ומאגר שמשקלו נמוך מ-2 אחוזים לא מצויר ולא מתקדם. הציור דל: סיבוב ב-setTransform, בלי מסננים. הוא מתחשב ב-devicePixelRatio עד 2, מוסר מה-ticker של GSAP כשהאיור מחוץ למסך (IntersectionObserver) ומשנה גודל עם ResizeObserver. הכיתוב: ארבע שורות מוערמות באותה תא רשת, שנעלמות ונכנסות בזו אחר זו עם הפסקה קצרה ביניהן כדי שלא ייראו שתי שורות על גבי שתיים. הרשימה המלאה גם נמצאת ב-sr-only. הצבעים: צבעי האיור עצמו (שמיים, עלווה, שלג) כתובים ישירות, כי הם חלק מהנוף ולא מהערכה של האתר, וכל השאר (כותרת, לשוניות, מסגרת) נשאר בטוקנים. כשמחליפים את האיור באיור אחר, מחליפים רק את גוף ה-SVG ואת בלוקי הצבעים. מצב סטטי: בלי GSAP ובהפחתת תנועה רואים את האביב בלבד, בלי חלקיקים. כיוון: האיור אינו תלוי בכיוון, והלשוניות והכיתוב מתהפכים לבד עם dir.

    const mk=[
      ()=>({x0:R(0,w),y:R(0,h),r:R(3.5,6),v:R(24,44),sw:R(10,26),ph:R(0,6.28),sp:R(.6,1.4),rot:R(0,6.28),rs:R(-2,2),c:Math.random()<.5?"#f7c4d2":"#fde3ea"}),
      ()=>({x0:R(0,w),y:R(0,h),r:R(1.2,2.6),v:-R(8,18),sw:R(8,22),ph:R(0,6.28),sp:R(.4,1),rot:0,rs:0,c:"#ffe08a"}),
      ()=>({x0:R(0,w),y:R(0,h),r:R(4,7),v:R(36,66),sw:R(14,34),ph:R(0,6.28),sp:R(.8,1.8),rot:R(0,6.28),rs:R(-3,3),c:LEAF[Math.floor(Math.random()*4)]}),
      ()=>({x0:R(0,w),y:R(0,h),r:R(1.4,3.4),v:R(26,58),sw:R(6,18),ph:R(0,6.28),sp:R(.5,1.2),rot:0,rs:0,c:"#fff"})
    ];
    function fit(){
      const r=art.getBoundingClientRect();w=r.width;h=r.height;dpr=Math.min(devicePixelRatio||1,2);
      cv.width=Math.round(w*dpr);cv.height=Math.round(h*dpr);ctx.setTransform(dpr,0,0,dpr,0,0);
      const f=Math.max(.45,Math.min(1,w*h/450000));
      pools=[34,26,32,80].map((n,k)=>Array.from({length:Math.round(n*f)},()=>mk[k]()));
    }
    function tick(t,dt){
      const d=Math.min(dt,50)/1000;time+=d;ctx.clearRect(0,0,w,h);
      for(let k=0;k<4;k++){
        if(W[k]<.02)continue;
        ctx.globalAlpha=W[k];
        for(const p of pools[k]){
          p.y+=p.v*d;
          if(p.v>0&&p.y>h+12){p.y=-12;p.x0=R(0,w);}else if(p.v<0&&p.y<-12){p.y=h+12;p.x0=R(0,w);}
          p.rot+=p.rs*d;
          const x=p.x0+Math.sin(time*p.sp+p.ph)*p.sw;
          ctx.fillStyle=p.c;ctx.beginPath();
          if(k===1){ctx.globalAlpha=W[k]*(.5+.5*Math.sin(time*1.6+p.ph));ctx.arc(x,p.y,p.r,0,6.283);}
          else if(k===3){ctx.arc(x,p.y,p.r,0,6.283);}
          else{const c=Math.cos(p.rot)*dpr,s=Math.sin(p.rot)*dpr;ctx.setTransform(c,s,-s,c,x*dpr,p.y*dpr);ctx.ellipse(0,0,p.r*(k===2?1.5:1),p.r*(k===2?.75:.55),0,0,6.283);}
          ctx.fill();ctx.setTransform(dpr,0,0,dpr,0,0);
        }
      }
      ctx.globalAlpha=1;
    }
    fit();
    const ro=new ResizeObserver(fit);ro.observe(art);
    const io=new IntersectionObserver(e=>{const v=e[0].isIntersecting;if(v&&!on){gsap.ticker.add(tick);on=true;}else if(!v&&on){gsap.ticker.remove(tick);on=false;}});io.observe(art);
    // s עובר מ-0 עד 3: אביב, קיץ, סתיו, חורף. כל שכבה מתגלה בין k+.3 ל-k+.7
    function draw(s){
      const o=[1,0,0,0];
      for(let k=1;k<4;k++)o[k]=sm(Math.max(0,Math.min(1,(s-(k-1)-.3)/.4)));
      const idx=Math.round(s);
      for(let k=0;k<4;k++){
        layers[k].style.opacity=o[k];W[k]=o[k]*(k<3?1-o[k+1]:1);
        tabs[k].classList.toggle("on",k===idx);
        const a=Math.max(0,Math.min(1,(.5-Math.abs(s-k))/.2));
        caps[k].style.opacity=a;caps[k].style.transform="translateY("+((1-a)*10).toFixed(1)+"px)";
      }
    }
    const st={s:0};
    gsap.to(st,{s:3,ease:"none",onUpdate:()=>draw(st.s),scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}});
    draw(0);
    return()=>{
      if(on){gsap.ticker.remove(tick);on=false;}
      io.disconnect();ro.disconnect();
      layers.slice(1).forEach(l=>l.remove());cv.remove();first.style.opacity="";
      tabs.forEach((t,k)=>t.classList.toggle("on",k===0));
      caps.forEach(c=>{c.style.opacity="";c.style.transform="";});
      root.classList.remove("is-live");
    };
  });
