G265 · אותו צילום בפסים ובתשע פלטות צבע

GSAP GSAP · ScrollTrigger scrub · sticky · mix-blend-mode תמונות וגלריותכרטיסים ממתין

אותו פורטרט חוזר בחמישה פסים אנכיים, וכל פס צבוע בשני גוונים אחרים מהפלטה של האתר, כשהפס האמצעי נשאר בצבע המלא. אחרי עצירה קצרה הפסים נמסים, ובמקומם נפתחת רשת של תשע גרסאות של אותו צילום, כל אחת בפלטה אחרת ועם רשת נקודות דקה. המספר ליד המסגרת עובר מחמש לתשע.

מתי משתמשים: מקום שבו צריך להשוות בין מראות של אותו דבר: פלטות צבע של מותג, גרסאות צבע של מוצר, אותו פורטרט בכמה סגנונות הדפסה. מתאים לדף של סטודיו לעיצוב או להדפסה, לחנות עם וריאציות צבע ולתיק עבודות. פעם אחת בעמוד, כשהבחירה בצבע היא הנושא. לא כשיש צילום אחד שאין לו שום אפשרות אחרת.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים בגלריה (21.10.2026) שבהם אותו צילום חוזר בפסים צבועים בפלטות שונות ואחר כך נפרש לרשת של גרסאות. נבנה מאפס מהצפייה. הדו-גוון הוא CSS בלבד על הצילום האמיתי: כל פאנל הוא figure עם רקע בגוון הבהיר (--c2), הצילום בשחור לבן עם mix-blend-mode:multiply מעליו, ושכבה בגוון הכהה (--c1) עם lighten מעל. הגוונים נבנים מהטוקנים של האתר (--ink, --accent, --muted, --bg, --card) עם color-mix, ולכן הפלטות מתחלפות עם העור של הפרויקט ואין צבע שנקבע ביד. הפאנל .pp-o בלי פילטר, בצבע המקורי. המארקאפ הוא המצב הסופי: רשת 3 על 3 עם הנקודות והמספר 9, ושכבת הפסים לא מוצגת בכלל. בלי GSAP או בהפחתת תנועה זה מה שרואים. כשהמהלך רץ נוספת is-live (במה נעוצה בגובה 300vh): הפסים נחשפים מלמעלה למטה, מימין לשמאל, נעצרים, ואז נמסים והאריחים צצים מהמרכז החוצה (stagger עם from:'center'), והמספר מחליף את החמש בתשע. צילום: פורטרט של אדם על רקע חלק עובד הכי טוב, כי הדו-גוון צובע גם את הרקע. בטלפון המסגרת ריבועית והפסים מראים פרוסה צרה מאותו צילום. נגישות: המסגרת היא role=img עם תיאור, הפאנלים בתוכה מוסתרים, והמספר הסופי בטקסט מוסתר.

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP הרשת של תשע הגרסאות כבר במארקאפ
  const root=document.querySelector(".pp"),stripes=[...root.querySelectorAll(".pp-stripes .pp-p")],tiles=[...root.querySelectorAll(".pp-grid .pp-p")];
  const n5=root.querySelector(".pp-num i:first-child"),n9=root.querySelector(".pp-num i:last-child");
  // בהפחתת תנועה הפונקציה לא רצה, והרשת נשארת
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top 70%",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    // 1. חמשת הפסים נחשפים מלמעלה, אחד אחרי השני מימין לשמאל
    gsap.set(stripes,{clipPath:"inset(0 0 100% 0)"});gsap.set(tiles,{scale:.4,autoAlpha:0});      // מצב פתיחה מפורש (ה-stagger לא תמיד מציג אותו מיד)
    tl.fromTo(stripes,{clipPath:"inset(0 0 100% 0)"},{clipPath:"inset(0 0 0% 0)",duration:.7,ease:"power3.out",stagger:.14},0);
    tl.to({},{duration:1});                                      // עצירה עם חמשת הפסים
    // 2. הפסים נמסים והתשעה צצים מהמרכז החוצה
    const t=tl.duration();
    tl.fromTo(tiles,{scale:.4,autoAlpha:0},{scale:1,autoAlpha:1,duration:.7,ease:"power3.out",stagger:{each:.1,from:"center",grid:[3,3]}},t);
    tl.to(stripes,{autoAlpha:0,duration:.5,ease:"power1.in"},t);
    tl.fromTo(n5,{autoAlpha:1,y:0},{autoAlpha:0,y:-14,duration:.35,ease:"power1.in"},t+.1);
    tl.fromTo(n9,{autoAlpha:0,y:14},{autoAlpha:1,y:0,duration:.45,ease:"power2.out"},t+.3);
    tl.to({},{duration:.8});
    return()=>{root.classList.remove("is-live");gsap.set([...stripes,...tiles,n5,n9],{clearProps:"all"});};
  });
})();
