גרף קו דק שמצויר בגלילה, ובקצהו נקודה זוהרת. כשהקו עובר על תקופה, פס שקוף נדלק מעליה ועליו תווית קצרה: פסח, עונת השיא, החופש הגדול. שני הפסים שמעל הירידות מודגשים בצבע המותג, ולצד הגרף מילה גדולה שמסכמת את הדפוס.

מתי משתמשים: סיפור של עונתיות או מגמה: פניות לפי חודש, מכירות לפי עונה, עומס בקליניקה. סקשן הוכחה, תוכן או הסבר. סדרה אחת של עשר עד שתים עשרה נקודות, ושלוש עד ארבע תקופות מסומנות שיש לכל אחת סיבה ידועה.

שחזור. השראה: סרטון בגלריה (21.10.2026) שבו קו דק עם נקודה זוהרת מצטייר, ואז פסים שקופים נדלקים מעל תאריכים נבחרים, שניים מהם מעל ירידות, ומילה גדולה ליד הגרף. נבנה מאפס מהצפייה. ציר הזמן מימין לשמאל: אוקטובר בימין וספטמבר בשמאל. הקו מצטייר מימין, והנקודה מסיימת בשמאל. באתר משמאל לימין מסדרים את הנקודות הפוך ומחליפים בסקריפט X0 ו-X1. הציור: הקו הוא path עם pathLength="1" ו-stroke-dasharray:1 2 (המרווח הכפול מונע נקודה בודדת בתחילת הציור), וההתקדמות היא stroke-dashoffset. הנקודה היא שכבת HTML שממוקמת מ-getPointAtLength ביחידות קונטיינר, ולכן היא מתכווצת עם הגרף ולא מתנתקת מהקו. הפסים: שכבת HTML עם left ו-width באחוזים מרוחב הגרף, מחושבים מאותם viewBox ונקודות. כל פס נדלק ברגע שהקו עובר במרכזו, והרגע מחושב בחיפוש בינארי על אורך הנתיב. הערך הסופי במארקאפ: בלי GSAP ובהפחתת תנועה רואים קו שלם, שלושה פסים עם תוויות והנקודה בקצה. נגישות: הגרף והנקודה מוסתרים מקורא מסך. התוויות הן טקסט, ופסקה מוסתרת מתארת את הדפוס במספרים. בטלפון הגרף מתכווץ והתוויות יורדות לשתי שורות. הסצנה נעוצה (is-live) כדי שאפשר יהיה להגיע לסוף.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP: קו שלם, שלושה פסים עם תוויות והנקודה בקצה (במארקאפ)
  const root=document.querySelector(".pb"),run=root.querySelector(".pb-run"),plot=root.querySelector(".pb-plot");
  const line=plot.querySelector(".pb-line"),bands=[...plot.querySelectorAll(".pb-band")];
  const X0=700,X1=20,fx0=plot.style.getPropertyValue("--fx"),fy0=plot.style.getPropertyValue("--fy");   // X0 אוקטובר (ימין), X1 ספטמבר (שמאל)
  // בהפחתת תנועה הפונקציה לא רצה, והגרף המלא נשאר
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const L=line.getTotalLength(),o={p:0};
    const place=()=>{
      const pt=line.getPointAtLength(L*o.p);line.style.strokeDashoffset=1-o.p;
      plot.style.setProperty("--fx",(pt.x/720).toFixed(4));plot.style.setProperty("--fy",(pt.y/300).toFixed(4));
    };
    // ההתקדמות שבה הקו עובר במרכז הפס: חיפוש בינארי על אורך הנתיב (ה-x יורד לאורכו)
    const when=b=>{
      const cx=(parseFloat(b.style.getPropertyValue("--l"))+parseFloat(b.style.getPropertyValue("--w"))/2)/100*720;
      let a=0,z=1;for(let i=0;i<24;i++){const m=(a+z)/2;if(line.getPointAtLength(L*m).x>cx)a=m;else z=m;}return z;
    };
    const D=4,T0=.4,tl=gsap.timeline({scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.5,invalidateOnRefresh:true}});
    place();
    tl.to(o,{p:1,duration:D,ease:"none",onUpdate:place},T0);
    bands.forEach(b=>tl.fromTo(b,{opacity:0},{opacity:1,duration:.5,ease:"power1.out"},T0+when(b)*D));
    tl.to({},{duration:.8});
    return()=>{
      tl.scrollTrigger.kill();tl.kill();root.classList.remove("is-live");line.style.strokeDashoffset="";
      plot.style.setProperty("--fx",fx0);plot.style.setProperty("--fy",fy0);gsap.set(bands,{clearProps:"opacity"});
    };
  });
})();
