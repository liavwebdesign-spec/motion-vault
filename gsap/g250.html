שחזור. השראה: סרטון בגלריה של prompt-motion.com (21.10.2026) שבו ריבוע מתאר קטן עובר על רשת גסה של פיקסלים שורה אחרי שורה, ופאנל שני מצטבר ממנו מפת תוצאה, עם מילה אחת גדולה שנותנת שם לפעולה. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: המפה מלאה, החלון נמצא במשבצת האחרונה והמונה אומר 144 מתוך 144. התמונה: תמונה אמיתית מהמאגר שהוקטנה ל-14 על 14 פיקסלים (PNG של כ-450 בתים, מוטמע), ו-image-rendering:pixelated מגדיל אותה למשבצות חדות בכל צפיפות מסך. המפה היא 12 על 12 ערכים (--v בין 0 ל-1) שחושבו מראש מהתמונה הזאת (גודל השיפוע בין שכנים, חלון של 3 על 3), ולכן החלון והמפה מתאימים לתמונה באמת. האנימציה: טיימליין עם scrub, וכל הציור קורה ב-onUpdate: אינדקס אחד n נגזר מהזמן, החלון זז לשורה ולעמודה של n דרך משתני CSS (--c, --r) והמשבצות עד n מוצגות. כך הכיוון האחורי בגלילה עובד בלי חשבון נוסף. הצמדה: .sw-run מקבל גובה רק אחרי ש-is-live נוסף ב-JS, ו-.sw-pin הוא sticky. נגישות: המפה והתמונה aria-hidden, ומשפט לקוראי מסך אומר מה הוצג. בהפחתת תנועה הכל נשאר במצב הסיום וגם אין הצמדה. באתר LTR הסדר בשורה מתהפך מעצמו, והחלון ממוקם בקואורדינטות פיזיות כי התמונה לא מתהפכת. טלפון: שני הפאנלים נערמים אנכית, עם המילה ביניהם, וכל הערימה נכנסת במסך אחד. כדי להחליף תמונה: ליצור PNG חדש של 14 על 14 ולחשב מחדש את 144 הערכים, אחרת החלון יסמן קצוות שלא קיימים.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP: המפה מלאה והחלון במשבצת האחרונה (במארקאפ)
  const root=document.querySelector(".sw"),run=root.querySelector(".sw-run"),win=root.querySelector(".sw-win"),
    cells=[...root.querySelectorAll(".sw-grid i")],word=root.querySelector(".sw-word"),cnt=root.querySelector(".sw-n");
  const T=cells.length,COLS=12,D=10;
  let shown=T;
  // כל הציור נגזר מזמן הטיימליין: n משבצות נסרקו, והחלון עומד על המשבצת ה-n
  const paint=t=>{
    const p=Math.min(1,Math.max(0,t/D)),n=Math.round(p*T),pos=Math.min(n,T-1);
    win.style.setProperty("--c",pos%COLS);win.style.setProperty("--r",Math.floor(pos/COLS));
    if(n!==shown){
      const a=Math.min(n,shown),b=Math.max(n,shown);
      for(let k=a;k<b;k++)cells[k].style.opacity=k<n?1:0;
      shown=n;
    }
    cnt.textContent=n;
    word.style.opacity=(.2+.8*p).toFixed(2);
  };
  // בהפחתת תנועה הפונקציה לא רצה, ואין הצמדה: המפה נשארת מלאה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const tl=gsap.timeline({scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.5,invalidateOnRefresh:true}});
    tl.set({},{},D);                          // זמן הטיימליין
    tl.eventCallback("onUpdate",()=>paint(tl.time()));
    cells.forEach(c=>{c.style.opacity=0;});shown=0;paint(0);
    return()=>{tl.scrollTrigger.kill();tl.kill();root.classList.remove("is-live");
      cells.forEach(c=>{c.style.opacity="";});win.style.removeProperty("--c");win.style.removeProperty("--r");
      word.style.opacity="";cnt.textContent=T;shown=T;};
  });
})();
