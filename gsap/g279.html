G279 · קווים שמתפצלים ואחד מהם נבחר

GSAP GSAP · ScrollTrigger scrub · sticky · SVG תהליך ושלבים ממתין

צומת אחד כבר הושלם, ומתוכו יוצאים ארבעה קווים דקים לארבע אפשרויות. האפשרויות נגלות אחת אחרי השנייה, ואז אחת נבחרת: הקו שלה מתעבה וצובע את עצמו, הצומת מתמלא בצבע ומקבל תווית, והשלוש האחרות נחלשות. הגלילה מובילה את הבחירה, וגלילה אחורה מחזירה את כל האפשרויות לפתוחות.

מתי משתמשים: סקשן שמראה איך מכמה כיוונים מגיעים לתוכנית אחת: בחירת מסלול התחלה, חבילה, גרסה מומלצת או כיוון עיצובי. מתאים כשיש שלוש עד ארבע אפשרויות אמיתיות ואחת מהן היא ההמלצה שלכם. לא כשיש רק שתי אפשרויות, ולא כשהבחירה שייכת למבקר ולא לכם. פעם אחת בעמוד.

אחרי שיחת האפיון מציעים ללקוח ארבע דרכים להתחיל

כל דרך מוצגת עם מה שהיא כוללת, והלקוח בוחר את זו שמתאימה לעסק ולתקציב שלו.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם מצומת שהושלם יוצאים קווים דקים לצמתי מועמדים, ואחד מהם מתמלא כצעד הבא וקו שלו מתעבה. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: כל הקווים מצוירים, האפשרות הנבחרת מלאה בצבע ועליה התווית, והאחרות חלשות. בלי GSAP ובהפחתת תנועה זה מה שמוצג. כשהמהלך רץ הסקשן נעוץ (sticky בתוך אזור של 300vh) ו-paint(p) אחת כותבת את כל המצב מההתקדמות: קווים דרך stroke-dashoffset (כל נתיב עם pathLength=1), כניסת האפשרויות וההחלשות דרך משתני CSS (--a, --dim, --on), והצבע של כל אפשרות מחושב ב-color-mix מהמשתנים האלה, ולכן אין אינטרפולציית צבע ב-JS וגלילה אחורה נכונה בלי קוד נוסף. הבחירה עצמה: נתיב נוסף (.fk-hot, עובי 4, בצבע accent) מצוייר על גבי הקו הדק של האפשרות הנבחרת. כדי לבחור אפשרות אחרת מחליפים את fk-pick ואת האינדקס FK_PICK בנתיב, והשאר מחושב. פריסה: שורות קבועות בגובה (88 בדסקטופ, 80 בטלפון) ואזור קווים ברוחב קבוע בפיקסלים (120 ו-40) בין העמודות, ולכן הקו תמיד נוגע בצומת בכל רוחב. בטלפון הדיאגרמה אנכית: צומת המקור למעלה וגזע עם ענפים לארבע האפשרויות, בסט SVG משלו. ל-LTR: העמודות זורמות לפי כיוון העמוד, ו-:dir(ltr) הופך את שני ה-SVG ב-scaleX(-1) ואת כיוון הכניסה (--dx). אין צורך לשנות דבר, ורק מי שמצייר נתיבים חדשים צריך לזכור שהם מצוירים כשהמקור מימין.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP: כל הקווים מצוירים והאפשרות הנבחרת מלאה (במארקאפ)
  const root=document.querySelector(".fk");if(!root)return;
  gsap.registerPlugin(ScrollTrigger);
  const run=root.querySelector(".fk-run"),ckp=root.querySelector(".fk-ckp"),opts=[...root.querySelectorAll(".fk-o")],
    chip=root.querySelector(".fk-chip"),pick=opts.findIndex(o=>o.classList.contains("fk-pick"));
  const cl=v=>Math.min(1,Math.max(0,v)),seg=(p,a,b)=>cl((p-a)/(b-a)),E=gsap.parseEase("power2.inOut"),O=gsap.parseEase("power2.out");
  const put=(el,k,v)=>el.style.setProperty(k,v.toFixed(3));
  const mm=gsap.matchMedia();
  // sel: סט הקווים הגלוי (דסקטופ או טלפון). הסט השני נשאר במצב הסיום ומוסתר ב-CSS
  const build=sel=>{
    const svg=root.querySelector(sel),base=[...svg.querySelectorAll(".fk-b")],hot=svg.querySelector(".fk-hot");
    const paint=p=>{
      const c=seg(p,0,.1);
      ckp.style.strokeDashoffset=(1-O(c)).toFixed(3);ckp.style.opacity=c>0?1:0;
      const dim=E(seg(p,.64,.8)),on=E(seg(p,.7,.84));
      base.forEach((b,i)=>{
        b.style.strokeDashoffset=(1-E(seg(p,.1+i*.07,.3+i*.07))).toFixed(3);
        if(i!==pick)b.style.opacity=(1-.6*dim).toFixed(3);
      });
      opts.forEach((o,i)=>{
        put(o,"--a",O(seg(p,.22+i*.07,.36+i*.07)));
        if(i===pick){put(o,"--dim",0);put(o,"--on",on);}else{put(o,"--dim",dim);put(o,"--on",0);}
      });
      hot.style.strokeDashoffset=(1-E(seg(p,.58,.76))).toFixed(3);
      put(chip,"--ch",O(seg(p,.82,.94)));
    };
    root.classList.add("is-live");paint(0);
    const o={p:0};
    const tw=gsap.to(o,{p:1,ease:"none",onUpdate:()=>paint(o.p),
      scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.4,invalidateOnRefresh:true}});
    return()=>{
      tw.scrollTrigger.kill();tw.kill();root.classList.remove("is-live");
      [ckp,chip,hot,...opts,...base].forEach(el=>el.removeAttribute("style"));
    };
  };
  // בהפחתת תנועה שני התנאים לא מתקיימים, ונשאר מצב הסיום
  mm.add("(prefers-reduced-motion: no-preference) and (min-width: 768px) and (min-height: 600px)",()=>build(".fk-sd"));
  mm.add("(prefers-reduced-motion: no-preference) and (max-width: 767px) and (min-height: 600px)",()=>build(".fk-sm"));
})();
