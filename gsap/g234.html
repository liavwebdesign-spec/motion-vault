G234 · אותיות שנוחתות על קווי הנחיה

GSAP GSAP · ScrollTrigger once · פיצול לאותיות טקסט וכותרותתהליך ושלבים ממתין

ארבעה קווי הנחיה אופקיים נמתחים לרוחב הסקשן, מהצד שבו מתחילה הקריאה: קו עליון לאותיות הגבוהות, קו לגובה האות, קו בסיס וקו תחתון לאותיות שיורדות. אחר כך אותיות הכותרת יורדות מלמעלה אחת אחרי השנייה, מסתובבות מעט ונוחתות על קו הבסיס, והקו נדלק בצבע המותג כשהמילה שלמה. בלי GSAP ובהפחתת תנועה המילה מוצגת עם הקווים הדהויים.

מתי משתמשים: סקשן שמציג מלאכה שקשורה לאותיות: קליגרפיה, מיתוג, הוצאה לאור, עיצוב גרפי, בית דפוס. מילה אחת גדולה, פעם אחת בעמוד, וכדאי שיהיו בה אות גבוהה (כמו ל) ואות יורדת (כמו ק), כדי שהקווים יסבירו משהו.

קליגרפיהקליגרפיה

סדנאות קליגרפיה עברית בסטודיו קטן בצפון תל אביב, בקבוצות של שמונה משתתפים. כל מפגש נפתח בחימום של קווים ואותיות בסיסיות.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם קווי הנחיה טיפוגרפיים נמתחים קודם, ואז אותיות נוחתות עליהם ובונות את המילה. נבנה מאפס מהצפייה. המארקאפ: אזור (.gr-fig) בגודל גופן אחד, ובתוכו ה-h2 עם line-height:1, ארבעה קווים (.gr-ln) וארבע תוויות (.gr-lb). מיקום הקווים נמדד, לא מנוחש: top של כל קו הוא --y באורך em מראש התיבה. התוויות הן בגופן קטן משלהן, ולכן הן נמדדות מגודל הגופן של האזור (--fs) ולא מ-em שלהן. ל-Google Sans בעברית, בתיבה של שורה אחת: גובה אות גבוהה .135, גובה אות רגילה .28, קו בסיס .77, קו תחתון .945. הערכים נמדדו עם measureText ועם בדיקת baseline בדפדפן. בגופן אחר מודדים מחדש ומחליפים את ארבעת הערכים. הפיצול: gr מפרקת לאותיות וחוברת סימני ניקוד וטעמים לאות שלפניהם (ללא SplitText). האות היא inline-block עם מקור טרנספורם בקו הבסיס. הטקסט לקורא מסך הוא sr-only בתוך ה-h2, והאותיות בעטיפה aria-hidden. הרצף (ציר זמן אחד): הקווים נמתחים (‏scaleX מ-0 ל-1, ‏power3.inOut, ‏.8 שניות, ‏.12 בין קווים) מהצד שבו מתחילה הקריאה, התוויות מופיעות, האותיות יורדות (‏yPercent מ-150- עם סיבוב אקראי עד 18 מעלות, ‏power4.out, ‏.75 שניות, ‏.09 בין אותיות), ובסוף קו הבסיס מתחזק מ-35% ל-100% בצבע --accent. RTL: מקור הקווים נקרא מ-direction, והם נמתחים מימין. באתר LTR הקוד מתהפך לבד, והתוויות עוברות לצד השני (inset-inline-end). בטלפון התוויות מוסתרות (הקווים צפופים מדי בשבילן) והקווים נשארים, והמילה שומרת על יחס em.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP המילה מוצגת עם הקווים הדהויים
  const root=document.querySelector(".gr"),vis=root.querySelector(".gr-vis"),lines=[...root.querySelectorAll(".gr-ln")],labels=[...root.querySelectorAll(".gr-lb")],baseLine=root.querySelector(".gr-ln.is-base");
  const isMark=ch=>{const c=ch.charCodeAt(0);return c>=1425&&c<=1479;};      // ניקוד וטעמים נשארים צמודים לאות
  const gr=s=>{const o=[];for(const ch of s){if(o.length&&isMark(ch))o[o.length-1]+=ch;else o.push(ch);}return o;};
  // בהפחתת תנועה הפונקציה לא רצה, והמילה נשארת עם הקווים הדהויים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const orig=vis.textContent,rtl=getComputedStyle(root).direction==="rtl";
    vis.textContent="";
    const L=gr(orig).map(g=>{const c=document.createElement("span");c.className="gr-c";c.textContent=g;vis.appendChild(c);return c;});
    gsap.set(lines,{scaleX:0,transformOrigin:rtl?"100% 50%":"0% 50%"});   // הקווים נמתחים מהצד שבו מתחילה הקריאה
    gsap.set(labels,{opacity:0});
    gsap.set(baseLine,{opacity:.35});
    gsap.set(L,{yPercent:-150,rotation:()=>gsap.utils.random(-18,18),opacity:0});
    const tl=gsap.timeline({paused:true});
    tl.to(lines,{scaleX:1,duration:.8,ease:"power3.inOut",stagger:.12})
      .to(labels,{opacity:1,duration:.4,ease:"none",stagger:.08},.5)
      .to(L,{yPercent:0,rotation:0,opacity:1,duration:.75,ease:"power4.out",stagger:.09},1)   // האותיות נוחתות על קו הבסיס
      .to(baseLine,{opacity:1,duration:.5,ease:"power1.out"},">-.3");                         // והקו נדלק כשהמילה שלמה
    const st=ScrollTrigger.create({trigger:root.querySelector(".gr-fig"),start:"top 80%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();vis.textContent=orig;gsap.set(lines.concat(labels,[baseLine]),{clearProps:"all"});};
  });
})();
