G277 · קו דהוי שנדלק כניאון

GSAP GSAP · ScrollTrigger · SVG stroke · drop-shadow הירו ופתיחהרקע ואווירה ממתין

איור קווי של כוס קפה מצויר לאט על רקע כהה בצבע עמום, ואז נדלק כמו שלט ניאון: הבהוב קצר, עוד הבהוב, וזוהר שמתייצב. ליד האיור הטקסט נשאר קבוע וקריא לכל אורך התנועה.

מתי משתמשים: אייקון אחד שמסמל את העסק ושמתאים לערב ולחושך: בית קפה פתוח עד מאוחר, בר, מועדון, חנות לילה, מסעדה. פעם אחת בעמוד, בסקשן סיום או בהירו. לא מתאים לעסק שמדבר בשקט, כי הזוהר מושך תשומת לב.

בית קפה שכונתי

פתוחים עד חצות, כל ערב

אחרי תשע בערב המטבח עובר לתפריט קל של כריכים ומאפים, והקפה ממשיך לצאת מהמכונה עד הסגירה. אפשר לשבת עם ספר או עם חברים.

ימים א׳ עד ה׳: 07:00 עד 24:00, ובשישי עד 16:00

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם קו דק עמום, דמות או כותרת, נדלק בצבע זוהר על רקע כהה. נבנה מאפס מהצפייה. הציור: כל קו באייקון הוא path עם pathLength=1, ולכן stroke-dashoffset מ-1 ל-0 מצייר אותו בלי למדוד אורכים. ההדלקה היא משתנה אחד: --lit בין 0 ל-1 על האייקון. הצבע הוא color-mix בין צבע ניאון (--neon, תערובת של accent עם הרקע) לצבע כבוי (--dim) לפי --lit, והזוהר הוא שני drop-shadow שהרדיוס והשקיפות שלהם נגזרים מאותו משתנה. לכן שינוי ההדלקה הוא טווין אחד, והכל נשאר מחובר לטוקנים של העור. הבהוב: שני הבהובים בלבד אחרי שהציור נגמר, והפער בין שתי הדלקות עולה על 0.7 שנייה, כלומר פחות מפעמיים בשנייה, הרבה מתחת לגבול של שלוש. אל תוסיפו הבהובים: מעבר לגבול זה הוא מפריע לאנשים עם רגישות לאור. הערך הסופי: המארקאפ הוא האייקון הדלוק (--lit:1, הקו מצויר במלואו). בלי GSAP ובהפחתת תנועה זה מה שרואים. צבע: משטח הרקע הוא var(--ink) והטקסט var(--bg), כך שהוא מתהפך בעור כהה. הניאון הוא תערובת של 60% accent, ובעור שבו ה-accent כהה מאוד הוא עדיין נראה, אבל כדאי לבדוק אותו. נגישות: האיור aria-hidden, והכל כתוב בטקסט. אין כאן טקסט בצבע ניאון, כי ניאון על כהה לא תמיד מגיע ל-AA.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP האייקון דלוק ומצויר במלואו (במארקאפ)
  const root=document.querySelector(".nl"),ic=root.querySelector(".nl-ic"),paths=gsap.utils.toArray("path",ic);
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    gsap.set(ic,{"--lit":0});gsap.set(paths,{strokeDashoffset:1});
    // ציור בצבע עמום, ואז שני הבהובים בלבד. ההדלקות מרוחקות יותר מ-0.7 שנייה זו מזו: פחות משתי הבהובים בשנייה
    gsap.timeline({scrollTrigger:{trigger:root,start:"top 72%",once:true}})
      .to(paths,{strokeDashoffset:0,duration:1.1,ease:"power2.inOut",stagger:.16})
      .to(ic,{"--lit":.85,duration:.1,ease:"none"},"+=.25")
      .to(ic,{"--lit":.18,duration:.06,ease:"none"},"+=.2")
      .to(ic,{"--lit":1,duration:.1,ease:"none"},"+=.34")
      .to(ic,{"--lit":.35,duration:.06,ease:"none"},"+=.3")
      .to(ic,{"--lit":1,duration:.6,ease:"power2.out"},"+=.4");
    return()=>{gsap.set(ic,{clearProps:"--lit"});gsap.set(paths,{clearProps:"strokeDashoffset"});};
  });
})();
