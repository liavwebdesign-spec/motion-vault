G22 · עיגול תהליך נצבע בגלילה

GSAP GSAP · DrawSVG · pin תהליך ושלביםמספרים ונתונים ממתין

קשת צבעונית נצבעת סביב עיגול גדול בקצב הגלילה, כל תחנה נדלקת בתורה והתוכן במרכז מתחלף.

מתי משתמשים: תהליך של 3-5 שלבים. האלטרנטיבה המעגלית לטיימליין האופקי.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
התחנה היא נקודה על המעגל, לא קופסה: .pdot ממורכז על הנקודה ב-translate(-50%,-50%), ולכן תווית לתחנה יושבת כאח שלו מחוץ לקופסה הזו, אחרת היא מזיזה את הנקודה מהקו (פיצ׳, 24.9.2026: התחנות צפו מחוץ למעגל). בשלוש תחנות או בכל מספר אחר המיקום מחושב ולא מנוחש: בזווית θ מלמעלה, top = 50 פחות 48.33·cosθ אחוז, left = 50 ועוד 48.33·sinθ אחוז (הרדיוס 290 מתוך 600). אחרי כל שינוי במספר התחנות מודדים שמרכז כל נקודה על הקו. בהפחתת תנועה, וכש-GSAP לא נטען, אין נעילה ואין עיגול: השלבים מוצגים כרשימה, כי ה-CSS מסתיר את שלבים 2 ואילך רק תחת .gsap-live.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger, DrawSVGPlugin);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP השלבים מוצגים כרשימה
  const pw=document.querySelector(".pw");
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    pw.classList.add("gsap-live");
    gsap.set(".pcirc .draw",{drawSVG:"0% 0%"});
    const ptl=gsap.timeline({scrollTrigger:{trigger:pw,start:"top top",end:"+=2000",scrub:true,pin:true,anticipatePin:1}});
    const STEPS=pw.querySelectorAll(".pstep").length;   // נגזר מה-HTML: שלב ונקודה נוספים, והקוד מתאים את עצמו
    for(let i=1;i<STEPS;i++){
      // הטקסט הקודם יוצא רגע לפני שהקשת מגיעה לתחנה, והבא נכנס יחד עם הדלקת התחנה:
      // שני טקסטים לא נכתבים זה על זה
      ptl.to(".pcirc .draw",{drawSVG:"0% "+(i*100/STEPS)+"%",duration:2})
         .to(".p"+i,{opacity:0,duration:.06},"-=0.06")
         .to(".d"+(i+1),{filter:"grayscale(0) brightness(1)",duration:.1},">")
         .to(".p"+(i+1),{opacity:1,duration:.06},"<");
    }
    ptl.to(".pcirc .draw",{drawSVG:"0% 100%",duration:2});
    return()=>pw.classList.remove("gsap-live");
  });
})();
