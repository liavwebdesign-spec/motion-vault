G218 · טקסט שמוצג כפסים בצבעים

GSAP GSAP · ScrollTrigger once · stagger טקסט וכותרותמספרים ונתונים ממתין

פסקה שלמה מוצגת כשורות של פסים מעוגלים, פס לכל מילה לפי אורכה. בכניסה למסך כל הפסים אפורים, ואז הם מקבלים את צבעם מילה אחרי מילה: מילים מוכרות בכהה, מילים ארוכות ונדירות באפור בהיר, ומונחים מקצועיים בצבע המותג. ליד הפסקה מספר שסופר את המונחים.

מתי משתמשים: ביקורת קריאות, בדיקת טקסט או הסבר איך קוראים אתר: מונחים שהלקוחות לא מכירים, מילים שדורשות הסבר, חלק מהטקסט שנקרא בפועל. סקשן הוכחה או שירות שבודק משהו בטקסט. מתאים כשצריך להראות נתון על טקסט בלי להכריח לקרוא אותו.

בדיקת קריאות של עמוד שירות

אנחנו מציעים פתרון אינטגרטיבי לניהול תהליכי עבודה, שמשלב אוטומציה של משימות חוזרות עם ממשק משתמש נוח. המערכת מתחברת למערכות הקיימות שלכם דרך API, ומאפשרת מעקב אחרי לידים ברמת הקמפיין. כך אפשר לראות כמה פניות הגיעו, מאיפה הן הגיעו ומה קרה איתן אחרי השיחה הראשונה. ההטמעה לוקחת שבועיים, כוללת הדרכה לצוות, ואחריה נשארים זמינים לכל שאלה. בטקסט הזה יש 7 מונחים מקצועיים שכדאי להסביר, מתוך 55 מילים.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שני סרטונים בגלריה (21.10.2026) שבהם פסקה מוצגת כפסים מעוגלים, מילים מוכרות בכהה ולא מוכרות בבהיר, וכמה קטעים מודגשים בצבע. נבנה מאפס מהצפייה. הפסים הם מילים: כל מילה היא span שרוחבו --n (מספר האותיות) כפול 9 פיקסלים, בתוך flex שעוטף שורות. בעברית הקריאה מימין לשמאל, ולכן הצביעה מתחילה בפינה הימנית העליונה ויורדת שורה אחר שורה. את הפסים אפשר לייצר מכל טקסט: מפצלים לפי רווחים, אורך המילה הוא --n, והמונחים המקצועיים מסומנים ברשימה. המחלקות: k מוכרת, u ארוכה ונדירה, h מונח מקצועי, והצבע הסופי הוא החוק ב-CSS. הערך הסופי במארקאפ: בלי GSAP ובהפחתת תנועה כל הפסים בצבעם הסופי והמספר הסופי. כשהמהלך רץ הסקריפט מוסיף is-run ומחזיר את כל הפסים לאפור, ואז מוסיף on מילה אחרי מילה (שלוש מאיות שנייה בין מילים). המעבר הוא transition של צבע ב-CSS ולא טווין של GSAP, כי GSAP לא קורא צבעים שנבנו ב-color-mix. המספר עולה בדיוק כשמונח מקצועי נצבע. נגישות: הפסים מוסתרים מקורא מסך, והפסקה המלאה והמספר הסופי כתובים בטקסט מוסתר, כך שהמשמעות לא תלויה בצבע. בטלפון הפסים נשברים לשורות צרות יותר והסיכום יורד מתחת ללוח.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP כל הפסים בצבעם הסופי והמספר הסופי (במארקאפ)
  const root=document.querySelector(".tb"),words=[...root.querySelectorAll(".tb-w")],cnt=root.querySelector(".tb-cnt");
  const total=parseInt(cnt.textContent,10);
  // בהפחתת תנועה הפונקציה לא רצה, והצבעים הסופיים נשארים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    let n=0;cnt.textContent=0;root.classList.add("is-run");      // כל הפסים אפורים עד שהקטע נכנס למסך
    const tl=gsap.timeline({paused:true});
    words.forEach((w,i)=>tl.call(()=>{w.classList.add("on");if(w.classList.contains("h")){n++;cnt.textContent=n;}},null,i*.03));
    const st=ScrollTrigger.create({trigger:root,start:"top 70%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();root.classList.remove("is-run");words.forEach(w=>w.classList.remove("on"));cnt.textContent=total;};
  });
})();
