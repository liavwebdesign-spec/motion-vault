G225 · מונה לאחור על מסך פתיחה קצר

GSAP GSAP timeline · SVG stroke · ScrollTrigger once הירו ופתיחהמספרים ונתונים ממתין

מסך פתיחה כהה ובמרכזו מספר ענק עם צל קשיח בצבע המותג, בתוך טבעת דקה עם סימני שעון. הוא סופר שלוש, שתיים, אחת, כשהטבעת נצבעת עם הזמן, ואז המסך כולו מתרומם ומגלה את האתר, שכל חלקיו נכנסים מיד אחריו. כל זה במהירות של שנייה וחצי.

מתי משתמשים: אתר חוויה, סטודיו, השקה או אירוע שמתחילים בדרמה קצרה לפני הדף. רק כשהדף באמת נטען לאט מהמסך, או פעם אחת בכל ביקור. באתר שבו המבקר מחפש תשובה מהר, אל תשתמשו בו.

לבדיקת מועדים
3
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שלושה סרטונים ב-prompt-motion.com (21.10.2026) שנפתחים במספר ענק עם צל קשיח בתוך עיגול דק לפני שהסצנה מתחילה. נבנה מאפס מהצפייה. לא חוסם: כל ההפעלה היא 3 x 0.42 שניות עם 0.6 שניות הרמה, ולפחות 40 אחוז מהזמן הזה כבר חושף את התוכן. המסך עם pointer-events:none תמיד, כך שגם בלחיצה מוקדמת לא נתקעים. בפרויקט: מפעילים בטעינה (play() ישיר) ובתנאי של ביקור ראשון (sessionStorage), לא מחכים ל-ScrollTrigger. כאן ה-ScrollTrigger נמצא רק כדי שהדמו ירוץ כשרואים אותו. בלי GSAP ובהפחתת תנועה המסך לא מופיע בכלל והאתר מוצג מיד (התנאי הוא display:none כברירת מחדל, והמחלקה מופעלת רק בסקריפט). הצל הקשיח הוא text-shadow ללא טשטוש, והטבעת נצבעת ב-stroke-dashoffset על נתיב עם pathLength=1. כפתור "הפעלה חוזרת" שייך לדמו בלבד: הוא כפתור אמיתי, עובד במקלדת, ואפשר להסיר אותו בפרויקט. נגישות: המסך כולו aria-hidden, והתוכן שמאחוריו תמיד קיים ב-DOM ונגיש גם בזמן הספירה.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP אין מסך פתיחה
  const root=document.querySelector(".cd"),site=root.querySelector(".cd-site"),over=root.querySelector(".cd-over"),num=root.querySelector(".cd-num"),
    arc=root.querySelector(".cd-arc"),btn=root.querySelector(".cd-replay"),parts=root.querySelectorAll(".cd-copy>*,.cd-photo");
  // בהפחתת תנועה הפונקציה לא רצה, ולכן המסך לא מופיע והאתר מוצג מיד
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const STEP=.42;let tl=null;
    const arm=()=>{root.classList.add("is-armed");num.textContent="3";gsap.set(over,{yPercent:0});gsap.set(arc,{strokeDashoffset:1});gsap.set(parts,{opacity:0});btn.hidden=true;};
    // בפרויקט קוראים ל-play() בטעינה, ובביקור ראשון בלבד
    const play=()=>{
      if(tl)tl.kill();
      tl=gsap.timeline({onComplete:()=>{root.classList.remove("is-armed");btn.hidden=false;}});
      [3,2,1].forEach((v,i)=>{
        tl.call(()=>{num.textContent=v;},null,i*STEP)
          .fromTo(num,{scale:1.16,opacity:0},{scale:1,opacity:1,duration:.2,ease:"power3.out"},i*STEP);
      });
      tl.to(arc,{strokeDashoffset:0,duration:STEP*3,ease:"none"},0)
        .to(over,{yPercent:-100,duration:.6,ease:"power3.inOut"},STEP*3+.04)
        .to(parts,{opacity:1,y:0,duration:.5,ease:"power2.out",stagger:.08},STEP*3+.2);
      gsap.set(parts,{y:24});
    };
    arm();
    const st=ScrollTrigger.create({trigger:site,start:"top 70%",once:true,onEnter:play});
    btn.addEventListener("click",()=>{arm();play();});
    return()=>{st.kill();if(tl)tl.kill();root.classList.remove("is-armed");btn.hidden=true;gsap.set([over,num,arc,parts],{clearProps:"all"});};
  });
})();
