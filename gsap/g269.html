G269 · פריטים צפים בעומק סביב כותרת מרכזית

GSAP GSAP · ScrollTrigger scrub · quickTo הירו ופתיחהרקע ואווירהתמונות וגלריות ממתין

שבבי טקסט וגזרי תמונה מרחפים סביב כותרת במרכז, בשלוש שכבות עומק. הרחוקים קטנים ומטושטשים, הקרובים גדולים ומטושטשים אף הם, והאמצעיים חדים. כל פריט מתנדנד לאט במקומו, בגלילה כל שכבה זזה בכמות אחרת, ועם העכבר השכבות הקרובות זזות יותר. בכניסה הפריטים מתקבצים מסביב לכותרת.

מתי משתמשים: הירו או סקשן סיום שיש בו כותרת אחת חזקה ועולם של דברים סביבה: מוצרים, שירותים, מנות, כלים, אנשים. עוזר לעסק שהמוצר שלו רב-פנים (מאפייה, חנות, סטודיו) להראות מגוון בלי רשת. לא מתחת לטקסט ארוך ולא ליד טופס.

להזמנת מאפים
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם אריחים, גזרים וצורות צפים בעומק סביב כותרת או מונה במרכז. נבנה מאפס מהצפייה. קרוב ל-g37 (ענן תמונות נעוץ), ושונה בכוונה: כאן אין הצמדה ואין מסלול גלילה, הכותרת היא הנושא והפריטים הם אטמוספרה סביבה, כולל שבבי טקסט ולא רק תמונות. המארקאפ הוא המצב הסופי: כל הפריטים במנוחה במקומם, וזה מה שרואים בלי GSAP ובהפחתת תנועה. מבנה: שלוש שכבות (.fl-far, .fl-mid, .fl-near), ובכל פריט שתי עטיפות: .fl-it נושא מיקום (--x, --y), סיבוב (--r) וטשטוש (--b), ו-.fl-bob בפנים נושא רק את הנדנוד. כך שום אנימציה לא דורסת אחרת: הסיבוב נעשה במאפיין rotate של CSS ולכן לא מתנגש ב-transform של GSAP. ארבע תנועות: (1) כניסה בקצב הגלילה (scrub, שקיפות וגודל, סדר אקראי); (2) הסטה אנכית של כל שכבה בגלילה, בקצב 3%, 7% ו-12%; (3) הסטה בעכבר עם quickTo לכל שכבה, 8, 20 ו-38 פיקסלים, רק במכשיר עם עכבר; (4) נדנוד sine עם yoyo לכל פריט, בטווח שגדל עם העומק, ונעצר כשהסקשן מחוץ למסך. טשטוש: filter:blur() על .fl-it הקבוע, לא על השכבה המונפשת, כדי לא לצייר מחדש שכבה גדולה בכל פריים. מובייל: כל פריט נושא מיקום נפרד לטלפון (--mx, --my) שמציב אותו ברצועה העליונה או התחתונה, מחוץ לטקסט. פריטים עם fl-m0 מוסתרים בטלפון, והתמונות מוקטנות ל-52% (--k). כיווניות: מיקום הפריטים פיזי (left) והקומפוזיציה סימטרית, אז באתר LTR אין מה להפוך. תמונות: מהמאגר (s14), עם width ו-height מפורשים. נגישות: כל השכבות aria-hidden והכותרת היא הטקסט היחיד שנקרא. בפרויקט מחליפים את התמונות והשבבים בתכנים של העסק, ושומרים על שלושה עד שישה פריטים בכל שכבה.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP כל הפריטים נשארים במנוחה, כמו במארקאפ
  const root=document.querySelector(".fl"),layers=[...root.querySelectorAll(".fl-layer")];
  const SC=[3,7,12],PT=[8,20,38],AM=[6,10,16];   // לכל שכבה (רחוקה, אמצעית, קרובה): הסטת גלילה באחוזים, הסטת עכבר בפיקסלים, טווח נדנוד
  // בהפחתת תנועה הפונקציה לא רצה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const items=[...root.querySelectorAll(".fl-it")],bobs=[];
    const rnd=n=>{const s=Math.sin(n*127.1+311.7)*43758.5453;return s-Math.floor(s);};   // אקראי קבוע: אותה קומפוזיציה בכל טעינה
    // כניסה: הפריטים מתקבצים מסביב לכותרת בקצב הגלילה
    gsap.fromTo(items,{autoAlpha:0,scale:.7},{autoAlpha:1,scale:1,ease:"power3.out",stagger:{each:.05,from:"random"},
      scrollTrigger:{trigger:root,start:"top 88%",end:"top 38%",scrub:.6}});
    layers.forEach((L,i)=>{
      // הסטה אנכית לפי עומק: ככל שהשכבה קרובה יותר היא זזה יותר
      gsap.fromTo(L,{yPercent:-SC[i]},{yPercent:SC[i],ease:"none",scrollTrigger:{trigger:root,start:"top bottom",end:"bottom top",scrub:true}});
      L.querySelectorAll(".fl-bob").forEach((b,j)=>{
        const s=i*17+j,t=gsap.to(b,{x:(rnd(s)-.5)*2*AM[i],y:(rnd(s+3)>.5?1:-1)*(AM[i]*.6+rnd(s+5)*AM[i]*.6),rotation:(rnd(s+7)-.5)*5,
          duration:3.2+rnd(s+9)*2.6,ease:"sine.inOut",yoyo:true,repeat:-1});
        t.progress(rnd(s+11));bobs.push(t);
      });
    });
    // נדנוד רק כשהסקשן על המסך
    ScrollTrigger.create({trigger:root,start:"top bottom",end:"bottom top",onToggle:s=>bobs.forEach(t=>s.isActive?t.resume():t.pause())});
    // עכבר: כל שכבה זזה נגד כיוון הסמן, והקרובה זזה הכי הרבה. במגע אין הסטה
    const fine=matchMedia("(hover:hover) and (pointer:fine)").matches;
    const qx=layers.map(L=>gsap.quickTo(L,"x",{duration:.9,ease:"power3.out"})),qy=layers.map(L=>gsap.quickTo(L,"y",{duration:.9,ease:"power3.out"}));
    const move=e=>{const r=root.getBoundingClientRect(),nx=(e.clientX-r.left)/r.width-.5,ny=(e.clientY-r.top)/r.height-.5;
      layers.forEach((L,i)=>{qx[i](-nx*PT[i]*2);qy[i](-ny*PT[i]*2);});};
    const leave=()=>layers.forEach((L,i)=>{qx[i](0);qy[i](0);});
    if(fine){root.addEventListener("pointermove",move);root.addEventListener("pointerleave",leave);}
    return()=>{root.removeEventListener("pointermove",move);root.removeEventListener("pointerleave",leave);
      gsap.set([...layers,...items,...root.querySelectorAll(".fl-bob")],{clearProps:"all"});};
  });
})();
