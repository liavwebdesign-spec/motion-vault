G193 · שני מונים במירוץ בזמנים שונים

GSAP GSAP · ScrollTrigger · counters מספרים ונתוניםכרטיסים ממתין

שתי שעוני עצר זו לצד זו, והמהירה מסתיימת אחרי שנייה אחת עם תג סיום ועלות קטנה, בזמן שהאיטית ממשיכה לספור והעלות שלה ממשיכה לגדול. ההבדל בין ארבע דקות לעשרים ואחת דקות נראה בעין לפני שמישהו קרא מילה.

מתי משתמשים: השוואה בין שתי דרכים לעשות אותו דבר, אנחנו מול הדרך הישנה: תבנית מול מאפס, אוטומציה מול ידני. סקשן אחד בעמוד שירות או מחיר. מתאים כשיש שני זמנים ושני מחירים ברורים. לא מתאים להשוואה של יותר משתי אפשרויות.

עם תבנית מוכנה ההצעה מוכנה תוך ארבע דקות ושתים עשרה שניות, בעלות של שלושה עשר שקלים. מאפס זה לוקח עשרים ואחת דקות ושלושים שניות, בעלות של שישים וחמישה שקלים.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שני סרטונים ב-prompt-motion.com עם שתי כרטיסיות שעון עצר שמסיימות בזמנים שונים מאוד (21.10.2026). נבנה מאפס מהצפייה. העיקרון: מונה אחד R.t רץ ליניארית מ-0 עד 5.63 שניות אמיתיות, ושתי השעונים נגזרים ממנו באותו מקדם K, כך שהמהירה מגיעה לסוף (04:12) ב-1.1 שניות והאיטית (21:30) ב-5.63. העלות היא סך השניות חלקי 60 כפול התעריף ומתעדכנת בכל פריים. הסרגל הוא scaleX עם נקודת מוצא בצד ההתחלה (ימין בעברית), והוא מתמלא עד אותו אחוז של הזמן. התג: התג של הכרטיס המהיר קופץ ב-expo.out ברגע שהשעון נעצר, והסטטוס של האיטית מתחלף מ"עדיין עובדים" ל"הושלם" רק בסוף. ההפעלה: נכנס ל-70% מגובה המסך, play, ויציאה אחורה מאפסת כדי שהמירוץ ירוץ שוב. המצב הסטטי: שני הערכים הסופיים והתג, כלומר בלי GSAP ובהפחתת תנועה רואים את התוצאה. נגישות: הכרטיסים aria-hidden ובמקומם משפט בטקסט מוסתר עם הערכים הסופיים, כדי שקורא מסך לא יקרא שעון שרץ. השעונים ב-dir="ltr" ו-tabular-nums כדי שהספרות לא יזוזו. באתר משמאל לימין מחליפים את נקודת המוצא של הסרגל ל-left.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP נשארים הערכים הסופיים
  const root=document.querySelector(".rc");
  const tm=[...root.querySelectorAll("[data-t]")],cs=[...root.querySelectorAll("[data-c]")],bars=[...root.querySelectorAll(".rc-bar i")];
  const st=root.querySelector("[data-s]"),badge=root.querySelector(".rc-bd");
  const orig={t:tm.map(e=>e.textContent),c:cs.map(e=>e.textContent),p:bars.map(b=>b.style.getPropertyValue("--p")),s:st.textContent};
  const FAST=252,SLOW=1290,END=5.63,K=SLOW/END;      // שניות מוצגות לכל שנייה אמיתית: המהירה נגמרת ב-1.1 והאיטית ב-5.63
  const two=n=>(n<10?"0":"")+n;
  const mmss=n=>two(Math.floor(n/60))+":"+two(Math.floor(n%60));
  // בהפחתת תנועה הפונקציה לא רצה, והערכים הסופיים נשארים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const R={t:0};
    const draw=()=>{
      const v=[Math.min(FAST,R.t*K),Math.min(SLOW,R.t*K)];
      v.forEach((x,i)=>{
        tm[i].textContent=mmss(x);
        cs[i].textContent="₪"+Math.round(x/60*3);
        bars[i].style.setProperty("--p",(x/SLOW).toFixed(4));
      });
      st.textContent=v[1]>=SLOW?orig.s:"עדיין עובדים";
    };
    const tl=gsap.timeline({scrollTrigger:{trigger:root,start:"top 70%",toggleActions:"play none none reset",onLeaveBack:draw}});
    tl.fromTo(R,{t:0},{t:END,duration:END,ease:"none",onUpdate:draw},0)
      .fromTo(badge,{opacity:0,scale:.6},{opacity:1,scale:1,duration:.5,ease:"expo.out"},FAST/K);      // התג קופץ כשהשעון המהיר נעצר
    draw();
    return()=>{tm.forEach((e,i)=>{e.textContent=orig.t[i];});cs.forEach((e,i)=>{e.textContent=orig.c[i];});bars.forEach((b,i)=>b.style.setProperty("--p",orig.p[i]));st.textContent=orig.s;gsap.set(badge,{clearProps:"all"});};
  });
})();
