G206 · כותרת מפוצלת בצבעים עם הבזק היפוך

GSAP GSAP · ScrollTrigger · oklch · עותקים מוסתרים טקסט וכותרותהירו ופתיחה ממתין

הכותרת נוחתת במכה אחת: שני עותקים בצבעי המותג נפרדים ממנה לשני הצדדים ורועדים כמה פיקסלים, ובאותו רגע הכרטיס כולו מהבהב להיפוך בהיר פעם אחת ונרגע. הכול נגמר בפחות מחצי שנייה ולא חוזר. בהובר אפשר להפעיל שוב את הרעידה, בלי ההבזק.

מתי משתמשים: משפט בודד שצריך להרגיש כמו מכה בנקודה אחת בעמוד: מוצר טכנולוגי, מערכת, קורס, שירות מהיר. כותרת קצרה אחת בכל עמוד. לא לשירותי בריאות, כספים או משפט, ולא לעסק שהלקוח רוצה בו אמון שקט.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני כותרות בגלריה של prompt-motion.com (21.10.2026) עם פיצול צבע אדום וציאן והבזק היפוך. נבנה מאפס מהצפייה. המארקאפ: כותרת אחת, ובתוכה שני עותקים (.gl-r, .gl-c) עם aria-hidden ושקיפות אפס. בלי GSAP ובהפחתת תנועה זו כותרת רגילה. הצבעים נגזרים מהעור: עותק אחד ב---accent, והשני במשלים שלו: oklch(from var(--accent) l c calc(h + 180)). דפדפן בלי צבע יחסי מקבל תערובת של accent ו-ink. אין אדום וציאן קבועים, ולכן זה עובד בכל עור. כללי הבזק (WCAG 2.3.1): פגיעה אחת היא חמישה היסטים קצרים בגודל יורד (7 מאיות שנייה כל אחד), הבזק היפוך אחד של הכרטיס בלבד ולא של המסך (filter: invert(1) hue-rotate(180deg) ל-90 מילישניות), וחזרה למנוחה. נעילה: lock מונע פגיעה נוספת במשך 2.5 שניות, ולכן גם גלילה מהירה למעלה ולמטה או הובר חוזר לא מייצרים יותר מהבזק אחד בכל 2.5 שניות. הובר מפעיל רק את הרעידה. הפעלה: כשהכרטיס נכנס למסך, ובכניסה חוזרת מלמטה. באתר משמאל לימין אין מה להפוך: ההיסטים סימטריים. נגישות: הטקסט היחיד שקורא מסך שומע הוא הכותרת עצמה. בהפחתת תנועה אין שום שכבה ואין הבזק.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP זו כותרת רגילה, בלי שכבות הצבע
  const root=document.querySelector(".gl"),box=root.querySelector(".gl-box"),title=root.querySelector(".gl-t"),
    R=root.querySelector(".gl-r"),C=root.querySelector(".gl-c");
  // בהפחתת תנועה הפונקציה לא רצה: אין היסטים ואין הבזק
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    let lock=0,tl=null;
    const rnd=(a,b)=>a+Math.random()*(b-a);
    // פגיעה אחת: חמישה היסטים אקראיים וקצרים, הבזק היפוך אחד של הכרטיס (בכניסה בלבד), וחזרה למנוחה
    const hit=flash=>{
      const now=performance.now();
      if(now<lock)return;lock=now+2500;      // לפחות 2.5 שניות בין פגיעות: אין הבזקים חוזרים
      if(tl)tl.kill();
      tl=gsap.timeline({onComplete:()=>gsap.set([R,C,box],{clearProps:"opacity,transform,filter"})});
      tl.set(title,{opacity:1},0);
      for(let i=0;i<5;i++){
        const k=1-i/6;
        tl.set(R,{opacity:.9,x:-rnd(8,20)*k,y:rnd(-4,4)*k},i*.07).set(C,{opacity:.9,x:rnd(8,20)*k,y:rnd(-4,4)*k},i*.07);
      }
      if(flash)tl.set(box,{filter:"invert(1) hue-rotate(180deg)"},.07).set(box,{clearProps:"filter"},.16);
      tl.to([R,C],{opacity:0,x:0,y:0,duration:.2,ease:"power2.out"},.4);
    };
    const hover=()=>hit(false);
    gsap.set(title,{opacity:0});
    box.addEventListener("pointerenter",hover);
    const st=ScrollTrigger.create({trigger:box,start:"top 82%",onEnter:()=>hit(true),onEnterBack:()=>hit(true)});
    return()=>{st.kill();box.removeEventListener("pointerenter",hover);if(tl)tl.kill();gsap.set([title,R,C,box],{clearProps:"all"});};
  });
})();
