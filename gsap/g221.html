G221 · מעבר מהיר בין מסכי מוצר עם טשטוש תנועה

GSAP GSAP · ScrollTrigger scrub · SVG feGaussianBlur תהליך ושלביםכרטיסים ממתין

חלון דפדפן אחד נעוץ במסך, ובתוכו שלושה מסכי מערכת בשורה. בגלילה המצלמה לא גוררת אלא מזנקת: המסך הנוכחי נזרק הצידה במהירות עם טשטוש תנועה אופקי, והבא נוחת במקומו חד ויציב. הטשטוש תלוי במהירות בלבד, אז כשעוצרים באמצע הוא נעלם.

מתי משתמשים: הסבר של זרימה אמיתית במערכת או באפליקציה: הזמנה, תשלום, ניהול. שלושה מסכים ברורים שלכל אחד תפקיד. מתאים להראות שהמערכת פשוטה ושהשלבים באים בזה אחר זה. לא לשימוש כשכל מסך מלא טקסט קטן שצריך לקרוא בזמן המעבר.

ההזמנה אושרה

קרוואן קטן, ארבעה לילותאיסוף ביום שלישי בעשר בבוקרפרטי האישור נשלחו למייל
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שני סרטוני מוצר ב-prompt-motion.com (21.10.2026) שבהם התצוגה עוברת בין מסכי ממשק בתנועת מצלמה מהירה עם טשטוש אופקי. נבנה מאפס מהצפייה. המסילה: שלושה מסכים בשורה אחת ברוחב 300%, והיא זזה ב-power-חזק של expo.inOut בין עצירות ארוכות, כך שרוב הגלילה היא עצירה על מסך והזינוק קצר. הטשטוש הוא מסנן SVG ולא blur(): feGaussianBlur עם stdDeviation="x 0" מטשטש רק לרוחב, כמו תנועת מצלמה. הוא מופעל על אזור החלון בלבד (לא על המסילה כולה, שרחבה פי שלושה), ורק כשיש מהירות. המהירות נמדדת מההפרש בין שני פריימים ומוחלקת ב-gsap.ticker, ולכן הטשטוש דועך לבד אחרי שהגלילה נעצרת, ולא נשאר תקוע באמצע. תקרה של 14 פיקסלים כדי שיישאר עדין. כיוון: המסך הראשון מימין והבא משמאל לו, כמו סדר הקריאה, והמסילה זזה ימינה (dir נקרא מה-direction של העמוד). באתר משמאל לימין זה מתהפך אוטומטית. בלי GSAP ובהפחתת תנועה שלושת המסכים זה מתחת לזה, בלי טשטוש. לשימוש בלחיצה במקום בגלילה: מחליפים את הטיימליין ב-gsap.to(o,{i:n,ease:"expo.inOut",duration:.5,onUpdate:place}). נגישות: החלון הוא איור (aria-hidden), והמשמעות בכותרת ובפסקה שמעליו.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP שלושת המסכים זה מתחת לזה
  const root=document.querySelector(".wp"),view=root.querySelector(".wp-view"),track=root.querySelector(".wp-track"),
    tabs=[...root.querySelectorAll(".wp-tabs span")],gb=document.getElementById("wp-gb"),n=tabs.length;
  // בהפחתת תנועה הפונקציה לא רצה, והמסכים נשארים זה מתחת לזה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const dir=getComputedStyle(root).direction==="rtl"?1:-1;      // בעברית המסך הבא משמאל, והמסילה זזה ימינה
    const o={i:0};let w=view.clientWidth,last=0,vel=0,on=-1;
    const place=()=>{
      gsap.set(track,{x:o.i*w*dir});
      const k=Math.round(o.i);
      if(k!==on){on=k;tabs.forEach((t,j)=>t.classList.toggle("on",j===k));}
    };
    const blur=s=>{gb.setAttribute("stdDeviation",s.toFixed(1)+" 0");view.style.filter=s>.4?"url(#wp-blur)":"";};
    // המהירות נמדדת מהפרש הפריימים, ולכן הטשטוש דועך לבד כשהגלילה נעצרת
    const tick=()=>{
      const dx=Math.abs(o.i-last)*w;last=o.i;vel+=(dx-vel)*.3;
      if(vel>.5||view.style.filter)blur(Math.min(vel*.45,14));
    };
    gsap.ticker.add(tick);
    const size=()=>{w=view.clientWidth;place();};
    ScrollTrigger.addEventListener("refresh",size);size();
    const tl=gsap.timeline({scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.5}});
    tl.to({},{duration:1});
    for(let j=1;j<n;j++)tl.to(o,{i:j,duration:.5,ease:"expo.inOut",onUpdate:place}).to({},{duration:1});
    return()=>{gsap.ticker.remove(tick);ScrollTrigger.removeEventListener("refresh",size);root.classList.remove("is-live");
      view.style.filter="";gsap.set(track,{clearProps:"transform"});tabs.forEach((t,j)=>t.classList.toggle("on",j===0));};
  });
})();
