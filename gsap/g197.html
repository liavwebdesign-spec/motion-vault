בהתחלה שני כרטיסי פנייה מונחים נטויים על הבמה, והמספר שלידם הוא 2. בגלילה כרטיסים חדשים נכנסים אחד אחרי השני, כל אחד עם תג קטן שנדלק ונכבה, והמספר סופר איתם עד 10. אחר כך הערימה המפוזרת מתהדקת למניפה צפופה, כל כרטיס מוזז מעט מקודמו, והחדש ביותר יושב למעלה.

מתי משתמשים: סקשן שמראה נפח של עבודה: פניות שנכנסות לאתר, הזמנות, לקוחות חדשים, קריאות שירות או מועמדים למשרה. מתאים כשיש ספירה אמיתית ואותו סוג פריט שחוזר, ורוצים שהמספר ירגיש כמו ערימה ולא כמו ספרה. פעם אחת בעמוד, ליד כותרת שאומרת מה סופרים. לא כשיש פריט אחד בלבד או כשכל הכרטיסים שונים זה מזה.

שחזור. השראה: כמה סרטונים בגלריה (21.10.2026) שבהם כרטיסים קטנים ומוטים נערמים בצביר ומונה סופר אותם. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: עשרה כרטיסים במניפה צפופה והמספר 10. המיקום של כל כרטיס במניפה כתוב עליו ב---fx, --fy, --fr (והסדר ב-z-index), ובלי GSAP או בהפחתת תנועה זה מה שרואים. כשהמהלך רץ הסקריפט מוסיף is-live (במה נעוצה בגובה 340vh) ובונה timeline אחד עם scrub. לכל כרטיס יש מיקום פיזור (טבלת LOOSE, לפי גיל הכרטיס) ומיקום מניפה, והמניפה נקראת מה-CSS ולא נכתבת פעמיים. שני הכרטיסים הוותיקים נמצאים מההתחלה, ושמונת האחרים קופצים בזה אחר זה ב-power3.out (בלי ספרינג, כי back נחסם), ובכל כניסה נדלק תג חדש קטן. המספר לא רץ בנפרד: הוא נגזר מזמן ה-timeline וסופר רק כרטיסים שכבר נכנסו, ולכן תמיד מסכים עם הערימה. ההידוק: רק אחרי שהאחרון נכנס כל הכרטיסים נעים למניפה, עם הפרש קטן בין כרטיס לכרטיס. ציפה: כל כרטיס מרחף מעט בעצמו (animation על הפנים ולא על המעטפת שמוזזת), ורק כשהמהלך חי. מובייל: המרחק האופקי של הפיזור מתכווץ לפי רוחב הבמה (kx), כך שהכרטיסים לא יוצאים מהמסך. כיווניות: הכרטיס החדש ביותר יושב משמאל ולמעלה, כלומר הזמן זורם מימין לשמאל. באתר LTR הופכים את הסימן של --fx ושל x בטבלת הפיזור. נגישות: האוואטרים והתג מוסתרים מקורא מסך, המספר הרץ מוסתר, ומשפט מוסתר אומר את הסכום הסופי.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP המניפה הצפופה והמספר הסופי כבר במארקאפ
  const root=document.querySelector(".ccl"),field=root.querySelector(".ccl-field"),cards=[...root.querySelectorAll(".ccl-card")];
  const num=root.querySelector(".ccl-cap b"),N=cards.length;
  const LOOSE=[[100,-70,-7],[-95,-15,6],[110,60,4],[-105,-115,-5],[10,-135,7],[-40,110,-6],[60,-10,3],[-110,70,5],[112,128,-4],[-20,-60,-3]];     // x, y וסיבוב של כל כרטיס בפיזור, לפי גיל (0 הוותיק)
  const fan=c=>{const s=getComputedStyle(c);return[parseFloat(s.getPropertyValue("--fx")),parseFloat(s.getPropertyValue("--fy")),parseFloat(s.getPropertyValue("--fr"))];};
  // בהפחתת תנועה הפונקציה לא רצה, והמניפה נשארת
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const kx=()=>Math.max(.25,Math.min(1,(field.clientWidth/2-cards[0].offsetWidth/2-14)/112));   // בטלפון הפיזור האופקי מצטמצם לרוחב הבמה
    const T0=.5,STEP=.8,MOVE=T0+STEP*(N-2)+.5;
    num.textContent="2";
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    tl.eventCallback("onUpdate",()=>{const t=tl.time();num.textContent=t<T0?2:Math.min(N,2+Math.floor((t-T0)/STEP)+1);});   // סופר רק כרטיסים שכבר נכנסו
    cards.forEach((c,i)=>{
      const j=N-1-i,L=LOOSE[j],F=fan(c),tag=c.querySelector(".ccl-tag"),at=T0+(j-2)*STEP;
      tl.fromTo(c,{x:()=>L[0]*kx(),y:L[1],rotation:L[2]},{x:F[0],y:F[1],rotation:F[2],duration:2.6,ease:"power3.inOut"},MOVE+j*.07);
      if(j>=2){
        tl.fromTo(c,{autoAlpha:0,scale:.7},{autoAlpha:1,scale:1,duration:.5,ease:"power3.out"},at);
        tl.fromTo(tag,{autoAlpha:0,scale:.6},{autoAlpha:1,scale:1,duration:.35,ease:"power3.out"},at+.25);
        if(j<N-1)tl.to(tag,{autoAlpha:0,duration:.3,ease:"power1.in"},at+STEP+.15);
      }
    });
    tl.to({},{duration:.5});      // עצירה קצרה בסוף, עם המניפה גמורה
    return()=>{root.classList.remove("is-live");num.textContent=N;};
  });
})();
