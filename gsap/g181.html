שדה חיפוש בצורת גלולה שמקליד לבד שאלה אחת עם סמן מהבהב. הכפתור עובר מאפור לצבע המותג ואז לספינר, מסגרת זוהרת סביב השדה מתבהרת ברגע השאלה, ומתחתיו נפתח כרטיס תשובה עם שורות שנכנסות בזו אחר זו. אפשר לראות מה השירות עושה בשתי שניות.

מתי משתמשים: אתר של שירות שעונה לשאלה אחת מוכרת: חיפוש תור פנוי, בדיקת זמינות, הערכת מחיר, בדיקת כתובת. תחילת העמוד או סקשן ״איך זה עובד״. השאלה והתשובה צריכות להיות אמיתיות, כאלה שהשירות באמת נותן, ולא ארוכות מארבע שורות. פעם אחת בעמוד.

שחזור. השראה: כמה סרטוני בינה מלאכותית וחיפוש בגלריה של prompt-motion.com (21.10.2026), שבהם שדה מקליד שאלה ומתחתיו מופיעה תשובה. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: השאלה מוקלדת, הכפתור צבוע ושלוש שורות התשובה מוצגות. בלי GSAP ובהפחתת תנועה זה מה שמוצג, והסמן המהבהב והספינר לא קיימים. כשהמהלך רץ ה-JS מרוקן את השדה ומסתיר את התשובה, ובכניסה לעמוד (top 70%, פעם אחת) ציר זמן אחד מקליד עם מונה ו-slice, צובע את הכפתור, לוחץ עליו, מראה ספינר ומכניס את התשובה בסטגר. המסגרת הזוהרת היא שיפוע חרוטי (conic-gradient) עם זווית שמסתובבת דרך @property, פעם כטבעת בגבול של הגלולה ופעם מטושטשת מאחוריה. העוצמה שלה עוברת דרך שלוש מחלקות (רגיל, בפוקוס, בשאלה). הסיבוב מופעל רק כשאין הפחתת תנועה. מקום התשובה שמור מראש, כך שהעמוד לא קופץ כשהיא מופיעה. החלק האינטראקטיבי כולו aria-hidden, והפסקה שמעליו מסבירה מה השירות עושה. RTL: השאלה בעברית מתקדמת משמאל לימין בהקלדה והסמן נשאר בסופה. לשאלה בלועזית מוסיפים dir=ltr על .qf-q. הצבעים של הזוהר הם גוונים של צבע המותג, בלי צבע חדש.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP השאלה כבר מוקלדת והתשובה מוצגת (במארקאפ)
  const root=document.querySelector(".qf"),field=root.querySelector(".qf-field"),typed=root.querySelector(".qf-typed"),
    btn=root.querySelector(".qf-btn"),card=root.querySelector(".qf-res"),parts=[...root.querySelectorAll(".qf-lead,.qf-row")];
  const full=typed.textContent;
  const finish=()=>{typed.textContent=full;field.classList.remove("is-focus","is-typing","is-ask");btn.classList.remove("is-busy");btn.classList.add("is-ready");};
  // בהפחתת תנועה הפונקציה לא רצה, והשאלה והתשובה נשארות במצב הסיום
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    typed.textContent="";btn.classList.remove("is-ready");
    gsap.set(card,{autoAlpha:0});gsap.set(parts,{autoAlpha:0,y:14});
    const k={n:0},tl=gsap.timeline({paused:true});
    tl.call(()=>field.classList.add("is-focus","is-typing"),null,.3)
      .to(k,{n:full.length,duration:1.5,ease:"none",onUpdate:()=>{typed.textContent=full.slice(0,Math.round(k.n));}})
      .call(()=>{field.classList.remove("is-typing");btn.classList.add("is-ready");})
      .to(btn,{scale:.94,duration:.12,ease:"power1.out"},"+=.55")
      .to(btn,{scale:1,duration:.2,ease:"power1.out"})
      .call(()=>{btn.classList.add("is-busy");field.classList.add("is-ask");})
      .call(()=>{btn.classList.remove("is-busy");field.classList.remove("is-focus");},null,"+=1.2")
      .to(card,{autoAlpha:1,duration:.4,ease:"power1.out"},"<.1")
      .to(parts,{autoAlpha:1,y:0,duration:.55,stagger:.14,ease:"power2.out"},"<.15")
      .call(()=>field.classList.remove("is-ask"),null,"+=.8");
    const st=ScrollTrigger.create({trigger:root,start:"top 70%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();gsap.set([btn,card,...parts],{clearProps:"all"});finish();};
  });
})();
