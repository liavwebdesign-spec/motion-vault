G204 · מנעול שנסגר עם קשת שיורדת

GSAP GSAP · ScrollTrigger once · SVG טעינה ופידבקתמונות וגלריות ממתין

מנעול גדול עם הקשת מורמת. כשהסקשן נכנס למסך הקשת יורדת ונעצרת בגוף המנעול, יש לו רעד קטן, טבעת אור אחת מתפשטת ממנו ודועכת, ואז כרטיס הפרטים נפתח לידו ושורות הכרטיס נכנסות אחת אחרי השנייה. ההבטחה נחתמת לפני שקוראים אותה.

מתי משתמשים: סקשן אבטחה, גיבוי או פרטיות: מערכת עם מידע של לקוחות, משרד שמחזיק מסמכים, שירות עם שחזור. רק כשיש מאחורי המנעול הבטחה אמיתית אחת שאפשר לפרט בשלוש שורות. פעם אחת בעמוד.

שחזור. השראה: סרטון אפליקציה ב-prompt-motion.com (21.10.2026) שבו מנעול נסגר והקשת יורדת אל גופו, ואחר כך מופיע כרטיס פרטים לידו. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: מנעול סגור והכרטיס גלוי. הקשת היא נתיב SVG נפרד שמצויר מאחורי הגוף, ולכן הרגליים שלה נעלמות בתוכו כשהיא יורדת, ואין צורך במסכה. כשהמהלך רץ הקשת מורמת ב-34 יחידות והכרטיס מוסתר, ורצה פעם אחת מ-ScrollTrigger כשהסקשן ב-70% מהמסך: הקשת יורדת ב-power4.in (מהירה בסוף), בפגיעה הגוף נלחץ ב-5% וחוזר, והקשת עולה 4 יחידות וחוזרת (שני טווינים של power2, בלי ease של קפיצה). בפגיעה גם יוצאת טבעת אחת מ-.7 ל-1.55 ודועכת, ההילה מאחור נדלקת, והכרטיס יוצא מכיוון המנעול ומתיישב. שורת אור אחת, בלי הבהוב. כיוון: המנעול הוא הראשון בקריאה (ימין) והכרטיס מגיע ממנו שמאלה. באתר משמאל לימין הופכים את סדר העמודות ואת x של הכרטיס לשלילי. נגישות: המנעול הוא איור (aria-hidden), וההבטחה כתובה בכותרת, בפסקה ובשורות הכרטיס, כטקסט אמיתי.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP המנעול סגור והכרטיס גלוי (במארקאפ)
  const root=document.querySelector(".pl"),q=s=>root.querySelector(s);
  const sh=q(".pl-sh"),body=q(".pl-body"),ring=q(".pl-ring"),glow=q(".pl-glow"),card=q(".pl-card");
  const rows=[...card.querySelectorAll(".pl-chip,.pl-rows>div")];
  // בהפחתת תנועה הפונקציה לא רצה, והמנעול נשאר סגור
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    gsap.set(sh,{y:-34});gsap.set(glow,{opacity:0});gsap.set(card,{autoAlpha:0,x:40});gsap.set(rows,{autoAlpha:0});
    const tl=gsap.timeline({paused:true});
    tl.to(sh,{y:0,duration:.3,ease:"power4.in"},0)
      // פגיעה: הגוף נלחץ קצת והקשת עולה קצת וחוזרת, בשני טווינים קצרים של power
      .to(body,{scaleY:.95,svgOrigin:"80 184",duration:.07,ease:"power2.out"},.3)
      .to(body,{scaleY:1,svgOrigin:"80 184",duration:.2,ease:"power2.out"},.37)
      .to(sh,{y:-4,duration:.08,ease:"power2.out"},.3)
      .to(sh,{y:0,duration:.16,ease:"power2.in"},.38)
      // טבעת אור אחת, בלי הבהוב
      .fromTo(ring,{scale:.7,opacity:.8,svgOrigin:"80 128"},{scale:1.55,opacity:0,svgOrigin:"80 128",duration:.9,ease:"power2.out"},.34)
      .to(glow,{opacity:.12,duration:.6,ease:"power1.out"},.34)
      .to(card,{autoAlpha:1,x:0,duration:.6,ease:"power3.out"},.5)
      .to(rows,{autoAlpha:1,duration:.4,stagger:.1,ease:"power2.out"},.7);
    const st=ScrollTrigger.create({trigger:root,start:"top 70%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();gsap.set([sh,body,ring,glow,card,...rows],{clearProps:"all"});};
  });
})();
