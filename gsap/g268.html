G268 · פסיפס צילומים שצץ מהמרכז עם מונים רצים

GSAP GSAP · ScrollTrigger once · stagger from center תמונות וגלריותמספרים ונתונים ממתין

פסיפס רחב של 160 אריחי צילום ממלא את המסגרת. האריחים צצים בגל מהמרכז החוצה, כל אחד בגודל התחלה ובקצב קצת אחרים, והשוליים מחשיכים בהדרגה. באמצע נכנס כרטיס עם שני מספרים גדולים שרצים למעלה ושורת הסבר מתחתיהם.

מתי משתמשים: סקשן שמראה היקף: ארכיון של סטודיו, ספריית תמונות, מאגר מוצרים, מספר הלקוחות והפרויקטים. הפסיפס אומר 'יש הרבה' לפני שמישהו קרא מספר, והמספרים נותנים את הפירוט. פעם אחת בעמוד, ליד הוכחה או 'אודות'. לא כשאין ארכיון אמיתי גדול, ולא כשהמספרים אינם נכונים.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון בגלריה (21.10.2026) שבו פסיפס של מאות אריחים קטנים צץ בגל מהמרכז עם שני מונים שרצים למיליונים ושורה מתחתיהם. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: פסיפס מלא של 160 אריחים (20 על 8, ובטלפון 10 על 16), כרטיס באמצע והמספרים הסופיים. בלי GSAP או בהפחתת תנועה זה מה שרואים. האריחים הם 16 צילומים אמיתיים שחוזרים עשר פעמים כל אחד, בסדר דטרמיניסטי שלא מחזיר אותו צילום ליד עצמו בשום אחד משני גדלי הרשת. כל אריח loading=lazy, והדפדפן טוען כל קובץ פעם אחת. כשהמהלך רץ timeline אחד, בהשהיה, מופעל פעם אחת על ידי ScrollTrigger עם once כשהפסיפס מגיע ל-72% מהמסך: כל אריח מתחיל בגודל קטן שונה (גזור מהאינדקס, לא ערך אקראי, כדי שאותו מסך ייראה אותו דבר בכל טעינה), וזמן הכניסה שלו הוא המרחק ממרכז הרשת (במידות אריחים, אחרי שמודדים כמה עמודות יש) ועוד סטייה קטנה. כך הגל לא נראה כמו טבעת מושלמת. המונים רצים ב-power2.out במקביל לגל, ונכתבים עם toLocaleString('he-IL'). הכרטיס נכנס אחרי תחילת הגל. ה-vignette הוא radial-gradient על --ink, ולכן מתאים לכל עור. ביצועים: 160 טווינים של transform ו-opacity בלבד. אם צריך יותר אריחים, מעבירים לקנבס ולא מוסיפים DOM. נגישות: הפסיפס קישוט (aria-hidden), המספרים מוסתרים בזמן הריצה ומשפט מוסתר אומר את הערכים הסופיים. בפרויקט: מחליפים בצילומי הלקוח ובמספרים אמיתיים.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP הפסיפס המלא והמספרים הסופיים כבר במארקאפ
  const root=document.querySelector(".fm"),grid=root.querySelector(".fm-grid"),tiles=[...grid.children],card=root.querySelector(".fm-card"),nums=[...root.querySelectorAll(".fm-nums b")];
  const to=nums.map(b=>parseInt(b.dataset.to,10)),fmt=n=>Math.round(n).toLocaleString("he-IL");
  const r=i=>{const x=Math.sin(i*127.1+311.7)*43758.5453;return x-Math.floor(x);};      // סטייה קבועה לכל אריח, בלי Math.random
  // בהפחתת תנועה הפונקציה לא רצה, והפסיפס והמספרים נשארים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const cols=getComputedStyle(grid).gridTemplateColumns.split(" ").length,rows=Math.ceil(tiles.length/cols),cx=(cols-1)/2,cy=(rows-1)/2;
    const o=nums.map(()=>({v:0}));
    nums.forEach(b=>{b.textContent="0";});
    const tl=gsap.timeline({paused:true});
    tiles.forEach((t,i)=>{
      const d=Math.hypot(i%cols-cx,Math.floor(i/cols)-cy),at=d*.055+r(i)*.2;      // הגל: המרחק מהמרכז ועוד סטייה קטנה
      tl.fromTo(t,{scale:.15+r(i+99)*.3,autoAlpha:0},{scale:1,autoAlpha:1,duration:.5+r(i+7)*.3,ease:"power3.out"},at);
    });
    nums.forEach((b,k)=>tl.to(o[k],{v:to[k],duration:2.2,ease:"power2.out",onUpdate:()=>{b.textContent=fmt(o[k].v);}},.2+k*.15));
    tl.fromTo(card,{autoAlpha:0,y:18,scale:.96},{autoAlpha:1,y:0,scale:1,duration:.7,ease:"power2.out"},.5);
    ScrollTrigger.create({trigger:root,start:"top 72%",once:true,onEnter:()=>tl.play()});
    return()=>{nums.forEach((b,k)=>{b.textContent=fmt(to[k]);});gsap.set([...tiles,card],{clearProps:"all"});};
  });
})();
