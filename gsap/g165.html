G165 · ראש הפעלה על סרגל שמחליף כרטיס

GSAP GSAP · ScrollTrigger scrub · sticky · 3D כרטיסיםתהליך ושלבים ממתין

כרטיס תצוגה אחד נעוץ במסך, ומתחתיו סרגל עם שנתות וראש הפעלה. כשגוללים הראש נע לאורך הסרגל, ובכל סימן הכרטיס מסתובב עד שהוא נעלם על הקצה, וחוזר עם התוכן הבא. הכיתוב והתווית שעל הסרגל מתחלפים יחד איתו. התוכן אחד, אבל רואים אותו נע בציר זמן.

מתי משתמשים: סקשן שמציג שלושה עד ארבעה סוגי עבודה, חבילות או שלבים מאותו סוג: סוגי פרויקטים של סטודיו, חבילות שירות, שימושים של מוצר. מתאים כשיש משהו להראות בכל אחד (לא רק כותרת), ורוצים שהקורא ירגיש שהוא עובר ביניהם בעצמו. פעם אחת בעמוד, ולא כשיש יותר מחמישה פריטים.

ארבעה דברים שאנחנו בונים לעסקים

כל פרויקט מתחיל בצורך אחר, ולכן לכל סוג יש מבנה, זמן עבודה ומחיר משלו.

  • אתר תדמיתמציג את העסק ומוביל מבקרים לפנייה.
  • דף נחיתהעמוד אחד לקמפיין, עם טופס אחד ברור.
  • חנות אונלייןקטלוג, עגלה ותשלום מאובטח.
  • מערכת ניהולפניות, לקוחות ותשלומים במקום אחד.

שחזור. השראה: כמה סרטוני מוצר ב-prompt-motion.com (21.10.2026), שבהם כרטיס תצוגה אחד מתחלף מעל סרגל זמן עם ראש הפעלה. נבנה מאפס מהצפייה. המארקאפ הוא רשימה: כל סוג עבודה הוא li עם התצוגה והכיתוב שלו. בלי GSAP ובהפחתת תנועה זה מה שרואים, כרטיסים זה לצד זה (ובטלפון זה מתחת לזה), והסרגל מוסתר. כשהמהלך רץ הסקריפט מוסיף is-live: הרשימה הופכת למקבץ כרטיסים באותו מקום, והסקשן נעשה גבוה עם שכבה דביקה (sticky) שנעוצה במסך. הציר: ראש ההפעלה זז לאורך הסרגל לפי ההתקדמות, והסימנים (.pr-mk) יושבים בדיוק במקומות שבהם הכרטיס מתחלף. הסרגל, הסימנים והתוויות הם span ו-i ולא כפתורים, כי זה איור ולא פקד. המעבר: הכרטיס היוצא מסתובב סביב הציר האנכי (rotationY עד 90 מעלות) עד שהוא על הקצה ולא נראה, והנכנס ממשיך מאותה נקודה מהצד השני ומסתובב חזרה עד שהוא שטוח. התוכן מוחלף ברגע שאין מה לראות, ולכן המעבר לא נראה כמו חיתוך. כל הכרטיסים נשארים ב-DOM בשקיפות 0 ולא ב-display:none, כדי שקורא מסך יקרא את כולם. כיוון: הראש זז מימין לשמאל, כי זה כיוון ההתקדמות בעמוד בעברית. הכיוון נקבע מ-direction של הסקשן, ולכן באתר משמאל לימין הוא מתהפך לבד. את התוויות מיישרים לשוליים עם :dir(ltr). מדידה: הרוחב נמדד מהסרגל בכל refresh (invalidateOnRefresh), ולכן זה נכון בכל רוחב מסך. כדי להחליף את התצוגות בתמונות או בסרטונים מחליפים את ה-.pr-shot בתוך כל כרטיס, והשאר נשאר.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשארת הרשימה: כרטיסים זה לצד זה, בלי סרגל
  const root=document.querySelector(".pr"),cards=[...root.querySelectorAll(".pr-card")],track=root.querySelector(".pr-track"),
    ph=root.querySelector(".pr-ph"),lbs=[...root.querySelectorAll(".pr-lb")],n=cards.length;
  // בהפחתת תנועה הפונקציה לא רצה, והרשימה נשארת
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const dir=getComputedStyle(root).direction==="rtl"?-1:1;           // הראש מתקדם בכיוון הקריאה
    const lead=.4,seg=1.2,flip=.34,tail=.4,at=k=>lead+k*seg;           // at(k): הרגע שבו הראש על הסימן k
    gsap.set(cards.slice(1),{opacity:0,rotationY:-90});
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.5,invalidateOnRefresh:true}});
    tl.fromTo(ph,{x:0},{x:()=>dir*track.clientWidth,duration:(n-1)*seg},lead);
    for(let k=1;k<n;k++){
      // הכרטיס היוצא מסתובב עד הקצה, והנכנס ממשיך משם: ההחלפה קורית כשאין מה לראות
      tl.to(cards[k-1],{rotationY:90,opacity:0,duration:flip/2,ease:"power2.in"},at(k)-flip/2)
        .fromTo(cards[k],{rotationY:-90,opacity:0},{rotationY:0,opacity:1,duration:flip/2,ease:"power2.out",immediateRender:false},at(k));
    }
    tl.to({},{duration:tail});
    let cur=0;
    tl.eventCallback("onUpdate",()=>{
      const i=Math.max(0,Math.min(n-1,Math.floor((tl.time()-lead)/seg+.001)));   // התווית מתחלפת על הסימן, באותו רגע שהכרטיס מסתובב
      if(i!==cur){cur=i;lbs.forEach((l,j)=>l.classList.toggle("on",j===i));}
    });
    return()=>{root.classList.remove("is-live");gsap.set([...cards,ph],{clearProps:"all"});lbs.forEach((l,j)=>l.classList.toggle("on",j===0));};
  });
})();
