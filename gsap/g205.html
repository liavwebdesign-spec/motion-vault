G205 · מילת הדגשה שנכנסת אחרונה בסגנון אחר

GSAP GSAP · ScrollTrigger · פיצול ידני למילים טקסט וכותרותהירו ופתיחה ממתין

כותרת בשתי שורות שנכנסת מילה אחרי מילה, והביטוי המודגש בכל שורה מגיע רגע אחרי כולן: בצבע המותג, בעיצוב אחר, ומתבהר מתוך טשטוש קל. השורה השנייה ממשיכה באותו מקצב. הכותרת נקראת כמשפט אחד, והדגש נוחת כמו הסיומת שלו.

מתי משתמשים: כותרת הירו או פתיחה של סקשן, כשיש בה ביטוי אחד שהוא הלב של המסר: מה הלקוח מקבל או למי זה מיועד. עובד באתר שקט ובאתר חוויתי, כי זו כניסה אחת. פעם אחת בעמוד, בכותרת הראשית. לא לכותרת ארוכה, ולא כשאין בה ביטוי אחד שנבדל מהשאר.

נגרייה משפחתית בשרון. מתכננים איתכם בבית, מייצרים בנגרייה ומתקינים בתוך שבוע.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני כותרות בגלריה של prompt-motion.com (21.10.2026), שבהם המילה האחרונה בשורה נכנסת אחרי השאר בצבע ובפונט אחרים. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: כל מילה היא span.aw-w והביטוי המודגש הוא span.aw-a, כך שבלי GSAP ובהפחתת תנועה רואים כותרת שלמה עם הדגש. פיצול ידני למילים, לא לתווים: עברית נשברת כשמפצלים תווים (חיבורי כתיב, סימני ניקוד, סדר הפוך), ולכן כל מילה במארקאפ. סדר ה-DOM הוא סדר הקריאה, אז ה-stagger נכנס מימין לשמאל מעצמו. התזמון: המילים הרגילות נכנסות ב-power3.out עם y וטשטוש קל, והדגש מתחיל כשהמילה האחרונה כמעט נחתה (>-.2), ארוך יותר ועם טשטוש עמוק יותר. השורה השנייה מתחילה רבע שנייה אחרי הדגש הראשון. clearProps:"filter" מנקה את הטשטוש בסוף, כדי שלא יישאר שכבת GPU על כל מילה. הדגש: צבע --accent, משקל 400 מול 700, וfont-style:italic. בעברית זה נטייה מלאכותית של הדפדפן, ולכן אם לאתר יש פונט שני לעברית (עם נטייה אמיתית או סריף) מחליפים את font-family של .aw-a ולא נוגעים בשאר. באתר משמאל לימין אין מה להפוך: התנועה אנכית בלבד, והסדר עוקב אחרי ה-DOM. נגישות: הטקסט כולו תמיד ב-DOM, ושקיפות אפס לא מסתירה אותו מקורא מסך. ההפעלה פעם אחת (once) כשהכותרת נכנסת למסך.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הכותרת כבר מוצגת במלואה, עם הדגש
  const root=document.querySelector(".aw"),lines=[...root.querySelectorAll(".aw-l")],all=root.querySelectorAll(".aw-w,.aw-a");
  // בהפחתת תנועה הפונקציה לא רצה, והכותרת נשארת במצב הסופי
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const tl=gsap.timeline({paused:true});
    lines.forEach((l,i)=>{
      const base=l.querySelectorAll(".aw-w"),acc=l.querySelector(".aw-a");
      // המילים הרגילות קודם. הדגש מתחיל כשהאחרונה שלהן כמעט נחתה, והשורה הבאה אחרי שהדגש נחת
      tl.fromTo(base,{opacity:0,y:30,filter:"blur(8px)"},{opacity:1,y:0,filter:"blur(0px)",duration:.65,ease:"power3.out",stagger:.075,clearProps:"filter"},i?">+.25":0)
        .fromTo(acc,{opacity:0,y:22,filter:"blur(14px)"},{opacity:1,y:0,filter:"blur(0px)",duration:.85,ease:"power2.out",clearProps:"filter"},">-.2");
    });
    const st=ScrollTrigger.create({trigger:root,start:"top 85%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();gsap.set(all,{clearProps:"opacity,transform,filter"});};
  });
})();
