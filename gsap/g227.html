G227 · סמל שנפתח כמניפה ונעצר על הלוגו

GSAP GSAP · SVG · ScrollTrigger (replay) הירו ופתיחהטקסט וכותרות ממתין

שישה עלים גדולים ושישה קטנים מונחים זה על זה כשהם מקופלים, ונפתחים כמניפה תוך כדי סיבוב עד שהם יוצרים סמל. אחרי שהסמל נעצר השם נחשף לידו מימין לשמאל והשורה התחתונה עולה מתחתיו. בסוף נשאר לוגו רגוע ושלם. כשהסקשן חוזר למסך, הכניסה רצה שוב.

מתי משתמשים: פתיחה של אתר או עמוד נחיתה, סיום של סרטון, או הלוגו בראש עמוד האודות. מתאים למותג שהסמל שלו בנוי מחזרה על אותה צורה (עלים, קרניים, טבעות). פעם אחת בעמוד. לא כמסך טעינה שמעכב את התוכן.

פריחה

קליניקה לטיפול משלים, הרצליה

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני אנימציית לוגו ב-prompt-motion.com (21.10.2026), שבהם סמל מסתובב ונפתח מתוך צורה מקופלת ואז השם מופיע לידו. נבנה מאפס מהצפייה. הסמל הוא SVG אחד: שתי שכבות של שישה עלים (גדולים בזווית 0, 60 ואילך, קטנים בזווית 30, 90 ואילך) סביב נקודה אחת (60,60). GSAP מסובב כל עלה מ-0 לזווית שלו עם stagger, בזמן שהקבוצה כולה מסתובבת ממינוס 140 מעלות ל-0 וגדלה. הזווית הסופית של כל עלה נקראת מהמארקאפ (transform=rotate(…)), כך שמשנים צורה או מספר עלים במקום אחד. מקור הסיבוב: svgOrigin:'60 60' ולא transformOrigin, כי הוא קבוע במרחב ה-SVG ולא תלוי בגודל התיבה של העלה. השם נחשף ב-clip-path מהקצה הימני (RTL) עם הזזה קטנה, בלי פיצול לאותיות בעברית. הערך הסופי במארקאפ: בלי GSAP ובהפחתת תנועה רואים את הלוגו השלם מיד. חזרה על המהלך: ScrollTrigger מפעיל מחדש בכניסה ובחזרה מלמטה, ומאפס כשחוזרים מעל נקודת הכניסה, כך שבכל כניסה רואים אותו מההתחלה. אין איפוס ביציאה כלפי מעלה, כדי שהלוגו לא יקפוץ לאחור כשהוא עוזב את המסך. באתר משמאל לימין מחליפים את inset(0 0 0 100%) ל-inset(0 100% 0 0) ואת ההזזה x:24 ל-x:-24. נגישות: הסמל aria-hidden, והשם והשורה התחתונה הם טקסט אמיתי.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;   // בלי GSAP הלוגו כבר שלם (במארקאפ)
  const root=document.querySelector(".pf"),svg=root.querySelector(".pf-mark"),fan=svg.querySelector(".pf-fan"),hub=svg.querySelector(".pf-hub");
  const outer=[...svg.querySelectorAll(".pf-o")],inner=[...svg.querySelectorAll(".pf-i")];
  const name=root.querySelector(".pf-name"),tag=root.querySelector(".pf-tag");
  // הזווית הסופית של כל עלה נקראת מהמארקאפ
  const ang=els=>els.map(e=>parseFloat((e.getAttribute("transform")||"").replace("rotate(",""))||0);
  const A1=ang(outer),A2=ang(inner),O={svgOrigin:"60 60"};
  // בהפחתת תנועה הפונקציה לא רצה, והלוגו השלם מוצג מיד
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const tl=gsap.timeline({paused:true,defaults:{ease:"power3.out"}});
    tl.fromTo(fan,{rotation:-140,scale:.5,opacity:0,...O},{rotation:0,scale:1,opacity:1,duration:1.5,...O},0)
      .fromTo(outer,{rotation:0,...O},{rotation:i=>A1[i],duration:1.2,stagger:.07,...O},.1)
      .fromTo(inner,{rotation:0,...O},{rotation:i=>A2[i],duration:1.2,stagger:.07,...O},.35)
      .fromTo(hub,{scale:0,...O},{scale:1,duration:.6,...O},.9)
      .fromTo(name,{clipPath:"inset(0 0 0 100%)",x:24},{clipPath:"inset(0 0 0 0%)",x:0,duration:.9,ease:"power3.inOut"},1)
      .fromTo(tag,{y:12,opacity:0},{y:0,opacity:1,duration:.8},1.55)
      .set([name,tag],{clearProps:"clipPath,transform,opacity"});
    const st=ScrollTrigger.create({trigger:root,start:"top 72%",end:"bottom 12%",
      onEnter:()=>tl.restart(),onEnterBack:()=>tl.restart(),onLeaveBack:()=>tl.pause(0)});
    return()=>{st.kill();tl.progress(1).kill();};
  });
})();
