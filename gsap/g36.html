G36 · מעבר סקשן בקשת שמתיישרת

GSAP GSAP · ScrollTrigger · MorphSVG רקע ואווירהניווט ומבנה ממתין

הגבול בין סקשן בהיר לכהה הוא קשת עגולה. ככל שגוללים אליו, הקשת מתיישרת לקו ישר. מעבר רך במקום חיתוך, בלי תמונה ובלי clip-path קופצני.

מתי משתמשים: מעבר מהירו לסקשן הבא, כניסה לציטוט או לסקשן אודות. פעם או פעמיים בעמוד. עובד על כל שני צבעי רקע.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger, MorphSVGPlugin);

(function(){
  if(typeof gsap==="undefined")return;
  const mm=gsap.matchMedia();
  mm.add("(prefers-reduced-motion: no-preference)",()=>{
    gsap.to("#arcA",{morphSVG:"#arcB",ease:"none",
      scrollTrigger:{trigger:".arc-svg",start:"top 95%",end:"top 25%",scrub:.6}});
  });
  // בלי תנועה הגבול כבר ישר
  mm.add("(prefers-reduced-motion: reduce)",()=>{gsap.set("#arcA",{morphSVG:"#arcB"});});
})();
