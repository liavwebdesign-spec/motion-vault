G229 · לוגו שמתכווץ לאריח והשם נכתב לידו

GSAP GSAP · steps ease · ScrollTrigger (replay) הירו ופתיחהטקסט וכותרות ממתין

אות המותג מופיעה גדולה במרכז על אריח צבעוני, מתכווצת ונעה לפינת השורה כמו אייקון של אפליקציה, ואז השם נכתב לידה בצעדים עם סמן מהבהב. בסוף הסמן נעלם והשורה התחתונה עולה. השם עצמו הוא טקסט אחד, לא אותיות נפרדות.

מתי משתמשים: מוצר דיגיטלי, אפליקציה, מערכת הזמנות או כל מותג שיש לו אייקון. פתיחה של הירו, מסך תודה או כותרת ברכה. עובד כשהאייקון והשם הם שני חלקים של אותו לוגו. פעם אחת בעמוד.

תורנו

קביעת תורים לעסקים קטנים

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטוני לוגו ב-prompt-motion.com (21.10.2026) שבהם האות מתכווצת לאריח אפליקציה והשם מוקלד אחריה עם סמן. נבנה מאפס מהצפייה. ההקלדה בלי פיצול: המילה נחשפת ב-clip-path עם ease:'steps(5)' (מספר הצעדים הוא מספר האותיות, ונקרא מהטקסט), והסמן נע באותו ערך ובאותו ease, כך שהוא תמיד בקצה החשוף. בעברית לא מפצלים לאותיות, ולכן אין SplitText. האריח מתכווץ ממרכז השורה: ה-x ההתחלתי נמדד מרוחב השורה פחות רוחב האריח, בזמן ההרצה, בלי ערך קשיח, ו-invalidate() לפני כל חזרה מחשב אותו מחדש אחרי שינוי גודל. הערך הסופי במארקאפ: האריח במקומו, המילה שלמה, הסמן שקוף. בלי GSAP ובהפחתת תנועה רואים את המצב הסופי. באתר משמאל לימין מחליפים את sgn ל-1 ואת inset(0 0 0 100%) ל-inset(0 100% 0 0). האות על האריח אפשר להחליף בסמל SVG.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;   // בלי GSAP הלוגו כבר במצבו הסופי
  const root=document.querySelector(".tw"),lock=root.querySelector(".tw-lock"),tile=root.querySelector(".tw-tile");
  const word=root.querySelector(".tw-word"),caret=root.querySelector(".tw-caret"),tag=root.querySelector(".tw-tag");
  const sgn=getComputedStyle(root).direction==="rtl"?-1:1,W=()=>word.offsetWidth,steps="steps("+(word.textContent.trim().length)+")";
  // בהפחתת תנועה הפונקציה לא רצה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const tl=gsap.timeline({paused:true});
    tl.fromTo(tile,{scale:2.4,x:()=>sgn*(lock.offsetWidth-tile.offsetWidth)/2,opacity:0},{opacity:1,duration:.5,ease:"power2.out"},0)
      .to(tile,{scale:1,x:0,duration:1,ease:"power3.inOut"},.7)
      .fromTo(word,{clipPath:"inset(0 0 0 100%)"},{clipPath:"inset(0 0 0 0%)",duration:.9,ease:steps},1.75)
      .fromTo(caret,{x:0},{x:()=>sgn*W(),duration:.9,ease:steps},1.75)
      .set(caret,{opacity:1},1.75)
      .fromTo(caret,{opacity:1},{opacity:0,duration:.22,repeat:4,yoyo:true,ease:"none",immediateRender:false},2.7)
      .fromTo(tag,{y:12,opacity:0},{y:0,opacity:1,duration:.8,ease:"power3.out"},2.9)
      .set(word,{clearProps:"clipPath"}).set(tile,{clearProps:"transform,opacity"}).set(tag,{clearProps:"transform,opacity"});
    const st=ScrollTrigger.create({trigger:root,start:"top 72%",end:"bottom 12%",
      onEnter:()=>tl.invalidate().restart(),onEnterBack:()=>tl.invalidate().restart(),onLeaveBack:()=>tl.pause(0)});
    return()=>{st.kill();tl.progress(1).kill();gsap.set(caret,{clearProps:"all"});};
  });
})();
