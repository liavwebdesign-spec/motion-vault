G185 · פריטים נכנסים למעבד ותוצרים יוצאים

GSAP GSAP · Timeline · ScrollTrigger once תהליך ושלביםטעינה ופידבק ממתין

ארבעה שבבי קבצים (PDF, DOCX, קישור, TXT) מרחפים בפינות כרטיס, ונמשכים בזה אחר זה אל אזור הגרירה המקווקו שבמרכזו. כשהאחרון נכנס האזור נצבע ונעלם, ובמקומו עולה שורת קובץ אחת: פס ההעלאה מתמלא, והתווית מתחלפת מ״מעלה״ ל״הועלה״ עם וי שמצטייר. ארבעה פריטים נכנסים, תוצאה אחת יוצאת.

מתי משתמשים: שלב העלאה, קליטה או הרשמה: מסמכים לפתיחת תיק, חומרים לאתר, קבצים לשירות שמעבד אותם. סקשן ״מה קורה כשמעלים״ או ״איך מתחילים״. צריך שלושה עד ארבעה סוגי קבצים אמיתיים ותוצאה אחת ברורה שאומרת שהכל נקלט.

הלקוח מעלה ארבעה קבצים ומקבל אישור אחד

חוזה, תלושי שכר, קישור לתיקיית תמונות והערות בכתב. הכל נכנס לתיק הלקוח במקום אחד, ופס ההעלאה מראה מתי זה הסתיים.

שחזור. השראה: כמה סרטוני קליטה בגלריה של prompt-motion.com (21.10.2026), שבהם שבבים של קבצים מתכנסים לאזור גרירה והוא מתחלף בשורת קובץ. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: שורת הקובץ מלאה, הפס מלא והוי מצויר, ואזור הגרירה והשבבים מוסתרים ב-CSS. בלי GSAP ובהפחתת תנועה רואים את הכרטיס המוגמר, והפסקה שליד אומרת מה בו. ציר הזמן (פעם אחת, כשהסקשן נכנס): השבבים מופיעים בפינות, נמשכים למרכז האזור עם הקטנה ב-power3.in (תחושת משיכה) ונעלמים בהגיעם, האזור נצבע ונעלם, שורת הקובץ עולה, הפס מתמלא ב-scaleX והתווית מתחלפת עם וי ב-stroke-dashoffset. המרחקים נמדדים ברגע ההפעלה ולכן נכונים בכל רוחב. שני מצבי הגוף (אזור ושורה) יושבים באותו תא בגריד, כך שגובה הכרטיס לא קופץ. להתאים: סוגי הקבצים והשמות הם התוכן, ואפשר שלושה שבבים במקום ארבעה. השבבים שומרים על ארבע פינות, ולכן עם שלושה מוחקים אחד מהם. RTL: הפס מתמלא מימין, ב-LTR משנים את transform-origin של .up-fill ל-left center. מסלולי השבבים נמדדים מה-DOM ולכן אין מה להפוך.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP רואים את הכרטיס המוגמר (במארקאפ)
  const root=document.querySelector(".up"),zone=root.querySelector(".up-zone"),file=root.querySelector(".up-file"),
    chips=[...root.querySelectorAll(".up-chip")],fill=root.querySelector(".up-fill"),
    sa=root.querySelector(".up-sa"),sb=root.querySelector(".up-sb"),tick=root.querySelector(".up-ok path");
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    let tl=null;
    zone.classList.remove("is-over");
    gsap.set(zone,{autoAlpha:1});gsap.set(file,{autoAlpha:0,y:8});gsap.set(fill,{scaleX:0});
    gsap.set(sa,{opacity:1});gsap.set(sb,{opacity:0});gsap.set(tick,{strokeDashoffset:1});gsap.set(chips,{autoAlpha:0,x:0,y:0,scale:1});
    const run=()=>{
      const z=zone.getBoundingClientRect(),zx=z.left+z.width/2,zy=z.top+z.height/2;   // מרכז אזור הגרירה, נמדד עכשיו
      tl=gsap.timeline();
      chips.forEach((c,i)=>{
        const r=c.getBoundingClientRect(),dx=zx-(r.left+r.width/2),dy=zy-(r.top+r.height/2);
        tl.fromTo(c,{autoAlpha:0,y:12},{autoAlpha:1,y:0,duration:.45,ease:"power2.out"},i*.1)
          .to(c,{x:dx,y:dy,scale:.35,duration:.8,ease:"power3.in"},.9+i*.12)
          .to(c,{autoAlpha:0,duration:.18,ease:"none"},1.52+i*.12);
      });
      tl.call(()=>zone.classList.add("is-over"),null,1.7)
        .to(zone,{autoAlpha:0,duration:.3,ease:"power1.out"},2.45)
        .to(file,{autoAlpha:1,y:0,duration:.4,ease:"power2.out"},2.5)
        .to(fill,{scaleX:1,duration:1.4,ease:"power1.inOut"},2.9)
        .to(sa,{opacity:0,duration:.2,ease:"none"},4.25)
        .to(sb,{opacity:1,duration:.2,ease:"none"},4.3)
        .to(tick,{strokeDashoffset:0,duration:.35,ease:"power2.out"},4.35);
    };
    const st=ScrollTrigger.create({trigger:root,start:"top 65%",once:true,onEnter:run});
    return()=>{st.kill();if(tl)tl.kill();gsap.set([zone,file,fill,sa,sb,tick,...chips],{clearProps:"all"});zone.classList.remove("is-over");};
  });
})();
