G242 · מפה מוטה שנעמדים עליה סמנים אחד אחרי השני

GSAP GSAP · ScrollTrigger scrub · sticky · CSS 3D תמונות וגלריותתהליך ושלבים ממתין

מפה שטוחה של אזור שירות, שבתחילת הגלילה מוצגת כמעט מלמעלה ואז מתכופפת אחורה לזווית של מבט אלכסוני. אחרי שהיא נוטה, חמישה סמנים נופלים עליה מלמעלה בזה אחר זה, כל אחד עם תווית של עיר, וטבעת נפתחת מתחתיו במקום שבו נחת. ברשימה שמתחת לכל עיר נדלקת כשהסמן שלה מגיע. הגולש רואה את תחום השירות נבנה ולא רק קורא עליו.

מתי משתמשים: עסק שהכיסוי הגיאוגרפי שלו הוא חלק מהמכירה: טכנאים, שליחויות, מכון בדיקות, מקום שיש לו סניפים, מי שמגיע ללקוח. מתאים לסקשן ״איפה אנחנו פועלים״ בעמוד שירות או בדף הבית. עד שבעה סמנים, פעם אחת בעמוד. לא לעסק שפועל ברחבי הארץ בלי הבדל, ולא כשאין מה להגיד על כל אזור.

שחזור. השראה: כמה סרטוני מוצר בגלריה של prompt-motion.com (21.10.2026) שבהם משטח מוטה בתלת ממד מקבל עליו סימונים אחד אחרי השני. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: מפה מוטה (56 מעלות), כל חמשת הסמנים עומדים עליה, וכל פריטי הרשימה דלוקים. בלי GSAP ובהפחתת תנועה זה מה שמוצג. כשהמהלך רץ הסקשן נעוץ (sticky בתוך אזור גלילה של 300vh), ו-p מ-0 עד 1 כותב רק משתני CSS: --rx ו---rz על השורש (הזווית של המפה, מ-16 מעלות כמעט שטוח ועד 56 במחצית הראשונה של הגלילה) ו---dz ו---rs על כל סמן (הגובה שממנו הוא נופל והגודל של הטבעת). כל הגיאומטריה ב-CSS: המפה היא rotateX(var(--rx)) rotateZ(var(--rz)), וכל סמן מקבל את ההיפוך המדויק translateZ(var(--dz)) rotateZ(-rz) rotateX(-rx), כך שהוא עומד זקוף מול הקורא בכל זווית, כשהבסיס שלו נשאר על הנקודה במפה. ה-translateZ נעשה לפני ההיפוך, לכן הסמן נופל לאורך הנורמל של המפה. שני כללי תלת ממד: (1) לא נותנים opacity לאלמנט שיש לו preserve-3d כי זה משטח אותו; השקיפות של התווית היא על עלה שטוח (.tbd-fl). (2) הטבעת יושבת על translateZ(1px) כדי לא להתחכך עם המפה. המפה היא SVG סכמטי בצבעי הערכה בלבד (חוף, כבישים, פארקים, רשת רחובות), aria-hidden, והמידע האמיתי נמצא ברשימה שמתחתיה, שנשארת קריאה בכל מצב. המיקום של סמן הוא --x ו---y באחוזים מהמפה. RTL: המפה גיאוגרפית (החוף משמאל), לכן אין מה להפוך. בנייד התוויות קטנות יותר והמפה נמוכה.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשארת המפה המוטה עם כל הסמנים והרשימה הדלוקה (במארקאפ)
  const root=document.querySelector(".tbd"),run=root.querySelector(".tbd-run"),ms=[...root.querySelectorAll(".tbd-m")],
    fls=ms.map(m=>m.querySelector(".tbd-fl")),chips=[...root.querySelectorAll(".tbd-list li")];
  const clamp=v=>Math.min(1,Math.max(0,v)),eT=gsap.parseEase("power2.inOut"),eD=gsap.parseEase("power3.out");
  // בהפחתת תנועה הפונקציה לא רצה, והמפה נשארת במצב הסיום
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    // הכל נגזר מ-p בלבד ונכתב כמשתני CSS, והזווית והזקיפות של הסמנים מחושבות שם
    const paint=p=>{
      const a=eT(clamp(p/.24));
      root.style.setProperty("--rx",(16+40*a).toFixed(2)+"deg");
      root.style.setProperty("--rz",(-4-18*a).toFixed(2)+"deg");
      ms.forEach((m,i)=>{
        const t=clamp((p-(.26+i*.13))/.11),e=eD(t);
        m.style.setProperty("--dz",((1-e)*240).toFixed(1)+"px");
        m.style.setProperty("--rs",e.toFixed(3));
        fls[i].style.opacity=clamp(t*4);
        chips[i].classList.toggle("is-on",t>=1);
      });
    };
    paint(0);
    const o={p:0};
    gsap.to(o,{p:1,ease:"none",onUpdate:()=>paint(o.p),
      scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.4,invalidateOnRefresh:true}});
    return()=>{
      root.classList.remove("is-live");["--rx","--rz"].forEach(v=>root.style.removeProperty(v));
      ms.forEach((m,i)=>{["--dz","--rs"].forEach(v=>m.style.removeProperty(v));fls[i].style.removeProperty("opacity");chips[i].classList.add("is-on");});
    };
  });
})();
