G201 · טבעת תמונות שנפתחת לשדה מפוזר

GSAP GSAP · ScrollTrigger scrub · CSS calc תמונות וגלריותטקסט וכותרות ממתין

כשישים אריחי תמונה יושבים בטבעת צפופה סביב כרטיס מט עם משפט, ובגלילה הטבעת נפתחת: האריחים נודדים החוצה במעגל, כל אחד לפי הסדר שלו, ונוחתים בשדה רחב עם גודל, סיבוב ועומק מטושטש שונים. המשפט שבאמצע נבנה מילה אחר מילה בזמן הפתיחה, ובגלילה לאחור הכול חוזר לטבעת.

מתי משתמשים: סקשן הוכחה חברתית או עבודות: לוגואים של לקוחות, תמונות מתיק העבודות, סרטונים קצרים או צוות קהילה, כשיש ארבעים עד שישים פריטים ומשפט אחד שמסכם אותם. מתאים להעביר מסר של היקף: הרבה עסקים, הרבה פרויקטים. פעם אחת בעמוד, עם תמונות שנראות טוב גם כשהן קטנות ומטושטשות.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון בגלריה (21.10.2026) שבו כשישים אריחי תמונה יושבים בטבעת צפופה סביב כיתוב, והטבעת נפתחת לשדה רחב ומפוזר. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: השדה המפוזר, והמשפט שבאמצע מלא. איך זה עובד: לכל אריח שני מיקומים כתובים עליו כמשתני CSS: טבעת (--rx, --ry, --rr) ופיזור (--sx, --sy, --sr, --ss, --bl). משתנה אחד על השורש, --p (0 עד 1), מזיז את כולם: לכל אריח יש --q משלו שנגזר מ---p ומהשהיה (--d), וממנו עקומה חלקה. המיקום, הסיבוב, הגודל והטשטוש הם calc(), ולכן אין לולאה בסקריפט וה-timeline רק מזיז את --p. יחידות: הפיזור ב-cqw ו-cqh של השדה (נמתח לפי השדה), והטבעת ב-cqmin (כדי להישאר עגולה). הפיזור מחושב מראש מרצף R2 (נקודות שמתפזרות בשווה בלי שורות ועמודות) ואחריו הדיפה הדדית של זוגות קרובים מדי, מחוץ לאליפסה הפנויה סביב המשפט (גרעין קבוע, לא רנדום בכל טעינה). רשת עם סטייה נראית כשורות במסך צר, ולכן לא משתמשים בה. כל אריח מקבל גודל וסיבוב משלו. התמונות: שמונה-עשרה תמונות אמיתיות מהמאגר (<img> עם width, height, loading=lazy, object-fit:cover ויחס 3:4), כל אחת לכל היותר ארבע פעמים, וכל עותק ממוקם הכי רחוק שאפשר מהקודמים שלו. אפשר להחליף ברשימה אחת בראש הקובץ. אריחים גדולים (קרובים) וקטנים (רחוקים) מטושטשים, וה-z-index לפי גודל. האריחים ממוינים לפי זווית, כך שכל אחד נודד למקום קרוב. הכיתוב: מילה אחר מילה ב-stagger בזמן שהטבעת נפתחת, על כרטיס מט (backdrop-filter). מובייל: אותו מהלך, והאריחים רחבים יותר יחסית לגובה, כך שהשדה נראה כערימה צפופה. כיווניות: אין כיוון אופקי מובנה, והמשפט נבנה לפי סדר ה-DOM (מימין לשמאל בעברית, ובאתר LTR מעצמו). נגישות: האריחים הם איור (aria-hidden, ו-alt ריק), והמשפט הוא כותרת רגילה.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשאר השדה המפוזר והמשפט מלא
  const root=document.querySelector(".rgf"),words=[...root.querySelectorAll(".rgf-w")];
  // בהפחתת תנועה הפונקציה לא רצה
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}});
    // --p על השורש מזיז את כל האריחים דרך calc(), לכן אין כאן לולאה על האריחים
    tl.fromTo(root,{"--p":0},{"--p":1,duration:4,ease:"power1.inOut"},.7)
      .from(words,{autoAlpha:0,y:14,duration:.5,stagger:.45,ease:"power3.out"},.2)
      .to({},{duration:.6});
    return()=>{root.classList.remove("is-live");};
  });
})();
