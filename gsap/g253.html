G253 · רשת אריחים שמתרוממת למפת גבהים

GSAP GSAP · ScrollTrigger scrub · CSS 3D · sticky מספרים ונתונים ממתין

רשת שטוחה של אריחים, כל אריח שעה ביום מסוים, שמתרוממת בגלילה למפת גבהים בהקרנה אלכסונית. העמודים עולים מהמרכז החוצה, הגובה והצבע מראים כמה פניות נכנסו, ובסוף השיא נדלק בצבע המותג עם תווית, והמספר הגדול סופר עד הערך שלו. רואים בדיוק איפה העומס ואיפה השקט, בלי לקרוא טבלה.

מתי משתמשים: פילוח של ביקוש לפי שני ממדים: שעה ויום במוקד שירות, חדר טיפולים, מסעדה או חנות. מתאים גם למחיר לפי עונה ושעה, או לכמות פרויקטים לפי שנה וסוג. כשיש שלושה ערכים בלבד עמודות רגילות עדיפות. פעם אחת בעמוד, וכשהשיא הוא ההחלטה העסקית.

שחזור. השראה: כמה סרטוני תנועה ב-prompt-motion.com (21.10.2026), רשת שטוחה של אריחים שמתרוממת למפת גבהים אלכסונית עם שיא מודגש אחד. נבנה מאפס מהצפייה. המבנה: שמונים וארבעה עמודים, כל עמוד קובייה אחת ב-CSS 3D שנמתחת בציר Z (scale3d(1,1,…)), ולכן הגובה הוא טרנספורם בלבד וללא חישוב פריסה. שלוש פאות (עליונה, קדמית, צדדית) מצוירות מ-b ומשני pseudo-elements, וכל פאה מקבלת גוון מצבע העמוד (--t). הקרנה: rotateX(58deg) rotateZ(45deg) על עולם בלי perspective, כלומר הקרנה אורתוגרפית. לכן מיקום כל תווית ושל תווית השיא הוא נוסחה ליניארית שמחושבת פעם אחת בבנייה ומוצבת באחוזים, והטקסט נשאר ישר. הנעה: GSAP מזיז משתנה אחד, --g (בין 0 ל-1), עם stagger על הרשת מהמרכז החוצה (grid:[12,7], from:"center"), ב-scrub בתוך סצנה נעוצה בגובה 240vh. ערך סופי בסימון: --g ברירת המחדל היא 1, ולכן בלי GSAP ובהפחתת תנועה רואים את המפה הגבוהה, את התווית ואת המספר. נגישות: המפה כולה aria-hidden, והנתונים מגיעים בטבלה מוסתרת (sr-only) עם כל 84 הערכים. למספר הגדול, שנספר, יש טקסט מוסתר קבוע. RTL: השעות והימים מתקדמים מימין לשמאל בתוך ההקרנה (ראשון ו-08:00 בצד הימני), והטקסטים ב-HTML. באתר LTR הופכים את סדר הימים בנתונים, והלגנדה עוברת מ-to left ל-to right. בטלפון אותה מפה: הטקסט מעליה, המספר לצד הכיתוב, והמפה מוגבלת בגובה כדי שהכול נכנס במסך נעוץ.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP המפה גבוהה במלואה (בסימון)
  const root=document.querySelector(".hm"),run=root.querySelector(".hm-run");
  const bars=[...root.querySelectorAll(".hm-b")],labs=[...root.querySelectorAll(".hm-lb")],tag=root.querySelector(".hm-tag");
  const nums=[...root.querySelectorAll("[data-to]")].map(el=>({el,to:Number(el.dataset.to),o:{v:0}}));
  const fmt=n=>Math.round(n).toLocaleString("he-IL");
  // בהפחתת תנועה הפונקציה לא רצה, ורואים את המצב הסופי מהסימון
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    nums.forEach(n=>{n.el.textContent="0";});
    const tl=gsap.timeline({scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    // הרשת השטוחה מופיעה, ואז העמודים עולים מהמרכז החוצה (12 שעות על 7 ימים), והמספר סופר לאורך כל העלייה
    tl.to(labs,{opacity:1,duration:.2,stagger:.015},0)
      .to(bars,{"--g":1,duration:.9,ease:"power2.out",stagger:{grid:[12,7],from:"center",amount:.9}},.15);
    nums.forEach(n=>{
      tl.to(n.o,{v:n.to,duration:1.8,ease:"power1.inOut",onUpdate:()=>{n.el.textContent=fmt(n.o.v);}},.15);
    });
    tl.fromTo(tag,{y:10,opacity:0},{y:0,opacity:1,duration:.3,ease:"power2.out"},">-.1")
      .to({},{duration:.35});
    return()=>{
      root.classList.remove("is-live");
      bars.forEach(b=>b.style.removeProperty("--g"));
      gsap.set([...labs,tag],{clearProps:"opacity,transform"});
      nums.forEach(n=>{n.el.textContent=fmt(n.to);});
    };
  });
})();
