G170 · חלקים שמתכנסים ונעולים למוצר אחד

GSAP GSAP · ScrollTrigger scrub · SVG · sticky תהליך ושלביםתמונות וגלריות ממתין

ארבעה חלקים של מנורת שולחן תלויים בנפרד על רקע כהה עם קווי מהירות חלשים. בגלילה הם מתקרבים זה לזה תוך כדי סיבוב ונעילים בנקודות החיבור, עם הבזק קטן ומילת חותמת. אחרי הנעילה המנורה נדלקת והתצוגה נכנסת אל המוצר המורכב.

מתי משתמשים: קטע שמציג הצעה שמורכבת מחלקים שאפשר לראות ולהצביע עליהם, כמו ערכה, מארז, מערכת מודולרית או חבילת בית. מתאים כשהמסר הוא שהחלקים משתלבים בלי מאמץ, ויש שלושה עד שישה חלקים עם שמות. לא כשיש יותר משמונה חלקים, ולא באתר שקט שהאפקט הזה יהיה חזק מדי בו. פעם אחת בעמוד.

מנורת שולחן שמגיעה בארבעה חלקים

הבסיס, המוט, הזרוע והאהיל נכנסים זה בזה ונעילים במקום. אין ברגים ואין כלים, והמנורה דולקת כבר בהרכבה הראשונה.

קליק
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: מוצר שמורכב מחלקים נפרדים על רקע כהה עם קווי מהירות חלשים, שבו החלקים מתקרבים, מסתובבים אל נקודות החיבור ונעילים בהבזק קטן ובמילת חותמת, והמצלמה נכנסת אל המוצר המורכב (21.10.2026). נבנה מאפס מהצפייה. המבנה: קטע דביק (sticky) בגובה מסך בתוך סקשן גבוה, עם עמודת טקסט ועמודה לאיור. האיור הוא SVG אחד שבו כל חלק הוא קבוצה <g> שמצוירת במיקום הסופי שלה, ולכן המצב הסטטי הוא המוצר המורכב בדיוק. את המיקום ההתחלתי מוסיפה רק ציר הזמן. כדי שהסיבוב ישב על נקודת החיבור, כל חלק עטוף בקבוצות מקוננות: קבוצת החלק נעה ב-x וב-y, בתוכה קבוצה עם transform קבוע מסוג translate(px py) שמניח את הראשית על נקודת החיבור, בתוכה .pa-r שמסתובבת סביב (0,0) (transformOrigin: של 0 0), ובתוכה קבוצה עם translate(-px -py) שמחזירה את הציור למקומו. כך אין svgOrigin ואין סחף ברענון. הזמנים: ארבעת החלקים יוצאים בהפרש של 4 נקודות אחוז זה מזה, ותנועת ההתקרבות של כל אחד לוקחת כ-44 אחוז מהגלילה (power2.inOut), כך שהם לא נוחתים ביחד. הסיבוב נפרד מהתנועה (power4.out על .pa-r, כ-28 אחוז, ומתחיל מעט אחרי התנועה), כדי שהחלק ישב על הציר רק ברגע האחרון, כמו נעילה. אין back ואין elastic (נאסר בספרייה), והתחושה של נעילה באה מהסיבוב המהיר ומהרעד הקצר של הקבוצה כולה. הנעילה: שלוש טבעות הבזק (.pa-ring, מתרחבות ונמוגות), מילת חותמת ב-HTML (.pa-stamp) שנכנסת מגדולה לקטנה, ורעד של 6 יחידות. אחריה נדלק האור: חרוט אור והשתקפות על הרצפה, שנחשפים ב-opacity. הקווים: repeating-conic-gradient עם מסכה רדיאלית שמרוקנת את המרכז, ובגלילה הם גדלים ונמוגים עד הנעילה. הסיום: הקבוצה .pa-zoom מתקרבת עד 1.32 (1.4 בטלפון), כך שהמצב האחרון הוא תקריב. הקבוצה .pa-world נשארת לרעד של הנעילה. צבעים: רק בטוקנים. הרקע הוא var(--ink) והחלקים var(--card), והמפרקים והאהיל var(--accent), ולכן גם העור ההפוך נראה נכון. כדי להציג מוצר אחר מחליפים את הצורות בתוך הקבוצות ואת הפיבוטים, ושומרים על ארבעה חלקים לכל היותר. מצב סטטי: בלי GSAP ובהפחתת תנועה רואים את המוצר המורכב והדלוק, בלי קווים ובלי חותמת. כיוון: האיור עצמו אינו מכוון. עמודת הטקסט יושבת בצד ההתחלה, והחותמת ממוקמת במאפיין left ולא לוגי, כי האיור עצמו לא מתהפך, ולכן בעמוד משמאל לימין רק עמודת הטקסט מתחלפת.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשאר המוצר המורכב והדלוק
  const root=document.querySelector(".pa"),q=s=>root.querySelector(s),qa=s=>[...root.querySelectorAll(s)];
  const world=q(".pa-world"),zoom=q(".pa-zoom"),base=q(".pa-base"),stem=q(".pa-stem"),arm=q(".pa-arm"),shade=q(".pa-shade"),
    rings=qa(".pa-ring"),cone=q(".pa-cone"),pool=q(".pa-pool"),lines=q(".pa-lines"),stamp=q(".pa-stamp"),labels=qa(".pa-lb"),
    R=el=>el.querySelector(".pa-r");
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const mob=()=>innerWidth<768;
    // הציר של כל סיבוב ושל ההתקרבות הוא (0,0) של הקבוצה הפנימית, וקבוצת העטיפה במאפיין transform קובעת היכן הוא יושב באיור
    gsap.set([R(base),R(stem),R(arm),R(shade),zoom,...rings],{transformOrigin:"0 0"});
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    // הקווים והתנועה: ארבעה חלקים מתקרבים בהפרש, וההסתובבות נפרדת מהתנועה
    tl.fromTo(lines,{opacity:.6,scale:.8},{opacity:0,scale:1.5,duration:.52,ease:"power1.in"},0)
      .fromTo(base,{x:-14,y:46},{x:0,y:0,duration:.44,ease:"power2.inOut"},0)
      .fromTo(R(base),{rotation:-9},{rotation:0,duration:.3,ease:"power4.out"},.14)
      .fromTo(stem,{x:-130,y:-60},{x:0,y:0,duration:.44,ease:"power2.inOut"},.04)
      .fromTo(R(stem),{rotation:-38},{rotation:0,duration:.28,ease:"power4.out"},.2)
      .fromTo(arm,{x:60,y:-120},{x:0,y:0,duration:.44,ease:"power2.inOut"},.08)
      .fromTo(R(arm),{rotation:40},{rotation:0,duration:.28,ease:"power4.out"},.24)
      .fromTo(shade,{x:170,y:-30},{x:0,y:0,duration:.42,ease:"power2.inOut"},.12)
      .fromTo(R(shade),{rotation:55},{rotation:0,duration:.26,ease:"power4.out"},.28)
      .to(labels,{opacity:0,duration:.1},.4)
      // הנעילה: הבזק, חותמת, רעד קצר
      .fromTo(rings,{scale:.3,opacity:.9},{scale:1.9,opacity:0,duration:.12,ease:"power2.out",stagger:.012},.56)
      .fromTo(stamp,{scale:2,opacity:0},{scale:1,opacity:1,duration:.05,ease:"power4.out"},.56)
      .to(stamp,{opacity:0,duration:.1},.72)
      .to(world,{y:6,duration:.03,ease:"power2.out"},.56)
      .to(world,{y:0,duration:.06,ease:"power2.inOut"},.59)
      // אחרי הנעילה המנורה נדלקת והתצוגה נכנסת פנימה
      .fromTo(cone,{opacity:0},{opacity:1,duration:.12,ease:"power1.out"},.62)
      .fromTo(pool,{opacity:0},{opacity:.22,duration:.12,ease:"power1.out"},.62)
      .to(zoom,{scale:()=>mob()?1.4:1.32,duration:.26,ease:"power2.inOut"},.74);
    return()=>{
      root.classList.remove("is-live");
      gsap.set([world,zoom,base,stem,arm,shade,R(base),R(stem),R(arm),R(shade),...rings,cone,pool,lines,stamp,...labels],{clearProps:"all"});
    };
  });
})();
