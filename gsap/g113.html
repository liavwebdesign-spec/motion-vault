G113 · גריד שמסתדר מחדש בגלילה

GSAP GSAP · ScrollTrigger · Flip כרטיסיםתמונות וגלריות ממתין

אותם שישה כרטיסים, שלוש פריסות: רשימה צפופה, גריד שווה, ובנטו עם כרטיס גיבור. בכל שלב גלילה הכרטיסים נוסעים למקומם החדש במקום לקפוץ.

מתי משתמשים: תיק עבודות, קטלוג, סקשן צוות. כשהתוכן קבוע והפריסה היא הסיפור. שלושה מצבים לכל היותר.

הדמו כאן עיצובי-ניטרלי בכוונה. כשהמהלך נכנס לפרויקט, מיובאת רק ההתנהגות: הצבעים, הרדיוסים, הפונטים והצללים יורשים את העיצוב של אותו פרויקט.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
Flip.getState לפני החלפת מחלקת הפריסה ו-Flip.from אחריה, עם absolute:false כדי לא לרוקן את המכל (המלכודת של g23 ו-b34). המעבר נורה על כניסת שלב, לא בסקראב, כי Flip לא אוהב scrub. במובייל שלוש הפריסות הן עמודה, שתי עמודות, ובנטו של 2.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

gsap.registerPlugin(ScrollTrigger, Flip);

(function(){
  const grid=document.querySelector(".rg-grid"),steps=gsap.utils.toArray(".rg-step"),layouts=["l1","l2","l3"];
  const reduce=matchMedia("(prefers-reduced-motion: reduce)").matches;
  let cur=0;
  function go(k){if(k===cur)return;cur=k;
    const state=Flip.getState(".rg-c");
    grid.classList.remove(...layouts);grid.classList.add(layouts[k]);
    if(reduce)return;
    // absolute:false: המכל שומר על גובהו והעמוד לא קופץ (המלכודת של g23 ו-b34)
    Flip.from(state,{duration:.8,ease:"power3.inOut",stagger:.03,absolute:false});
  }
  steps.forEach((s,i)=>ScrollTrigger.create({trigger:s,start:"top 60%",end:"bottom 40%",onEnter:()=>go(i),onEnterBack:()=>go(i)}));
})();
