עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון אחד ב-prompt-motion.com של נגן שמכיל מסך שמנגן את אותו נגן, והמצלמה צוללת פנימה שוב ושוב (21.10.2026). נבנה מאפס מהצפייה. העיקרון: כל מסך הוא עותק של החלון, מוקטן ב---s ומוזז לחריץ ב---sx ו---sy (שברים מגודל החלון). ארבעת המסכים הם אחים שטוחים בתוך .ds-world ולא מקוננים בפועל: מסך k מוקטן ב-s בחזקת k ומוזז בסכום סדרה הנדסית, ולכן החישוב הוא translate ו-scale נקיים בלי חיתוכים. הזום: כל המסכים מקבלים מקדם אחד z = (1/s)^p סביב נקודת השבת של ההעתקה sx/(1-s), שנמצאת ב-transform-origin של העולם. כש-p הוא מספר שלם, מסך מספר p ממלא בדיוק את החלון. תנועה: מונה אחד p (0 עד 3) נע בשלוש קפיצות עם עצירה בין כל שתיים, כדי שכל שלב יספיק להיקרא. חיסכון: מסך שכוסה לגמרי על ידי הבא אחריו מקבל visibility:hidden. חדות: בלי will-change ובלי 3D, כדי שהטקסט יצויר מחדש חד בכל גודל. היחידות: כל מסך הוא container (cqw), ולכן גודל הפונט והריווחים נגזרים מרוחב המסך ולא מהדף. במובייל המסכים בפורמט עומד, החריץ במרכז מתחת לטקסט, והמשתנים --s, --sx, --sy מוחלפים ב-CSS (הקוד קורא אותם מחדש בכל רענון). המצב הסטטי: כל ארבעת המסכים מקוננים בגודל מלא ללא תנועה, ומתחתיהם הרשימה הממוספרת של השלבים, שהיא גם התוכן לקורא מסך. אין כפתורים אמיתיים בתוך המסכים, כי הם איור. באתר משמאל לימין מחליפים את צד החריץ (--sx הופך ל-1 - s - sx) ואת מיקום הטקסט.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP נשארים ארבעת המסכים המקוננים
  const root=document.querySelector(".ds"),view=root.querySelector(".ds-view"),world=root.querySelector(".ds-world");
  const lv=[...world.querySelectorAll(".ds-l")],steps=[...root.querySelectorAll(".ds-steps li")];
  // בהפחתת תנועה הפונקציה לא רצה, והמסכים נשארים מקוננים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const P={p:0};
    let s=.4;
    const rd=()=>{s=parseFloat(getComputedStyle(view).getPropertyValue("--s"))||.4;};      // החריץ שונה בטלפון
    const draw=()=>{
      const z=Math.pow(1/s,P.p);      // z = (1/s)^p: כשp שלם, מסך מספר p ממלא את החלון
      world.style.transform="scale("+z.toFixed(4)+")";
      lv.forEach((l,k)=>{l.style.visibility=(k<lv.length-1&&z*Math.pow(s,k+1)>1.02)?"hidden":"";});      // מסך שהבא אחריו כיסה אותו לגמרי
      const cur=Math.round(P.p);
      steps.forEach((li,i)=>li.classList.toggle("is-on",i===cur));
    };
    rd();
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}});
    tl.eventCallback("onUpdate",draw);
    tl.to({},{duration:.5});
    for(let k=1;k<=3;k++){
      tl.fromTo(P,{p:k-1},{p:k,duration:1.2,ease:"power2.inOut",immediateRender:false});
      tl.to({},{duration:k<3?.7:.8});
    }
    const sync=()=>{rd();draw();};
    ScrollTrigger.addEventListener("refresh",sync);
    return()=>{ScrollTrigger.removeEventListener("refresh",sync);root.classList.remove("is-live");world.style.transform="";lv.forEach(l=>{l.style.visibility="";});steps.forEach((li,i)=>li.classList.toggle("is-on",i===0));};
  });
})();
