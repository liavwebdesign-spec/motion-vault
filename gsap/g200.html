שחזור. השראה: סרטון בגלריה (21.10.2026) של ספר פתוח בתלת מימד שדף ממנו מתהפך מעבר לשדרה ומגלה את הדפים הבאים. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: ספר פתוח על הזוג הראשון, ובו כל עשרת העמודים כבר במארקאפ. בטלפון הם הופכים לגריד של שני טורים (display:contents על העלים), כי ספר ברוחב 340 פיקסלים לא נקרא. המבנה: חמישה זוגות עמודים. עמוד התחלה בימין ועמוד סיום בשמאל נשארים במקומם, וארבעה עלים (.bkp-leaf) נושאים כל אחד עמוד משמאל בחזית ועמוד מימין מאחור. כיוון הדפדוף: בעברית קוראים מימין לשמאל, לכן הערימה שעוד לא נקראה בשמאל והעלה מתהפך משמאל לימין. כל עלה יושב בחצי השמאלי עם transform-origin בשדרה (קצה ימין), ו-rotateY מ-0 ל-180 מעביר אותו לצד השני. הפנים האחורית מסובבת מראש ב-180, ו-backface-visibility:hidden על שתי הפנים. עומק הערימה: כל עלה מוזז מעט ב-Z, והסדר מתהפך באמצע התנועה (כשהדף אנכי ולא רואים אותו) כדי שהעלים לא יחתכו זה את זה. התלתל: משתנה --tw על העלה עולה ויורד סביב אמצע התנועה, ומחזק את צל השדרה ואת טשטוש העמוד (filter על העמוד עצמו ולא על העלה, כי filter על אלמנט עם preserve-3d משטח אותו). מובייל: גריד של עמודים שמופיעים בגלילה. כיווניות: באתר LTR הופכים הכול: הערימה בימין, transform-origin:0 50%, rotateY ל-180 שלילי, והצללים של העמודים. תמונות: בכל עמוד תמונה אמיתית מהמאגר (object-fit:cover, loading=lazy, עם width ו-height), ומתחתיה שם, שורת תיאור ומחיר. התמונה יושבת בתוך העמוד, לכן הטשטוש של התלתל והצל חלים עליה יחד עם הטקסט. נגישות: כל עמוד הוא טקסט רגיל (התמונה עם alt ריק, כי השם כתוב מתחתיה), והצללים הם איור (aria-hidden).

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הספר פתוח על הזוג הראשון, וכל העמודים במארקאפ
  const root=document.querySelector(".bkp"),leaves=[...root.querySelectorAll(".bkp-leaf")],L=leaves.length,DZ=.6;
  const mm=gsap.matchMedia();
  // בהפחתת תנועה שתי הפונקציות לא רצות, והספר נשאר פתוח
  mm.add("(prefers-reduced-motion: no-preference) and (min-width: 768px)",()=>{
    root.classList.add("is-live");
    const D=1.6,GAP=.5;
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}});
    leaves.forEach((lf,i)=>{
      const t=i*(D+GAP)+.3;
      tl.fromTo(lf,{rotationY:0},{rotationY:180,duration:D,ease:"power2.inOut"},t)
        .fromTo(lf,{"--tw":0},{"--tw":1,duration:D/2,ease:"power1.in"},t)         // התלתל: צל וטשטוש עולים עד אמצע הדף
        .to(lf,{"--tw":0,duration:D/2,ease:"power1.out"},t+D/2)
        .fromTo(lf,{z:-i*DZ},{z:-(L-1-i)*DZ,duration:.01,ease:"none",immediateRender:false},t+D/2);   // הסדר בערימה מתהפך כשהדף אנכי
    });
    tl.to({},{duration:.5});
    return()=>{root.classList.remove("is-live");};
  });
  // בטלפון הגריד נשאר, והעמודים מופיעים בגלילה
  mm.add("(prefers-reduced-motion: no-preference) and (max-width: 767px)",()=>{
    gsap.from(root.querySelectorAll(".bkp-pg"),{autoAlpha:0,y:24,duration:1,stagger:{amount:2},ease:"power2.out",scrollTrigger:{trigger:root.querySelector(".bkp-book"),start:"top 88%",end:"bottom 65%",scrub:.5}});
  });
})();
