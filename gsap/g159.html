G159 · נקודה שקופצת מכותרת לכותרת

GSAP GSAP · ScrollTrigger scrub טקסט וכותרות ממתין

כל כותרת בעמוד נגמרת בנקודה בצבע המותג, אבל זו אותה נקודה. כשגוללים היא ניתקת מהכותרת, קופצת בקשת, נמתחת בנפילה, נמעכת לרגע כשהיא נוחתת, ומתיישבת כנקודה בסוף הכותרת הבאה. סימן פיסוק שמלווה את הקורא לאורך הסיפור.

מתי משתמשים: עמוד תדמית או סטודיו עם כמה כותרות קצרות וחזקות, מרוחקות זו מזו, ומותג שיש לו נקודה או עיגול בלוגו. הנקודה הופכת לחתימה שעוברת בין הסקשנים ונוחתת בסוף על הקריאה לפעולה.

לקביעת שיחה
עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם נקודת המותג עוברת ממילה למילה. נבנה מאפס מהצפייה. המארקאפ: בסוף כל כותרת span.bd-dot ריק, inline-block עגול שיושב על קו הבסיס כמו נקודה. בלי GSAP ובהפחתת תנועה כל כותרת מציגה את הנקודה שלה, וזה המצב הסופי. כשהמהלך רץ הנקודות שבכותרות נעשות שקופות (ממשיכות לתפוס מקום, כדי למדוד אותן), ונקודה אחת (.bd-ball) עוברת ביניהן. בלי התנגשות בין קטעים: כל קטע (כותרת לכותרת) מזיז מספר משלו מ-0 ל-1 בסקראב, ופונקציה אחת מציירת את הנקודה לפי הקטע הפעיל. אם כמה טריגרים היו מזיזים את אותה נקודה, גלילה מהירה למעלה הייתה משאירה אותה בכותרת הלא נכונה. הקשת נשארת במסך: קטע מתחיל כשהכותרת הנוכחית ב-30% מהמסך ונגמר כשהבאה ב-65%, ולכן בבסיס הנקודה נעה בתוך המסך מ-30% ל-65%, והקפיצה (פרבולה, עד 18% מהגובה) מעליו. במנוחה היא יושבת בקואורדינטות של העמוד וגוללת איתו בלי קפיצות. מתיחה בנפילה ומעיכה בנחיתה שומרות על שטח קבוע (scaleX = 1 / scaleY). דרישה: לפחות חצי מסך בין כותרות, אחרת הקטעים חופפים. בפרויקט: הצבע הוא --accent, והגודל .2em מהכותרת, כך שבכותרת גדולה יותר הנקודה גדלה בדרך.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP כל כותרת מציגה את הנקודה שלה
  const root=document.querySelector(".bd"),dots=[...root.querySelectorAll(".bd-dot")],ball=root.querySelector(".bd-ball"),body=ball.firstElementChild;
  if(dots.length<2)return;
  // בהפחתת תנועה הפונקציה לא רצה, והנקודות נשארות בכותרות
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const prog=dots.slice(1).map(()=>({p:0}));
    const at=d=>{const r=d.getBoundingClientRect(),o=root.getBoundingClientRect();return{x:r.left-o.left,y:r.top-o.top,s:r.width};};
    const lerp=(a,b,t)=>a+(b-a)*t;
    const draw=()=>{
      let k=-1;prog.forEach((o,i)=>{if(o.p>0)k=i;});      // הקטע האחרון שהתחיל
      const p=k<0?0:prog[k].p,a=at(dots[Math.max(k,0)]),b=k<0?a:at(dots[k+1]);
      let sy=1;
      if(p>.55&&p<.95)sy=1+.28*Math.min((p-.55)/.3,1);   // נמתחת בנפילה
      else if(p>=.95&&p<1){const t=(p-.95)/.05;sy=t<.5?lerp(1.28,.7,t*2):lerp(.7,1,(t-.5)*2);}   // נמעכת בנחיתה
      const s=lerp(a.s,b.s,p),hop=Math.min(innerHeight*.18,240)*4*p*(1-p);
      gsap.set(ball,{x:lerp(a.x,b.x,p),y:lerp(a.y,b.y,p),width:s,height:s});
      gsap.set(body,{y:-hop,scaleY:sy,scaleX:1/sy});
    };
    prog.forEach((o,i)=>gsap.to(o,{p:1,ease:"none",onUpdate:draw,
      scrollTrigger:{trigger:dots[i],start:"top 30%",endTrigger:dots[i+1],end:"top 65%",scrub:.5}}));
    ScrollTrigger.addEventListener("refresh",draw);draw();
    return()=>{ScrollTrigger.removeEventListener("refresh",draw);root.classList.remove("is-live");gsap.set([ball,body],{clearProps:"all"});};
  });
})();
