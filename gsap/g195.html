שני לוחות זה לצד זה. בלוח הראשון רשימת פריטים במחיר לכל אחד, והשורות מתווספות בגלילה, ארבע ואז תשע, כשהסכום למטה מטפס וסרגל הסכום מתארך. בלוח השני מסלול חודשי קבוע במחיר שלא זז, עם משפט שסופר את אותם פריטים ושבב חיסכון שמופיע ברגע שהרשימה עוברת את המסלול. רואים את המחיר מתנפח ליד מחיר שנשאר במקום.

מתי משתמשים: עמוד מחירים של שירות שנמכר גם לפי יחידה וגם כחבילה חודשית: עיצוב פוסטים, צילום, תיקונים, הדרכות, משלוחים. מראה במקום להסביר למה כדאי לעבור למסלול, ועובד כשהמחיר ליחידה והמחיר הקבוע אמיתיים ומוצגים באותו מבנה. פעם אחת בעמוד, ליד הכפתור שמזמין לבחור מסלול.

שחזור. השראה: שני סרטונים בגלריה (21.10.2026) שבהם שורות מחיר מתווספות בצד אחד והסכום מתנפח, בזמן שהצד השני נשאר על מחיר אחד. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: כל תשע השורות, הסכום ₪1,980, הסרגל המלא, המחיר הקבוע ושבב החיסכון (₪780). בלי GSAP ובהפחתת תנועה רואים אותו. המחירים נקראים מהמארקאפ: data-unit (מחיר לפריט) ו-data-plan (המסלול) על שורש הרכיב, והסקריפט מחשב מהם את הסכום לכל מספר שורות. הכול תלוי במספר שלם אחד (k) שגלילה מזיזה מ-0 ל-9, ולכן השורות, הסכום, הסרגל, משפט הספירה והשבב תמיד מסכימים. העצירה בארבעה: ה-timeline עוצר שם רגע, כי עד ארבעה פריטים תשלום לפי פריט זול יותר מהמסלול, ורק אחר כך הוא עובר אותו. בלי קפיצת פריסה: כל תשע השורות תופסות מקום מההתחלה, והשורות שעוד לא נוספו שקופות (opacity ו-transform בלבד, עם easing מהרשימה). מעבר ה-CSS מופעל רק אחרי ציור המצב ההתחלתי (is-anim), כדי שלא יהיה הבהוב בטעינה. הבמה נעוצה (is-live, גובה 320vh), כדי שיהיה מרווח גלילה להגיע עד תשע שורות. כיווניות: הסרגל מתמלא מימין לשמאל (transform-origin:100%), ובאתר LTR הופכים אותו ל-0%, ומחליפים את סדר שני הלוחות אם רוצים את המסלול בצד ימין. נגישות: הסכום והמשפט הרצים מוסתרים מקורא מסך, ופסקה מוסתרת מקריאה את שני המחירים הסופיים. מובייל: הלוחות נערמים בעמודה אחת.

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP כל השורות והסכום הסופי כבר במארקאפ
  const root=document.querySelector(".ppv"),rows=[...root.querySelectorAll(".ppv-row")],N=rows.length;
  const unit=+root.dataset.unit,plan=+root.dataset.plan;
  const total=root.querySelector(".ppv-total"),bar=root.querySelector(".ppv-pay .ppv-bar i");
  const save=root.querySelector(".ppv-save"),saveN=save.querySelector("b"),counts=[...root.querySelectorAll("[data-count]")];
  const fmt=n=>n.toLocaleString("en-US"),word=k=>k===1?"פוסט אחד":k+" פוסטים";
  // מצב אחד לכל מספר שורות: שורות, סכום, סרגל, משפט הספירה והשבב נגזרים כולם מ-k
  const paint=k=>{
    rows.forEach((r,i)=>r.classList.toggle("is-off",i>=k));
    total.textContent="₪"+fmt(k*unit);
    counts.forEach(c=>{c.textContent=word(k);});
    bar.style.setProperty("--w",(k/N).toFixed(3));
    const d=k*unit-plan;save.classList.toggle("is-off",d<=0);if(d>0)saveN.textContent="₪"+fmt(d);
  };
  // בהפחתת תנועה הפונקציה לא רצה, והמצב הסופי נשאר
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const o={n:0},go=()=>paint(Math.round(o.n));
    paint(0);
    root.classList.add("is-live");                              // במה נעוצה: הגובה נקבע לפני שה-ScrollTrigger נוצר
    requestAnimationFrame(()=>root.classList.add("is-anim"));   // הטרנזישן מופעל רק אחרי ציור המצב ההתחלתי
    gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}})
      .to(o,{n:4,duration:4,onUpdate:go})
      .to({},{duration:1.4})                    // עצירה בארבעה: עד כאן לפי פריט זול יותר
      .to(o,{n:N,duration:5,onUpdate:go});
    return()=>{root.classList.remove("is-live","is-anim");paint(N);};
  });
})();
