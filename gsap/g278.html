15 שנים של שיפוצי דירות בתל אביב

ליווינו כמה מאות דירות מהתכנון ועד המפתח, ורובן הגיעו אלינו בהמלצה של שכנים או של לקוחות קודמים.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: סרטון ב-prompt-motion.com (21.10.2026) שבו ספרה ענקית בקו מתאר עומדת ליד תווית ונמלאת בהדרגה. נבנה מאפס מהצפייה. הספרה היא טקסט אמיתי: color:transparent, קו מתאר מ--webkit-text-stroke בצבע ה-ink, ומילוי מגרדיאנט אנכי בצבע ה-accent שנחתך לצורת האות עם background-clip:text. המילוי הוא מיקום רקע: הגרדיאנט גבוה פי 2.2 מהספרה, חצאו התחתון צבוע וחצאו העליון שקוף עם קצה רך באמצע, ו---p (בין 0 ל-1) מזיז אותו מלמעלה למטה דרך background-position. כך הצבע עולה בתוך האות בלי לחשב אחוזים בתוך הגרדיאנט עצמו. משתנה שני: --so הוא עוצמת קו המתאר (מ-20% ל-100%). טיימליין אחד מקושר לגלילה (scrub) מזיז את שניהם: הקו מתחזק ברבע הראשון, והמילוי עולה לאורך כולו ב-power1.inOut. נעיצה קצרה: הסקשן נעוץ (pin:true) כשהוא מגיע למרכז המסך ונשאר שם 75% מגובה המסך, וכל המילוי קורה בזמן הנעיצה. כך הספרה נכנסת למסך כקו בלבד, באמצע הדרך היא חצי מלאה, ובסוף מלאה לפני שהעמוד ממשיך. בלי הנעיצה המילוי היה נגמר כשהספרה עוד בדרך למרכז. הנעיצה נוצרת רק בתוך matchMedia ולכן בלי GSAP ובהפחתת תנועה אין מסלול גלילה מיותר. הערך הסופי במארקאפ: --p:1 ו---so:1, כלומר ספרה מלאה עם קו מלא. זה מה שרואים בלי GSAP ובהפחתת תנועה. הספרה לא סופרת ממספר למספר: זה ההבדל מ-g157, שבו המספר והרשת סופרים יחד. כאן המספר הסופי קבוע, והמילוי הוא שמראה התקדמות. גודל ומשקל: המשקל הכבד (900) נותן שטח מילוי לאות, ובמשקל דק הקו והמילוי כמעט לא נראים. נגישות: הספרה המצוירת aria-hidden, והמספר מופיע גם בכותרת, בטקסט מוסתר חזותית, כך שקורא מסך קורא אותו פעם אחת. באתר משמאל לימין מחליפים רק את סדר העמודות. הספרות נשארות משמאל לימין בשני הכיוונים (direction:ltr).

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP הספרה מלאה והקו מלא (במארקאפ)
  const num=document.querySelector(".ol-n");
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    // הסקשן נעוץ קצר במרכז המסך, והמילוי עולה לאורך כל הנעיצה: בכניסה רק קו, באמצע חצי, ביציאה מלא
    gsap.timeline({scrollTrigger:{trigger:".ol",start:"center center",end:"+=75%",pin:true,scrub:.4}})
      .fromTo(num,{"--so":.2},{"--so":1,duration:.25,ease:"none"},0)
      .fromTo(num,{"--p":0},{"--p":1,duration:1,ease:"power1.inOut"},0);
    return()=>{gsap.set(num,{clearProps:"--p,--so"});};
  });
})();
