G224 · מחיקה מעגלית כמו מחוג שמחליפה מספר

GSAP GSAP · ScrollTrigger scrub · conic-gradient mask תהליך ושלביםמספרים ונתונים ממתין

עיגול גדול עם טבעת דקה וסימני שעון, ובמרכזו מספר ענק. בגלילה מחוג דק מסתובב סביב המרכז, ומאחוריו המספר הבא מחליף את הקודם בצבע אחר, פלח אחר פלח, כמו שעון שנסגר. קו אור אופקי מהבהב על פני המסך בכל סיבוב, והטקסט שלצד העיגול מתחלף יחד איתו.

מתי משתמשים: תהליך עבודה בשלושה שלבים ברורים: ייעוץ, תכנון, ביצוע. כל מקום שבו המספר הוא מידע ולא קישוט, ושלושה שלבים באים אחד אחרי השני. מתאים גם לספירה לאחור של הצעה או מבצע.

משכנתא בשלושה שלבים, מהשיחה הראשונה ועד החתימה

  1. שלב 1

    שיחת היכרות

    מספרים לנו על הדירה, על ההכנסות ועל ההחזר החודשי שנוח לכם. השיחה אורכת חצי שעה.

  2. שלב 2

    השוואה בין הבנקים

    מבקשים הצעות משישה בנקים ומציגים אותן בטבלה אחת, עם ההחזר החודשי והריבית הכוללת.

  3. שלב 3

    חתימה וליווי

    מלווים אתכם עד החתימה בבנק, ואחריה בודקים מדי שנה אם כדאי למחזר.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שלושה סרטונים ב-prompt-motion.com (21.10.2026) שבהם טבעת דקה מקיפה מספר גדול ופלח מעגלי מחליף אותו במספר הבא, בליווי הבזק אור אופקי. נבנה מאפס מהצפייה. המספרים הם שלוש שכבות זו על זו, לכל אחת דיסק רקע אטום משלה ו-mask-image:conic-gradient(#000 var(--a), transparent 0). השכבה העליונה נחתכת לפלח שגדל מ-0 ל-360 מעלות, ומאחוריו נראית הקודמת. דיסק אטום הכרחי: בלעדיו הספרה הישנה הייתה נראית דרך הספרה החדשה. --a מתעדכן מ-GSAP (style.setProperty), לא דרך @property, ולכן זה עובד גם בדפדפנים בלי תמיכה בהנפשת משתנים. מחוג וקשת: קו דק מסתובב סביב אותה נקודה (rotate), וקשת צבועה על הטבעת (stroke-dasharray) נמתחת איתו. בלי GSAP ובהפחתת תנועה רואים את המספר האחרון (3) ואת שלושת השלבים כרשימה, כל אחד עם התווית "שלב" שלו. כשהמהלך רץ הבמה נעוצה והשלבים נערמים באותו תא בגריד ועולים בתור. כיוון: המחוג מסתובב עם כיוון השעון גם באתר בעברית, כי שעונים לא מתהפכים. נגישות: העיגול והמספרים הם איור (aria-hidden), והמספר נמצא בתווית של כל שלב ברשימה.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP העיגול מציג את המספר האחרון
  const root=document.querySelector(".ck"),n2=root.querySelector(".ck-n2"),n3=root.querySelector(".ck-n3"),hand=root.querySelector(".ck-hand"),
    arc=root.querySelector(".ck-arc"),flare=root.querySelector(".ck-flare"),li=[...root.querySelectorAll(".ck-steps li")];
  // בהפחתת תנועה הפונקציה לא רצה, והמספר האחרון והרשימה מוצגים
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const o={a:0,b:0};
    const draw=()=>{
      n2.style.setProperty("--a",o.a+"deg");n3.style.setProperty("--a",o.b+"deg");
      const cur=o.b>0?o.b:o.a,run=cur>0&&cur<360;
      hand.style.rotate=cur+"deg";hand.style.opacity=run?1:0;
      arc.setAttribute("stroke-dasharray",(run?cur/360:0)+" 1");arc.style.opacity=run?1:0;
      flare.style.opacity=run?Math.sin(Math.PI*cur/360)*.9:0;
    };
    gsap.set(li.slice(1),{autoAlpha:0});draw();
    const tl=gsap.timeline({scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6}});
    tl.to({},{duration:.4})
      .to(o,{a:360,duration:.9,ease:"power2.inOut",onUpdate:draw})
      .to(li[0],{autoAlpha:0,duration:.25,ease:"power1.in"},"<.15")
      .to(li[1],{autoAlpha:1,duration:.3,ease:"power1.out"},"<.3")
      .to({},{duration:.7})
      .to(o,{b:360,duration:.9,ease:"power2.inOut",onUpdate:draw})
      .to(li[1],{autoAlpha:0,duration:.25,ease:"power1.in"},"<.15")
      .to(li[2],{autoAlpha:1,duration:.3,ease:"power1.out"},"<.3")
      .to({},{duration:.7});
    return()=>{root.classList.remove("is-live");[n2,n3].forEach(e=>e.style.removeProperty("--a"));
      hand.style.cssText="";flare.style.cssText="";arc.style.cssText="";arc.setAttribute("stroke-dasharray","0 1");gsap.set(li,{clearProps:"all"});};
  });
})();
