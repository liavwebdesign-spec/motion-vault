G175 · דיאגרמת מערכת שנבנית על רשת

GSAP GSAP · ScrollTrigger · SVG תהליך ושלביםכרטיסים ממתין

על רקע רשת שרטוט נבנית דיאגרמה של עץ: קופסה אחת מימין, ממנה יוצאים קווים בזוויות ישרות אל שלוש קופסאות, ומכל אחת קו ישר אל כלי בקצה. הקווים מצטיירים בזה אחר זה, הקופסאות נפתחות כשהקו מגיע אליהן, ואחרי שהכול נבנה נקודות מאירות ממשיכות לרוץ לאורך המסלולים, מהשורש עד העלים.

מתי משתמשים: סקשן שמסביר איך השירות מחובר מבפנים: אתר שמזין יומן ותשלום, מוצר שמתחבר לכמה מערכות, תהליך עבודה שמתפצל לשלושה ענפים. מתאים לעסק שמוכר חיבור בין מערכות, ולמי שרוצה להראות סדר ולא רק לספר עליו. שורש אחד, שלושה ענפים ושלושה עלים. יותר מזה הופך לתרשים שאי אפשר לקרוא.

שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם דיאגרמת מערכת על רקע רשת שרטוט נבנית בעצמה: קופסאות נפתחות, קווים מצטיירים בזוויות ישרות, ונקודות רצות על הקשרים. נבנה מאפס מהצפייה. הקווים לא נמדדים: הקופסאות הן שורות קבועות בגריד, והקווים הם עמודות בגודל קבוע בפיקסלים בין העמודות, ולכן הם תמיד צמודים לקופסאות בכל רוחב. הפינות המעוגלות וצמתי הפיצול (.ar-sd) מצוירים בקואורדינטות פיקסל אחת מול אחת (גובה שורה 88, רוחב עמודה 72), בלי הקטנה אוטומטית של ה-SVG. שתי תצורות: בדסקטופ השורש בצד ימין והעלים בשמאל. בטלפון (.ar-sm) השורש למעלה, ציר אנכי בצד, וכל ענף הוא שורה של קופסה וכלי. שני סטי ה-SVG כתובים במארקאפ, ו-CSS מסתיר אחד מהם. ה-JS בונה ציר זמן רק על הסט הגלוי (gsap.matchMedia עם שני תנאים). הנקודות נעות לפי getPointAtLength. אחרי שהבנייה נגמרת הן רצות בלולאה, ונעצרות כשהדיאגרמה מחוץ לחלון. כיוון: המסלולים מתחילים מימין לשמאל, כדי להתאים לעברית. באתר משמאל לימין מהפכים את כל ה-SVG בקבצים (transform:scaleX(-1)) ואת סדר העמודות. המצב הסופי במארקאפ: כל הקווים מצוירים והקופסאות פתוחות. בלי GSAP או בהפחתת תנועה זה מה שרואים (בלי הנקודות). נגישות: הקווים aria-hidden והתוכן קופסאות עם טקסט רגיל. ביצועים: רק opacity, transform ו-stroke-dashoffset.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  const root=document.querySelector(".ar");if(!root)return;
  if(typeof gsap==="undefined")return;       // בלי GSAP: הדיאגרמה גמורה, בלי הנקודות
  gsap.registerPlugin(ScrollTrigger);
  const tree=root.querySelector(".ar-tree"),mm=gsap.matchMedia(),q=(s,r=root)=>[...r.querySelectorAll(s)];
  // בונה את ציר הזמן על סט הקווים הגלוי: mob = תצורת הטלפון
  const build=mob=>{
    const spine=root.querySelector(mob?".ar-sm":".ar-sd"),sp=q(".ar-p",spine),sd=q(".ar-dot",spine);
    const rb=root.querySelector(".ar-root"),mids=q(".ar-m1,.ar-m2,.ar-m3"),leaves=q(".ar-f1,.ar-f2,.ar-f3"),links=q(".ar-l");
    const lp=links.map(l=>l.querySelector(".ar-p")),la=links.map(l=>l.querySelector(".ar-a")),ld=links.map(l=>l.querySelector(".ar-dot"));
    const all=[...sp,...lp];
    gsap.set(all,{strokeDashoffset:1});gsap.set([rb,...mids,...leaves],{autoAlpha:0,y:12});gsap.set(la,{autoAlpha:0});
    const mover=(path,dot)=>{const o={p:0},L=path.getTotalLength();return gsap.timeline()
      .set(dot,{autoAlpha:1}).fromTo(o,{p:0},{p:1,duration:path===lp[0]||lp.includes(path)?.7:1.1,ease:"none",onUpdate:()=>{const pt=path.getPointAtLength(o.p*L);dot.setAttribute("cx",pt.x);dot.setAttribute("cy",pt.y);}}).set(dot,{autoAlpha:0});};
    const flow=gsap.timeline({paused:true,repeat:-1,repeatDelay:1});
    sp.forEach((p,i)=>flow.add(mover(p,sd[i]),i*.45).add(mover(lp[i],ld[i]),i*.45+1.1));
    let built=false,st=null;
    const tl=gsap.timeline({scrollTrigger:{trigger:tree,start:"top 78%",once:true},onComplete(){built=true;st=ScrollTrigger.create({trigger:tree,start:"top bottom",end:"bottom top",onToggle:s=>s.isActive?flow.play():flow.pause()});flow.play();}});
    tl.to(rb,{autoAlpha:1,y:0,duration:.5,ease:"power2.out"});
    sp.forEach((p,i)=>{
      const t0=.5+i*.35;
      tl.to(p,{strokeDashoffset:0,duration:.8,ease:"power1.inOut"},t0)
        .to(mids[i],{autoAlpha:1,y:0,duration:.4,ease:"power2.out"},t0+.7)
        .to(lp[i],{strokeDashoffset:0,duration:.5,ease:"power1.inOut"},t0+1)
        .to(la[i],{autoAlpha:1,duration:.2},t0+1.4)
        .to(leaves[i],{autoAlpha:1,y:0,duration:.4,ease:"power2.out"},t0+1.4);
    });
    return()=>{flow.kill();if(st)st.kill();gsap.set([...all,rb,...mids,...leaves,...la,...sd,...ld],{clearProps:"all"});};
  };
  mm.add("(prefers-reduced-motion: no-preference) and (min-width: 768px)",()=>build(false));
  mm.add("(prefers-reduced-motion: no-preference) and (max-width: 767px)",()=>build(true));
})();
