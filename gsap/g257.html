RTL: היום הראשון בצד ימין והפס זז שמאלה, כי הסימן של התזוזה מתהפך ב-:dir(rtl). באתר LTR הסימן חוזר לחיובי. בטלפון הנתונים הקטנים עוברים לשורה אחת מתחת למספר, והעמודות צרות יותר.

  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP נשאר המצב הסופי מהסימון: העמודה האחרונה דולקת
  const root=document.querySelector(".hs"),run=root.querySelector(".hs-run");
  const bars=[...root.querySelectorAll(".hs-b")],N=bars.length,ph=root.querySelector(".hs-ph");
  const V=[19,23,20,10,4,21,23,27,19,24,11,5,24,28,34,38,31,16,7,33,24,30,30,28,12,5,30,33,29,27],D=["יום שלישי, 1 בספטמבר","יום רביעי, 2 בספטמבר","יום חמישי, 3 בספטמבר","יום שישי, 4 בספטמבר","יום שבת, 5 בספטמבר","יום ראשון, 6 בספטמבר","יום שני, 7 בספטמבר","יום שלישי, 8 בספטמבר","יום רביעי, 9 בספטמבר","יום חמישי, 10 בספטמבר","יום שישי, 11 בספטמבר","יום שבת, 12 בספטמבר","יום ראשון, 13 בספטמבר","יום שני, 14 בספטמבר","יום שלישי, 15 בספטמבר","יום רביעי, 16 בספטמבר","יום חמישי, 17 בספטמבר","יום שישי, 18 בספטמבר","יום שבת, 19 בספטמבר","יום ראשון, 20 בספטמבר","יום שני, 21 בספטמבר","יום שלישי, 22 בספטמבר","יום רביעי, 23 בספטמבר","יום חמישי, 24 בספטמבר","יום שישי, 25 בספטמבר","יום שבת, 26 בספטמבר","יום ראשון, 27 בספטמבר","יום שני, 28 בספטמבר","יום שלישי, 29 בספטמבר","יום רביעי, 30 בספטמבר"];
  const num=root.querySelector("[data-num]"),dt=root.querySelector("[data-dt]"),sum=root.querySelector("[data-sum]"),avg=root.querySelector("[data-avg]"),flag=root.querySelector("[data-flag]");
  const fin=[num,dt,sum,avg,flag].map(el=>el.textContent);
  const cum=V.reduce((a,v,i)=>{a.push((a[i-1]||0)+v);return a;},[]);
  let cur=-1;
  // עמודה פעילה אחת, מה שלפניה "עבר", והמספרים כולם נגזרים מהאינדקס
  const show=i=>{
    if(i===cur)return;
    cur=i;
    bars.forEach((b,j)=>{b.classList.toggle("is-on",j===i);b.classList.toggle("is-past",j<i);});
    num.textContent=V[i];dt.textContent=D[i];sum.textContent=cum[i];avg.textContent=(cum[i]/(i+1)).toFixed(1);flag.textContent=i+1;
  };
  // בהפחתת תנועה הפונקציה לא רצה, ורואים את המצב הסופי מהסימון
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    root.classList.add("is-live");
    const st={x:0};
    show(0);
    gsap.timeline({scrollTrigger:{trigger:run,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}})
      .to(bars,{"--g":1,duration:.2,ease:"power2.out",stagger:{amount:.16,from:"start"}},0)
      .to(st,{x:N-1,duration:1,ease:"none",onUpdate:()=>{ph.style.setProperty("--x",st.x.toFixed(3));show(Math.round(st.x));}},.14)
      .to({},{duration:.1});
    return()=>{
      root.classList.remove("is-live");
      ph.style.removeProperty("--x");
      bars.forEach((b,j)=>{b.classList.toggle("is-on",j===N-1);b.classList.toggle("is-past",j<N-1);});
      [num,dt,sum,avg,flag].forEach((el,k)=>{el.textContent=fin[k];});
      cur=-1;
    };
  });
})();
