לוח משימות עם שלוש עמודות: לביצוע, בעבודה והושלם. בגלילה חמש משימות עוברות מעמודה לעמודה אחת אחרי השנייה, ובכל מעבר הכרטיס מתרומם קצת ומתיישר במקומו החדש בעמודה, כשהכרטיסים שנשארו מתכנסים למעלה. לצד הלוח גרף שמתחיל גבוה ויורד, ומונה של משימות פתוחות יורד איתו עד שהוא מגיע לאפס. הגולש רואה פרויקט שמתקדם ולא רק קורא על תהליך.

מתי משתמשים: עסק שהמכירה שלו היא תהליך עבודה שקוף: סוכנות, אולפן, חברת שיפוצים, מי שמוסר פרויקט בשלבים ורוצה להראות שיש סדר. מתאים לסקשן ״איך אנחנו עובדים״ או ״מה קורה אחרי שסוגרים״. חמש עד שש משימות, פעם אחת בעמוד. לא כשהתהליך אצלכם לא באמת מתנהל בלוח, ולא עם משימות שהשם שלהן לא אומר כלום ללקוח.

      שחזור. השראה: סרטוני מוצר בגלריה של prompt-motion.com (21.10.2026) שבהם כרטיסים עוברים בין עמודות של לוח וגרף ירידה מתעדכן לידם. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: כל חמשת הכרטיסים בעמודת ״הושלם״ (בזרימה רגילה, בתוך <ul>), שתי העמודות האחרות ריקות, הגרף מצויר עד אפס והמונה מראה 0. בלי GSAP ובהפחתת תנועה זה מה שמוצג. כשהמהלך רץ הסקשן נעוץ (sticky בתוך אזור גלילה של 300vh), הכרטיסים הופכים ל-position:absolute והמיקום של כל אחד נכתב כ-transform מ-paint(p) אחת. המיקום נמדד ולא מקודד: ב-refresh של ScrollTrigger נמדדות שלוש הרשימות (getBoundingClientRect מול הלוח), ולכן הכיוון (RTL), הרוחב והמובייל לא דורשים שום היפוך. הגיאומטריה: לכל כרטיס s הוא סכום שני קטעים מוחלקים (לביצוע אל בעבודה, בעבודה אל הושלם), ומשקלי השיוך לעמודות הם w0=1-s, w1=1-|s-1|, w2=s-1 (חסומים ב-0). ה-x הוא ממוצע משוקלל של שמאל העמודות, וה-y הוא ממוצע משוקלל של ״שורה ראשונה בעמודה + (סכום משקלי הכרטיסים שלפניו באותה עמודה) כפול גובה שורה״. לכן כרטיסים שנשארים מתכנסים למעלה בהחלקה רציפה, בלי קפיצה, והכל נכון גם בגלילה לאחור. הגרף: remAt(q) הוא סכום (1 פחות השלמת כל כרטיס), והקו מצויר מחדש ב-61 נקודות מ-0 עד p. הנקודה בקצה ממוקמת באותה נוסחה. ב-RTL כל ה-SVG מתהפך אופקית ב-CSS (:dir(rtl)), כך שהזמן זורם בכיוון הקריאה, ובו אין טקסט. הכרטיס המורם: --lift (סינוס של ההתקדמות במעבר) מוסיף צל וסיבוב של עד 2 מעלות, ו---d צובע את הכרטיס ואת הסימון לפי מידת ההשלמה. להחליף למקרה אחר: מערכי A ו-B הם זמני ההתחלה של שני המעברים לכל כרטיס (D הוא האורך), והמשימות נקראות מהמארקאפ. בנייד העמודות נשארות שלוש, הכרטיסים מציגים רק כותרת, והגרף עובר לצד המונה.

  const A=[.04,.14,.30,.44,.58],B=[.20,.32,.46,.60,.78],D=.1;
  const X=q=>10+280*q,Y=r=>10+(1-r/N)*140;

    dotX=dot.getAttribute("cx"),dotY=dot.getAttribute("cy");
  let geo=null,last=[-1,-1,-1];
  function measure(){
    // מדידה מול הלוח: ככה RTL, רוחב ומובייל לא דורשים היפוך
    const bb=board.getBoundingClientRect();
    geo={cols:lists.map(l=>{const r=l.getBoundingClientRect();return{x:r.left-bb.left,y:r.top-bb.top,w:r.width};}),
      h:cards[0].offsetHeight,gap:parseFloat(getComputedStyle(root).getPropertyValue("--cg"))||12};
    cards.forEach(c=>{c.style.width=geo.cols[0].w+"px";});
  }
  const remAt=q=>{let r=0;for(let i=0;i<N;i++)r+=1-ease(clamp((q-B[i])/D));return r;};
  function trace(p){
    const n=Math.max(1,Math.ceil(p*60));let d="M"+X(0).toFixed(1)+" "+Y(N).toFixed(1);
    for(let k=1;k<=n;k++){const q=Math.min(p,k/60);d+=" L"+X(q).toFixed(1)+" "+Y(remAt(q)).toFixed(1);}
    return d;
  }
  function paint(p){
    if(!geo)return;
    const st=geo.h+geo.gap;
    // משקלי שיוך רציפים של כל כרטיס לכל עמודה
    const W=cards.map((_,i)=>{
      const m1=clamp((p-A[i])/D),m2=clamp((p-B[i])/D),e1=ease(m1),e2=ease(m2),s=e1+e2;
      const lift=Math.max((m1>0&&m1<1)?Math.sin(Math.PI*m1):0,(m2>0&&m2<1)?Math.sin(Math.PI*m2):0);
      return{s,e2,lift,w:[Math.max(0,1-s),Math.max(0,1-Math.abs(s-1)),Math.max(0,s-1)]};
    });
    const counts=[0,0,0];
    cards.forEach((c,i)=>{
      const w=W[i].w;let x=0,y=0;
      for(let k=0;k<3;k++){
        let ahead=0;for(let j=0;j<i;j++)ahead+=W[j].w[k];   // כמה כרטיסים לפניו באותה עמודה
        x+=w[k]*geo.cols[k].x;y+=w[k]*(geo.cols[k].y+ahead*st);
      }
      c.style.transform="translate("+x.toFixed(1)+"px,"+y.toFixed(1)+"px) rotate("+(-2*W[i].lift).toFixed(2)+"deg) scale("+(1+.03*W[i].lift).toFixed(3)+")";
      c.style.setProperty("--d",W[i].e2.toFixed(3));
      c.style.setProperty("--lift",W[i].lift.toFixed(3));
      c.style.zIndex=W[i].lift>0?2:1;
      counts[Math.round(W[i].s)]++;
    });
