G198 · קיר כרטיסים שמתעקל לגליל

GSAP GSAP · ScrollTrigger scrub · CSS 3D תמונות וגלריותכרטיסים ממתין

קיר שטוח של שלושים ושניים כרטיסי וידאו אנכיים מתעקל בגלילה לגליל חצי עגול שפונה אל הגולש, עם השתקפות רכה מתחתיו. אחר כך הגליל מסתובב סביב עצמו סיבוב שלם, ובאמצע, כשהוא הכי מהיר, הפרצופים נמרחים ללהקה מטושטשת וחוזרים לחדות.

מתי משתמשים: סקשן עדויות או סרטוני לקוחות כשיש הרבה סרטונים אנכיים קצרים (בין שתים עשרה לשלושים ושניים) ורוצים שהקיר ירגיש כמו גלריה שמסתובבת סביב הגולש. מתאים גם לגלריית עבודות או לצוות גדול. פעם אחת בעמוד, ורק עם תמונות שמחזיקות גם כשהן קטנות ומוטות. במסך צר זה גריד רגיל.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים בגלריה (21.10.2026) שבהם קיר של כרטיסים אנכיים מתעקל לגליל ומסתובב. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: גריד שטוח של 32 כרטיסים (שמונה בשורה, ובטלפון ארבעה). זה מה שרואים בלי GSAP, בהפחתת תנועה ובטלפון, ובטלפון הכרטיסים רק מופיעים בגלילה אחד אחרי השני (scrub אחד לכל הרשימה). כשהמהלך רץ (מסך רחב) הסקריפט מוסיף is-live: הכרטיסים עוברים לשתי שורות של 16, הבמה נעוצה בגובה 360vh, ולכל כרטיס נכתב transform בכל פריים מחישוב אחד. הגיאומטריה: נקודה על הקיר במרחק קשת s מהמרכז מקבלת זווית α = κ·s, מיקום x = sin(α)/κ, z = (cos(α)-1)/κ וסיבוב rotateY(α), כאשר κ = p / R ו-R הוא רדיוס הגליל המלא (היקף של 16 כרטיסים עם רווח). כש-p=0 הקיר שטוח לגמרי וכש-p=1 הוא גליל. אותה נוסחה מסובבת את הגליל כשמוסיפים קבוע a לזווית. מה יושב על מה: perspective, -webkit-box-reflect ו-filter על המעטפת ולא על הרשימה, כי filter, overflow או contain על אלמנט עם preserve-3d משטחים אותו. ההשתקפות עובדת בכרום ובספארי, ובפיירפוקס פשוט אין השתקפות והכול עובד. הכרטיסים שבצד האחורי מוסתרים (backface-visibility), והקצוות מוצללים לפי הזווית (--sh, שכבה בצבע הרקע). המריחה: טשטוש של המעטפת עולה ויורד סביב אמצע הסיבוב, כשהמהירות הכי גבוהה בגלל power2.inOut. גודל: רוחב הכרטיס --cw נמדד מהכרטיס עצמו, ולכן משנים רק ב-CSS. כיווניות: הכרטיס הראשון ברשימה יושב בימין והסיבוב מזיז את הקיר ימינה, כמו דפדוף בעברית. הסימן נקרא מ-direction, כך שבאתר LTR אין מה לשנות. התמונות: תשע-עשרה תמונות אמיתיות של אנשים ושל ידיים בעבודה מהמאגר, עם object-fit:cover ו-object-position שמכוון לפנים בכל תמונה, כי כרטיס אנכי חותך תמונה לרוחב. הסדר 7i מודולו 19 מבטיח ששני כרטיסים שכנים, בקיר ובגריד, גם באלכסון, אף פעם לא מציגים אותה תמונה (המחזור ארוך מכל מרחק בין שכנים). width ו-height על כל תמונה (בלי קפיצת פריסה), loading=lazy ו-decoding=async, ורקע var(--card) עד שהתמונה נטענת. נגישות: שם הלקוח הוא טקסט, ואייקון ההפעלה והתמונה (alt ריק, כי השם כבר כתוב) מוסתרים מקורא מסך. אם מחליפים את הכרטיסים בסרטונים אמיתיים, צריך לעצור אותם מחוץ למסך.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP נשאר הגריד השטוח, וכל הכרטיסים נראים
  const root=document.querySelector(".cyl"),persp=root.querySelector(".cyl-persp"),list=root.querySelector(".cyl-list"),cards=[...list.children];
  const NC=16,GAP=12,TAU=Math.PI*2,rtl=getComputedStyle(root).direction==="rtl"?1:-1;
  const mm=gsap.matchMedia();
  // בהפחתת תנועה שתי הפונקציות לא רצות, והגריד השטוח נשאר
  mm.add("(prefers-reduced-motion: no-preference) and (min-width: 768px)",()=>{
    root.classList.add("is-live");
    const o={p:0,a:0};
    // p: כמה הקיר מתעקל (0 שטוח, 1 גליל מלא), a: זווית הסיבוב של הגליל
    const place=()=>{
      const w=cards[0].offsetWidth,h=cards[0].offsetHeight,pitch=w+GAP,R=NC*pitch/TAU,k=o.p/R;
      cards.forEach((c,i)=>{
        const col=i%NC,row=(i/NC)|0,s=-rtl*(col-(NC-1)/2)*pitch,y=(row-.5)*(h+GAP);
        let x=s,z=0,ang=0,sh=0;
        if(k>1e-6){const al=k*s+o.a;x=Math.sin(al)/k;z=(Math.cos(al)-1)/k;ang=al*180/Math.PI;sh=o.p*(1-Math.max(.2,Math.cos(al)))*.75;}
        c.style.transform="translate3d("+x.toFixed(1)+"px,"+y.toFixed(1)+"px,"+z.toFixed(1)+"px) rotateY("+ang.toFixed(2)+"deg)";
        c.style.setProperty("--sh",sh.toFixed(3));
      });
    };
    place();
    const ra=3.4;
    const tl=gsap.timeline({defaults:{ease:"none"},scrollTrigger:{trigger:root,start:"top top",end:"bottom bottom",scrub:.6,invalidateOnRefresh:true}});
    tl.to(o,{p:1,duration:3,ease:"power2.inOut",onUpdate:place})
      .to(o,{a:rtl*TAU,duration:6,ease:"power2.inOut",onUpdate:place},ra)
      .fromTo(persp,{filter:"blur(0px)"},{filter:"blur(7px)",duration:1.6,ease:"power2.in"},ra+1.4)   // המריחה: עולה עד אמצע הסיבוב
      .to(persp,{filter:"blur(0px)",duration:1.6,ease:"power2.out"},ra+3)
      .to({},{duration:.5});
    ScrollTrigger.addEventListener("refresh",place);
    return()=>{ScrollTrigger.removeEventListener("refresh",place);root.classList.remove("is-live");cards.forEach(c=>{c.style.transform="";c.style.removeProperty("--sh");});};
  });
  // בטלפון אין גליל: הגריד נשאר, והכרטיסים מופיעים בגלילה
  mm.add("(prefers-reduced-motion: no-preference) and (max-width: 767px)",()=>{
    gsap.from(cards,{autoAlpha:0,y:28,duration:1,stagger:{amount:3},ease:"power2.out",scrollTrigger:{trigger:list,start:"top 88%",end:"bottom 65%",scrub:.5}});
  });
})();
