G152 · תמונה שנפתחת בתוך הכותרת

GSAP GSAP · Flip · ScrollTrigger טקסט וכותרותהירו ופתיחהתמונות וגלריות ממתין

כותרת עריכתית שבין שתי מילים שלה יושב פער סגור. כשהכותרת נכנסת למסך נפתחת בפער קפסולת תמונה, והמילים משני צידיה מתרחקות בהחלקה, כך שהמשפט ממשיך להיקרא כשורה אחת עם תמונה בתוכה.

מתי משתמשים: הירו עריכתי או משפט מניפסט באמצע העמוד: סטודיו, קליניקה, מסעדה, כל עסק שיש לו תמונות טובות והמשפט שלו חשוב לא פחות מהן. תמונה אחת או שתיים בכותרת, פעם אחת בעמוד.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
איך זה בנוי. אין כאן אנימציית רוחב. הכותרת היא שתי שורות flex ממורכזות, ובכל שורה יש slot בין שתי מילים. במצב הסגור ה-slot ברוחב 0 (קלאס אחד, בלי transition), Flip.getState מצלם את מיקום המילים, הקלאס is-open מחזיר ל-slot את הרוחב האמיתי, ו-Flip.from מנפיש את המילים מהמקום הישן לחדש ב-transform בלבד. הקפסולה עצמה כבר יושבת בגודלה הסופי ונחשפת ב-clip-path: inset() מהמרכז, עם סקייל קטן של התמונה בפנים (1.35 ל-1). השורה ממורכזת, ולכן שני צידי הפער זזים בדיוק חצי רוחב כל אחד והקפסולה גדלה מנקודה שלא זזה: אין שנייה שבה היא חורגת על מילה. ה-slot השני נפתח 140ms אחרי הראשון. מקור, ומה שונה. פורט של MediaBetweenText מ-Fancy Components: במקור הרוחב מונפש ב-motion, וכאן הוא הוחלף ב-FLIP כדי לא להנפיש layout ולא לגרום ל-reflow בכל פריים. הכניסה נקבעת ב-ScrollTrigger פעם אחת (start: "top 75%"), ובמצביע עדין ריחוף על הכותרת פותח אותה אם עוד לא נפתחה, ואחר כך מגדיל את התמונות מעט (1.08). הכותרת אינה רכיב אינטראקטיבי ולכן אין לה פוקוס: הפתיחה במקלדת קורית בגלילה, כמו לכל משתמש. התמונות דקורטיביות (alt="" ו-aria-hidden), והכותרת נושאת aria-label עם המשפט המלא, כדי שקורא מסך לא יאחד מילים ממקטעים שונים. לא נמצא בספרייה קובץ וידאו שמתאים לקפסולה ברוחב כזה, ולכן אין ענף וידאו. בטלפון (מתחת ל-768). מילים שנסגרות ונפתחות בשורה שנשברת קופצות בין שורות, ולכן שם הפער שמור מראש, השורות נשברות כרגיל, והקפסולה נחשפת בלבד (clip-path וסקייל, בלי הזזת מילים). גובה הקפסולה .8em, בכל גודל. בהפחתת תנועה, בלי GSAP או כשה-CDN חסום: הקפסולות פתוחות מההתחלה והמשפט שלם, ואין שום דבר שמוסתר ב-CSS בציפייה ל-JS (מצב הסגור נקבע רק בקלאס ש-JS מוסיף). מה שונה מ-g64 ומ-g108. ב-g64 אותה מילה מודפסת פעמיים והתמונה עוברת בין השכבות בסקראב. ב-g108 שתי שורות מתרחקות בסקראב וחושפות תמונה בגודל מלא בין השורות. כאן התמונה קטנה ויושבת בתוך זרימת המשפט כמו מילה, הפתיחה היא אירוע אחד (לא מחוברת לגלילה), הטקסט כולו נשאר גלוי וקריא לפני ואחרי, ויש עד שני פערים באותה כותרת. שלוש שאלות הגימור. (1) הגרסה הבנאלית היא תמונה רחבה מתחת לכותרת, או תמונה שמתנפחת מעל המילים וחוסמת אותן. כאן התמונה נוספת למשפט בלי לקחת ממנו שום אות, והמילים עצמן זזות כדי לפנות לה מקום. (2) המיקרו-אינטראקציה שנושאת מצב: המילים ש"נדחפות" בהחלקה מראות שהקפסולה היא חלק מהשורה, ובריחוף התמונה מתקרבת מעט, כך שהכותרת מגיבה ליד. (3) הפעולה שהיא מקלה: המבקר מבין במבט אחד מה העסק עושה, בלי לחכות לקרוסלה או לגלול לגלריה. מקור: Fancy Components, Daniel Petho, https://fancycomponents.dev/docs/components/blocks/media-between-text, רישיון MIT

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger, Flip);

(function(){
  const root=document.querySelector(".mbt");
  if(!root||typeof gsap==="undefined")return;      // בלי GSAP הקפסולות פתוחות והמשפט שלם
  const slots=[...root.querySelectorAll(".mbt-slot")];
  const pills=slots.map(s=>s.querySelector(".mbt-pill")),imgs=slots.map(s=>s.querySelector("img"));
  const CLOSED="inset(0% 50% 0% 50% round 999px)",OPEN="inset(0% 0% 0% 0% round 999px)";
  const fine=matchMedia("(hover: hover) and (pointer: fine)");
  // בהפחתת תנועה הפונקציה לא רצה בכלל: הכותרת נשארת פתוחה
  gsap.matchMedia().add({
    desk:"(min-width:768px) and (prefers-reduced-motion: no-preference)",
    mob:"(max-width:767px) and (prefers-reduced-motion: no-preference)"
  },ctx=>{
    // Flip רק בשולחן עבודה: שם כל שורה בנויה ב-nowrap, ולכן מילה אף פעם לא קופצת בין שורות. בטלפון הפער שמור והקפסולה רק נחשפת
    const flip=ctx.conditions.desk&&typeof Flip!=="undefined";
    let opened=false;
    if(flip)root.classList.add("mbt-flip");
    gsap.set(pills,{clipPath:CLOSED});
    gsap.set(imgs,{scale:1.35});
    function openSlot(i){
      const slot=slots[i],delay=i*.14;
      if(flip){
        // 1. מצלמים את המילים בשורה, 2. פותחים את הפער (רוחב אמיתי), 3. Flip מנפיש את ההפרש ב-transform בלבד
        const state=Flip.getState(slot.parentElement.querySelectorAll(".mbt-w"));
        slot.classList.add("is-open");
        Flip.from(state,{duration:.65,delay,ease:"power3.out"});
      }
      gsap.fromTo(pills[i],{clipPath:CLOSED},{clipPath:OPEN,duration:.65,delay,ease:"power3.out"});
      // clearProps מחזיר לתמונה את ה-transition של ה-CSS, כדי שריחוף ימשיך לעבוד אחרי הכניסה
      gsap.to(imgs[i],{scale:1,duration:.9,delay,ease:"power3.out",onComplete(){gsap.set(imgs[i],{clearProps:"transform"});}});
    }
    const open=()=>{if(opened)return;opened=true;slots.forEach((_,i)=>openSlot(i));};
    if(typeof ScrollTrigger!=="undefined")ScrollTrigger.create({trigger:root,start:"top 75%",once:true,onEnter:open});
    else open();
    // מצביע עדין: ריחוף על הכותרת פותח אותה אם עוד לא נפתחה (למשל כשהיא כבר במסך והגלילה לא הגיעה לנקודת הכניסה)
    const hov=()=>{if(fine.matches)open();};
    root.addEventListener("pointerenter",hov);
    return()=>{
      root.removeEventListener("pointerenter",hov);
      root.classList.remove("mbt-flip");
      slots.forEach(s=>s.classList.remove("is-open"));
      gsap.set(imgs,{clearProps:"transform"});
    };
  });
})();
