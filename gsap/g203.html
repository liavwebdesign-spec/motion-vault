G203 · כרטיס סיום: לוגו, כותרת וכפתור בזה אחר זה

GSAP GSAP · ScrollTrigger once הירו ופתיחהטקסט וכותרות ממתין

כרטיס הסגירה של העמוד נבנה בסדר קבוע: המשטח נרגע במקומו, הסמל והשם מתיישבים במרכז, הכותרת עולה מילה אחרי מילה והקו מתחת לביטוי המרכזי נמתח, ואז הכפתור נכנס עם נחיתה קטנה ושורת הפרטים הקטנים מופיעה אחרון. בסוף רואים הכול ביחד, ואין ספק איפה ללחוץ.

מתי משתמשים: סקשן הסגירה של דף נחיתה, וואן-פייג'ר או אתר תדמית, במקום פוטר עם כפתור. מתאים לכל עסק שיש לו שם, משפט אחד וקריאה אחת לפעולה. פעם אחת בעמוד, בסוף. עובד גם באתר שקט, כי כל מה שזז הוא כניסה אחת בסדר קריאה.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטוני מוצר ב-prompt-motion.com (21.10.2026) שנגמרים בכרטיס סיום: הסצנה הקודמת מתפנה, הלוגו מתיישב במרכז, כותרת של שורה אחת נבנית מתחתיו, וכפתור בצורת גלולה נכנס אחרון. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: סמל, שם, כותרת, כפתור ושורת פרטים, הכול גלוי. בלי GSAP ובהפחתת תנועה זה מה שרואים. כשהמהלך רץ טיימליין אחד (paused) נורה פעם אחת מ-ScrollTrigger כשהכרטיס ב-72% מהמסך: המשטח עולה 24px, הסמל גדל מ-.6, השם יוצא מאחורי הסמל, מילות הכותרת עולות בהפרש של .07 שנייה, הקו מתחת לביטוי המרכזי נמתח, הכפתור מגיע ב-.9 ונוחת ב-1.04 וחוזר ל-1 (שני שלבים של power, בלי ease של קפיצה), ושורת הפרטים נוספת אחרון. הביטוי המרכזי מסומן בקו עבה בצבע --accent ולא בטקסט צבעוני, כדי שהקריאות לא תיפול בעור עם accent בהיר. הקו הוא i נפרד כדי שאפשר יהיה לטוון אותו. הכיוון: הקו נמתח מימין לשמאל (transform-origin ימין), והשם יוצא מהסמל שמאלה. באתר משמאל לימין הופכים את transform-origin של הקו ל-0 50% ואת x של השם לשלילי. נגישות: הכותרת היא h2 אחד עם רווחים אמיתיים בין המילים, והקו מוסתר מקורא מסך. הכפתור הוא קישור אמיתי עם focus-visible.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined"||typeof ScrollTrigger==="undefined")return;      // בלי GSAP הכרטיס כבר שלם (במארקאפ)
  const root=document.querySelector(".ec"),q=s=>root.querySelector(s);
  const mark=q(".ec-mark"),word=q(".ec-word"),words=root.querySelectorAll(".ec-w"),ul=q(".ec-ul"),btn=q(".ec-btn"),fine=q(".ec-fine");
  const all=[root,mark,word,...words,ul,btn,fine];
  // בהפחתת תנועה הפונקציה לא רצה, והכרטיס נשאר שלם
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const tl=gsap.timeline({paused:true,defaults:{ease:"power3.out"}});
    tl.from(root,{autoAlpha:0,y:24,duration:.6})
      .from(mark,{autoAlpha:0,scale:.6,transformOrigin:"50% 50%",duration:.5},.25)
      .from(word,{autoAlpha:0,x:16,duration:.5},.4)
      .from(words,{autoAlpha:0,y:22,duration:.55,stagger:.07},.65)
      .fromTo(ul,{scaleX:0},{scaleX:1,duration:.5,ease:"power2.inOut"},1.15)
      // הכפתור נוחת בשני שלבים: קצת מעל הגודל המלא וחזרה. בלי ease של קפיצה
      .fromTo(btn,{autoAlpha:0,scale:.9},{autoAlpha:1,scale:1.04,duration:.3},1.3)
      .to(btn,{scale:1,duration:.18,ease:"power2.inOut"})
      .from(fine,{autoAlpha:0,y:8,duration:.5},">-.05");
    const st=ScrollTrigger.create({trigger:root,start:"top 72%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();gsap.set(all,{clearProps:"all"});};
  });
})();
