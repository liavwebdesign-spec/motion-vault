שבעה אריחי ימים בשורה. מסגרת קטנה קופצת מיום ליום, מהר בהתחלה ובהאטה הולכת וגוברת, ונעצרת על יום אחד. שאר הימים מתעמעמים, היום שנבחר נצבע, והמשפט שמתחת מתחלף לתאריך ולשעות הפנויות. קו דק מתמלא לאורך הסריקה.

מתי משתמשים: שירות עם תורים או מועדים: קליניקה, ייעוץ, פגישות, מועד השקה, אירוע. התנועה אומרת בלי מילים ״בדקנו בשבילכם ובחרנו את היום הקרוב ביותר״. לא מתאים כשהלקוח בוחר יום בעצמו: סריקה שנגמרת בבחירה של המערכת מבטיחה בחירה אוטומטית, ואם אין כזו זה מטעה.

שחזור. השראה: סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם שורה של שבעה אריחי ימים ומסגרת מתאר שקופצת ביניהם, מאטה ונעצרת על אחד בעוד השאר מתעמעמים. נבנה מאפס מהצפייה. המארקאפ הוא מצב הסיום: האריח השלישי מסומן (is-pick), האחרים מעומעמים והמשפט מציג תאריך ושעות. בזמן הסריקה הסקריפט מוסיף is-scan לשורש, ואז כל האריחים שווים, וההסרה שלו בסוף היא שצובעת את הנבחר ומעמעמת את השאר (במעבר CSS על צבע ושקיפות). המנגנון: מונה אחד p עובר מ-0 ל-16 ב-power3.out, והאריח הנוכחי הוא Math.round(p) % 7, ולכן המסגרת קופצת בצעדים שלמים, מאטה ונעצרת. 16 מודולו 7 הוא 2, כלומר האריח השלישי. כדי לעצור ביום אחר משנים את END כך שהשארית תהיה האינדקס הרצוי ומסמנים את is-pick באריח הזה. המסגרת ממוקמת לפי offsetLeft ו-offsetWidth של האריח, ולכן אין תלות בכיוון הדף. באתר LTR רק מסדרים את הימים במארקאפ משמאל לימין, והקו נפרש מתחילת השורה מעצמו. נגישות: הלוח aria-hidden, ומשפט לקוראי מסך נותן את היום והשעות. בהפחתת תנועה נשאר מצב הסיום. טלפון: האריחים מצטמצמים לכ-42 פיקסלים, האות והסטטוס נשארים קריאים. כל מה שמופיע בימים הוא דוגמה: מחליפים בנתוני היומן האמיתיים.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP: היום השלישי נבחר, המשפט מציג תאריך ושעות (במארקאפ)
  const root=document.querySelector(".wd"),days=[...root.querySelectorAll(".wd-d")],fr=root.querySelector(".wd-fr"),
    l1=root.querySelector(".wd-l1"),l2=root.querySelector(".wd-l2"),bar=root.querySelector(".wd-bar i");
  const N=days.length,END=16;               // END מודולו 7 הוא האינדקס של היום הנבחר (2 = שלישי)
  const done1=l1.textContent,done2=l2.textContent;
  // בהפחתת תנועה הפונקציה לא רצה, והלוח נשאר במצב הסיום
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    const s={p:0};let last=-1;
    root.classList.add("is-scan");
    l1.textContent="בודקים את השבוע";l2.textContent="עוברים על היומן ועל הימים הפנויים";
    gsap.set(bar,{width:"0%"});
    // המסגרת קופצת לאריח לפי המונה, ולכן בצעדים שלמים ובהאטה
    const hop=()=>{const i=Math.round(s.p)%N;if(i===last)return;last=i;gsap.set(fr,{x:days[i].offsetLeft,width:days[i].offsetWidth});};
    const tl=gsap.timeline({paused:true});
    tl.call(hop,null,0);
    tl.to(fr,{opacity:1,duration:.25,ease:"power1.out"},0);
    tl.to(s,{p:END,duration:3.4,ease:"power3.out",onUpdate:hop},.2);
    tl.to(bar,{width:"100%",duration:3.4,ease:"power3.out"},.2);
    tl.call(()=>root.classList.remove("is-scan"),null,">+=.2");               // הנבחר נצבע, השאר מתעממים
    tl.to([l1,l2],{opacity:0,duration:.2,ease:"power1.in"},"<+=.2");
    tl.call(()=>{l1.textContent=done1;l2.textContent=done2;},null,">");
    tl.to([l1,l2],{opacity:1,duration:.35,ease:"power1.out"},">");
    tl.to(fr,{opacity:0,duration:.45,ease:"power1.in"},"<+=.2");
    const st=ScrollTrigger.create({trigger:root.querySelector(".wd-board"),start:"top 65%",once:true,onEnter:()=>tl.play()});
    return()=>{st.kill();tl.kill();root.classList.remove("is-scan");gsap.set([fr,bar,l1,l2],{clearProps:"all"});
      l1.textContent=done1;l2.textContent=done2;};
  });
})();
