B81 · טבעת מילים תלת ממדית שמסתובבת סביב נקודה

התנהגויות GSAP · CSS 3D · quickTo טקסט וכותרותרקע ואווירהhover ומיקרו ממתין

שירותי העסק כתובים על טבעת מוטה בתלת ממד, והיא מסתובבת לאט סביב נקודה בצבע המותג. הצד הקרוב בהיר וגדול, הצד הרחוק עמום והפוך. במחשב הטבעת נוטה מעט אחרי העכבר.

מתי משתמשים: הירו או סקשן שירותים של עסק עם חמישה עד שמונה שירותים קצרים של מילה או שתיים: מיתוג, אתרים, צילום. מהלך חתימה אחד בעמוד. פחות מתאים כשצריך לקרוא כל שירות בנחת, כי רק הצד הקרוב קריא בכל רגע.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם מילים על טבעת מוטה מקיפות נקודה במרכז. נבנה מאפס מהצפייה. המארקאפ הוא שורת טקסט אחת: זה מה שרואים בלי GSAP ובהפחתת תנועה, וגם מה שקורא מסך קורא, כי כשהטבעת נבנית השורה נשארת בעץ הנגישות (sr-only) והטבעת עצמה aria-hidden. הבנייה ב-JS: כל אות היא span ממוקם במרכז, ואחרי שהפונט נטען נמדד הרוחב האמיתי של כל אות. הגודל נקבע ב-CSS: הקוטר clamp(280px,42vw,560px) וגודל האות clamp(18px,2vw,28px) יושבים כמשתנים על .rg, וה-JS קורא אותם מאלמנט מדידה בלתי נראה (.rg-sz), כך שהטבעת תמיד הנושא של הסקשן ונכנסת לרוחב בטלפון. הרדיוס הוא חצי מהקוטר, ושורת השירותים חוזרת כמה פעמים שנכנס בהיקף. מה שנשאר עד היקף מלא מתחלק שווה בין הרווחים (ואם הרווח גדול מדי האות גדלה עד 30%), והאותיות נפרשות לפי רוחבן הנמדד, בלי הנחה שכולן שוות. הטבעת נבנית מחדש בשינוי גודל ובטעינת הפונט (ResizeObserver). בעברית בלי ניקוד פיצול לאותיות בטוח: האותיות הסופיות (ם ן ך ף ץ) הן תווים נפרדים ולא נשברות. התלת ממד: perspective על הבמה (פי 1.7 מהקוטר, כך שהעומק נראה אותו דבר בכל גודל), ושתי שכבות עם preserve-3d: הטיה (rotationX של מינוס 26 מעלות מסתכלת על הטבעת מלמעלה) וטבעת שמסתובבת ב-rotationY (36 שניות, ease:"none"). כל אות היא rotateY(זווית) translateZ(רדיוס). השקיפות של כל אות מחושבת מקוסינוס הזווית בעולם, כך שהצד הרחוק עמום. הצד האחורי הפוך מעצמו, וזה חלק מהמראה. כיוון: האותיות נפרשות מימין לשמאל והצד הקרוב נע שמאלה. באתר משמאל לימין מחליפים את הסימן של הזווית (ang) ושל rotationY. עכבר: quickTo על ההטיה, רק במכשיר עם עכבר. ביצועים: הסיבוב נעצר כשהבמה מחוץ למסך (IntersectionObserver). הפחתת תנועה: הטבעת נבנית רק בתנועה רגילה, ובהפחתת תנועה נשארת השורה.

  const root=document.querySelector(".rg"),fb=root.querySelector(".rg-fallback");
  if(!fb)return;

    let spans=[],ang=[],alive=true,key="";const TILT=-26;
    gsap.set(tilt,{rotationX:TILT});
    // שקיפות לפי זווית בעולם: הצד הקרוב מלא, הרחוק עמום
    const paint=()=>{

      for(let i=0;i<spans.length;i++)spans[i].style.opacity=(.16+.84*Math.pow(Math.max(Math.cos(ang[i]+r),0),1.4)).toFixed(3);

    // n עותקים של השורה, בגודל אות fs

      return out;
    };
    const keyNow=()=>sz.offsetWidth+":"+parseFloat(getComputedStyle(sz).fontSize);
    const build=()=>{
      key=keyNow();
      // הקוטר וגודל האות מגיעים מה-CSS (clamp), והטבעת נבנית לפיהם

      let T1=0;mk(1,fsT).forEach(s=>{T1+=s.offsetWidth;});

      spans=mk(n,fs);
      const w=spans.map(s=>s.offsetWidth);
      let total=0;w.forEach(x=>{total+=x;});
      // מה שנשאר (או חסר) עד היקף מלא מתחלק שווה בין הרווחים, והטבעת נסגרת בדיוק
      const extra=S?(C-total)/(n*S):0;
      let cum=0;ang=[];
      // האות הראשונה בחזית והבאות שמאלה (זווית יורדת), כך שהקריאה ימין לשמאל. באתר LTR: זווית עולה

    };

    if(matchMedia("(hover: hover) and (pointer: fine)").matches){root.addEventListener("pointermove",move);root.addEventListener("pointerleave",leave);}
    return()=>{alive=false;tw.kill();io.disconnect();ro.disconnect();root.removeEventListener("pointermove",move);root.removeEventListener("pointerleave",leave);stage.remove();root.classList.remove("is-ring");};
  });
})();
