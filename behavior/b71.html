B71 · רשימה שהשורה בה נפתחת לרצועה זורמת

התנהגויות Vanilla JS · GSAP · CSS marquee ניווט ומבנהhover ומיקרותמונות וגלריות ממתין

רשימה גדולה של שורות. כשהסמן נכנס לשורה נכנסת אליה מהקצה הקרוב אליו רצועה בצבע המותג, ובתוכה השם של השורה ותמונה קטנה רצים הצידה בלולאה. ביציאה הרצועה יוצאת דרך הקצה הקרוב ליציאה.

מתי משתמשים: אינדקס פרויקטים, רשימת שירותים, תפריט טיפולים, תפריט מסעדה, חלוקה לקטגוריות. ארבע עד שבע שורות, כל שורה היא קישור אמיתי. במסך מגע הרשימה סטטית, עם תמונה בכל שורה.

איך זה בנוי. בכל שורה (קישור) יש מסכה, ובה רצועה (.fm-band) ובתוכה שכבה פנימית (.fm-bandin) שמחזיקה את המרקי. הרצועה זזה ב-translateY(±101%) והשכבה הפנימית זזה בדיוק בהפוך (∓101%), כך שהתוכן נראה עומד במקום בזמן שהמסכה מחליקה מעליו. הערך הוא ערך אחד שמונף ב-GSAP (0.6 שניות, expo.out), והשניים נכתבים ממנו בכל פריים, ולכן הקיזוז מדויק גם באמצע תנועה. ב-101 ולא ב-100 כדי שלא תישאר שורת פיקסלים בקצה. הקצה הקרוב. מחשבים את מרחק הסמן מהקצה העליון והתחתון של השורה (clientY מול getBoundingClientRect) ובוחרים את הקטן. הרצועה נכנסת מהצד הזה ויוצאת דרך הצד הקרוב ליציאה. אם נכנסים שוב באמצע יציאה, ההחלקה ממשיכה מהמקום שבו היא עומדת, בלי קפיצה. במקלדת, :focus-visible פותח את הרצועה תמיד מלמעלה וסוגר אותה למעלה; פוקוס שבא מלחיצת עכבר לא פותח שוב. המרקי. ב-JS בלבד: מהשם ומהתמונה של השורה בונים יחידות (שם ותמונה) מספיק כדי שחצי מהמסלול יכסה את הרוחב, ומכפילים אותו. הלולאה היא translateX(50%) על transform בלבד, 15 שניות לינארי, והיא עוצרת כשהשורה לא פתוחה. הכיוון נקבע לעברית: התוכן נע ימינה, והטקסט החדש נכנס משמאל, מהכיוון שאליו הקריאה מתקדמת; ב-html[dir="ltr"] הוא הפוך. הרצועה כולה aria-hidden: קורא מסך שומע רק את הטקסט האמיתי של הקישור. התמונות במרקי דקורטיביות. בלי קווים. השורות מופרדות במרווח ובמשטח עדין (var(--card)), לא בקו שיער. מגע, הפחתת תנועה, בלי JS. במסך מגע או בלי סמן עדין אין ריחוף, ולכן הרצועה לא נבנית בכלל, וכל שורה מציגה את התמונה שלה בקפסולה בצד: קישור רגיל, ולחיצה מנווטת. בהפחתת תנועה הרצועה מופיעה ונעלמת מיד בלי החלקה, והמרקי לא רץ. בלי GSAP, או כשה-CDN חסום, זו רשימת קישורים רגילה עם התמונות. בטלפון השם קטן יותר, ושורת הפרטים יורדת מתחתיו. מה שונה מ-b54 ומ-g43. ב-b54 שכבת הריחוף נכנסת מהכיוון של הסמן על כרטיס בגריד, והיא אינה זורמת ואין בה תוכן שרץ. ב-g43 חלון תמונה צף עוקב אחרי הסמן מחוץ לשורה ומתחלף בין שורות, והשורה עצמה לא משתנה. כאן אין חלון צף: הרצועה היא חלק מהשורה, נושאת בתוכה את השם והתמונה, ולכל שורה יש כניסה ויציאה משלה לפי הקצה. שלוש שאלות הגימור. (1) הגרסה הבנאלית: שורה שמשנה צבע ברקע בריחוף, או רשימה עם תמונה שקופצת. כאן התנועה נכנסת מהכיוון שממנו הגיעה היד ויוצאת לשם שאליו היא הולכת, והתמונה נמצאת בשורה, לא מעליה. (2) המיקרו-אינטראקציה שנושאת מצב: שאר השורות נחלשות כשאחת פתוחה, והרצועה זורמת, כך שברור איזו שורה נבחרה וגם שהיא ניתנת ללחיצה. (3) הפעולה שהיא מקלה: בחירת פרויקט או שירות מתוך רשימה, עם הצצה לתמונה לפני הלחיצה, בלי לטעון עמוד. שחזור, השראה: React Bits FlowingMenu (https://www.reactbits.dev/components/flowing-menu); הקוד נכתב מחדש

(function(){
  const root=document.querySelector(".fm");
  // דמו: לשורות אין יעד, ולכן הלחיצה לא קופצת לראש העמוד. בפרויקט אלה קישורים אמיתיים
  root.querySelectorAll(".fm-row").forEach(r=>r.addEventListener("click",e=>e.preventDefault()));
  if(typeof gsap==="undefined")return;                                   // בלי GSAP: רשימת קישורים רגילה עם התמונות
  if(!matchMedia("(hover: hover) and (pointer: fine)").matches)return;   // מגע: גרסה סטטית, בלי רצועה
  root.classList.add("is-live");
  const rows=[...root.querySelectorAll(".fm-row")];
  const rm=()=>matchMedia("(prefers-reduced-motion: reduce)").matches;   // נבדק בכל אירוע, ולכן שינוי ההעדפה נתפס
  const OFF=101;

  // המרקי נבנה מהשם והתמונה של השורה, כך שהמארקאפ נשאר רשימה נקייה
  function build(row){
    const label=row.querySelector(".fm-label").textContent.trim();
    const src=row.querySelector(".fm-thumb img").getAttribute("src");
    const pos=row.querySelector(".fm-thumb img").style.objectPosition;
    const track=row.querySelector(".fm-track");
    const unit=()=>{
      const u=document.createElement("span");u.className="fm-unit";
      const t=document.createElement("span");t.textContent=label;
      const p=document.createElement("span");p.className="fm-pill";
      const i=document.createElement("img");i.alt="";i.src=src;if(pos)i.style.objectPosition=pos;
      p.appendChild(i);u.append(t,p);return u;
    };
    track.textContent="";
    track.appendChild(unit());
    const w=track.firstChild.getBoundingClientRect().width||300;
    const n=Math.min(14,Math.max(2,Math.ceil(row.clientWidth*1.15/w)));
    const half=document.createDocumentFragment();
    for(let k=0;k<n;k++)half.appendChild(unit());
    track.textContent="";
    track.appendChild(half.cloneNode(true));track.appendChild(half);
  }
  rows.forEach(build);
  let lastW=root.clientWidth,tm;
  new ResizeObserver(()=>{if(root.clientWidth===lastW)return;lastW=root.clientWidth;clearTimeout(tm);tm=setTimeout(()=>rows.forEach(build),150);}).observe(root);

  rows.forEach(row=>{
    const band=row.querySelector(".fm-band"),inner=row.querySelector(".fm-bandin");
    const st={y:OFF,tw:null,key:false};
    // הרצועה והשכבה הפנימית נכתבות מאותו ערך, בקיזוז מדויק: התוכן עומד והמסכה מחליקה
    const paint=()=>{band.style.transform="translate3d(0,"+st.y+"%,0)";inner.style.transform="translate3d(0,"+(-st.y)+"%,0)";};
    paint();
    const go=(y,done)=>{
      if(st.tw)st.tw.kill();
      if(rm()){st.y=y;paint();if(done)done();return;}
      st.tw=gsap.to(st,{y,duration:.6,ease:"expo.out",onUpdate:paint,onComplete(){if(done)done();}});
    };
    // -1 = הקצה העליון, 1 = התחתון: הקרוב לסמן
    const edge=cy=>{const r=row.getBoundingClientRect();return (cy-r.top)<(r.bottom-cy)?-1:1;};
    const show=side=>{
      row.classList.add("is-on");
      // אם הרצועה בחוץ מתחילים מהקצה שנבחר; אם היא באמצע יציאה, ממשיכים משם בלי קפיצה
      if(Math.abs(st.y)>=OFF-1){st.y=OFF*side;paint();}
      go(0);
    };
    const hide=side=>go(OFF*side,()=>row.classList.remove("is-on"));
    row.addEventListener("pointerenter",e=>{if(e.pointerType==="touch")return;root.classList.add("is-hover");show(edge(e.clientY));});
    row.addEventListener("pointerleave",e=>{if(e.pointerType==="touch")return;hide(edge(e.clientY));});
    // מקלדת: נכנסת תמיד מלמעלה ויוצאת למעלה. פוקוס מלחיצת עכבר אינו focus-visible, ולכן לא פותח פעמיים
    row.addEventListener("focus",()=>{if(!row.matches(":focus-visible"))return;st.key=true;root.classList.add("is-hover");show(-1);});
    row.addEventListener("blur",()=>{if(!st.key)return;st.key=false;if(!row.matches(":hover"))hide(-1);if(!root.matches(":hover"))root.classList.remove("is-hover");});
  });
  root.addEventListener("pointerleave",()=>root.classList.remove("is-hover"));
})();
