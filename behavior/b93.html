B93 · מילה שמחליפה משפחת גופן במקום

התנהגויות GSAP · ScrollTrigger toggle · hover · Bellefair טקסט וכותרותhover ומיקרו ממתין

מילה גדולה כתובה בגופן סנס ישר וחד, ושלושה קווי הנחיה מאחוריה (קו עליון, קו בסיס וקו תחתון) לא זזים. כשהמילה מגיעה לאמצע המסך, או כשמרחפים מעליה, היא מתחלפת בטשטוש רך לאותה מילה בגופן סריף חם ואישי, באותו גובה ועל אותו קו בסיס. תווית קטנה מתחת מראה באיזה קול המילה נמצאת עכשיו. בלי GSAP ובהפחתת תנועה מוצג רק הגופן הראשון.

מתי משתמשים: כשרוצים להראות שני קולות של אותו עסק, הרשמי והאישי. מילה אחת בכותרת, באתר של סטודיו, יועץ או חנות בוטיק. כשהגופנים דומים מדי זה לזה אין מה להראות, וכשהשורה מכילה כמה מילים הדגש הולך לאיבוד.

אנחנואנחנואנחנו

גופן סנס: הקול המקצועיגופן סריף: הקול האישי

סטודיו קטן לעיצוב גרפי, ארבעה אנשים ושולחן אחד גדול. כל פרויקט מתחיל בשיחה ארוכה ובכוס קפה.

שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם אותה מילה מתחלפת בין גופן סנס חמור לגופן סריף אישי, וקווי ההנחיה נשארים במקומם. נבנה מאפס מהצפייה. המארקאפ: אותה מילה פעמיים בתא אחד של grid (.gw-a סנס, .gw-b סריף) עם align-items:baseline, כך שקווי הבסיס שלהם נצמדים והשורה לא קופצת. הגופן השני הוא Bellefair, שמוגדר ב-@font-face בקובץ (size-adjust:112%, וגם font-size:.95em כדי שגובה האות יתאים לגופן הראשון). לגופן סריף אחר נבדקים שני דברים: שיש בו עברית, ושגובה האות שלו מתאים לקו העליון (אחרת מכווננים את font-size של .gw-b). קווי ההנחיה הם ארבעה span ב-em מראש תיבת השורה, והם לא נוגעים בטקסט, ולכן לא זזים בהחלפה. הערכים (עליון .30, בסיס .79, תחתון .965) נמדדו בשורה אחת של Google Sans ו-Bellefair יחד. ההחלפה: opacity ו-filter:blur(10px) ב-power2.inOut, ‏.7 שניות, בשני העותקים בכיוונים הפוכים (ובתוויות). כשאין GSAP או שיש הפחתת תנועה, העותק השני נשאר שקוף ובלי טשטוש. הטריגרים: ScrollTrigger עם onToggle כשהמילה בין 62% ל-38% מהמסך (מצב סריף), ו-pointerenter ו-pointerleave של עכבר (מגע לא מחליף). המצב הוא hover || scroll, ולכן אין התנגשות בין השניים. נגישות: ה-h2 מחזיק sr-only עם המילה, והעותקים בעטיפה aria-hidden. אין פקד לחיץ, ולכן אין מה להפעיל במקלדת. RTL: אין תנועה אופקית, ולכן אין מה להחליף.

if (typeof gsap !== "undefined") gsap.registerPlugin(ScrollTrigger);

(function(){
  if(typeof gsap==="undefined")return;      // בלי GSAP מוצג רק הגופן הראשון
  const root=document.querySelector(".gw"),stack=root.querySelector(".gw-stack"),a=root.querySelector(".gw-a"),b=root.querySelector(".gw-b"),ta=root.querySelector(".gw-ta"),tb=root.querySelector(".gw-tb");
  // בהפחתת תנועה הפונקציה לא רצה, והגופן הראשון נשאר
  gsap.matchMedia().add("(prefers-reduced-motion: no-preference)",()=>{
    let toB=false,hov=false,scr=false;
    const apply=()=>{
      const want=hov||scr;if(want===toB)return;toB=want;
      const o={duration:.7,ease:"power2.inOut",overwrite:"auto"};
      gsap.to(a,Object.assign({opacity:want?0:1,filter:"blur("+(want?10:0)+"px)"},o));
      gsap.to(b,Object.assign({opacity:want?1:0,filter:"blur("+(want?0:10)+"px)"},o));
      gsap.to(ta,{opacity:want?0:1,duration:.5,ease:"none",overwrite:"auto"});
      gsap.to(tb,{opacity:want?1:0,duration:.5,ease:"none",overwrite:"auto"});
    };
    const st=ScrollTrigger.create({trigger:stack,start:"top 62%",end:"bottom 38%",onToggle:s=>{scr=s.isActive;apply();}});
    const over=e=>{if(e.pointerType==="mouse"){hov=true;apply();}};
    const out=e=>{if(e.pointerType==="mouse"){hov=false;apply();}};
    stack.addEventListener("pointerenter",over);stack.addEventListener("pointerleave",out);
    return()=>{
      st.kill();stack.removeEventListener("pointerenter",over);stack.removeEventListener("pointerleave",out);
      gsap.killTweensOf([a,b,ta,tb]);gsap.set([a,b,ta,tb],{clearProps:"opacity,filter"});
    };
  });
})();
