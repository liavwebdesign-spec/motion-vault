B86 · טבעת פינג בנקודת הלחיצה

התנהגויות CSS animation · pointer position · vanilla JS טעינה ופידבקhover ומיקרו ממתין

בכל לחיצה על כפתור יוצאת טבעת דקה בדיוק מהנקודה שנלחצה, מתרחבת מעבר לכפתור ודועכת, והכפתור נלחץ פנימה קצת. בלחיצה במקלדת הטבעת יוצאת מהמרכז. שלושה כפתורים בגדלים וצבעים שונים, והטבעת נראית על כל אחד מהם.

מתי משתמשים: כפתורי הקריאה לפעולה הראשיים: קביעת שיחה, שליחה, הזמנה. נותן לכל לחיצה תחושה של משהו שהגיב, בלי להסתיר את הכפתור. לא על כל כפתור בעמוד, ולא על כפתור שמעביר מיד לעמוד אחר.

נקבע שיחה על האתר שלכם

שיחה של עשרים דקות, בטלפון או בזום, בלי התחייבות.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שני סרטוני מוצר ב-prompt-motion.com (21.10.2026) שבהם סמן נוחת על כפתור וטבעת דקה מתרחבת ודועכת מנקודת הלחיצה בזמן שהכפתור נלחץ פנימה. נבנה מאפס מהצפייה. המנגנון: ב-pointerdown נשמרים מיקום הלחיצה (--px, --py, בפיקסלים מתוך הכפתור) ומוסיפים is-ping, והטבעת היא ::after שמונפש ב-CSS בלבד (scale מ-0 ל-3.2 ו-opacity מ-.8 ל-0, 700ms). לפני כל לחיצה מורידים את המחלקה ומכריחים חישוב מחדש כדי שלחיצה מהירה תתחיל טבעת חדשה, ובסוף האנימציה המחלקה יורדת. הטבעת נראית על כל רקע: היא טבעת כפולה, חיצונית בצבע --accent ופנימית בצבע שמוגדר לכפתור (--ring-in), כך שגם על כפתור בצבע המותג וגם על כפתור כהה אחת משתיים נראית. לחיצת מקלדת (Enter או רווח) מגיעה כ-click בלי מיקום, והטבעת יוצאת מהמרכז. הלחיצה פנימה היא :active עם scale .97. הפחתת תנועה: בלי טבעת. הבדיקה היא ברגע הלחיצה, ולכן היא מגיבה גם לשינוי הגדרה באמצע. גלישה: המעטפת היא overflow:clip, כדי שהטבעת לא תרחיב את העמוד לרוחב בטלפון. כפתור אמיתי: <button> (או <a> שמוסיפים לו data-ping), עם focus-visible. בפרויקט: מוסיפים data-ping לכל קריאה לפעולה, וקובעים --ring-in לפי הרקע של הכפתור.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

(function(){
  const reduce=matchMedia("(prefers-reduced-motion: reduce)");
  document.querySelectorAll("[data-ping]").forEach(b=>{
    // הטבעת מתחילה מנקודת הלחיצה. מורידים ומחזירים את המחלקה, כדי שלחיצה מהירה תתחיל טבעת חדשה
    const ping=(x,y)=>{
      if(reduce.matches)return;
      b.style.setProperty("--px",x+"px");b.style.setProperty("--py",y+"px");
      b.classList.remove("is-ping");void b.offsetWidth;b.classList.add("is-ping");
    };
    b.addEventListener("pointerdown",e=>{
      if(e.pointerType==="mouse"&&e.button!==0)return;
      const r=b.getBoundingClientRect();ping(e.clientX-r.left,e.clientY-r.top);
    });
    // Enter או רווח מגיעים כ-click בלי מיקום: הטבעת יוצאת מהמרכז
    b.addEventListener("click",e=>{if(e.detail===0)ping(b.offsetWidth/2,b.offsetHeight/2);});
    b.addEventListener("animationend",e=>{if(e.animationName==="pgRing")b.classList.remove("is-ping");});
  });
})();
