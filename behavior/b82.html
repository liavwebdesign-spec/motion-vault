B82 · כרטיס הולוגרפי עם פסי נתונים

התנהגויות CSS 3D · pointer · mix-blend-mode hover ומיקרוכרטיסיםטעינה ופידבק ממתין

כרטיס חבר צוות במסגרת מוזהבת, והעיצוב מעליו מבריק כמו כרטיס אספנות: שכבת צבעי קשת נעה על פני הכרטיס ומשנה גוון בהתאם לנטייה, והכרטיס עצמו נוטה אחרי העכבר עם הברקה שעוקבת אחריו. כשהכרטיס נכנס למסך שלושה פסי נתונים מתמלאים עד 98, 100 ו-99. במגע הכרטיס מתנדנד לאט מעצמו.

מתי משתמשים: כרטיס צוות, כרטיס עדות מרכזי, הגשת נותן שירות מוביל או מוצר אחד שרוצים שירגיש מיוחד. רק כשיש מה להראות בנתונים (שביעות רצון, ותק, פרויקטים), ורק על כרטיס אחד או שניים בעמוד, כי ברק על כל כרטיס מבטל את האפקט. לא בעמוד רשמי של משרד או קליניקה.

שחזור. השראה: סרטון בגלריה (21.10.2026) של דמות על כרטיס אספנות עם רדיד קשת ופסי נתונים. נבנה מאפס מהצפייה. המארקאפ הוא המצב הסופי: כרטיס שטוח, הרדיד באמצע ופסי הנתונים מלאים (--v על כל פס). בלי JS ובהפחתת תנועה זה מה שרואים. ההברקה: שכבת linear-gradient של צבעי קשת בגודל 260% עם mix-blend-mode:color-dodge, ו-background-position מתוך --fx ו---fy שזזים עם הנטייה. הבסיס חייב להיות בגוון בינוני (אצלנו תערובת של --accent), כי color-dodge על שחור נשאר שחור. שכבה שנייה, soft-light, היא נקודת האור שעוקבת אחרי העכבר. הנטייה: האזנה אחת (pointermove על המסגרת), ערך יעד ערך נוכחי עם החלקה בכל פריים, והלולאה נעצרת כשהכרטיס נח. עד 14 מעלות, והכיוון הוא שהצד שמצביעים עליו מתרחק. perspective יושב על האב ולא על הכרטיס. מגע: touch-action:pan-y כדי שלא לחסום גלילה, ובמכשיר בלי hover ((hover: none)) הכרטיס מתנדנד לבד כשהוא במסך (נעצר מחוץ למסך). הפסים: IntersectionObserver אחד מוסיף is-in, והמילוי הוא transform:scaleX עם טרנזישן מהרשימה והשהיה לכל פס. הפחתת תנועה: בלי נטייה, בלי הנדנוד ובלי מילוי הפסים. התמונה: תמונה אמיתית מהמאגר בחלון הכרטיס (object-fit:cover, ו-object-position שמכוון לפנים). התמונה עצמה יושבת מעל הרדיד של הכרטיס, כי color-dodge על תמונה בהירה שוטף אותה ללבן והיא נראית דהויה. במקומו יש לה שכבת ברק משלה בתוך החלון (overlay עם אותו גרדיאנט ואותם --fx ו---fy), שמוסיפה צבע בלי לשטוף את הפנים. כיווניות: הפסים מתמלאים מימין לשמאל (transform-origin:100%), ובאתר LTR הופכים ל-0%. נגישות: שכבות הרדיד וההברקה והאיור מוסתרים מקורא מסך, והמספרים והשם והתפקיד הם טקסט רגיל.

JS

(function(){
  const root=document.querySelector(".hfc"),card=root.querySelector(".hfc-card");
  const mq=matchMedia("(prefers-reduced-motion: reduce)"),touch=matchMedia("(hover: none)");
  let off=null;
  function enable(){
    root.classList.add("is-prep");
    const io=new IntersectionObserver(es=>{if(es.some(e=>e.isIntersecting)){root.classList.add("is-in");}},{threshold:.4});
    io.observe(card);
    let tx=0,ty=0,cx=0,cy=0,raf=0,held=false,seen=true;
    const MAX=14;
    const apply=()=>{
      card.style.setProperty("--rx",(-cy*2*MAX).toFixed(2)+"deg");
      card.style.setProperty("--ry",(cx*2*MAX).toFixed(2)+"deg");
      card.style.setProperty("--fx",(50+cx*60).toFixed(1)+"%");
      card.style.setProperty("--fy",(50+cy*60).toFixed(1)+"%");
      card.style.setProperty("--gx",(50+cx*100).toFixed(1)+"%");
      card.style.setProperty("--gy",(35+cy*100).toFixed(1)+"%");
    };
    const tick=t=>{
      raf=0;
      if(touch.matches&&!held&&seen){tx=Math.sin(t/1700)*.34;ty=Math.cos(t/2300)*.28;}   // בלי עכבר: נדנוד איטי
      cx+=(tx-cx)*.12;cy+=(ty-cy)*.12;apply();
      if(Math.abs(tx-cx)>.001||Math.abs(ty-cy)>.001||(touch.matches&&seen))raf=requestAnimationFrame(tick);
    };
    const kick=()=>{if(!raf)raf=requestAnimationFrame(tick);};
    const aim=e=>{
      const r=card.getBoundingClientRect();
      tx=Math.max(-.8,Math.min(.8,(e.clientX-r.left)/r.width-.5));
      ty=Math.max(-.8,Math.min(.8,(e.clientY-r.top)/r.height-.5));kick();
    };
    const rest=()=>{held=false;tx=0;ty=0;kick();};
    const down=e=>{held=true;aim(e);};
    const vis=new IntersectionObserver(es=>{seen=es.some(e=>e.isIntersecting);if(seen)kick();});
    vis.observe(card);
    root.addEventListener("pointermove",aim);root.addEventListener("pointerleave",rest);
    root.addEventListener("pointerdown",down);root.addEventListener("pointerup",rest);root.addEventListener("pointercancel",rest);
    kick();
    return()=>{
      io.disconnect();vis.disconnect();cancelAnimationFrame(raf);
      root.removeEventListener("pointermove",aim);root.removeEventListener("pointerleave",rest);
      root.removeEventListener("pointerdown",down);root.removeEventListener("pointerup",rest);root.removeEventListener("pointercancel",rest);
      ["--rx","--ry","--fx","--fy","--gx","--gy"].forEach(p=>card.style.removeProperty(p));
      root.classList.remove("is-prep","is-in");
    };
  }
  const sync=()=>{if(off){off();off=null;}if(!mq.matches)off=enable();};
  mq.addEventListener("change",sync);sync();
})();
