B105 · שירותים שהשם שלהם מתערבל ומתייצב

התנהגויות JS · requestAnimationFrame · IntersectionObserver טקסט וכותרותhover ומיקרו ממתין

רשימה של חמישה שירותים בכותרות גדולות. כשהעכבר עובר על שורה, השם שלה מתערבל לאותיות, ספרות וסימנים אקראיים ומתייצב בחזרה אות אחרי אות, מתחילת המילה, כמו קוד שמתפענח. השורה מקבלת משטח, והחץ זז לכיוון הקריאה.

מתי משתמשים: רשימת שירותים, תחומים או מוצרים של סטודיו, משרד פרסום, חברת תוכנה או סייבר, שבה המילים קצרות (מילה עד שתיים בשורה). משהו אחד כזה בעמוד. לא לטקסט רץ, לא לכותרת ארוכה, ולא באתר שמוכר שקט ואמון, כמו קליניקה, משרד עורכי דין או בית אבות.

הדמו

מה אנחנו עושים

חמישה תחומים וצוות אחד. אפשר להתחיל מאחד מהם, או לעבור את כולם יחד.

  • אסטרטגיה

    מי הלקוח, מה הוא צריך לשמוע ואיפה הוא נמצא. כל פרויקט מתחיל כאן.

  • מיתוג

    שם, לוגו, צבעים ושפה, שנראים אותו דבר בכל מקום שהלקוח פוגש אתכם.

  • אתרים

    אתר תדמית או דף נחיתה שנטען מהר, נקרא בטלפון ומביא פניות.

  • חנויות

    חנות עם סליקה, משלוחים ומלאי, שמנהלים לבד מהיום הראשון.

  • קמפיינים

    מודעות בפייסבוק, באינסטגרם ובגוגל, עם מדידה של כל שקל.

שחזור. השראה: animtd.store (26.10.2026), סרטון של רשימה שהמילים בה מתערבלות במעבר עכבר. נבנה מאפס מהצפייה. הטקסט האמיתי לא נוגע: כל שם נשאר מחרוזת אחת בתוך .cx-t, וה-JS מוסיף מעליו שכבה (.cx-ov, aria-hidden) עם שני חלקים: מה שכבר התייצב, ושאר המילה באותיות אקראיות בצבע ה-accent. בזמן הערבול הטקסט האמיתי שקוף, ולכן הפריסה לא זזה וקורא מסך שומע את המילה הנכונה. זו לא חלוקה לאותיות: אין span לכל אות ואין אנימציה לכל אות (התורה אוסרת לפרק עברית לאותיות), אלא החלפה של מחרוזת שלמה בכל פריים. הקצב: המשך תלוי באורך המילה (320ms ועוד 70ms לאות, עד 950ms), והאותיות מתייצבות מתחילת המילה, כלומר מימין לשמאל בעברית. המאגר האקראי הוא אותיות עבריות, ספרות וכמה סימנים, כדי שזה ייראה כמו פענוח ולא כמו פונט שבור. מגע: אין ריחוף, ולכן כל שורה מתפענחת פעם אחת כשהיא נכנסת למסך (IntersectionObserver). מקלדת: פוקוס על השורה מפעיל את אותו ערבול, ויש טבעת פוקוס ברורה. הפחתת תנועה: אין ערבול בכלל; נשארים המשטח והחץ. באתר משמאל לימין החץ מתהפך והאותיות מתייצבות משמאל, בלי שינוי בקוד. קווים: אין קווי הפרדה בין השורות; ההפרדה היא המרווח והמשטח בריחוף.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

JS

(function(){
  const root=document.querySelector("[data-cx]");
  if(!root)return;
  const still=matchMedia("(prefers-reduced-motion: reduce)");
  const fine=matchMedia("(hover: hover) and (pointer: fine)");
  // אותיות, ספרות וכמה סימנים: נראה כמו קוד שמתפענח ולא כמו פונט שבור
  const POOL="אבגדהוזחטיכלמנסעפצקרשת0123456789#%&*+=?";
  root.querySelectorAll(".cx-w").forEach(w=>{
    const txt=w.querySelector(".cx-t").textContent;
    const ov=document.createElement("span"),done=document.createElement("span"),rest=document.createElement("i");
    ov.className="cx-ov";ov.setAttribute("aria-hidden","true");ov.append(done,rest);w.append(ov);
    const DUR=Math.min(950,320+txt.length*70);
    let raf=0,t0=0;
    const stop=()=>{cancelAnimationFrame(raf);raf=0;w.classList.remove("is-mix");done.textContent="";rest.textContent="";};
    const tick=now=>{
      if(!t0)t0=now;
      const p=Math.min(1,(now-t0)/DUR),n=Math.floor(p*txt.length);   // n אותיות כבר התייצבו, מתחילת המילה
      let r="";
      for(let i=n;i<txt.length;i++)r+=txt[i]===" "?" ":POOL[Math.floor(Math.random()*POOL.length)];
      done.textContent=txt.slice(0,n);rest.textContent=r;
      if(p<1)raf=requestAnimationFrame(tick);else stop();
    };
    w.mix=()=>{if(still.matches)return;cancelAnimationFrame(raf);t0=0;w.classList.add("is-mix");raf=requestAnimationFrame(tick);};
  });
  const rows=[...root.querySelectorAll(".cx-row")];
  rows.forEach(row=>{
    const w=row.querySelector(".cx-w");
    row.addEventListener("pointerenter",()=>{if(fine.matches)w.mix();});
    row.addEventListener("focus",()=>w.mix());
  });
  // במסך מגע אין ריחוף: כל שורה מתפענחת פעם אחת כשהיא נכנסת למסך
  if(!fine.matches&&"IntersectionObserver" in window){
    const io=new IntersectionObserver(es=>es.forEach(e=>{
      if(!e.isIntersecting)return;
      io.unobserve(e.target);
      setTimeout(()=>e.target.querySelector(".cx-w").mix(),120);
    }),{threshold:.6});
    rows.forEach(r=>io.observe(r));
  }
})();
