B112 · טבעת תמונות שמסתובבת עם ההמלצות

התנהגויות CSS 3D · JS כרטיסיםתמונות וגלריותניווט ומבנה ממתין

חמש תמונות מסודרות בטבעת תלת ממדית, אחת מלפנים ושתיים בצדדים, מוטות לאחור. לצדן המלצה של הורה. לחיצה על החץ או על הטבעת מסובבת אותה לתמונה הבאה, וההמלצה מתחלפת: הישנה מיטשטשת ונעלמת, והחדשה מתחדדת למקום. ליד שם הממליץ תווית של סוג הביקור.

מתי משתמשים: המלצות של לקוחות כשלכל המלצה יש תמונה אמיתית (מהמקום, מהפרויקט, מהלקוח): מרפאה, גן, סטודיו, קבלן, צלם. ארבע עד שש המלצות. לא כשאין תמונות, שם רשימה או קיר המלצות, ולא באתר שמוכר רצינות שקטה (משפט, פיננסים), כי הסיבוב בולט.

דנהאמא של עומרמרשם דיגיטלי

שחזור. השראה: animtd.store (26.10.2026), סרטון של טבעת כרטיסים תלת ממדית שמסתובבת עם המלצות. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: b62 הוא קאברפלואו שטוח (כרטיסים בשורה שמוטים), ו-g201 הוא טבעת תמונות שנפתחת לשדה בגלילה. כאן הטבעת היא גליל אמיתי שמסתובב כגוף אחד בלחיצה, ומסונכרן עם ההמלצה. הגיאומטריה: כל תמונה מסובבת ב-rotateY של 360 חלקי מספר התמונות ונדחפת החוצה ב-translateZ של הרדיוס; הטבעת כולה נסוגה באותו רדיוס, כך שהתמונה הקדמית בגודל הטבעי שלה. הרדיוס מחושב מרוחב התמונה ב-CSS (calc), ולכן הטבעת מתכווצת עם המסך. backface-visibility:hidden מסתיר את התמונות שמאחור. הסיבוב: ה-JS סופר צעדים ולא אינדקס, כך שהמעבר מהאחרונה לראשונה ממשיך באותו כיוון ולא חוזר אחורה סיבוב שלם. משך 1.1 שניות בעקומת in-out המאושרת. ההמלצה: כל ההמלצות באותו תא גריד; היוצאת מקבלת טשטוש ושקיפות, והנכנסת מתחדדת אחרי רבע שנייה, כשהטבעת כבר בתנועה. אזור ההמלצות הוא aria-live, כי ההחלפה תמיד ביוזמת המשתמש. מקלדת: כפתורי חץ עם תווית, וחצי המקלדת כשהפוקוס עליהם; הטבעת עצמה aria-hidden, כי היא חוזרת על התמונה. בלי JS כל ההמלצות מוצגות זו מתחת לזו והטבעת עומדת. הפחתת תנועה: בלי סיבוב ובלי טשטוש, רק החלפה. בטלפון הטבעת מעל ההמלצה.

JS

(function(){
  const root=document.querySelector("[data-rq]");
  if(!root)return;
  const ring=root.querySelector(".rq-ring"),ps=[...root.querySelectorAll(".rq-p")],qs=[...root.querySelectorAll(".rq-q")];
  const count=root.querySelector(".rq-count b"),n=qs.length,A=360/n;
  let step=0;
  root.classList.add("is-live");          // בלי JS כל ההמלצות זו מתחת לזו
  root.querySelector(".rq-nav").hidden=false;
  const show=()=>{
    const i=((step%n)+n)%n;
    // סופרים צעדים ולא אינדקס: מהאחרונה לראשונה הטבעת ממשיכה באותו כיוון
    ring.style.setProperty("--rot",(-step*A)+"deg");
    ps.forEach((p,k)=>p.classList.toggle("is-on",k===i));
    qs.forEach((q,k)=>q.classList.toggle("is-on",k===i));
    count.textContent=i+1;
  };
  const move=d=>{step+=d;show();};
  root.querySelector(".rq-prev").addEventListener("click",()=>move(-1));
  root.querySelector(".rq-next").addEventListener("click",()=>move(1));
  root.querySelector(".rq-scene").addEventListener("click",()=>move(1));
  root.querySelector(".rq-nav").addEventListener("keydown",e=>{
    if(e.key!=="ArrowLeft"&&e.key!=="ArrowRight")return;
    e.preventDefault();
    const rtl=getComputedStyle(root).direction==="rtl";
    move((e.key==="ArrowLeft")===rtl?1:-1);   // החץ שמצביע לכיוון הקריאה הוא «הבא»
  });
})();
