B110 · מניפת תמונות שמתחלפת עם הרשימה

התנהגויות CSS transform · JS כרטיסיםתמונות וגלריותhover ומיקרו ממתין

חמש תמונות מונחות כמו מניפה של תמונות מודפסות, אחת מלפנים והשאר מוטות מאחוריה. לידן רשימת ימי המסלול. לחיצה על יום, על «היום הבא» או על התמונה זורקת את התמונה הקדמית הצידה ומחזירה אותה לסוף הערימה, והתמונה של היום הנבחר עולה קדימה. ברשימה היום הנבחר מסומן ונפתח עם משפט עליו.

מתי משתמשים: מסלול, תוכנית או סיפור בכמה תחנות שיש לכל אחת תמונה: טיול, סדנה בכמה מפגשים, תפריט טעימות, שלבי בנייה, פרויקט לפני ואחרי. ארבע עד שש תחנות. לא כשהתמונות לא באותו יחס וסגנון, ולא כשחשוב לראות את כל התמונות במלואן בבת אחת, שם גריד.

הדמו

חמישה ימים, ואן אחד, כל החוף

ככה נראה המסלול הכי מוזמן שלנו. בחרו יום כדי לראות איפה עוצרים.

שחזור. השראה: animtd.store (26.10.2026), סרטון של ערימת תמונות במניפה שמתחלפת בלחיצה. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: g67 מחליף כרטיסים בגלילה, ו-h22 הוא הירו עם ערימה. כאן ההחלפה בלחיצה, והערימה מסונכרנת עם רשימה שהיא הניווט האמיתי. המניפה: לכל תמונה data-k (0 מלפנים עד 4 מאחור), ולכל ערך יש תנוחה קבועה ב-CSS: הזזה, הטיה והקטנה הולכות וגדלות, ובהירות יורדת מעט. החלפה היא רק החלפת ערכי data-k, וה-transition (0.7 שנייה, in-out מאושר) עושה את התנועה. הזריקה: התמונה הקדמית מקבלת קודם is-throw (עפה הצידה מעל כולם, 0.38 שנייה), ורק אז עוברת לסוף הערימה, כדי שלא תיעלם מאחורי האחרות באמצע הדרך. לחיצות מהירות סוגרות את הזריקה הקודמת לפני החדשה. הרשימה: כפתורים עם aria-pressed, בתוך ol כי הימים הם סדר אמיתי; היום הנבחר נפתח ב-grid-template-rows, ושם היום מוכרז ב-aria-live. הפחתת תנועה: בלי זריקה ובלי מעבר, התמונה פשוט מתחלפת. בטלפון הערימה מעל הרשימה, קטנה יותר. בלי JS המניפה מוצגת כמו שהיא, והיום הראשון פתוח.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית.

(function(){
  const root=document.querySelector("[data-cf]");
  if(!root)return;
  const cards=[...root.querySelectorAll(".cf-card")],its=[...root.querySelectorAll(".cf-it")],live=root.querySelector(".cf-sr");
  const still=matchMedia("(prefers-reduced-motion: reduce)");
  const n=cards.length;let cur=0,pend=null;
  // זריקה שעוד באוויר נסגרת לפני הבאה: התמונה נוחתת במקום שלה לפי הסדר הנוכחי
  const settle=()=>{if(!pend)return;clearTimeout(pend.t);pend.c.classList.remove("is-throw");pend.c.dataset.k=(cards.indexOf(pend.c)-cur+n)%n;pend=null;};
  const go=j=>{
    j=(j+n)%n;if(j===cur)return;
    settle();
    const out=cards[cur];cur=j;
    cards.forEach((c,i)=>{if(c!==out||still.matches)c.dataset.k=(i-j+n)%n;});
    if(!still.matches){out.classList.add("is-throw");pend={c:out,t:setTimeout(settle,380)};}
    its.forEach((b,i)=>b.setAttribute("aria-pressed",String(i===j)));
    live.textContent=its[j].querySelector(".cf-day").textContent+": "+its[j].querySelector("b").textContent;
  };
  its.forEach((b,i)=>b.addEventListener("click",()=>go(i)));
  root.querySelector(".cf-next").addEventListener("click",()=>go(cur+1));
  root.querySelector(".cf-deck").addEventListener("click",()=>go(cur+1));
})();
