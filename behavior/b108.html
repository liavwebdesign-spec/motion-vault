B108 · תפריט וילון עם תמונה לכל קישור

התנהגויות CSS clip-path · JS · inert ניווט ומבנהתמונות וגלריותhover ומיקרו ממתין

לחיצה על «תפריט» מורידה מלמעלה וילון כהה על כל המסך. הקישורים נכנסים בזה אחר זה בכתב גדול, ולידם תמונה. כשהעכבר עובר על קישור, השאר מתעמעמים, ליד הקישור מופיע חץ, והתמונה מתחלפת לפרויקט שמייצג אותו, עם כיתוב קצר.

מתי משתמשים: אתר תדמית שהעבודות הן המוצר: אדריכלות, עיצוב פנים, צילום, סטודיו, מלון בוטיק. תפריט של ארבעה עד שישה קישורים. לא לאתר עם תפריט עמוק או חנות עם קטגוריות, שם מגה תפריט, ולא כשאין תמונה טובה לכל קישור. בטלפון התמונה יורדת ונשאר התפריט הגדול.

שחזור. השראה: animtd.store (26.10.2026), סרטון של תפריט מסך מלא שבו כל קישור מחליף תמונה בצד. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: b23 הוא תפריט מסך מלא בלי תמונות, ו-g43 הוא תמונה שעוקבת אחרי הסמן ברשימה. כאן התמונה קבועה במקום, ומתחלפת לפי הקישור. הווילון: clip-path:inset(0 0 100% 0) שנפתח ל-inset(0) ב-0.8 שנייה בעקומת in-out המאושרת (מעבר על פני המסך), ואחריו הקישורים עולים בדירוג של 60ms. התמונה: כל התמונות בערימה אחת, והפעילה מקבלת is-on (שקיפות והקטנה קלה מ-1.06). המיקוד: כשהתפריט פתוח, העמוד וההדר מקבלים inert (מלכודת פוקוס בלי ספרייה), הפוקוס עובר לקישור הראשון, Escape וכפתור הסגירה מחזירים אותו לכפתור «תפריט». פוקוס מהמקלדת על קישור עושה את מה שעושה הריחוף. בדמו הווילון מכסה את הבמה (position:absolute); בפרויקט הוא position:fixed, וה-inert עובר לכל מה שמחוץ לתפריט, כולל main והפוטר. בלי JS התפריט מוצג כרשימה רגילה מתחת להדר. הפחתת תנועה: בלי וילון ובלי עלייה, התפריט מופיע בעמעום קצר. באתר משמאל לימין החץ מתהפך.

(function(){
  const root=document.querySelector("[data-mc]");
  if(!root)return;
  const tg=root.querySelector(".mc-tg"),x=root.querySelector(".mc-x"),panel=root.querySelector(".mc-p");
  const page=root.querySelector(".mc-page"),bar=root.querySelector(".mc-bar"),list=panel.querySelector(".mc-links");
  const links=[...list.querySelectorAll("a")],imgs=[...panel.querySelectorAll(".mc-pv img")],caps=[...panel.querySelectorAll(".mc-pv figcaption span")];
  root.classList.add("is-live");panel.inert=true;     // בלי JS התפריט הוא רשימה רגילה מתחת להדר
  const show=i=>[links,imgs,caps].forEach(g=>g.forEach((el,k)=>el.classList.toggle("is-on",k===i)));
  const set=on=>{
    root.classList.toggle("is-open",on);
    panel.inert=!on;page.inert=on;bar.inert=on;       // מלכודת פוקוס: מה שמאחורי הווילון לא נגיש
    tg.setAttribute("aria-expanded",String(on));
    if(on){show(0);list.classList.remove("is-dim");setTimeout(()=>links[0].focus({preventScroll:true}),80);}
    else tg.focus({preventScroll:true});
  };
  tg.addEventListener("click",()=>set(true));
  x.addEventListener("click",()=>set(false));
  root.addEventListener("keydown",e=>{if(e.key==="Escape"&&root.classList.contains("is-open"))set(false);});
  links.forEach((a,i)=>{
    a.addEventListener("pointerenter",()=>{show(i);list.classList.add("is-dim");});
    a.addEventListener("focus",()=>{if(a.matches(":focus-visible")){show(i);list.classList.add("is-dim");}});
    a.addEventListener("click",e=>{e.preventDefault();set(false);});   // בדמו; בפרויקט הקישור מנווט
  });
  list.addEventListener("pointerleave",()=>list.classList.remove("is-dim"));
})();
