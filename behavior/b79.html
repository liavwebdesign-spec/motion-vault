B79 · עיגולים שמתמזגים כמו טיפות

התנהגויות SVG goo filter · pointer · rAF רקע ואווירההירו ופתיחה ממתין

עיגולי צבע מקיפים כתם מרכזי, מתמזגים אליו ונפרדים ממנו כמו טיפות נוזל עם גשר דק ביניהם, ועיגול אחד יוצא מהמסלול ועוקב אחרי הסמן. המיזוג נעשה בפילטר SVG אחד (טשטוש ומטריצת צבע שמחדדת את הקצה), לכן אין חישוב צורה: רק מזיזים עיגולים. בתוך הכתם יושב סמל המותג, והוא לא זז.

מתי משתמשים: אלמנט עיצובי ליד כותרת הירו של מותג שהמוצר שלו רך או נוזלי: קוסמטיקה, מזון, משקאות, בריאות, מוצרי תינוקות. מתאים גם למצב טעינה או למסך ריק. לא לאתר של בנק או של חברת ביטוח, ולא כרקע מאחורי פסקה (המיזוג מושך את העין מהטקסט).

מעבדת טל, חיפה

קוסמטיקה שמכינים בכמויות קטנות

סבונים, קרמים וסרומים מהמעבדה שלנו. על כל אריזה רשימת מרכיבים קצרה, וכל אצווה נבדקת לפני שהיא יוצאת.

לצפייה במוצרים
שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) שבהם עיגולים מתמזגים ונפרדים כמו טיפות עם גשר נוזלי ביניהם, ואחד מהם עוקב אחרי הסמן. נבנה מאפס מהצפייה. המנגנון: קבוצת עיגולים בצבע אחד עם filter של feGaussianBlur (stdDeviation 14) ואחריו feColorMatrix שמכפיל את ערוץ האלפא ב-22 ומחסר 9. כל מה שאחרי הטשטוש מעל סף מסוים הופך לאטום והשאר נעלם, ובין שני עיגולים קרובים נוצר גשר. התזוזה: style.transform = translate(...) על כל עיגול, והערכים נגזרים מ-cx ו-cy שבסימון. לכן בלי JS ובהפחתת תנועה רואים את הכתם הממוזג עצמו, ולא מצב ריק. כל עיגול מקיף את המרכז לפי זווית ורדיוס שמתחילים בדיוק במיקום שבסימון, והרדיוס נושם לאט, כך שעיגולים נוגעים בכתם ומתנתקים ממנו לסירוגין. הסמן: העיגול האחרון נמשך לנקודת הסמן במערכת הקואורדינטות של ה-viewBox עם החלקה (.12). המשקל שלו עולה בכניסה לסקשן ויורד ביציאה, והוא חוזר למסלול לבד. לא מגיב למגע. ביצועים: שמונה עיגולים, פילטר על אזור של 480 על 480, והלולאה רצה רק כשהסקשן על המסך ובלשונית פעילה. הפילטר יקר כשהקבוצה גדולה מאוד, ולכן לא למלא בו מסך שלם. מזהה הפילטר (goo-b79) חייב להיות ייחודי בעמוד. RTL: הפריסה בגריד לפי כיוון המסמך והאיור בצד הסוף. אין מה להפוך באתר LTR. הטבעות המנוקדות מסתובבות ב-CSS בלבד, ומושהות כשהסקשן מחוץ למסך.

(function(){
  const root=document.querySelector(".gm"),art=root.querySelector(".gm-art"),svg=art.querySelector("svg");
  const cs=[].slice.call(art.querySelectorAll(".gm-c")),N=cs.length,C=200;
  const reduced=matchMedia("(prefers-reduced-motion: reduce)").matches;
  // כל עיגול זוכר את המיקום שלו בסימון: זווית ורדיוס סביב המרכז, וההזזה היא ההפרש ממנו
  const P=cs.map((c,i)=>{const x=+c.getAttribute("cx"),y=+c.getAttribute("cy");
    return{c,x,y,a:Math.atan2(y-C,x-C),r:Math.hypot(x-C,y-C),sp:(i%2?1:-1)*(.16+.05*i),ra:10+4*(i%3),rf:.5+.17*i};});
  const M={x:P[N-1].x,y:P[N-1].y,tx:P[N-1].x,ty:P[N-1].y,w:0,on:false},t0=performance.now();
  let running=false,vis=false,raf=0;
  function frame(now){
    const t=(now-t0)/1000;
    for(let i=1;i<N;i++){
      const p=P[i],rad=p.r+p.ra*Math.sin(t*p.rf),ang=p.a+t*p.sp;
      let x=C+rad*Math.cos(ang),y=C+rad*Math.sin(ang);
      if(i===N-1){   // העיגול האחרון נמשך אל הסמן, ובלי סמן חוזר למסלול
        M.x+=(M.tx-M.x)*.12;M.y+=(M.ty-M.y)*.12;M.w+=((M.on?1:0)-M.w)*.08;
        x+=(M.x-x)*M.w;y+=(M.y-y)*M.w;
      }
      p.c.style.transform="translate("+(x-p.x).toFixed(2)+"px,"+(y-p.y).toFixed(2)+"px)";
    }
    P[0].c.style.transform="scale("+(1+.035*Math.sin(t*1.1)).toFixed(3)+")";
    raf=running?requestAnimationFrame(frame):0;
  }
  function sync(){running=!reduced&&vis&&!document.hidden;if(running&&!raf)raf=requestAnimationFrame(frame);}
  root.addEventListener("pointermove",e=>{
    if(e.pointerType==="touch")return;
    const r=svg.getBoundingClientRect();if(!r.width)return;
    M.tx=Math.max(40,Math.min(360,(e.clientX-r.left)/r.width*400));
    M.ty=Math.max(40,Math.min(360,(e.clientY-r.top)/r.height*400));
    M.on=true;
  });
  root.addEventListener("pointerleave",()=>{M.on=false;});
  document.addEventListener("visibilitychange",sync);
  new IntersectionObserver(es=>{vis=es[0].isIntersecting;if(vis&&!reduced)art.classList.add("is-on");sync();},{rootMargin:"10% 0px"}).observe(root);
})();
