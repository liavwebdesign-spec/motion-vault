B80 · פאנל סליידרים שמשנה את התמונה

התנהגויות input range · CSS filter · GSAP quickTo תמונות וגלריותטעינה ופידבק ממתין

פאנל זכוכית עם שלושה מחוונים אמיתיים ליד תמונה גדולה. המחוון הראשון בוחר בין ארבעה גוונים של אותו חדר ומערבב ביניהם בהדרגה, השני מכוון את העוצמה, והשלישי מסיט את גוון הזוהר שמאחורי התמונה. הפאנל הוא הסיבה והתמונה התוצאה, ובכניסה למסך המחוונים זזים פעם אחת מעצמם כדי להראות את הקשר, עד שהמבקר נוגע בהם.

מתי משתמשים: אזור שבו המבקר מכוון תוצאה לפני שהוא פונה: גוון תאורה בחדר, גימור של חומר, רמת חבילה, תצוגה מקדימה של עיצוב. מתאים כשיש שניים או שלושה פרמטרים שבאמת משנים את התמונה, ושלוש עד חמש גרסאות מוכנות שלה. לא כשיש רק אפשרות אחת, ולא בעמוד שהכפתור הראשי שלו צריך להישאר במרכז.

טבעי
100%
שחזור. השראה: כמה סרטוני מוצר ב-prompt-motion.com (21.10.2026), שבהם פאנל זכוכית עם מחוונים משנה תמונה שלידו. נבנה מאפס מהצפייה. הפקדים אמיתיים: שלושה input type=range עם label ו-output, ולכן מקלדת, מגע וקורא מסך עובדים בלי קוד נוסף (חצים, Home, End, PageUp). aria-valuetext נותן למחוון הראשון את שם הגוון במקום מספר, ויש טבעת פוקוס גם על המחוון וגם על הידית (:focus-visible). התמונה: ארבעה עותקים של אותה תמונה בשכבות, כל אחד עם גוון משלו ב-filter. המחוון הראשון הוא מיקום בין הגרסאות: השכבה k נראית לפי clamp(pos - k, 0, 1), ולכן בין שתי גרסאות יש מיזוג רציף ולא קפיצה. כדי לעבוד עם צילומים אמיתיים מחליפים את ה-src של כל שכבה ומוחקים את ה-filter שלה. העוצמה והזוהר: העוצמה היא brightness על כל המקבץ, והזוהר הוא הילה מטושטשת בצבע המותג שמסתובבת ב-hue-rotate. שניהם נקבעים ממשתני CSS (--br, --hue) על הסקשן, ולכן אין חישוב צבע ב-JS. ההחלקה: gsap.quickTo מוביל את המצב המוצג אחרי הערך האמיתי של הפקד, בלי השהיה בתוויות. בלי GSAP או בהפחתת תנועה כל שינוי מתבצע מיד, והפקדים ממשיכים לעבוד. הדגמה עצמית: בכניסה למסך המחוונים זזים פעם אחת וחוזרים לברירת המחדל, ומפסיקים ברגע שנוגעים בפאנל. ברירת המחדל במארקאפ: הגוון הטבעי, עוצמה 100 אחוז. מילוי המסלול: גרדיאנט שמתחיל מהצד שבו הערך הנמוך, ולכן הוא מתהפך לבד באתר משמאל לימין (:dir(ltr)). את הפאנל מיישרים לצד ההתחלה עם inset-inline-start. בטלפון הפאנל יורד מתחת לתמונה.

JS

(function(){
  const root=document.querySelector(".sp");
  if(!root)return;
  const L=[...root.querySelectorAll(".sp-img")].slice(1);                 // השכבות שמתמזגות מעל הבסיס
  const inV=root.querySelector("#sp-v"),inB=root.querySelector("#sp-b"),inG=root.querySelector("#sp-g");
  const outV=root.querySelector("#sp-vo"),outB=root.querySelector("#sp-bo");
  const NAMES=["קריר","טבעי","חם","ערב"];
  const reduce=matchMedia("(prefers-reduced-motion: reduce)");
  const smooth=typeof gsap!=="undefined"&&!reduce.matches;               // בלי GSAP או בהפחתת תנועה: שינוי מיידי
  const cur={v:+inV.value,b:+inB.value,g:+inG.value};                     // המצב המוצג (מוחלק), לא הערך של הפקד
  function paint(){
    L.forEach((im,k)=>{im.style.opacity=Math.max(0,Math.min(1,cur.v/100-k)).toFixed(3);});
    root.style.setProperty("--br",(.6+cur.b/100*.8).toFixed(3));
    root.style.setProperty("--hue",((cur.g-50)*2.4).toFixed(1)+"deg");
  }
  function labels(){                                                      // תוויות ומילוי מסלול: תמיד מהערך האמיתי, בלי השהיה
    const i=Math.max(0,Math.min(NAMES.length-1,Math.round(inV.value/100)));
    outV.textContent=NAMES[i];inV.setAttribute("aria-valuetext",NAMES[i]);
    outB.textContent=Math.round((.6+inB.value/100*.8)*100)+"%";
    [inV,inB,inG].forEach(el=>el.style.setProperty("--p",((el.value-el.min)/(el.max-el.min)*100).toFixed(2)+"%"));
  }
  const q=smooth?{
    "sp-v":gsap.quickTo(cur,"v",{duration:.55,ease:"power3.out",onUpdate:paint}),
    "sp-b":gsap.quickTo(cur,"b",{duration:.55,ease:"power3.out",onUpdate:paint}),
    "sp-g":gsap.quickTo(cur,"g",{duration:.55,ease:"power3.out",onUpdate:paint})}:null;
  const key={"sp-v":"v","sp-b":"b","sp-g":"g"};
  let intro=null;
  const stopIntro=()=>{if(intro){intro.kill();intro=null;}};
  root.addEventListener("input",e=>{
    stopIntro();labels();
    const k=key[e.target.id];if(!k)return;
    if(q)q[e.target.id](+e.target.value);else{cur[k]=+e.target.value;paint();}
  });
  ["pointerdown","keydown"].forEach(t=>root.addEventListener(t,stopIntro));
  // הדגמה עצמית אחת בכניסה למסך: המחוונים זזים וחוזרים לברירת המחדל, ונעצרים בנגיעה הראשונה
  if(smooth&&"IntersectionObserver" in window){
    const sync=()=>{inV.value=cur.v;inB.value=cur.b;inG.value=cur.g;labels();paint();};
    const io=new IntersectionObserver(es=>{
      if(!es.some(e=>e.isIntersecting))return;
      io.disconnect();
      intro=gsap.timeline({delay:.5,onUpdate:sync,onComplete:()=>{intro=null;}})
        .to(cur,{v:250,duration:1.4,ease:"power2.inOut"})
        .to(cur,{b:78,g:82,duration:1,ease:"power2.inOut"},"-=.4")
        .to(cur,{v:100,b:50,g:50,duration:1.3,ease:"power2.inOut"},"+=.3");
    },{threshold:.45});
    io.observe(root);
  }
  labels();paint();
})();
