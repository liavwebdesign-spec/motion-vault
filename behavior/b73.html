B73 · קיר לוגואים שמתחלפים במקום

התנהגויות vanilla JS · WAAPI · CSS mask מספרים ונתוניםרקע ואווירה ממתין

שורה של שש משבצות לוגו, וכל משבצת מחליפה את הלוגו שלה בעצמה כל כמה שניות: הישן עולה למעלה ונמוג בטשטוש קל, החדש עולה מלמטה ומתחדד. המשבצות לא מתחלפות יחד, ובסך הכל מופיעים 18 לקוחות בשטח של שישה.

מתי משתמשים: רצועת לקוחות בעמוד הבית או בדף שירות, כשיש יותר לקוחות ממה שנכנס בשורה אחת (12 עד 18 לוגואים) והשטח מוגבל. מתחת לכותרת הראשית או לפני הצעת המחיר. לא כשיש עד שישה לקוחות: אז מציגים את כולם, בלי תנועה.

מקור: Cult UI logo-carousel (nolly-studio, https://github.com/nolly-studio/cult-ui, registry/default/ui/logo-carousel.tsx) ו-Eldora UI logo-cloud-04 (karthikmudunuri, https://github.com/karthikmudunuri/eldoraui, apps/www/registry/blocks/logo-cloud-04), רישיון MIT לשניהם. נכתב מחדש בלי React ובלי Framer Motion: שעון גלובלי אחד, Web Animations, ולוגואים כמסכת CSS. הבדל מהמהלכים הקרובים: b01 (מרקי) מזיז את כל הרצועה אופקית, ולכן לא ניתן לקרוא לוגו אחד בלי שהוא בורח; lm10 הוא קראפט של צבע, בלי תנועה, ושישה לוגואים בלבד. כאן אין תנועה אופקית, כל לוגו עומד במקום ונקרא בנחת, ושש משבצות מכילות 18 לקוחות: יותר לוגואים בפחות שטח. איך זה עובד: 18 הלוגואים מחולקים round-robin בין שש המשבצות (משבצת 1 מקבלת את הלוגו 1, 7 ו-13). שעון אחד על requestAnimationFrame, שמתקדם רק כשהוא באמת צריך: מחוץ למסך (IntersectionObserver), בלשונית מוסתרת, בריחוף עכבר ובפוקוס מקלדת, ועל כפתור ההשהיה הוא עוצר לגמרי. כל משבצת מתחלפת כל 2.4 שניות, בהיסט של 260ms מהשכנה, בסדר שאינו סריקה משמאל לימין, כך שאף פעם שתי משבצות לא מתחלפות יחד. ההחלפה: הישן עולה 40% למעלה ונמוג בטשטוש של 4px תוך 0.3 שנייה, החדש עולה מלמטה ומתחדד תוך 0.4 שנייה. הלוגואים הם מסכה (mask) בצבע currentColor, כמו ב-lm10, ולכן הם לובשים את צבע הטקסט של הסקשן בכל עור, כהה או בהיר, בלי קובץ נוסף. בדמו הם SVG מוטבע; בפרויקט אמיתי url("logos/x.svg"), כל לוגו בכל צבע. נגישות: כפתור השהיה גלוי (WCAG 2.2.2, תוכן שמתחלף מעבר לחמש שניות), 44 פיקסלים. רשימה מוסתרת-חזותית עם שמות כל הלקוחות לקורא מסך, והמשבצות עצמן aria-hidden, כך שקורא המסך לא שומע החלפות. בתנועה מופחתת אין החלפה: הרשת הסטטית מציגה את כל 18 הלוגואים (שלוש שורות של שישה, ובמובייל שש שורות של שלושה) והכפתור נעלם. בלי JS: הלוגו הראשון של כל משבצת גלוי. חזרה מלשונית מוסתרת לא יוצרת פרץ החלפות, כי הזמן בין פריימים נחתך. בטלפון: שלוש עמודות על שתי שורות. שלוש תשובות הגימור. (1) הגרסה הבנאלית: רצועת לוגואים נעה בלופ, שבה שום לוגו לא נקרא; כאן כל לוגו עומד במקום ומתחלף בנחת, ואין צורך בחצי רצועה כדי להראות 18 לקוחות. (2) מה נושא המשכיות: היציאה למעלה והכניסה מלמטה באותו כיוון, וההיסט בין המשבצות, שנותנים תחושה של גלגל אחד ולא של שש נורות מהבהבות; והכפתור שמראה מצב. (3) איזו פעולה זה מקל: לקרוא את הלוגו שמעניין אותך בלי לרדוף אחריו, ולעצור את הכל כשצריך.

(function(){
  const wrap=document.querySelector(".cw"),wall=wrap.querySelector(".cw-wall"),cells=[...wall.children],btn=wrap.querySelector(".cw-pause");
  const rq=matchMedia("(prefers-reduced-motion: reduce)"),hasA=typeof Element.prototype.animate==="function";
  const E="cubic-bezier(.2,.6,.2,1)",PERIOD=2400,STEP=260,FIRST=700;   // כל משבצת כל 2.4 שניות, בהיסט של 260ms מהשכנה
  // סדר ההחלפה: לא סריקה מצד לצד. מיון לפי השבר של i*0.618 נותן סדר מפוזר לכל מספר משבצות
  const seq=cells.map((_,i)=>i).sort((a,b)=>((a*.618)%1)-((b*.618)%1));
  const st=cells.map(c=>({logos:[...c.querySelectorAll(".cw-logo")],cur:0,at:0}));
  seq.forEach((ci,k)=>{st[ci].at=FIRST+k*STEP;});
  let elapsed=0,last=0,raf=0,inView=false,userPaused=false,hov=false,foc=false;
  // השעון רץ רק כשכל התנאים מתקיימים. אחרת requestAnimationFrame לא נקרא בכלל, ואין עבודה ברקע
  const live=()=>hasA&&!rq.matches&&inView&&!document.hidden&&!userPaused&&!hov&&!foc;
  function swap(s){
    if(s.logos.length<2)return;
    const old=s.logos[s.cur];s.cur=(s.cur+1)%s.logos.length;const nu=s.logos[s.cur];
    nu.classList.add("is-on");
    // הישן: למעלה, נמוג, מיטשטש. החדש: מלמטה, מתחדד. אחרי שהישן יצא מסירים ממנו את המצב
    const out=old.animate([{transform:"translateY(0%)",opacity:1,filter:"blur(0px)"},{transform:"translateY(-40%)",opacity:0,filter:"blur(4px)"}],{duration:300,easing:E,fill:"forwards"});
    out.finished.then(()=>{old.classList.remove("is-on");out.cancel();}).catch(()=>{});
    nu.animate([{transform:"translateY(40%)",opacity:0,filter:"blur(4px)"},{transform:"translateY(0%)",opacity:1,filter:"blur(0px)"}],{duration:400,delay:60,easing:E,fill:"backwards"});
  }
  function tick(t){
    raf=0;
    if(!live())return;
    // חיתוך של הפרש הזמן: חזרה מלשונית מוסתרת או מהשהיה לא יוצרת פרץ החלפות
    const dt=last?Math.min(t-last,100):16;last=t;elapsed+=dt;
    st.forEach(s=>{if(elapsed>=s.at){swap(s);s.at+=PERIOD;}});
    raf=requestAnimationFrame(tick);
  }
  function update(){
    if(live()){if(!raf){last=0;raf=requestAnimationFrame(tick);}}
    else{if(raf){cancelAnimationFrame(raf);raf=0;}last=0;}
  }
  if("IntersectionObserver" in window)new IntersectionObserver(es=>{inView=es[es.length-1].isIntersecting;update();},{threshold:.2}).observe(wall);
  else{inView=true;update();}
  document.addEventListener("visibilitychange",update);
  rq.addEventListener("change",update);
  // ריחוף עכבר ופוקוס מקלדת עוצרים זמנית (לא מגע: אצבע שנחה על המסך אינה "ריחוף")
  wrap.addEventListener("pointerenter",e=>{if(e.pointerType==="mouse"){hov=true;update();}});
  wrap.addEventListener("pointerleave",()=>{hov=false;update();});
  wrap.addEventListener("focusin",()=>{if(wrap.querySelector(":focus-visible")){foc=true;update();}});
  wrap.addEventListener("focusout",e=>{if(!wrap.contains(e.relatedTarget)){foc=false;update();}});
  btn.addEventListener("click",()=>{
    userPaused=!userPaused;foc=false;                // לחיצה מפורשת גוברת על העצירה האוטומטית של הפוקוס
    btn.classList.toggle("is-paused",userPaused);
    btn.setAttribute("aria-label",userPaused?"הפעלת החלפת הלוגואים":"השהיית החלפת הלוגואים");
    update();
  });
})();
