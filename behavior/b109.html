B109 · שובל תמונות אחרי העכבר

התנהגויות JS · Web Animations API · pointer תמונות וגלריותhover ומיקרורקע ואווירה ממתין

כותרת גדולה באמצע הסקשן, ומסביבה קולאז' שקט של חמש תמונות מהאירוע. כשמזיזים את העכבר, הקולאז' מתפנה והעכבר משאיר אחריו שובל של תמונות: כל אחת קופצת במקום, מוטה מעט, ונעלמת אחרי רגע. בטלפון כל הקשה משאירה תמונה אחת.

מתי משתמשים: אירוע, פסטיבל, מסעדה, סטודיו לצילום או מותג אופנה, שיש לו הרבה תמונות טובות ומעט מה להסביר. סקשן אחד בעמוד, בדרך כלל הפתיח או ההזמנה לאירוע. לא לעסק שמוכר שקט ומקצועיות (רפואה, משפט, פיננסים), ולא כשהתמונות לא באותה איכות, כי השובל מראה את כולן מהר.

פסטיבל הדרום, באמצע אוגוסט

שלושה ימים של מוזיקה, אוכל ואנשים

עשרים הופעות על שתי במות, שוק אוכל של שפים מהאזור ומתחם לילדים שפתוח עד השקיעה. הכרטיס לשלושת הימים כולל חניה והסעה.

לכרטיסים
שחזור. השראה: animtd.store (26.10.2026), סרטון של שובל תמונות שנשאר אחרי הסמן. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: g43 הוא תמונה אחת שעוקבת אחרי הסמן ברשימה. כאן הסמן משאיר עקבות של הרבה תמונות, בכל השטח. המנגנון: מאגר של עשר תמונות שנוצרות פעם אחת וממוחזרות בסבב. pointermove נקרא פעם אחת לפריים, ותמונה חדשה נולדת רק כשהסמן זז יותר מ-90px מהקודמת, כדי שהשובל יהיה קצב ולא מריחה. כל תמונה מונפשת ב-Web Animations API: קופצת מ-0.6 ל-1, מוטה עד 8 מעלות לכל צד, נשארת רגע ונעלמת תוך 1.3 שניות בעקומה החתומה. הקריאות: השובל מתחת לטקסט (z-index), והטקסט על הרקע עצמו. במנוחה רואים קולאז' סטטי של חמש תמונות בפינות, כך שהסקשן לא ריק לפני שמזיזים את העכבר; בתזוזה הוא מתעמעם, ושנייה אחרי שהעכבר יוצא הוא חוזר. מגע: הקשה על שטח פנוי משאירה תמונה אחת במקום ההקשה, וגלילה רגילה לא נפגעת. הפחתת תנועה: אין שובל; נשאר הקולאז'. התמונות aria-hidden עם alt ריק, כי הן אווירה והתוכן הוא הכותרת והטקסט.

JS

(function(){
  const root=document.querySelector("[data-it]");
  if(!root||matchMedia("(prefers-reduced-motion: reduce)").matches)return;   // נשאר הקולאז' הסטטי
  const trail=root.querySelector(".it-trail");
  const SRC=[...root.querySelectorAll(".it-col img")].map(i=>i.getAttribute("src"));
  const pool=[];
  for(let k=0;k<10;k++){const m=new Image();m.alt="";m.decoding="async";m.src=SRC[k%SRC.length];trail.append(m);pool.push(m);}
  let n=0,lx=null,ly=0,px=0,py=0,raf=0,back=0;
  const spawn=(x,y)=>{
    const m=pool[n++%pool.length],r=(Math.random()*16-8).toFixed(1);
    const at=s=>"translate(-50%,-50%) scale("+s+") rotate("+r+"deg)";
    m.style.left=x+"px";m.style.top=y+"px";
    m.getAnimations().forEach(a=>a.cancel());
    m.animate([{opacity:0,transform:at(.6)},{opacity:1,transform:at(1),offset:.2},{opacity:1,transform:at(1),offset:.55},{opacity:0,transform:at(.82)}],
      {duration:1300,easing:"cubic-bezier(.2,.6,.2,1)",fill:"both"});
  };
  const local=e=>{const b=root.getBoundingClientRect();return[e.clientX-b.left,e.clientY-b.top];};
  const frame=()=>{raf=0;if(lx===null||Math.hypot(px-lx,py-ly)>90){lx=px;ly=py;spawn(px,py);}};
  root.addEventListener("pointermove",e=>{
    if(e.pointerType&&e.pointerType!=="mouse")return;
    [px,py]=local(e);
    clearTimeout(back);root.classList.add("is-moving");     // הקולאז' מפנה מקום לשובל
    if(!raf)raf=requestAnimationFrame(frame);
  });
  root.addEventListener("pointerleave",()=>{lx=null;clearTimeout(back);back=setTimeout(()=>root.classList.remove("is-moving"),1000);});
  // מגע: כל הקשה על שטח פנוי משאירה תמונה אחת
  root.addEventListener("pointerdown",e=>{
    if(!e.pointerType||e.pointerType==="mouse"||e.target.closest("a,button"))return;
    const [x,y]=local(e);spawn(x,y);
  });
})();
