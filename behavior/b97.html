B97 · השתקפות מים מתנדנדת מתחת לתמונה

התנהגויות SVG feTurbulence + feDisplacementMap · IntersectionObserver · CSS mask הירו ופתיחהתמונות וגלריותרקע ואווירה ממתין

תמונת שקיעה על הים נחתכת בקו האופק, ומתחתיה מופיע העתק הפוך שלה, כמו השתקפות במים שקטים. ההעתק מטושטש לאט לאורך הגובה ומתעוות בגלים אופקיים דקים שמשתנים כל הזמן, כך שהצבעים של השמיים נמרחים לפסים על המים. כשהסקשן נחשף, ההשתקפות עולה לאט מקו האופק. הכותרת יושבת על השמיים, מעל שכבה כהה רכה.

מתי משתמשים: הירו של עסק שיושב ליד מים או שמוכר רוגע: חדרי אירוח, ספא, חוף אירועים, יאכטות, צימרים, סדנאות. עובד רק עם תמונה שיש בה אופק ברור ושמיים חלקים. לא לתמונת אנשים או מוצר, ולא באתר שצריך להרגיש טכני.

שמונה חדרים מול הים

לכל חדר מרפסת שפונה מערבה, וארוחת הבוקר מוגשת עד אחת עשרה, במרחק דקה מהחוף.

בדיקת זמינות
שחזור. השראה: שני סרטונים ב-prompt-motion.com (21.10.2026) שבהם החלק התחתון של הפריים משקף את הסצנה שמעליו בגלים אופקיים. נבנה מאפס מהצפייה, על תמונת שקיעה מהמאגר (s08/hero.webp). החיתוך: קו האופק בתמונה הוא ב-71% מהגובה, ולכן כל התמונה מונחת בתחתית אזור השמיים כך ש-29% התחתונים שלה יוצאים מהקופסה והאופק יושב בדיוק על הקצה. הרוחב הוא max(100cqw, גובה / .473), והגובה נגזר מ---hs, ולכן בטלפון התמונה גדלה ונחתכת מהצדדים במקום לאבד את האופק. ההשתקפות: אותה קופסה, הפוכה ב-scaleY(-1), כך שקו האופק יושב בקצה העליון שלה, וחלון קטן יותר (60% מ---hs) חותך אותה. מסכה מדללת אותה כלפי מטה. העיוות: מסנן SVG אחד: feTurbulence בתדר נמוך אופקית וגבוה אנכית נותן פסים אופקיים, feColorMatrix קובע את הערוץ הירוק ל-.5, כך שהעיוות הוא אופקי בלבד ואין חור שקוף באופק, ו-feDisplacementMap בעוצמה 22 מזיז את הפיקסלים. הקופסה רחבה ב-32px מכל צד, והתמונה בתוכה רחבה ב-64px מהשמיים, כדי שהעיוות לא ימשוך שקיפות מהקצה. התנועה: baseFrequency משתנה בגל איטי ב-20 עדכונים בשנייה (15 בטלפון), והמסנן מחושב מחדש בכל עדכון, ולכן האזור נשמר נמוך (כ-250px) והלולאה עוצרת מחוץ למסך, בלשונית רקע ובהפחתת תנועה. זו העלות הגבוהה ביותר בקבוצה: מסנן SVG רץ על המעבד, ולכן לא מגדילים את אזור המים. המצב הסטטי: ההשתקפות עם אותו מסנן, בלי עדכונים, וזה מה שרואים בלי JS ובהפחתת תנועה. החשיפה: is-ready ו-is-in מעלים את ההשתקפות ב-opacity. נגישות: התמונות aria-hidden עם alt ריק, והמסר בכותרת ובפסקה.

(function(){
  const root=document.querySelector(".hw"),turb=document.getElementById("hw-turb");
  if(!turb||matchMedia("(prefers-reduced-motion: reduce)").matches)return;   // בלי JS ובהפחתת תנועה: ההשתקפות הסטטית
  const small=matchMedia("(max-width:767px)").matches,GAP=small?66:50;
  let vis=false,raf=0,last=0;
  // ההשתקפות עולה מקו האופק כשהסקשן נחשף
  root.classList.add("is-ready");
  // הגלים: התדר עצמו משתנה לאט, וכך הפסים נמרחים ונעים. עדכון ב-20 פעמים בשנייה מספיק, כי המסנן מחושב מחדש בכל עדכון
  function tick(now){
    raf=0;if(!vis||document.hidden)return;
    if(now-last>=GAP){last=now;const t=now/1000;
      turb.setAttribute("baseFrequency",(.0055+.0012*Math.sin(t*.35)).toFixed(5)+" "+(.092+.012*Math.sin(t*.5+1)).toFixed(5));}
    raf=requestAnimationFrame(tick);
  }
  const go=()=>{if(vis&&!document.hidden&&!raf)raf=requestAnimationFrame(tick);};
  new IntersectionObserver(es=>{vis=es[0].isIntersecting;if(vis)root.classList.add("is-in");go();},{rootMargin:"10% 0px"}).observe(root);
  document.addEventListener("visibilitychange",go);
})();
