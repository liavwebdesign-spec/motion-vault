B61 · משיכה מגנטית לעבר הסמן

התנהגויות vanilla JS hover ומיקרו ממתין

כפתור שנמשך בעדינות לעבר הסמן כשהוא מתקרב, עד שליש מהמרחק, וחוזר למקומו כשהסמן מתרחק. מספיק כדי להרגיש שהכפתור "רוצה" שילחצו עליו.

מתי משתמשים: CTA ראשי אחד בעמוד, אייקונים חברתיים, לוגו בהדר. דסקטופ בלבד, ולעולם לא על טקסט רץ.

מקור: ReactBits Magnet. נכתב מחדש: המרחק נמדד מהמרכז, המשיכה היא שבר מהמרחק (לא מיקום מוחלט), על translate בלבד, וחזרה עם טרנזישן ארוך יותר מהמשיכה. אזור המשיכה (40 פיקסלים סביב האלמנט) מחושב ב-JS מתוך תנועת העכבר במסמך, ולא כ-padding סביב הכפתור: כך העטיפה .mg צמודה לכפתור ולא מזיזה את הפריסה כשעוטפים בה CTA קיים, ואין שכבה שקופה שחוסמת לחיצות על קישור סמוך. במגע ובתנועה מופחתת הסקריפט לא רץ בכלל.

JS

(function(){
  if(!matchMedia("(hover:hover) and (pointer:fine)").matches||matchMedia("(prefers-reduced-motion: reduce)").matches)return;
  const PULL=.35,REACH=40;   // PULL: כמה מהמרחק הכפתור עובר לעבר הסמן (1 = דבוק לסמן, כבר לא אלגנטי). REACH: אזור המשיכה סביב האלמנט
  const zones=[...document.querySelectorAll(".mg")].map(z=>({z,el:z.firstElementChild,on:false}));
  let x=-1e5,y=-1e5,raf=0;
  function tick(){
    raf=0;
    zones.forEach(o=>{
      const r=o.z.getBoundingClientRect();   // העטיפה לא זזה, רק הילד, לכן המלבן שלה הוא מקום המנוחה
      if(x>r.left-REACH&&x<r.right+REACH&&y>r.top-REACH&&y<r.bottom+REACH){
        o.on=true;o.z.classList.remove("leave");
        o.el.style.setProperty("--x",((x-(r.left+r.width/2))*PULL).toFixed(1)+"px");o.el.style.setProperty("--y",((y-(r.top+r.height/2))*PULL).toFixed(1)+"px");
      }else if(o.on){o.on=false;o.z.classList.add("leave");o.el.style.setProperty("--x","0px");o.el.style.setProperty("--y","0px");}
    });
  }
  const req=()=>{if(!raf)raf=requestAnimationFrame(tick);};
  document.addEventListener("mousemove",e=>{x=e.clientX;y=e.clientY;req();},{passive:true});
  document.addEventListener("mouseleave",()=>{x=y=-1e5;req();});
  addEventListener("scroll",req,{passive:true});   // גלילה בלי תזוזת עכבר משנה את המרחק
})();
