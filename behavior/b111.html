B111 · מדריכים עם זרקור שעובר ביניהם

התנהגויות JS · CSS animation · IntersectionObserver כרטיסיםתמונות וגלריותתהליך ושלבים ממתין

רשימה של ארבעה מדריכים, ולידה תמונה גדולה של המדריך הנבחר. השורה הנבחרת מקבלת משטח, נפתחת ומראה משפט על המדריך, ובתחתית שלה סרגל דק מתמלא. כשהוא מתמלא, הזרקור עובר לבד למדריך הבא והתמונה מתחלפת. עכבר על הרכיב עוצר את ההחלפה, ויש כפתור לעצור אותה לגמרי.

מתי משתמשים: מרכז ידע, בלוג, מדריכים, מקרי לקוח או מוצרים מובילים, כשיש שלושה עד חמישה פריטים ותמונה טובה לכל אחד. מתאים לעמוד הבית של עסק שמוכר ידע (חשמלאי, רואה חשבון, סוכן ביטוח). לא לרשימה של עשרה מאמרים ומעלה, שם רשת כרטיסים, ולא כשהתמונות חלשות, כי התמונה הגדולה היא חצי מהרכיב.

מרכז הידע

מדריכים לפני שמתקינים עמדת טעינה

ארבעה מדריכים קצרים שכתבו החשמלאים שלנו, לפי השאלות שאנחנו שומעים הכי הרבה בטלפון.

שחזור. השראה: animtd.store (26.10.2026), סרטון של רשימת מאמרים שבה השורה הפעילה גדלה ומקבלת פס הדגשה. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: טאבים רגילים מחליפים תוכן רק בלחיצה. כאן יש שעון גלוי: הסרגל בתחתית השורה הפעילה הוא אנימציית CSS של שש שניות, וה-JS עובר לפריט הבא רק באירוע animationend שלו. לכן עצירה היא פשוט animation-play-state:paused, והסרגל והזמן לא יכולים להתפצל. עוצר כש: העכבר על הרכיב, הפוקוס בתוכו (מי שקורא בטאב לא יאבד את המקום), הרכיב מחוץ למסך (IntersectionObserver), או שלחצו «עצירת ההחלפה». בחירה: לחיצה על שורה, וריחוף בעכבר; השורה היא כפתור עם aria-pressed, ומי שלא רואה את התמונה מקבל את כל המידע מהשורה עצמה. התמונות בערימה אחת, והפעילה מתחלפת בעמעום והקטנה קלה; הכרטיסים הלא פעילים מקבלים inert, כך שהקישור «לקריאת המדריך» הוא תמיד של המדריך הנבחר. בלי JS כל המשפטים פתוחים וכל הכרטיסים זה מתחת לזה. הפחתת תנועה: אין החלפה אוטומטית, אין סרגל ואין כפתור עצירה; הבחירה ידנית. בטלפון התמונה מעל הרשימה, ברוחב מלא.

(function(){
  const root=document.querySelector("[data-as]");
  if(!root)return;
  const its=[...root.querySelectorAll(".as-it")],cards=[...root.querySelectorAll(".as-card")],pp=root.querySelector(".as-pp");
  const fine=matchMedia("(hover: hover) and (pointer: fine)"),still=matchMedia("(prefers-reduced-motion: reduce)");
  let cur=0;
  root.classList.add("is-live");          // בלי JS כל המשפטים פתוחים וכל הכרטיסים גלויים
  cards.forEach((c,i)=>{c.inert=i!==0;});
  if(!still.matches)pp.hidden=false;
  const go=j=>{
    if(j===cur)return;cur=j;
    its.forEach((b,i)=>b.setAttribute("aria-pressed",String(i===j)));
    cards.forEach((c,i)=>{c.classList.toggle("is-on",i===j);c.inert=i!==j;});
  };
  its.forEach((b,i)=>{
    b.addEventListener("click",()=>go(i));
    b.addEventListener("pointerenter",()=>{if(fine.matches)go(i);});
    // הסרגל של השורה הפעילה הוא השעון: כשהוא מתמלא עוברים למדריך הבא, וכשהוא עוצר גם ההחלפה עוצרת
    b.querySelector(".as-bar").addEventListener("animationend",()=>{if(i===cur)go((cur+1)%its.length);});
  });
  const hold=on=>root.classList.toggle("is-hold",on);
  root.addEventListener("pointerenter",()=>hold(true));
  root.addEventListener("pointerleave",()=>hold(false));
  root.addEventListener("focusin",()=>hold(true));
  root.addEventListener("focusout",e=>{if(!root.contains(e.relatedTarget))hold(false);});
  pp.addEventListener("click",()=>{
    const on=!root.classList.contains("is-paused");
    root.classList.toggle("is-paused",on);
    pp.querySelector("span").textContent=on?"המשך ההחלפה":"עצירת ההחלפה";
  });
  // מחוץ למסך השעון עומד, כדי שמי שגולל חזרה ימצא את אותו מדריך
  if("IntersectionObserver" in window){
    root.classList.add("is-off");
    new IntersectionObserver(es=>es.forEach(e=>root.classList.toggle("is-off",!e.isIntersecting)),{threshold:.35}).observe(root);
  }
})();
