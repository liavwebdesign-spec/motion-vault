B100 · לוגו שבנוי ממטריצת תווים

התנהגויות Canvas דגימה · pre · pointer הירו ופתיחהטקסט וכותרות ממתין

שם המותג מצויר מתווים קטנים: הלוגו נדגם לרשת של תאים, וכל תא מקבל תו לפי כמה הוא מלא, מנקודה קטנה ועד @. כשהלוח נכנס למסך הוא מתגבש מצד ההתחלה, ורצועת תווים אקראיים רצה לפני הצורה ומשאירה אותה מאחור. אחר כך התווים מנצנצים בעדינות. מעבר עכבר או נגיעה מערבלים את התווים באזור עגול והם חוזרים למקומם.

מתי משתמשים: חתימה של מותג עם שם קצר וחזק, עד חמש אותיות: בית קפה, סטודיו, מותג טכנולוגי. מתאים לרצועת מותג בפתיחה או באודות, כשרוצים להראות שהמותג בנוי ביד ובסדר. לא ללוגו עם פרטים דקים, כי הוא לא נשאר קריא בתווים.

קפה קלוי בקטנה, חיפה

    
קולים בכל שבוע כמה שקי קפה בלבד, וארוזים ביום הקלייה עצמו. אפשר לבוא לקחת מהסדנה או להזמין למשלוח.

  





שחזור. השראה: שני סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם סימן מצויר כרשת של תווים באותו צבע על רקע כהה, והתווים מתחלפים עד שהצורה מתייצבת. נבנה מאפס מהצפייה. הדגימה: המילה מצוירת על קנבס נסתר באותו פונט ובאותו משקל כמו ה-h2, ומחולקת לתאים. הכיסוי של כל תא (0 עד 1) בוחר תו מהסולם  .:-=+*#%@. המדידה רצה אחרי document.fonts.ready, מתאימה את מספר העמודות לרוחב (60 עד 120, בטלפון עמודות צפופות וקטנות כדי שהצורה תישאר ברורה) ואת גודל הגופן לרוחב האמיתי של תו במונוספייס, כך שהתווים ממלאים את הלוח בכל מסך. שתי שכבות pre: אחת בצבע הטקסט לתווים שהתייצבו ואחת ב---muted לרעש, באותה רשת, כי שורת טקסט אחת לא יכולה להחזיק שני צבעים בלי span לכל תו. התנועה: חזית רצה מצד ההתחלה (בעברית מימין) בשנייה וחצי. מאחוריה התווים סופיים ולפניה רצועה של 12 עמודות עם תווים אקראיים. אחרי ההתייצבות מתחלפים כאחוז אחד מהתווים בכל פריים (עד כ-11 פריימים בשנייה), וחזרה למסך מפעילה את החזית שוב. מעבר עכבר או נגיעה מחממים תאים ברדיוס של 9 תאים, והם מתקררים תוך כמה פריימים. הלולאה רצה רק כשהלוח על המסך ובטאב גלוי. בלי JS מוצג השם בכתב רגיל במשקל כבד, ובהפחתת תנועה מצויר פעם אחת הפריים המתייצב. נגישות: שתי שכבות התווים aria-hidden, וה-h2 עם השם האמיתי נשאר בדף כשהלוח פעיל (שקוף, באותו מקום), כך שקורא מסך ממשיך לקרוא אותו. הלוח מוגדר direction: ltr, כדי שהעמודות לא יתהפכו ב-RTL. באתר LTR הקוד קורא את direction של הסקשן ומריץ את החזית משמאל.



  
קוד להדבקה

  
הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

(function(){
  const root=document.querySelector(".al"),panel=root.querySelector(".al-panel"),name=root.querySelector(".al-name"),art=root.querySelector(".al-art"),pa=art.children[0],pb=art.children[1];
  const reduced=matchMedia("(prefers-reduced-motion: reduce)").matches,rtl=getComputedStyle(root).direction==="rtl",word=name.textContent.trim();
  const RAMP=" .:-=+*#%@",BAND=12,HR=9,NL=String.fromCharCode(10);
  let COLS=0,ROWS=0,cw=8,lh=13,T=null,HEAT=null,front=0,sweepT=0,sweeping=false,vis=false,raf=0,last=0,ready=false,lastW=0;
  // המילה נדגמת פעם אחת: מצוירת על קנבס נסתר באותו פונט, והכיסוי של כל תא בוחר תו מהסולם
  function build(){
    const W=Math.floor(art.clientWidth);if(!W)return;lastW=W;
    COLS=Math.max(60,Math.min(120,Math.round(W/5)));
    const mcs=getComputedStyle(pa),span=document.createElement("span");
    span.style.cssText="position:absolute;visibility:hidden;white-space:pre;font:"+mcs.fontWeight+" 100px "+mcs.fontFamily;
    span.textContent="M".repeat(40);document.body.appendChild(span);
    const ratio=span.getBoundingClientRect().width/40/100;span.remove();
    const f=W/(COLS*ratio);cw=f*ratio;lh=f;pa.style.fontSize=pb.style.fontSize=f+"px";
    const ncs=getComputedStyle(name),m=document.createElement("canvas").getContext("2d");
    m.font=ncs.fontWeight+" 100px "+ncs.fontFamily;
    const fs=Math.min(W*.92/(m.measureText(word).width||300)*100,W*.6);
    m.font=ncs.fontWeight+" "+fs+"px "+ncs.fontFamily;
    const mt=m.measureText(word),asc=mt.actualBoundingBoxAscent,desc=mt.actualBoundingBoxDescent;
    ROWS=Math.max(6,Math.ceil((asc+desc)/lh)+2);
    const bw=Math.round(COLS*cw),bh=Math.round(ROWS*lh),big=document.createElement("canvas");big.width=bw;big.height=bh;
    const bc=big.getContext("2d");bc.font=m.font;bc.textAlign="center";bc.textBaseline="alphabetic";bc.fillStyle="black";
    bc.fillText(word,bw/2,(bh-(asc+desc))/2+asc);
    const bd=bc.getImageData(0,0,bw,bh).data;T=new Uint8Array(COLS*ROWS);HEAT=new Float32Array(COLS*ROWS);
    for(let r=0;r<ROWS;r++)for(let c=0;c<COLS;c++){
      let sum=0,n=0;const x0=Math.floor(c*cw),x1=Math.min(bw,Math.floor((c+1)*cw)),y0=Math.floor(r*lh),y1=Math.min(bh,Math.floor((r+1)*lh));
      for(let y=y0;y<y1;y++)for(let x=x0;x<x1;x++){sum+=bd[(y*bw+x)*4+3];n++;}
      const a=n?sum/n/255:0;T[r*COLS+c]=a<.08?0:Math.min(9,1+Math.floor(a*8.99));      // אינדקס בסולם, 0 = תא ריק
    }
    compose();
  }
  // שתי שכבות: a לתווים שהתייצבו, b לרעש שלפני החזית ולמעבר העכבר
  function compose(){
    let sa="",sb="";const rnd=Math.random;
    for(let r=0;r<ROWS;r++){
      let la="",lb="";
      for(let c=0;c<COLS;c++){
        const k=r*COLS+c,t=T[k],cr=rtl?COLS-1-c:c;let ca=" ",cb=" ";
        if(sweeping&&cr>front){}
        else if(sweeping&&cr>front-BAND){if(t||rnd()<.06)cb=RAMP[1+Math.floor(rnd()*9)];}
        else if(HEAT[k]>.2){if(t||rnd()<.25)cb=RAMP[1+Math.floor(rnd()*9)];}
        else if(t){ca=RAMP[t];if(!reduced&&!sweeping&&rnd()<.012)ca=RAMP[Math.max(1,Math.min(9,t+(rnd()<.5?-1:1)))];}
        la+=ca;lb+=cb;
      }
      sa+=la+(r<ROWS-1?NL:"");sb+=lb+(r<ROWS-1?NL:"");
    }
    pa.textContent=sa;pb.textContent=sb;
  }
  function frame(now){
    raf=0;if(!vis||document.hidden||!ready)return;
    if(now-last>=(sweeping?33:90)){
      last=now;
      if(sweeping){front=(now-sweepT)/1600*(COLS+BAND);if(front>=COLS+BAND)sweeping=false;}
      for(let k=0;k<HEAT.length;k++)if(HEAT[k]){HEAT[k]*=.8;if(HEAT[k]<.03)HEAT[k]=0;}
      compose();
    }
    raf=requestAnimationFrame(frame);
  }
  const kick=()=>{if(!raf&&vis&&!document.hidden&&ready)raf=requestAnimationFrame(frame);};
  // מעבר עכבר או נגיעה מחממים תאים ברדיוס HR, והם מתקררים בפריימים הבאים
  const warm=e=>{
    if(!ready)return;const r=art.getBoundingClientRect(),cx=(e.clientX-r.left)/cw,cy=(e.clientY-r.top)/lh;
    for(let y=Math.max(0,Math.floor(cy-HR));y<=Math.min(ROWS-1,Math.ceil(cy+HR));y++)for(let x=Math.max(0,Math.floor(cx-HR));x<=Math.min(COLS-1,Math.ceil(cx+HR));x++)
      if(Math.hypot(x+.5-cx,y+.5-cy)<HR)HEAT[y*COLS+x]=1;
    kick();
  };
  (document.fonts&&document.fonts.ready||Promise.resolve()).then(()=>{
    build();ready=true;
    new ResizeObserver(()=>{if(Math.floor(art.clientWidth)!==lastW){build();kick();}}).observe(art);
    if(reduced){root.classList.add("is-live");return;}      // פריים אחד מתייצב, בלי לולאה ובלי מאזינים
    panel.addEventListener("pointermove",e=>{if(e.pointerType!=="touch")warm(e);});
    panel.addEventListener("pointerdown",warm);
    document.addEventListener("visibilitychange",kick);
    new IntersectionObserver(es=>{
      vis=es[0].isIntersecting;
      if(vis){sweeping=true;sweepT=performance.now();front=0;compose();root.classList.add("is-live");}
      kick();
    },{rootMargin:"10% 0px"}).observe(panel);
  });
})();
