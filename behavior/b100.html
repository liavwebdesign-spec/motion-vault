B100 · לוגו שבנוי ממטריצת תווים

התנהגויות Canvas דגימה · pre · pointer הירו ופתיחהטקסט וכותרות ממתין

שם המותג מצויר מתווים קטנים על לוח כהה: הלוגו נדגם לרשת צפופה של תאים, וכל תא מקבל תו לפי כמה הוא מלא, מנקודה קטנה ועד @. התווים המלאים בהירים והקצוות מתעמעמים אל הרקע, ולכן הצורה רכה ולא נראית כמו מסך מסוף. כשהלוח נכנס למסך הוא מתגבש מצד ההתחלה, ורצועת תווים בצבע המבטא רצה לפני הצורה ומשאירה אותה מאחור. אחר כך גל אור איטי חוצה את האותיות והמרקם של התווים נראה גם בתוכן. מעבר עכבר או נגיעה מערבלים את התווים באזור עגול והם חוזרים למקומם. כפתור בפינת הלוח משהה את הכול.

מתי משתמשים: חתימה של מותג עם שם קצר וחזק, עד חמש אותיות: בית קפה, סטודיו, מותג טכנולוגי. מתאים לרצועת מותג בפתיחה או באודות, כשרוצים להראות שהמותג בנוי ביד ובסדר, ולרגע כהה ומנוגד באמצע עמוד בהיר. לא ללוגו עם פרטים דקים, כי הוא לא נשאר קריא בתווים.

קפה קלוי בקטנה, חיפה

    
קולים בכל שבוע כמה שקי קפה בלבד, וארוזים ביום הקלייה עצמו. אפשר לבוא לקחת מהסדנה או להזמין למשלוח.

  





שחזור. השראה: שני סרטונים בגלריה של prompt-motion.com (21.10.2026) שבהם סימן מצויר כרשת של תווים באותו צבע על רקע כהה, והתווים מתחלפים עד שהצורה מתייצבת. נבנה מאפס מהצפייה. שופר ב-25.10.2026 אחרי שנדחה: הלוח היה לבן עם תווים שחורים ונראה כמו מסוף ישן, היו שוליים צבעוניים סביב כל תו, והרזולוציה הייתה נמוכה (כ-14 שורות). הדגימה: המילה מצוירת על קנבס נסתר באותו פונט ובאותו משקל כמו ה-h2, ומחולקת לתאים. הכיסוי של כל תא (0 עד 1) בוחר תו מהסולם  .:-=+*#%@. המדידה רצה אחרי document.fonts.ready, מתאימה את מספר העמודות לרוחב (64 עד 184, כ-5.4 פיקסלים לעמודה ובטלפון 4.7) ואת גודל הגופן לרוחב האמיתי של תו במונוספייס, כך שהתווים ממלאים את הלוח בכל מסך. רזולוציה: line-height הוא 0.8 ולא 1, כך שהתא כמעט ריבוע ויש יותר מפי שניים שורות (כ-34 בדסקטופ). גובה התא נקרא מה-CSS ולא מוגדר פעמיים. צבע: הלוח הוא --ink עם זוהר רך של --accent במרכז, והתווים בצבע --bg, ולכן הוא הפוך לעמוד בכל עור (בעור כהה הלוח בהיר). ארבע שכבות pre באותה רשת, כי שורת טקסט אחת לא יכולה להחזיק כמה צבעים בלי span לכל תו, והצבע עוקב אחרי הצפיפות של התו: % ו-@ ב---bg, # ו-* בגוון ביניים, התווים הדקים (+ ומטה) בגוון חלש, ורעש בצבע המבטא מעורבב ב---bg. הקצוות מתעמעמים אל הרקע והצורה נראית רכה. שוליים צבעוניים: will-change: opacity על .al-art ו--webkit-font-smoothing: antialiased מונעים החלקה של תתי-פיקסלים שיוצרת שוליים אדומים וכחולים סביב תווים בהירים על כהה. התנועה: חזית רצה מצד ההתחלה (בעברית מימין) בשנייה וחצי. מאחוריה התווים סופיים ולפניה רצועה של 14 עמודות עם תווים אקראיים. אחרי ההתייצבות גל איטי של אור חוצה את האותיות (הוא מוזז לפי שעון פנימי, כ-11 פריימים בשנייה): התווים המלאים מתעמעמים בהדרגה מ-@ עד + ושוב מתבהרים, כך שהמרקם של התווים נראה גם בתוך האות. חזרה למסך מפעילה את החזית שוב. מעבר עכבר או נגיעה מחממים תאים ברדיוס שנמדד בפיקסלים (כ-8 אחוז מרוחב הלוח, בין 48 ל-88), כך שהאזור עגול ולא אליפסה, והם מתקררים תוך כמה פריימים. הלולאה רצה רק כשהלוח על המסך ובטאב גלוי. השהיה: כפתור בפינת הלוח (מופיע רק כשהתנועה פועלת) עוצר את הלולאה. שעון החזית נפרד, ולכן הפעלה מחדש ממשיכה מאותה נקודה, ובזמן השהיה מעבר עכבר לא מזיז כלום. בלי JS מוצג השם בכתב רגיל במשקל כבד, ובהפחתת תנועה מצויר פעם אחת הפריים המתייצב, בלי כפתור. נגישות: ארבע שכבות התווים aria-hidden, וה-h2 עם השם האמיתי נשאר בדף כשהלוח פעיל (שקוף, באותו מקום), כך שקורא מסך ממשיך לקרוא אותו. הלוח מוגדר direction: ltr, כדי שהעמודות לא יתהפכו ב-RTL. באתר LTR הקוד קורא את direction של הסקשן ומריץ את החזית משמאל.

(function(){

  // המילה נדגמת פעם אחת: מצוירת על קנבס נסתר באותו פונט, והכיסוי של כל תא בוחר תו מהסולם
  function build(){

    }

  }

  }
  const kick=()=>{if(!raf&&vis&&!document.hidden&&ready&&!paused)raf=requestAnimationFrame(frame);};
  // מעבר עכבר או נגיעה מחממים תאים ברדיוס HR (בפיקסלים), והם מתקררים בפריימים הבאים. בזמן השהיה אין תגובה
  const warm=e=>{
    if(!ready||paused)return;const r=art.getBoundingClientRect(),px=e.clientX-r.left,py=e.clientY-r.top;
    for(let y=Math.max(0,Math.floor((py-HR)/lh));y<=Math.min(ROWS-1,Math.ceil((py+HR)/lh));y++)for(let x=Math.max(0,Math.floor((px-HR)/cw));x<=Math.min(COLS-1,Math.ceil((px+HR)/cw));x++)

    kick();
  };
  (document.fonts&&document.fonts.ready||Promise.resolve()).then(()=>{
    build();ready=true;
    new ResizeObserver(()=>{if(Math.floor(art.clientWidth)!==lastW){build();kick();}}).observe(art);
    if(reduced){root.classList.add("is-live");return;}      // פריים אחד מתייצב, בלי לולאה, בלי כפתור ובלי מאזינים
    root.classList.add("is-motion");
    pz.addEventListener("click",()=>{
      paused=!paused;
      pzl.textContent=paused?"הפעלת התנועה":"השהיית התנועה";
      pzi.setAttribute("d",paused?"M8 5l11 7-11 7z":"M9 5v14M15 5v14");
      if(!paused){tPrev=performance.now();kick();}
    });
    panel.addEventListener("pointermove",e=>{if(e.pointerType!=="touch")warm(e);});
    panel.addEventListener("pointerdown",warm);
    document.addEventListener("visibilitychange",kick);
    new IntersectionObserver(es=>{
      vis=es[0].isIntersecting;
      if(vis){

        root.classList.add("is-live");
      }
      kick();
    },{rootMargin:"10% 0px"}).observe(panel);
  });
})();
