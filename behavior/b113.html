B113 · גלריה עם רצועת תמונות שמתקדמת לבד

התנהגויות JS · CSS animation · IntersectionObserver תמונות וגלריותניווט ומבנה ממתין

תמונה גדולה עם כיתוב, ומתחתיה רצועה של ארבע תמונות קטנות. הקטנה הפעילה בולטת ועולה מעט לכיוון התמונה הגדולה, ובתחתיתה סרגל מתמלא; כשהוא מתמלא, התמונה הגדולה מתחלפת לבאה בעמעום והתרחקות קלה, והכיתוב מתחלף איתה. לחיצה על תמונה קטנה עוברת אליה. מונה קטן בפינה, וכפתור לעצירה.

מתי משתמשים: מקום או חוויה שרוצים להראות בכמה רגעים: בר, מסעדה, צימר, אולם, סטודיו, נכס להשכרה. שלוש עד שש תמונות באותו יחס, ולכל אחת משפט. לא לקטלוג מוצרים, שם c25 בלי החלפה אוטומטית, ולא לתמונות שצריך לבחון בעיון (תוכנית, מפרט), כי ההחלפה לא מחכה.

שחזור. השראה: animtd.store (26.10.2026), סרטון של גלריה עם רצועת ממוזערות שמתקדמת לבד. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: c25 הוא תמונה ראשית ופס ממוזערות שמתחלף רק בלחיצה. כאן הגלריה מתקדמת לבד עם שעון גלוי בתוך הממוזערת הפעילה, והתמונה הגדולה נכנסת בהתרחקות (מ-1.06 ל-1). השעון: כמו ב-b111, הסרגל הוא אנימציית CSS של חמש שניות, וה-JS עובר לתמונה הבאה רק ב-animationend. עוצר כש: העכבר על הגלריה, הפוקוס בתוכה, הגלריה מחוץ למסך, או בכפתור העצירה (WCAG 2.2.2). הממוזערת הפעילה עולה ב-transform (הזזה למעלה בלבד, בלי הגדלה), לא ברוחב, כך שאין קפיצות בפריסה; האחרות בשקיפות .55. הריווח (תוקן 26.10 לפי ליאב): הרווח בין התמונה הגדולה לרצועה הוא הרווח בין הממוזערות ועוד גובה ההרמה (--fs-gap ו---fs-lift), כך שהפעילה נעצרת במרחק מהתמונה הגדולה ששווה בדיוק לרווח בינה לשכנות. בגרסה הקודמת היא עלתה עם הגדלה ל-1.06, כמעט נגעה בתמונה הגדולה וחרגה מהקו שלה בקצה. כל ממוזערת היא כפתור עם aria-pressed ותווית. הכיתוב, המונה והעצירה יושבים על משטח בצבע הכרטיס עם טשטוש, ולא על שכבה כהה שקופה: שכבה מ-ink מתבהרת באתר כהה, והטקסט נעלם מעל תמונה כהה (נמצא בבדיקת העורות). בלי JS מוצגות התמונה והכיתוב הראשונים, והממוזערות לא עושות כלום. הפחתת תנועה: בלי החלפה אוטומטית ובלי כפתור עצירה; הבחירה בלחיצה. בטלפון אותו מבנה, והרצועה נשארת בשורה אחת.

(function(){
  const root=document.querySelector("[data-fs]");
  if(!root)return;
  const ths=[...root.querySelectorAll(".fs-th")],imgs=[...root.querySelectorAll(".fs-main img")],caps=[...root.querySelectorAll(".fs-cap>div")];
  const num=root.querySelector(".fs-count b"),pp=root.querySelector(".fs-pp"),view=root.querySelector(".fs-view");
  let cur=0;
  root.classList.add("is-live");
  if(!matchMedia("(prefers-reduced-motion: reduce)").matches)pp.hidden=false;
  const go=j=>{
    if(j===cur)return;cur=j;
    ths.forEach((b,i)=>b.setAttribute("aria-pressed",String(i===j)));
    [imgs,caps].forEach(g=>g.forEach((el,i)=>el.classList.toggle("is-on",i===j)));
    num.textContent=j+1;
  };
  ths.forEach((b,i)=>{
    b.addEventListener("click",()=>go(i));
    // הסרגל בממוזערת הפעילה הוא השעון: מתמלא, ועוברים לתמונה הבאה
    b.querySelector(".fs-bar").addEventListener("animationend",()=>{if(i===cur)go((cur+1)%ths.length);});
  });
  const hold=on=>root.classList.toggle("is-hold",on);
  view.addEventListener("pointerenter",()=>hold(true));
  view.addEventListener("pointerleave",()=>hold(false));
  view.addEventListener("focusin",()=>hold(true));
  view.addEventListener("focusout",e=>{if(!view.contains(e.relatedTarget))hold(false);});
  pp.addEventListener("click",()=>{
    const on=!root.classList.contains("is-paused");
    root.classList.toggle("is-paused",on);
    pp.setAttribute("aria-label",on?"המשך ההחלפה":"עצירת ההחלפה");
  });
  if("IntersectionObserver" in window){
    root.classList.add("is-off");
    new IntersectionObserver(es=>es.forEach(e=>root.classList.toggle("is-off",!e.isIntersecting)),{threshold:.35}).observe(root);
  }
})();
