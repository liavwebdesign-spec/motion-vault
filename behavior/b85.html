B85 · כפתור שהופך לפס התקדמות ולאישור

התנהגויות GSAP width · CSS state roll · aria-live טעינה ופידבק ממתין

כפתור "אישור ההצעה" רחב. בלחיצה הוא מתכווץ לגלולה: התווית מתגלגלת למעלה ומוחלפת באייקון, במילה "מאשרים" ובאחוז שעולה, ופס דק בתוכה מתמלא. בסוף הגלולה מתכווצת עוד, הצבע מתהפך, וי מצטייר ועולה המילה "אושרה", והכרטיס ננעל.

מתי משתמשים: פעולה שיש לה אורך אמיתי: אישור הצעת מחיר או הזמנה, שליחת קבצים, תשלום, חתימה על מסמך. כשהכפתור הוא הרגע שהמבקר מחליט, והוא צריך לראות שמשהו קורה ושזה נגמר. לא כשהפעולה מיידית, ולא כשהאחוזים לא משקפים תהליך אמיתי.

עוד מסלול גלילה. נסה גם חזרה למעלה ↑
שחזור. השראה: שני סרטוני מוצר ב-prompt-motion.com (21.10.2026): כפתור הזמנה שהופך לגלולה עם אחוז עולה ופס דק, וכפתור אישור שמשנה צבע, מקבל וי ומילה ננעלת. נבנה מאפס מהצפייה. שלושה מצבים בכפתור אחד (is-busy, is-done, ובלי מחלקה זה המצב הרגיל). שלוש שכבות טקסט יושבות באותו תא בגריד, והמעבר ביניהן הוא גלילה אנכית של transform ו-opacity בלבד. הרוחב של כל מצב קבוע ב-CSS (רגיל, 260px, 184px), ו-GSAP מטווין את ה-width מהערך הנוכחי ליעד ב-power3.inOut ומנקה אחר כך, כי מעבר CSS על רוחב אסור במאגר. בלי GSAP הרוחב מתחלף מיד. הכפתור מעוגן לצד ההתחלה (ימין), ולכן הוא מתכווץ לתוך עצמו. ההתקדמות רצה 2.6 שניות בעקומה של פתיחה מהירה, אמצע איטי וסיום מהיר, כדי שתרגיש כמו עבודה אמיתית. הפס הוא scaleX בלבד. באתר אמיתי מחברים את האחוז להתקדמות אמיתית, או לא מציגים אחוז. הצבע הסופי הוא --ink עם טקסט --bg (אין טוקן ירוק, והמצב הנעול צריך להיראות שונה בכל עור). נעילה: הכפתור עובר ל-aria-disabled ולא ל-disabled, כך שהפוקוס נשאר עליו ושם הכפתור מתעדכן (aria-label: מאשרים, אושרה). מעל זה aria-busy בזמן ההתקדמות, אזור role="status" שמכריז את שני השלבים, וכפתור "להדגמה חוזרת" שהוא inert עד האישור. בהפחתת תנועה אין גלילת טקסט, אין אחוז ואין פס: אחרי חצי שנייה המצב עובר ל"אושרה". באתר משמאל לימין: מחליפים את transform-origin של הפס ל-0 50%.

JS

(function(){
  const root=document.querySelector(".ap"),q=s=>root.querySelector(s);
  const btn=q(".ap-btn"),s1=q(".ap-s1"),pct=q(".ap-pct"),bar=q(".ap-bar i"),reset=q(".ap-reset"),live=q(".ap-live");
  const reduce=matchMedia("(prefers-reduced-motion: reduce)");
  let state="idle",raf=0,timer=0;
  const canTween=()=>typeof gsap!=="undefined"&&!reduce.matches;
  // המצב קובע מחלקות. הרוחב של כל מצב ב-CSS, ו-GSAP רק מטווין אליו מהערך הנוכחי (מעבר CSS על רוחב אסור במאגר)
  function go(s){
    const from=btn.offsetWidth;
    state=s;
    btn.classList.toggle("is-busy",s==="busy");btn.classList.toggle("is-done",s==="done");
    root.classList.toggle("is-locked",s==="done");
    btn.setAttribute("aria-disabled",s==="idle"?"false":"true");
    if(s==="busy")btn.setAttribute("aria-busy","true");else btn.removeAttribute("aria-busy");
    // שם הכפתור מתעדכן עם המצב: שכבת הטקסט הרגילה מוסתרת כשהיא לא פעילה
    if(s==="idle")btn.removeAttribute("aria-label");else btn.setAttribute("aria-label",s==="busy"?"מאשרים את ההצעה":"ההצעה אושרה");
    s1.setAttribute("aria-hidden",String(s!=="idle"));
    reset.inert=s!=="done";
    const to=btn.offsetWidth;
    if(canTween()&&from!==to)gsap.fromTo(btn,{width:from},{width:to,duration:.55,ease:"power3.inOut",clearProps:"width"});
  }
  function finish(){go("done");live.textContent="ההצעה אושרה";}
  // עקומה של פתיחה מהירה, אמצע איטי וסיום מהיר, כדי שזה ירגיש כמו עבודה אמיתית
  const curve=k=>k<.45?k/.45*.42:k<.8?.42+(k-.45)/.35*.3:.72+(k-.8)/.2*.28;
  btn.addEventListener("click",()=>{
    if(state!=="idle")return;
    live.textContent="מאשרים את ההצעה";
    go("busy");
    if(reduce.matches){pct.hidden=true;timer=setTimeout(finish,500);return;}
    const t0=performance.now(),dur=2600;
    const step=t=>{
      const k=Math.min((t-t0)/dur,1),p=curve(k);
      pct.textContent=Math.round(p*100)+"%";bar.style.transform="scaleX("+p+")";
      if(k<1)raf=requestAnimationFrame(step);else timer=setTimeout(finish,260);
    };
    raf=requestAnimationFrame(step);
  });
  reset.addEventListener("click",()=>{
    cancelAnimationFrame(raf);clearTimeout(timer);
    go("idle");pct.hidden=false;pct.textContent="0%";bar.style.transform="";live.textContent="";
    btn.focus({preventScroll:true});
  });
})();
