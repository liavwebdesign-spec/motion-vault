B95 · מסגרת שמחליפה יחס בלחיצה והתמונה נחתכת מחדש תוך כדי תנועה

התנהגויות GSAP Flip · radiogroup · container query units תמונות וגלריותטעינה ופידבק ממתין

תמונה בתוך מסגרת, ולידה ארבע אפשרויות עם סמל קטן של היחס: רחב, ריבוע, אינסטגרם וסטורי. בלחיצה המסגרת עוברת בהחלקה ליחס החדש, והתמונה שבתוכה נחתכת מחדש בזמן אמת במקום להימתח או לקפוץ. כיתוב בפינה מתחלף ואומר איפה משתמשים בפורמט הזה. הגולש רואה את אותו צילום בכל הפורמטים בלי לדמיין.

מתי משתמשים: עסק שמוסר אותו תוצר בכמה פורמטים: סרטוני תדמית, צילום מוצר, באנרים, תמונות לרשתות חברתיות. מתאים לסקשן ״מה מקבלים״ או לדף שירות. ארבע אפשרויות לכל היותר, פעם אחת בעמוד. לא לתוצר שיש לו פורמט אחד, ולא כשאין צילום אחד שנראה טוב בכל החיתוכים.

16:9 אתר ויוטיוב
שחזור. השראה: כמה סרטוני מוצר בגלריה של prompt-motion.com (21.10.2026) שבהם מסגרת מחליפה יחס מסך והתוכן שבתוכה מסתדר מחדש. נבנה מאפס מהצפייה. זה מהלך עם בקרה אמיתית: role="radiogroup" של ארבעה button עם aria-checked, tabindex מתגלגל, חיצים (החצים ימינה ושמאלה מתהפכים לפי direction), ו-:focus-visible ברור. אזור המגע הוא 64px. גודל המסגרת מחושב ב-CSS בלבד: הבמה היא container-type:size, והמסגרת היא min(100cqw, 100cqh * var(--r)) על min(100cqh, 100cqw / var(--r)), כש---r הוא היחס כמספר. לכן המסגרת תמיד המלבן הגדול ביותר מהיחס הזה שנכנס לבמה, בכל רוחב. המעבר הוא Flip.from(state, {scale:false}): Flip מנפיש width ו-height אמיתיים ולא scale, והתמונה (object-fit:cover) נחתכת מחדש בכל פריים, בלי עיוות. קליקים מהירים מבטלים את המעבר הקודם (Flip.killFlipsOf) ומתחילים ממה שמוצג. בלי Flip או בהפחתת תנועה היחס מתחלף מיד, בלי מעבר, והבקרה עובדת כרגיל. בלי JS מוצגת המסגרת ביחס הרחב, והבקרה מוחלפת ברשימה פשוטה של ארבעת הפורמטים (במארקאפ), כדי שלא יוצגו כפתורים שלא עושים כלום. RTL: ה-<bdi dir="ltr"> שומר שיחס כמו 9:16 לא יתהפך. תמונה: מהמאגר, מרובעת, כדי שכל החיתוכים יראו פנים.

if (typeof gsap !== "undefined") gsap.registerPlugin(Flip);

(function(){
  const root=document.querySelector(".arf"),frame=root.querySelector(".arf-frame"),opts=[...root.querySelectorAll(".arf-opt")],
    tag=root.querySelector(".arf-tag bdi"),use=root.querySelector(".arf-use");
  const rm=matchMedia("(prefers-reduced-motion: reduce)");
  const rtl=getComputedStyle(root).direction==="rtl"?-1:1;       // החצים ימינה ושמאלה מתהפכים עם כיוון העמוד
  let cur=0;
  root.classList.add("is-live");
  function pick(i,focus){
    if(i===cur)return;
    // החלפת היחס עצמה רצה תמיד, גם בלי Flip וגם בהפחתת תנועה. רק המעבר מותנה
    const animate=typeof Flip!=="undefined"&&typeof gsap!=="undefined"&&!rm.matches;
    if(animate)Flip.killFlipsOf(frame);
    const state=animate?Flip.getState(frame):null;
    cur=i;
    opts.forEach((o,j)=>{o.setAttribute("aria-checked",j===i?"true":"false");o.tabIndex=j===i?0:-1;});
    const o=opts[i];
    frame.style.setProperty("--r",o.dataset.r);
    tag.textContent=o.dataset.l;use.textContent=o.dataset.u;
    if(focus)o.focus();
    if(!state)return;
    // scale:false מנפיש width ו-height אמיתיים, והתמונה נחתכת מחדש בכל פריים
    Flip.from(state,{duration:.6,ease:"power3.inOut",scale:false});
    gsap.fromTo([tag,use],{y:8,autoAlpha:0},{y:0,autoAlpha:1,duration:.4,delay:.2,ease:"power2.out",stagger:.06,overwrite:true});
  }
  opts.forEach((o,i)=>{
    o.addEventListener("click",()=>pick(i,false));
    o.addEventListener("keydown",e=>{
      const k=e.key;let j=-1;
      if(k==="ArrowDown")j=i+1;else if(k==="ArrowUp")j=i-1;
      else if(k==="ArrowRight")j=i+rtl;else if(k==="ArrowLeft")j=i-rtl;
      else if(k==="Home")j=0;else if(k==="End")j=opts.length-1;
      if(j<0)return;
      e.preventDefault();pick((j+opts.length)%opts.length,true);
    });
  });
})();
