B114 · מספר ענק שסופר ועוגה שמתמלאת

התנהגויות SVG · JS · IntersectionObserver מספרים ונתונים ממתין

מספר ענק אחד שסופר מאפס כשהסקשן נכנס למסך, ומתחתיו מה הוא אומר. לצדו עוגה בשלושה צבעים שהפלחים שלה נמתחים בזה אחר זה, עם מקרא. בשורה שמתחת שלושה נתונים נוספים שסופרים יחד איתו. הכל קורה פעם אחת.

מתי משתמשים: עסק שהמספרים שלו הם הטיעון: מתקין סולארי, קבלן, משרד רואי חשבון, חברת הובלות, עמותה. מספר אחד שהוא הגאווה (כמה פרויקטים, כמה לקוחות), ופילוח אחד שמסביר אותו. לא כשהמספרים קטנים או עגולים מדי ("מעל 50 לקוחות"), כי הספירה מבליטה אותם, ולא יותר מפעם אחת בעמוד.

שחזור. השראה: animtd.store (26.10.2026), סרטון של לוח נתונים שהמספרים בו סופרים ודיאגרמה מתמלאת. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: b02 הוא מספר שסופר, לבד. כאן יש היררכיה (מספר אחד ענק ושלושה קטנים) ועוגה שמסבירה את המספר. הספירה: ה-JS מאפס את המספרים רק אחרי שהוא נטען (בלי JS רואים את הערך הסופי), וסופר 1.6 שניות עם האטה בסוף (ease-out בחזקה שלישית), בעיצוב מספרים עברי (Intl.NumberFormat). tabular-nums שומר על רוחב קבוע, כך שהשורה לא רוקדת בזמן הספירה. קורא מסך מקבל את הערך הסופי מטקסט נסתר, והמספר הסופר aria-hidden. העוגה: שלושה עיגולי SVG עם pathLength=100, כך ש-stroke-dasharray הוא אחוזים פשוטים; כל פלח מתחיל בסוף הקודם (stroke-dashoffset), עם רווח קטן ביניהם. המצב ההתחלתי 0 100 חל רק עם is-live, והמתיחה היא transition עם השהיה של רבע שנייה בין הפלחים. הצבעים מהטוקנים (accent, ערבוב שלו עם הכרטיס, ink), כך שהעוגה מתחלפת עם העור. הפחתת תנועה: העוגה מוצגת מלאה מההתחלה, והמספרים עדיין סופרים, כי מספר שמשתנה הוא טקסט ולא תנועה (כמו ב-b02). בטלפון הכל בטור, והנתונים הקטנים אחד מתחת לשני.

JS

(function(){
  const root=document.querySelector("[data-sc]");
  if(!root)return;
  root.classList.add("is-live");
  // בהפחתת תנועה העוגה מלאה מההתחלה, והמספרים עדיין סופרים: מספר שמשתנה הוא טקסט ולא תנועה (motion.md 4, כמו ב-b02)
  if(matchMedia("(prefers-reduced-motion: reduce)").matches)root.classList.add("is-in");
  const nums=[...root.querySelectorAll(".sc-n")],fmt=new Intl.NumberFormat("he-IL");
  const put=k=>nums.forEach(b=>{b.textContent=fmt.format(Math.round(+b.dataset.to*k))+(b.dataset.suf||"");});
  put(0);                                  // מאפסים רק עכשיו: בלי JS רואים את הערך הסופי
  const run=()=>{
    root.classList.add("is-in");           // הפלחים של העוגה נמתחים ב-CSS
    const t0=performance.now(),D=1600;
    const tick=now=>{
      const p=Math.min(1,(now-t0)/D);
      put(1-Math.pow(1-p,3));              // מאט לקראת הסוף, כמו מונה שמגיע למקום
      if(p<1)requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  };
  if(!("IntersectionObserver" in window))return run();
  const io=new IntersectionObserver(es=>{if(es.some(e=>e.isIntersecting)){io.disconnect();run();}},{threshold:.4});
  io.observe(root);
})();
