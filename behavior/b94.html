B94 · כרטיס מוצר שעובר בין טעינה, מצב ריק ומצב מלא

התנהגויות Tabs · CSS transitions · IntersectionObserver טעינה ופידבקכרטיסים ממתין

כרטיס של פורטל לקוחות עם רשימת מסמכים, ולידו שלוש לשוניות: בטעינה, ריק, מלא. בכניסה לעמוד הכרטיס עובר לבד פעם אחת: שורות אפורות עם ברק שעובר עליהן, אחר כך מסך ריק עם הסבר, ואז רשימה שנכנסת שורה אחרי שורה. אחרי זה הגולש בוחר לבד איזה מצב לראות. הוא מבין שהמערכת חשבה גם על הרגע שבו אין עדיין כלום.

מתי משתמשים: שירות שיש לו ממשק ללקוח: פורטל, אפליקציה, מערכת הזמנות, לוח מעקב. מתאים לסקשן ״מה הלקוח רואה״ כשהאמון תלוי בכך שגם המצבים הלא יפים מטופלים. שלוש לשוניות, פעם אחת בעמוד. לא למוצר שאין לו ממשק, ולא כשיש רק מצב אחד אמיתי להראות.

שחזור. השראה: כמה סרטוני מוצר בגלריה של prompt-motion.com (21.10.2026) שבהם אותו מסך עובר בין שלד טעינה, מצב ריק ומצב מלא. נבנה מאפס מהצפייה. זה מהלך עם בקרה אמיתית: שלוש לשוניות button בתוך role="tablist", aria-selected ו-tabindex מתגלגל, חיצים ו-Home/End (החצים ימינה ושמאלה מתהפכים לפי direction של העמוד), ו-:focus-visible ברור. אזור המגע הוא 44px לפחות. המארקאפ הוא מצב הסיום: הרשימה המלאה בלבד, והלשוניות מוסתרות עד שה-JS עולה, כדי שלא יוצגו כפתורים שלא עושים כלום. כשה-JS חי שלושת הלוחות נערמים באותו תא של grid (לכן הגובה הוא של הגבוה ביותר, בלי קפיצה בין מצבים), והלוח הפעיל מקבל .on. הלוחות הלא פעילים הם inert, כך שקורא מסך ו-Tab מדלגים עליהם. המעבר הוא opacity ו-transform בלבד, ושורות הרשימה נכנסות עם transition-delay לפי --i. הברק של השלד הוא translateX על אלמנט מדומה, ולא תנועה של מאפיין פריסה; הוא רץ מימין לשמאל (ב-LTR הכיוון מתהפך לבד דרך :dir(ltr)). ההפעלה האוטומטית רצה פעם אחת בלבד, כשהכרטיס נכנס לחצי מהמסך (שלד, מצב ריק, מלא) ונעצרת מיד אם הגולש נגע בלשונית, כדי לא להזיז תוכן מתחת לידיים שלו. בהפחתת תנועה אין הפעלה אוטומטית, אין מעברים ואין ברק, והכרטיס נפתח על המצב המלא. הכפתור במצב הריק הוא span לא אינטראקטיבי (מסך דוגמה).

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית. הרכיב הזה מסתמך על dir="rtl".

(function(){
  const root=document.querySelector(".ust"),tabs=[...root.querySelectorAll(".ust-tab")],panes=[...root.querySelectorAll(".ust-pane")],
    cnt=[...root.querySelectorAll(".ust-cnt span")],sk=root.querySelector(".ust-sk");
  const reduced=matchMedia("(prefers-reduced-motion: reduce)").matches;
  const rtl=getComputedStyle(root).direction==="rtl"?-1:1;       // החצים ימינה ושמאלה מתהפכים עם כיוון העמוד
  let touched=false,t1=0,t2=0;
  const show=i=>{
    tabs.forEach((t,j)=>{t.setAttribute("aria-selected",j===i?"true":"false");t.tabIndex=j===i?0:-1;});
    panes.forEach((p,j)=>{p.classList.toggle("on",j===i);p.inert=j!==i;});
    cnt.forEach((c,j)=>c.classList.toggle("on",j===i));
    sk.setAttribute("aria-busy",i===0?"true":"false");
  };
  const stop=()=>{touched=true;clearTimeout(t1);clearTimeout(t2);};
  root.classList.add("is-live");
  tabs.forEach((t,i)=>{
    t.addEventListener("click",()=>{stop();show(i);});
    t.addEventListener("keydown",e=>{
      const k=e.key;let j=-1;
      if(k==="ArrowRight")j=i+rtl;else if(k==="ArrowLeft")j=i-rtl;else if(k==="ArrowDown")j=i+1;else if(k==="ArrowUp")j=i-1;
      else if(k==="Home")j=0;else if(k==="End")j=tabs.length-1;
      if(j<0)return;
      e.preventDefault();stop();j=(j+tabs.length)%tabs.length;show(j);tabs[j].focus();
    });
  });
  show(2);
  if(reduced||!("IntersectionObserver" in window))return;      // בהפחתת תנועה אין הפעלה אוטומטית
  // מעבר אחד בלבד כשהכרטיס נכנס למסך, ונעצר ברגע שהגולש נוגע בלשונית
  show(0);
  const io=new IntersectionObserver(es=>{
    if(!es[0].isIntersecting)return;
    io.disconnect();
    t1=setTimeout(()=>{if(touched)return;show(1);t2=setTimeout(()=>{if(!touched)show(2);},2400);},1800);
  },{threshold:.5});
  io.observe(root.querySelector(".ust-card"));
})();
