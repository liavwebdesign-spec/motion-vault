B104 · הדפס כפול עם סטייה

התנהגויות CSS · mix-blend-mode · pointer · custom properties הירו ופתיחהhover ומיקרורקע ואווירה ממתין

כותרת וצורות מודפסות כמו בדפוס בשני צבעים שלא התיישר בדיוק: מאחורי הכותרת השחורה נראית הדפסה שנייה בצבע המותג עם קווי מילוי, והעיגולים מתכהים איפה שהם חופפים. כשהעכבר זז על הסקשן, ההדפסה השנייה זזה איתו ומגדילה את הסטייה.

מתי משתמשים: פוסטר, כרטיס פתיחה או אירוע של סטודיו, בית דפוס, חנות עיצוב, גלריה או פסטיבל. כותרת אחת וצורות גדולות בסקשן אחד. עובד על רקע בהיר בלבד: על רקע כהה החפיפה נבלעת. במסך מגע נשארת סטייה קבועה, והמהלך עדיין נראה.

סדנת הדפס בשכונה

בסדנה של שלוש שעות מכינים פוסטר בהדפס ריזו: בוחרים שני צבעים, מדפיסים כל שכבה בנפרד ורואים איך הם נחתכים. כל משתתף יוצא עם חמישה עותקים.

לסדנה הקרובה
שחזור. השראה: סרטון ב-prompt-motion.com (21.10.2026) שבו טקסט וצורות מודפסים בסגנון ריזו, עם הדפסה שנייה מוזזת, קווי מילוי וגרעין, שבה החפיפות כהות יותר. נבנה מאפס מהצפייה. שתי לוחות הדפסה: לוח ראשון בצבע ה-ink (הכותרת, והעיגול הראשון ב-accent) ולוח שני בצבע אחר (העיגול המשורטט והעותק של הכותרת, בקווי מילוי באלכסון). העותק של הכותרת הוא span נוסף עם aria-hidden, ממוקם מתחת לטקסט האמיתי, בצבע accent עם קווי מילוי (repeating-linear-gradient שנחתך לאות עם background-clip:text), ומוזז בכמה פיקסלים. הכותרת עצמה נשארת טקסט אחד ברור. החפיפה: mix-blend-mode:multiply על העיגולים, ולכן איפה שהם נחתכים הצבע מתכהה כמו בדפוס אמיתי. זה עובד על רקע בהיר, ובעור כהה החפיפה נבלעת ברקע. הגרעין הוא אריח SVG קטן (feTurbulence) שמשמש כמסכה על שכבה בצבע הכרטיס: הוא בוקע חורים קטנים בצבע, כמו נייר שהדיו לא תפס בו. העכבר: pointermove בקריאה אחת לפריים כותב את --dx ו---dy (בין מינוס אחד לאחד) על הסקשן, והלוח השני נע ב-translate לפי שניהם, עם מעבר קצר של 0.5 שנייה בעקומה החתומה, ונחזר למקום ביציאה. ללא תנועה: בהפחתת תנועה אין האזנה לעכבר ונשארת הסטייה הקבועה. בטלפון העיגולים מעל הטקסט והכל סטטי. באתר משמאל לימין מחליפים את הצד של העיגולים. נגישות: העיגולים והעותק של הכותרת aria-hidden.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית.

JS

(function(){
  const root=document.querySelector("[data-rz]");
  const mq=window.matchMedia("(prefers-reduced-motion: reduce)");
  if(!root||mq.matches)return;                  // בהפחתת תנועה נשארת הסטייה הקבועה
  let raf=0,nx=0,ny=0;
  const apply=()=>{raf=0;root.style.setProperty("--dx",nx.toFixed(3));root.style.setProperty("--dy",ny.toFixed(3));};
  const queue=()=>{if(!raf)raf=requestAnimationFrame(apply);};
  const move=e=>{
    const r=root.getBoundingClientRect();
    nx=Math.max(-1,Math.min(1,((e.clientX-r.left)/r.width-.5)*2));
    ny=Math.max(-1,Math.min(1,((e.clientY-r.top)/r.height-.5)*2));
    queue();
  };
  const leave=()=>{nx=0;ny=0;queue();};
  root.addEventListener("pointermove",move);
  root.addEventListener("pointerleave",leave);
  mq.addEventListener("change",e=>{if(e.matches){root.removeEventListener("pointermove",move);root.removeEventListener("pointerleave",leave);leave();}});
})();
