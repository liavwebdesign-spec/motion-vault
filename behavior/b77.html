B77 · שדה ריבועים שגל עובר עליו ומנפח את המילה

התנהגויות Canvas 2D · pointer · מסכת טקסט רקע ואווירההירו ופתיחה ממתין

רשת של כמה אלפי ריבועים קטנים מכסה את הרקע, וגל עגול עובר עליה בלי הפסקה: כל ריבוע מסתובב, מתכווץ ומתבהר לפי המרחק שלו מנקודת המוצא. בתוך המילה הריבועים גדולים יותר ובצבע המותג, ולכן היא נקראת כמו הדפס רשת שהגל מנפח ומרגיע. הסמן מזיז את נקודת המוצא של הגל, ובלי סמן היא נודדת לאט לאורך הקצה.

מתי משתמשים: פתיחה של עמוד עם מילה אחת חזקה: שם המותג, הנושא או המילה שמגדירה את העסק, עד שש אותיות. מתאים לסטודיו, משרד מקצועי או מוצר, ורק פעם אחת בעמוד. לא מתחת לפסקאות ולא בעמוד שכבר יש בו רקע נע.

דיוק

משרד מדידות והנדסה, תל אביב

מודדים בשטח, מחשבים במשרד, ומוסרים תוכנית שאפשר לבנות לפיה בלי לחזור ולשאול.

שחזור. השראה: כמה סרטונים ב-prompt-motion.com (21.10.2026) עם רשת ריבועים או נקודות שגל אלכסוני או עגול עובר עליה, ובאחד מהם מילה שנפתרת מתוך רעש של הדפס רשת. נבנה מאפס מהצפייה. ההבדל מ-b68: שם הנקודות נדלקות סביב הסמן, כאן הגל רץ לבד, והסמן רק מזיז את נקודת המוצא שלו. המילה היא מסכה: המילה נשארת טקסט אמיתי ב-h2 (קוראי מסך ומנועי חיפוש), והקוד מצייר אותה פעם אחת על קנבס נסתר באותו פונט ואותו משקל, ומשתמש בו כדי לדעת איזה ריבוע יושב בתוך אות. כשהקנבס נחשף ה-h2 הופך שקוף והריבועים מחליפים אותו. המדידה רצה אחרי document.fonts.ready, אחרת המילה נמדדת בפונט החלופי. הגל: לכל ריבוע w = 0.5 + 0.5 * sin(d / 32 - t) כש-d הוא המרחק מנקודת המוצא. מחוץ למילה w קובע גודל, שקיפות וסיבוב (עד 90 מעלות). בתוך המילה w קובע גודל בין 0.66 ל-1.16 מהתא, כך שבשיא הגל הריבועים מתחברים לאות מלאה. קל: עד 6500 תאים (המרווח גדל עד שהכמות נכנסת), 10 קבוצות צבע ושקיפות עם מסלול אחד לכל קבוצה, DPR עד 2, הלולאה רק כשהשדה על המסך. RTL: נקודת המוצא ברירת מחדל היא הפינה הימנית העליונה והיא נודדת לכיוון פנים. באתר LTR הקוד קורא את direction ועובר לפינה השמאלית. בלי JS נשאר רקע של רשת נקודות בגרדיאנט והמילה בטקסט רגיל. בהפחתת תנועה מצויר פריים אחד עם הגל באמצע המסלול, בלי לולאה.

(function(){
  const root=document.querySelector(".dw"),field=root.querySelector(".dw-field"),cv=root.querySelector(".dw-cv"),ctx=cv.getContext("2d"),h2=root.querySelector(".dw-word");

  let W=0,H=0,g=10,n=0,cols=0,rows=0,PX,PY,IN,HS,TH,BK,COL=[],running=false,vis=false,raf=0,ro=null;
  const O={x:0,y:0,tx:0,ty:0,on:false},t0=performance.now();

  const probe=document.createElement("canvas").getContext("2d");

  // המילה מצוירת פעם אחת על קנבס נסתר, באותו פונט, ומשם נקרא איזה תא יושב בתוך אות
  function mask(){

    c.font=cs.fontWeight+" 100px "+cs.fontFamily;const w100=c.measureText(word).width||300,fs=Math.min(H*.86,W*.8/w100*100);
    c.font=cs.fontWeight+" "+fs+"px "+cs.fontFamily;c.textAlign="center";c.textBaseline="middle";c.fillStyle="#000";c.fillText(word,W/2,H/2+fs*.04);

  function build(){

  }
  function paint(t){
    const hm=g*.5;
    for(let i=0;i<n;i++){
      const w=.5+.5*Math.sin(Math.hypot(PX[i]-O.x,PY[i]-O.y)/LAM-t);
      if(IN[i]){HS[i]=(.66+.5*w)*hm;BK[i]=5+Math.min(4,Math.floor(w*4.99));TH[i]=0;}
      else{HS[i]=(.12+.58*w*w+.1*w)*hm;BK[i]=Math.min(4,Math.floor(w*4.99));TH[i]=w*1.5708;}

    }

  }

  field.addEventListener("pointerdown",e=>{if(!reduced)at(e);});
  field.addEventListener("pointerleave",()=>{O.on=false;});

    ro=new ResizeObserver(()=>{build();});ro.observe(field);
    if(reduced)field.classList.add("is-on");   // פריים אחד, בלי לולאה
  });
})();
