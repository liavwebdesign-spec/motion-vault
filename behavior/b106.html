B106 · כרטיס חבילה שמתהפך ומראה מה כלול

התנהגויות CSS 3D · JS · inert כרטיסיםhover ומיקרו ממתין

שלושה כרטיסי חבילה, ובכל אחד שם, מחיר ומשפט אחד. לחיצה על «מה כלול» הופכת את הכרטיס כמו קלף, ובגב שלו מופיעים כל הפריטים בחבילה וכפתור לבדיקת תאריך. «חזרה» הופך אותו בחזרה. החבילה המומלצת כהה ומסומנת.

מתי משתמשים: עמוד מחירים או סקשן חבילות עם שלוש עד ארבע חבילות, שהרשימה של כל אחת ארוכה מדי לכרטיס: צלם, מאמן, ספק אירועים, סטודיו, שירות חודשי. הכרטיס נשאר קצר ונקי, ומי שרוצה פרטים הופך אותו. לא כשהלקוח צריך להשוות שורה מול שורה בין החבילות, כי בכרטיס הפוך רואים חבילה אחת בכל פעם; שם טבלה או C20 עם רשימות פתוחות.

שחזור. השראה: animtd.store (26.10.2026), סרטון של כרטיס תמחור שמתהפך ומראה את הפירוט מאחור. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: C20 מציג את כל הרשימות פתוחות, ו-c27 (מה כלול ומה לא) נפסל ב-25.10. כאן הרשימה מאחורי הכרטיס, והכרטיס עצמו נשאר מחיר ומשפט. המבנה: שני הצדדים (.fp-front, .fp-back) יושבים באותו תא גריד (grid-area), ולכן גובה הכרטיס הוא גובה הצד הגבוה, ושלושת הכרטיסים באותו גובה. בלי JS שני הצדדים מוצגים זה מתחת לזה, וכל המידע נגיש. עם JS (is-live) הם עוברים לאותו תא, backface-visibility:hidden, והגב מסובב ב-180 מעלות. ההיפוך הוא rotateY של העטיפה ב-0.7 שנייה בעקומת in-out המאושרת, עם perspective על הכרטיס. נגישות: הצד שלא נראה מקבל inert, כך שאי אפשר להגיע אליו בטאב ולא ללחוץ עליו; הפוקוס עובר לכפתור «חזרה» אחרי ההיפוך, וחוזר ל«מה כלול» אחרי החזרה; Escape הופך בחזרה; לכפתור יש aria-expanded. המחיר בתוך unicode-bidi:isolate עם dir=ltr, כדי שהסימן והפסיק לא יקפצו בשורה עברית. הפחתת תנועה: אין סיבוב; הצדדים מתחלפים בעמעום של רבע שנייה. בטלפון הכרטיסים אחד מתחת לשני, וההיפוך זהה.

JS

(function(){
  const root=document.querySelector("[data-fp]");
  if(!root)return;
  root.classList.add("is-live");          // בלי JS שני הצדדים גלויים זה מתחת לזה
  root.querySelectorAll(".fp-card").forEach(card=>{
    const front=card.querySelector(".fp-front"),back=card.querySelector(".fp-back");
    const open=card.querySelector(".fp-open"),close=card.querySelector(".fp-close");
    back.inert=true;
    const set=on=>{
      card.classList.toggle("is-flip",on);
      front.inert=on;back.inert=!on;
      open.setAttribute("aria-expanded",String(on));
      // הפוקוס עובר לצד שנראה, אחרת הוא נשאר על כפתור שהסתובב לגב
      (on?close:open).focus({preventScroll:true});
    };
    open.addEventListener("click",()=>set(true));
    close.addEventListener("click",()=>set(false));
    card.addEventListener("keydown",e=>{if(e.key==="Escape"&&card.classList.contains("is-flip"))set(false);});
  });
})();
