B115 · קווי מסלול שנמתחים על תמונה

התנהגויות SVG · JS · requestAnimationFrame תמונות וגלריותניווט ומבנה ממתין

תמונה גדולה של שקיעה מעל הים, ועליה שלוש קשתות שנמתחות מתל אביב לשלושה יעדים כשהכרטיס נכנס למסך. בקצה כל קשת נקודה ותווית עם שם היעד וזמן הטיסה. לצד התמונה רשימת היעדים: בחירה ביעד מבליטה את הקשת שלו, מעמעמת את האחרות, ונקודה טסה לאורכה מהמוצא ליעד.

מתי משתמשים: סוכנות נסיעות, חברת שילוח או הובלות, רשת סניפים, שירות שמגיע לכמה אזורים, או כל עסק שהסיפור שלו הוא "מכאן לשם". שלושה עד חמישה יעדים. לא כמפה אמיתית לניווט (הקשתות הן איור, לא גאוגרפיה), ולא כשיש עשרות יעדים, שם רשימה עם חיפוש.

שלושה יעדים, עד שעתיים טיסה

בחרו יעד ותראו כמה זמן לוקחת הטיסה ומה כולל המחיר. את כל השאר, מהמלון ועד ההסעה משדה התעופה, אנחנו מסדרים.

לתכנן חופשה
תל אביב לרנקה45 דקות טיסה רודוסשעה ו-20 דקות טיסה אתונהשעתיים טיסה
שחזור. השראה: animtd.store (26.10.2026), סרטון של כרטיס תמונה שעליו נמתחים קווי מסלול עם תוויות יעד. נבנה מאפס מהצפייה. במה זה שונה ממה שיש: g172 הוא גלובוס נקודות בקנבס עם קשתות. כאן הקשתות יושבות על תמונה אמיתית, והרשימה לידן היא הממשק: בחירה ביעד מבליטה את הקשת ושולחת עליה נקודה. המפה היא SVG בגודל 600 על 400, והכרטיס נעול על יחס 3:2 בכל רוחב, כך שהתוויות (HTML במיקום באחוזים) יושבות בדיוק על קצות הקשתות. הקשת של כל מסלול היא עקומה ריבועית שנקודת הבקרה שלה מחושבת מהאמצע, בניצב כלפי מעלה, ב-24% מאורך המסלול. המתיחה: pathLength=1 ו-stroke-dashoffset מ-1 ל-0, בדירוג של 0.2 שנייה בין הקשתות; התוויות נכנסות אחריהן. הנקודה היא אלמנט HTML שה-JS מזיז ב-transform לפי נוסחת העקומה (בלי getPointAtLength, שמתנהג שונה בין דפדפנים כש-pathLength מוגדר), ב-1.5 שניות עם האצה והאטה, כמו המראה ונחיתה. הבחירה: כפתורים עם aria-pressed; בעכבר גם ריחוף. אחרי שהקשתות נמתחו, היעד הראשון נבחר לבד פעם אחת. ניגודיות: הקשתות בצבע הרקע עם צל רך, והתוויות על כרטיס אטום, כך שהן קריאות על כל תמונה. בלי JS הקשתות והתוויות מוצגות, בלי בחירה. הפחתת תנועה: בלי מתיחה ובלי טיסה; הנקודה קופצת ליעד. בטלפון התמונה מעל הרשימה, וזמני הטיסה יורדים מהתוויות (הם נשארים ברשימה).

JS

(function(){
  const root=document.querySelector("[data-rl]");
  if(!root)return;
  const map=root.querySelector(".rl-map"),dot=root.querySelector(".rl-dot");
  const paths=[...root.querySelectorAll(".rl-path")],pins=[...root.querySelectorAll(".rl-pin:not(.is-o)")],its=[...root.querySelectorAll(".rl-it")];
  const still=matchMedia("(prefers-reduced-motion: reduce)"),fine=matchMedia("(hover: hover) and (pointer: fine)");
  const Q=paths.map(p=>p.dataset.q.split(" ").map(Number));
  let cur=-1,raf=0,last=null;
  root.classList.add("is-live");          // בלי JS הקשתות והתוויות כבר מצוירות
  its.forEach(b=>b.setAttribute("aria-pressed","false"));
  // נקודה על העקומה הריבועית, ביחידות המפה (600x400), מומרת לפיקסלים של הכרטיס
  const at=(q,t)=>{const u=1-t;return[u*u*q[0]+2*u*t*q[2]+t*t*q[4],u*u*q[1]+2*u*t*q[3]+t*t*q[5]];};
  const place=xy=>{last=xy;dot.style.transform="translate("+(xy[0]/600*map.clientWidth)+"px,"+(xy[1]/400*map.clientHeight)+"px) translate(-50%,-50%)";};
  const fly=i=>{
    cancelAnimationFrame(raf);
    if(still.matches)return place(at(Q[i],1));
    const t0=performance.now(),D=1500;
    const tick=now=>{
      const t=Math.min(1,(now-t0)/D),k=t<.5?4*t*t*t:1-Math.pow(-2*t+2,3)/2;   // מאיץ בהמראה ומאט בנחיתה
      place(at(Q[i],k));
      if(t<1)raf=requestAnimationFrame(tick);
    };
    raf=requestAnimationFrame(tick);
  };
  const go=i=>{
    if(i===cur)return;cur=i;
    root.classList.add("has-sel");
    [paths,pins].forEach(g=>g.forEach((el,k)=>el.classList.toggle("is-on",k===i)));
    its.forEach((b,k)=>b.setAttribute("aria-pressed",String(k===i)));
    fly(i);
  };
  its.forEach((b,i)=>{
    b.addEventListener("click",()=>go(i));
    b.addEventListener("pointerenter",()=>{if(fine.matches)go(i);});
  });
  addEventListener("resize",()=>{if(last)place(last);});
  // כשהכרטיס נכנס למסך הקשתות נמתחות, ובסוף היעד הראשון נבחר לבד
  const start=()=>{root.classList.add("is-in");setTimeout(()=>{if(cur<0)go(0);},still.matches?0:1500);};
  if(!("IntersectionObserver" in window))return start();
  const io=new IntersectionObserver(es=>{if(es.some(e=>e.isIntersecting)){io.disconnect();start();}},{threshold:.4});
  io.observe(map);
})();
