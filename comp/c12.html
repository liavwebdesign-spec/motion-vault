בדסקטופ עם פחות מחמישה פריטים: C4 גריד שמציג את כולם.

הדמו כאן ניטרלי בכוונה ומגדיר מבנה בלבד: גריד, יחסים וקריסה. צבעים, רדיוסים, גופנים וריווח נלקחים מהעור של הפרויקט. מתג הרוחב למעלה מצמצם את המכולה ולא את חלון הדפדפן, ולכן הקריסה שרואים כאן היא הקריסה האמיתית.

הדמו

רוחב תצוגה:

גלול הצידה, הפריט הבא מציץ

בדסקטופ רק כשבאמת יש חמישה פריטים ומעלה. פחות מזה, לפרוש בגריד (פסיקת הסתרת התוכן). בדסקטופ יש חיצי הקודם והבא בשורת הכותרת, כי גלגלת עכבר לא גוללת הצידה. בקצה החץ נעשה חיוור (aria-disabled), והגלילה מכבדת הפחתת תנועה. בטלפון החיצים יורדים והאגודל גולל. שלושה פריטים וקצת נראים בבת אחת, בגובה 4/5, כדי שהשורה לא תתפוס מסך שלם. המסדרון עצמו מקבל tabindex="0", role="region" ושם, כדי שמקלדת תוכל לגלול אותו וקורא מסך יידע שזו גלריה.

קוד להדבקה

הקוד כאן עומד בפני עצמו. כל טוקן עיצוב נושא ברירת מחדל, מחלקות העזר של המאגר מוטמעות בו, ותגי הסקריפט מסודרים לפי הסדר הנכון. פרויקט שמגדיר --accent או --line משלו יורש אותם אוטומטית.

JS

document.querySelectorAll(".c12w").forEach(function(w){
  var t=w.querySelector(".c12"),p=w.querySelector("[data-dir=prev]"),n=w.querySelector("[data-dir=next]");
  if(!t||!p||!n)return;
  var rm=matchMedia("(prefers-reduced-motion: reduce)");
  function step(){var f=t.firstElementChild;return f?f.getBoundingClientRect().width+(parseFloat(getComputedStyle(t).columnGap)||0):t.clientWidth*.8}
  function sign(){return getComputedStyle(t).direction==="rtl"?-1:1}
  function upd(){var max=t.scrollWidth-t.clientWidth,x=Math.abs(t.scrollLeft);p.setAttribute("aria-disabled",x<=2?"true":"false");n.setAttribute("aria-disabled",x>=max-2?"true":"false")}
  function go(d){t.scrollBy({left:d*step()*sign(),behavior:rm.matches?"auto":"smooth"})}
  p.addEventListener("click",function(){go(-1)});
  n.addEventListener("click",function(){go(1)});
  t.addEventListener("scroll",upd,{passive:true});
  addEventListener("resize",upd);
  upd();
});
